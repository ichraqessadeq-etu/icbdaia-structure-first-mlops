```python
# fraud_detection_pipeline.py

import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.linear_model import LogisticRegression
from sklearn.metrics import accuracy_score
import joblib
import logging

logging.basicConfig(level=logging.INFO)

class FraudDetectionPipeline:
    def __init__(self, data_path):
        self.data_path = data_path
        self.model = LogisticRegression()

    def load_data(self):
        logging.info("Loading data...")
        # For demonstration, we'll create a mock dataset
        self.data = pd.read_csv(self.data_path)

    def preprocess_data(self):
        logging.info("Preprocessing data...")
        # Mock preprocessing: Assume the last column is the target
        self.X = self.data.iloc[:, :-1]
        self.y = self.data.iloc[:, -1]

    def split_data(self, test_size=0.2):
        logging.info("Splitting data...")
        self.X_train, self.X_test, self.y_train, self.y_test = train_test_split(
            self.X, self.y, test_size=test_size, random_state=42
        )

    def train_model(self):
        logging.info("Training model...")
        self.model.fit(self.X_train, self.y_train)

    def evaluate_model(self):
        logging.info("Evaluating model...")
        predictions = self.model.predict(self.X_test)
        accuracy = accuracy_score(self.y_test, predictions)
        logging.info(f"Model accuracy: {accuracy}")
        return accuracy

    def save_model(self, model_path):
        logging.info(f"Saving model to {model_path}...")
        joblib.dump(self.model, model_path)

    def run_pipeline(self, model_path):
        self.load_data()
        self.preprocess_data()
        self.split_data()
        self.train_model()
        accuracy = self.evaluate_model()
        self.save_model(model_path)
        return accuracy

if __name__ == "__main__":
    pipeline = FraudDetectionPipeline(data_path='fraud_data.csv')
    pipeline.run_pipeline(model_path='fraud_detection_model.pkl')
```

In this module:
- We load data from a CSV file.
- Preprocess it (assume simple split for features and target).
- Split the data into training and testing sets.
- Train a Logistic Regression model.
- Evaluate and save the model.
