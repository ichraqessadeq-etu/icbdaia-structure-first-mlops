# test_fraud_detection_pipeline.py

import pytest
import pandas as pd
from sklearn.datasets import make_classification
from fraud_detection_pipeline import FraudDetectionPipeline
import os

@pytest.fixture
def sample_data():
    """Create a sample dataset for testing."""
    X, y = make_classification(n_samples=100, n_features=5, random_state=42)
    df = pd.DataFrame(X, columns=[f'feature_{i}' for i in range(X.shape[1])])
    df['is_fraud'] = y
    file_path = 'test_fraud_data.csv'
    df.to_csv(file_path, index=False)
    yield file_path
    os.remove(file_path)

@pytest.fixture
def pipeline(sample_data):
    """Initialize the FraudDetectionPipeline with the sample data."""
    model_path = 'test_fraud_model.pkl'
    pipeline = FraudDetectionPipeline(data_path=sample_data, model_path=model_path)
    yield pipeline
    if os.path.exists(model_path):
        os.remove(model_path)

def test_load_data(pipeline):
    """Test loading of data."""
    X, y = pipeline.load_data()
    assert not X.empty, "Feature set should not be empty"
    assert not y.empty, "Target set should not be empty"
    assert 'is_fraud' not in X.columns, "'is_fraud' should not be in features"

def test_split_data(pipeline):
    """Test data splitting."""
    pipeline.load_data()
    X_train, X_test, y_train, y_test = pipeline.split_data()
    assert len(X_train) > 0 and len(X_test) > 0, "Train and test sets should not be empty"
    assert len(y_train) > 0 and len(y_test) > 0, "Train and test labels should not be empty"

def test_train_model(pipeline):
    """Test model training."""
    pipeline.load_data()
    pipeline.split_data()
    model = pipeline.train_model()
    assert model is not None, "Model should not be None after training"
    assert hasattr(model, 'coef_'), "Trained model should have coefficients"

def test_evaluate_model(pipeline):
    """Test model evaluation."""
    pipeline.load_data()
    pipeline.split_data()
    pipeline.train_model()
    accuracy, precision, recall = pipeline.evaluate_model()
    assert 0 <= accuracy <= 1, "Accuracy should be between 0 and 1"
    assert 0 <= precision <= 1, "Precision should be between 0 and 1"
    assert 0 <= recall <= 1, "Recall should be between 0 and 1"

def test_save_model(pipeline):
    """Test saving of the model."""
    pipeline.load_data()
    pipeline.split_data()
    pipeline.train_model()
    pipeline.save_model()
    assert os.path.exists(pipeline.model_path), "Model file should exist after saving"

### b) GitHub Actions Workflow

```yaml
# .github/workflows/python-app.yml

name: Python application

on:
  push:
    branches: [ main ]
  pull_request:
    branches: [ main ]

jobs:
  build:

    runs-on: ubuntu-latest

    steps:
    - name: Checkout code
      uses: actions/checkout@v2

    - name: Set up Python
      uses: actions/setup-python@v2
      with:
        python-version: '3.x'

    - name: Install dependencies
      run: |
        python -m pip install --upgrade pip
        pip install -r requirements.txt
        pip install pytest scikit-learn pandas

    - name: Run tests
      run: |
        pytest test_fraud_detection_pipeline.py
```
