# test_fraud_detection_pipeline.py

import pytest
import pandas as pd
import numpy as np
from fraud_detection_pipeline import load_data, preprocess_data, train_model, evaluate_model, save_model
from sklearn.ensemble import RandomForestClassifier
from sklearn.preprocessing import StandardScaler
import os

# Sample data
@pytest.fixture
def sample_data():
    data = {
        'feature1': [1.0, 2.0, 3.0, np.nan, 5.0],
        'feature2': [2.0, 3.5, np.nan, 5.0, 6.0],
        'is_fraud': [0, 1, 0, 0, 1]
    }
    return pd.DataFrame(data)

def test_load_data(tmp_path):
    # Create a temporary CSV file
    file_path = tmp_path / "test_data.csv"
    df = pd.DataFrame({'col1': [1, 2], 'col2': [3, 4]})
    df.to_csv(file_path, index=False)

    # Test load_data
    loaded_df = load_data(file_path)
    pd.testing.assert_frame_equal(df, loaded_df)

def test_preprocess_data(sample_data):
    X_scaled, y, scaler = preprocess_data(sample_data)
    
    assert isinstance(X_scaled, np.ndarray)
    assert isinstance(y, pd.Series)
    assert isinstance(scaler, StandardScaler)
    assert X_scaled.shape[0] == sample_data.shape[0]  # Number of samples should match
    assert y.equals(sample_data['is_fraud'])

def test_train_model(sample_data):
    X_scaled, y, _ = preprocess_data(sample_data)
    model = train_model(X_scaled, y)
    
    assert isinstance(model, RandomForestClassifier)
    assert hasattr(model, 'predict')

def test_evaluate_model(sample_data):
    X_scaled, y, _ = preprocess_data(sample_data)
    model = train_model(X_scaled, y)
    accuracy, precision, recall = evaluate_model(model, X_scaled, y)
    
    assert 0 <= accuracy <= 1
    assert 0 <= precision <= 1
    assert 0 <= recall <= 1

def test_save_model(tmp_path):
    model = RandomForestClassifier()
    scaler = StandardScaler()
    model_path = tmp_path / "model.pkl"
    scaler_path = tmp_path / "scaler.pkl"
    
    save_model(model, scaler, model_path, scaler_path)
    
    assert os.path.exists(model_path)
    assert os.path.exists(scaler_path)
