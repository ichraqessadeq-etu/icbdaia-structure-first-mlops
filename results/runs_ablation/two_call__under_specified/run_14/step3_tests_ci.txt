# test_fraud_detection_pipeline.py

import pytest
import pandas as pd
import numpy as np
from fraud_detection_pipeline import FraudDetectionPipeline

@pytest.fixture
def sample_data(tmp_path):
    """Create a sample dataset for testing."""
    data = {
        'feature1': np.random.rand(100),
        'feature2': np.random.rand(100),
        'feature3': np.random.rand(100),
        'is_fraud': np.random.randint(0, 2, size=100)
    }
    df = pd.DataFrame(data)
    file_path = tmp_path / "sample_data.csv"
    df.to_csv(file_path, index=False)
    return file_path

def test_load_data(sample_data):
    """Test loading data functionality."""
    pipeline = FraudDetectionPipeline(data_path=sample_data)
    pipeline.load_data()
    assert hasattr(pipeline, 'data'), "Data not loaded correctly"
    assert not pipeline.data.empty, "Loaded data is empty"

def test_preprocess_data(sample_data):
    """Test data preprocessing functionality."""
    pipeline = FraudDetectionPipeline(data_path=sample_data)
    pipeline.load_data()
    pipeline.preprocess_data()
    assert hasattr(pipeline, 'X'), "Features not extracted"
    assert hasattr(pipeline, 'y'), "Target not extracted"
    assert len(pipeline.X) == len(pipeline.y), "Features and target size mismatch"

def test_train_model(sample_data):
    """Test model training functionality."""
    pipeline = FraudDetectionPipeline(data_path=sample_data)
    pipeline.load_data()
    pipeline.preprocess_data()
    pipeline.train_model()
    assert hasattr(pipeline, 'model'), "Model not trained"

def test_evaluate_model(sample_data):
    """Test model evaluation functionality."""
    pipeline = FraudDetectionPipeline(data_path=sample_data)
    pipeline.run_pipeline()
    metrics = pipeline.evaluate_model()
    assert isinstance(metrics, dict), "Metrics should be a dictionary"
    for metric in ['accuracy', 'precision', 'recall', 'f1_score']:
        assert metric in metrics, f"{metric} not in evaluation metrics"
        assert 0 <= metrics[metric] <= 1, f"{metric} should be between 0 and 1"

def test_save_model(sample_data, tmp_path):
    """Test model saving functionality."""
    model_path = tmp_path / "fraud_model.pkl"
    pipeline = FraudDetectionPipeline(data_path=sample_data, model_path=model_path)
    pipeline.run_pipeline()
    assert model_path.exists(), "Model file not saved"

### (b) GitHub Actions Workflow

```yaml
# .github/workflows/python-package.yml

name: Python Package

on:
  push:
    branches: [main]
  pull_request:
    branches: [main]

jobs:
  build:

    runs-on: ubuntu-latest

    strategy:
      matrix:
        python-version: [3.8, 3.9, 3.10]

    steps:
    - uses: actions/checkout@v2

    - name: Set up Python ${{ matrix.python-version }}
      uses: actions/setup-python@v2
      with:
        python-version: ${{ matrix.python-version }}

    - name: Install dependencies
      run: |
        python -m pip install --upgrade pip
        pip install pandas scikit-learn joblib pytest

    - name: Test with pytest
      run: |
        pytest --tb=short --disable-warnings
```
