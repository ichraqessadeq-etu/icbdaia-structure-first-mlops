import pytest
import pandas as pd
from fraud_detection_pipeline import FraudDetectionPipeline

    os.remove('temp_data.csv')

def test_preprocess_data(pipeline, sample_data):
    """Test preprocessing data."""
    X_train, X_test, y_train, y_test = pipeline.preprocess_data(sample_data)
    assert X_train.shape[0] + X_test.shape[0] == sample_data.shape[0]
    assert y_train.shape[0] + y_test.shape[0] == sample_data.shape[0]

def test_train_model(pipeline, sample_data):
    """Test model training."""
    X_train, X_test, y_train, y_test = pipeline.preprocess_data(sample_data)
    pipeline.train_model(X_train, y_train)

### b) GitHub Actions Workflow

```yaml
# .github/workflows/test.yml

name: Python Package

on:
  push:
    branches: [ "main" ]
  pull_request:
    branches: [ "main" ]

jobs:
  build:
    runs-on: ubuntu-latest

    steps:
    - name: Checkout repository
      uses: actions/checkout@v2

    - name: Set up Python
      uses: actions/setup-python@v2
      with:
        python-version: '3.8'

    - name: Install dependencies
      run: |
        python -m pip install --upgrade pip
        pip install pytest scikit-learn joblib pandas

    - name: Test with pytest
      run: |
        pytest --maxfail=1 --disable-warnings -v
```
