To create a CI/CD pipeline for a fraud detection model, we need to break it down into several components. The pipeline module will typically include steps for data loading, preprocessing, model training, evaluation, and deployment packaging. Below is a simplified version of a Python module for this purpose:

```python
import os
import joblib
import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.ensemble import RandomForestClassifier
from sklearn.metrics import classification_report, accuracy_score
from sklearn.preprocessing import StandardScaler

class FraudDetectionPipeline:
    def __init__(self, data_path, model_output_path):
        self.data_path = data_path
        self.model_output_path = model_output_path
        self.model = RandomForestClassifier(n_estimators=100, random_state=42)
        self.scaler = StandardScaler()

    def load_data(self):
        # Load the dataset
        try:
            data = pd.read_csv(self.data_path)
            print(f"Data loaded successfully from {self.data_path}")
            return data
        except FileNotFoundError:
            print(f"File not found at {self.data_path}")
            raise

    def preprocess_data(self, data):
        # Preprocess the data: handle missing values, feature scaling, etc.
        data = data.dropna()  # Drop missing values for simplicity
        X = data.drop('label', axis=1)  # Assuming 'label' is the target column
        y = data['label']
        X_scaled = self.scaler.fit_transform(X)
        print("Data preprocessing completed.")
        return train_test_split(X_scaled, y, test_size=0.2, random_state=42)

    def train_model(self, X_train, y_train):
        # Train the model
        self.model.fit(X_train, y_train)
        print("Model training completed.")

    def evaluate_model(self, X_test, y_test):
        # Evaluate the model
        predictions = self.model.predict(X_test)
        accuracy = accuracy_score(y_test, predictions)
        report = classification_report(y_test, predictions)
        print(f"Model Accuracy: {accuracy}")
        print("Classification Report:")
        print(report)
        return accuracy, report

    def save_model(self):
        # Save the trained model to the output path
        if not os.path.exists(self.model_output_path):
            os.makedirs(self.model_output_path)
        model_file_path = os.path.join(self.model_output_path, 'fraud_detection_model.pkl')
        joblib.dump(self.model, model_file_path)
        print(f"Model saved at {model_file_path}")

    def run_pipeline(self):
        data = self.load_data()
        X_train, X_test, y_train, y_test = self.preprocess_data(data)
        self.train_model(X_train, y_train)
        accuracy, report = self.evaluate_model(X_test, y_test)
        self.save_model()
        return accuracy, report

# Example usage
if __name__ == '__main__':
    pipeline = FraudDetectionPipeline(data_path='data/fraud_data.csv', model_output_path='models/')
    pipeline.run_pipeline()
```

### Explanation:

- **Data Loading**: The `load_data` method reads a CSV file containing the dataset.
- **Preprocessing**: The `preprocess_data` method handles basic preprocessing like dropping missing values and scaling features.
- **Training**: The `train_model` method trains a Random Forest classifier.
- **Evaluation**: The `evaluate_model` method evaluates the model using accuracy and classification report.
- **Saving**: The `save_model` method saves the trained model using `joblib`.
- **Pipeline Execution**: The `run_pipeline` method orchestrates the entire process from loading data to saving the model.
