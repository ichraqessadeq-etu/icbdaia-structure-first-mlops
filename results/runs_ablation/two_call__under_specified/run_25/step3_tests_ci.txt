# test_fraud_detection_pipeline.py
import pytest
import pandas as pd
import numpy as np
from sklearn.datasets import make_classification
from fraud_detection_pipeline import FraudDetectionPipeline

@pytest.fixture
def sample_data(tmpdir):
    """Generate a sample dataset and save it as a CSV file."""
    X, y = make_classification(n_samples=100, n_features=20, n_informative=2, n_redundant=0, random_state=42)
    data = pd.DataFrame(X, columns=[f'feature_{i}' for i in range(20)])
    data['is_fraud'] = y
    file_path = tmpdir.join("sample_data.csv")
    data.to_csv(file_path, index=False)
    return str(file_path)

@pytest.fixture
def pipeline(sample_data):
    """Initialize the FraudDetectionPipeline with sample data."""
    return FraudDetectionPipeline(data_path=sample_data)

def test_load_data(pipeline):
    """Test loading data functionality."""
    pipeline.load_data()
    assert hasattr(pipeline, 'data')
    assert not pipeline.data.empty

def test_preprocess_data(pipeline):
    """Test data preprocessing functionality."""
    pipeline.load_data()
    pipeline.preprocess_data()
    assert hasattr(pipeline, 'X_train')
    assert hasattr(pipeline, 'X_test')
    assert hasattr(pipeline, 'y_train')
    assert hasattr(pipeline, 'y_test')

def test_train_model(pipeline):
    """Test model training functionality."""
    pipeline.load_data()

    pipeline.train_model()
    assert hasattr(pipeline.model, 'feature_importances_')

    pipeline.load_data()

    pipeline.train_model()
    pipeline.save_model(model_path)
    assert model_path.exists()
    pipeline.load_model(model_path)

### (b) GitHub Actions Workflow

```yaml
# .github/workflows/python-app.yml
name: Python application

on:
  push:
    branches:
      - main
  pull_request:
    branches:
      - main

jobs:
  build:
    runs-on: ubuntu-latest

    steps:
    - name: Check out repository
      uses: actions/checkout@v2

    - name: Set up Python
      uses: actions/setup-python@v2
      with:
        python-version: '3.8'

    - name: Install dependencies
      run: |
        python -m pip install --upgrade pip
        pip install pytest scikit-learn pandas joblib

    - name: Test with pytest
      run: |
        pytest
```

In this setup:

- **Testing**: The pytest test file `test_fraud_detection_pipeline.py` includes tests for each critical function within the pipeline, leveraging fixtures to set up sample data and environment.
- **GitHub Actions**: The workflow is designed to trigger on pushes and pull requests to the `main` branch. It sets up a Python environment, installs dependencies, and runs the pytest tests. Adjust the Python version and dependencies as per your project's requirements.
