### (a) Pytest Test File

```python
# test_fraud_detection_pipeline.py

import pytest
import os
import pandas as pd
from fraud_detection_pipeline import FraudDetectionPipeline

# Create a sample dataframe to use for testing
@pytest.fixture
def sample_data(tmpdir):
    data = {
        'feature1': [0.1, 0.2, 0.3, 0.4],
        'feature2': [1.0, 0.9, 0.8, 0.7],
        'label': [0, 1, 0, 1]
    }
    df = pd.DataFrame(data)
    file_path = tmpdir.join('sample_data.csv')
    df.to_csv(file_path, index=False)
    return str(file_path)

def test_load_data(sample_data):
    pipeline = FraudDetectionPipeline(data_path=sample_data)
    pipeline.load_data()
    assert pipeline.data is not None
    assert not pipeline.data.empty
    assert pipeline.data.shape == (4, 3)

def test_preprocess_data(sample_data):
    pipeline = FraudDetectionPipeline(data_path=sample_data)
    pipeline.load_data()
    pipeline.preprocess_data()
    assert hasattr(pipeline, 'X')
    assert hasattr(pipeline, 'y')
    assert pipeline.X.shape[1] == 2  # Two features
    assert len(pipeline.y) == 4

def test_split_data(sample_data):
    pipeline = FraudDetectionPipeline(data_path=sample_data)
    pipeline.load_data()
    pipeline.preprocess_data()
    pipeline.split_data(test_size=0.5)
    assert len(pipeline.X_train) == 2
    assert len(pipeline.X_test) == 2
    assert len(pipeline.y_train) == 2
    assert len(pipeline.y_test) == 2

def test_train_model(sample_data):
    pipeline = FraudDetectionPipeline(data_path=sample_data)
    pipeline.load_data()
    pipeline.preprocess_data()
    pipeline.split_data(test_size=0.5)
    pipeline.train_model()
    assert hasattr(pipeline.model, 'feature_importances_')

def test_evaluate_model(sample_data, capsys):
    pipeline = FraudDetectionPipeline(data_path=sample_data)
    pipeline.load_data()
    pipeline.preprocess_data()
    pipeline.split_data(test_size=0.5)
    pipeline.train_model()
    pipeline.evaluate_model()
    captured = capsys.readouterr()
    assert "Model accuracy:" in captured.out
    assert "Classification Report:" in captured.out

def test_save_model(sample_data, tmpdir):
    model_path = tmpdir.join('fraud_model.pkl')
    pipeline = FraudDetectionPipeline(data_path=sample_data, model_path=str(model_path))
    pipeline.load_data()
    pipeline.preprocess_data()
    pipeline.split_data(test_size=0.5)
    pipeline.train_model()
    pipeline.save_model()
    assert os.path.exists(model_path)
```

### (b) GitHub Actions Workflow

```yaml
# .github/workflows/python-test.yml

name: Python Tests

on: [push, pull_request]

jobs:
  build:
    runs-on: ubuntu-latest

    steps:
    - name: Checkout code
      uses: actions/checkout@v2

    - name: Set up Python
      uses: actions/setup-python@v2
      with:
        python-version: '3.8'

    - name: Install dependencies
      run: |
        python -m pip install --upgrade pip
        pip install pytest pandas scikit-learn joblib

    - name: Run tests
      run: |
        pytest
```

This setup assumes that you have a GitHub repository where the `fraud_detection_pipeline.py` script and the test file `test_fraud_detection_pipeline.py` are located. The GitHub Actions workflow will automatically trigger on pushes and pull requests and will run the tests using pytest.
