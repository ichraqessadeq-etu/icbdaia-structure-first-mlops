# test_fraud_detection_pipeline.py

import pytest
import pandas as pd
from sklearn.ensemble import RandomForestClassifier
from fraud_detection_pipeline import (
    load_data,
    preprocess_data,
    split_data,
    train_model,
    evaluate_model,
    save_model
)

@pytest.fixture
def sample_data():
    # Create a small sample dataframe
    data = {
        'feature1': [0.1, 0.2, 0.3, 0.4],
        'feature2': [1, 0, 1, 0],
        'Class': [0, 1, 0, 1]
    }
    return pd.DataFrame(data)

def test_load_data(sample_data, tmp_path):
    # Save sample data to a CSV file
    file_path = tmp_path / "sample_data.csv"
    sample_data.to_csv(file_path, index=False)
    # Test load_data function
    loaded_data = load_data(file_path)
    pd.testing.assert_frame_equal(sample_data, loaded_data)

def test_preprocess_data(sample_data):
    # Test preprocess_data function
    X, y = preprocess_data(sample_data)
    assert X.shape[1] == sample_data.shape[1] - 1
    assert all(y == sample_data['Class'])

def test_split_data(sample_data):
    # Test split_data function
    X, y = preprocess_data(sample_data)
    X_train, X_test, y_train, y_test = split_data(X, y, test_size=0.5, random_state=42)
    assert len(X_train) == 2
    assert len(X_test) == 2
    assert len(y_train) == 2
    assert len(y_test) == 2

def test_train_model(sample_data):
    # Test train_model function
    X, y = preprocess_data(sample_data)
    X_train, X_test, y_train, y_test = split_data(X, y, test_size=0.5, random_state=42)
    model = train_model(X_train, y_train)
    assert isinstance(model, RandomForestClassifier)

def test_evaluate_model(sample_data):
    # Test evaluate_model function
    X, y = preprocess_data(sample_data)
    X_train, X_test, y_train, y_test = split_data(X, y, test_size=0.5, random_state=42)
    model = train_model(X_train, y_train)
    accuracy, precision, recall = evaluate_model(model, X_test, y_test)

### (b) GitHub Actions Workflow

Create a file `.github/workflows/python-app.yml` with the following content:

```yaml
name: Python application

on:
  push:
    branches: [main]
  pull_request:
    branches: [main]

jobs:
  build:

    runs-on: ubuntu-latest

    steps:
    - name: Checkout repository
      uses: actions/checkout@v2

    - name: Set up Python
      uses: actions/setup-python@v2
      with:
        python-version: 3.8

    - name: Install dependencies
      run: |
        python -m pip install --upgrade pip
        pip install pytest scikit-learn pandas joblib

    - name: Run tests
      run: |
        pytest
```

This setup ensures that every time a change is pushed to the `main` branch or a pull request is made against it, the tests are automatically run in a clean environment.
