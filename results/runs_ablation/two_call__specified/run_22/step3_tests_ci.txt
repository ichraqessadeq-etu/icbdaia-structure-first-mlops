# test_pipeline.py
import pytest
from unittest.mock import patch, MagicMock
import pandas as pd
from pipeline_module import load_data, preprocess_data, split_data, train_model, evaluate_model, log_experiment

# Mock data
MOCK_DATA = pd.DataFrame({
    'feature1': [0.1, 0.2, 0.2, 0.4],
    'feature2': [0.3, 0.1, 0.4, 0.2],
    'Class': [0, 1, 0, 1]
})

def test_load_data(mocker):
    mocker.patch('pandas.read_csv', return_value=MOCK_DATA)
    data = load_data('fake_path.csv')
    pd.testing.assert_frame_equal(data, MOCK_DATA)

def test_preprocess_data():
    X, y = preprocess_data(MOCK_DATA)
    assert list(X.columns) == ['feature1', 'feature2']
    assert list(y) == [0, 1, 0, 1]

def test_split_data():
    X, y = preprocess_data(MOCK_DATA)
    X_train, X_test, y_train, y_test = split_data(X, y, test_size=0.5, random_state=42)
    assert len(X_train) == 2
    assert len(X_test) == 2
    assert len(y_train) == 2
    assert len(y_test) == 2

def test_train_model():
    X, y = preprocess_data(MOCK_DATA)
    X_train, X_test, y_train, y_test = split_data(X, y, test_size=0.5, random_state=42)
    model = train_model(X_train, y_train)
    assert model is not None

def test_evaluate_model():
    X, y = preprocess_data(MOCK_DATA)
    X_train, X_test, y_train, y_test = split_data(X, y, test_size=0.5, random_state=42)
    model = train_model(X_train, y_train)
    accuracy, report = evaluate_model(model, X_test, y_test)
    assert isinstance(accuracy, float)
    assert isinstance(report, str)

@patch('mlflow.start_run')
def test_log_experiment(mock_start_run):
    mock_run = MagicMock()
    mock_start_run.return_value.__enter__.return_value = mock_run
    log_experiment(0.95, "some report", MagicMock())
    mock_run.log_metric.assert_called_with("accuracy", 0.95)

### (b) Complete GitHub Actions Workflow

```yaml
# .github/workflows/python-app.yml
name: Python application

on:
  push:
    branches: [ "main" ]
  pull_request:
    branches: [ "main" ]

jobs:
  build:

    runs-on: ubuntu-latest

    steps:
    - uses: actions/checkout@v2

    - name: Set up Python
      uses: actions/setup-python@v2
      with:
        python-version: '3.8'

    - name: Install dependencies
      run: |
        python -m pip install --upgrade pip
        pip install pandas scikit-learn mlflow pytest pytest-mock

    - name: Run tests
      run: |
        pytest --doctest-modules --junitxml=junit/test-results.xml
```

- **Test File**: The test file uses `pytest` along with `pytest-mock` for mocking and `unittest.mock.patch` for some specific parts like `mlflow`. It tests each function in the pipeline module to verify functionality.

- **GitHub Actions Workflow**: The workflow is triggered on pushes and pull requests to the main branch. It sets up Python, installs dependencies, and runs the tests using `pytest`. The results are formatted in JUnit XML for better integration with different CI/CD systems.
