Here's a complete Python pipeline script for a fraud detection model using scikit-learn's `RandomForestClassifier`, leveraging MLflow for experiment tracking. This script handles data loading, model training, evaluation, and logging to MLflow. It also includes basic error handling.

```python
import numpy as np
import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.ensemble import RandomForestClassifier
from sklearn.metrics import classification_report, accuracy_score
from sklearn.utils import shuffle
from sklearn.utils.class_weight import compute_sample_weight
import mlflow
import mlflow.sklearn
import logging

# Configure logging
logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)

def load_data(file_path):
    try:
        data = pd.read_csv(file_path)
        logger.info("Data loaded successfully.")
        return data
    except Exception as e:
        logger.error(f"Failed to load data: {e}")
        raise

def preprocess_data(data):
    try:
        X = data.drop('Class', axis=1)
        y = data['Class']
        logger.info("Data preprocessed successfully.")
        return X, y
    except Exception as e:
        logger.error(f"Failed to preprocess data: {e}")
        raise

def split_data(X, y, test_size=0.2, random_state=42):
    try:
        X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=test_size, random_state=random_state, stratify=y)
        logger.info("Data split into train and test sets.")
        return X_train, X_test, y_train, y_test
    except Exception as e:
        logger.error(f"Failed to split data: {e}")
        raise

def train_model(X_train, y_train):
    try:
        sample_weights = compute_sample_weight('balanced', y_train)
        model = RandomForestClassifier(random_state=42)
        model.fit(X_train, y_train, sample_weight=sample_weights)
        logger.info("Model trained successfully.")
        return model
    except Exception as e:
        logger.error(f"Failed to train model: {e}")
        raise

def evaluate_model(model, X_test, y_test):
    try:
        y_pred = model.predict(X_test)
        accuracy = accuracy_score(y_test, y_pred)
        report = classification_report(y_test, y_pred)
        logger.info("Model evaluated successfully.")
        logger.info(f"Accuracy: {accuracy}\n")
        logger.info(f"Classification Report:\n{report}")
        return accuracy, report
    except Exception as e:
        logger.error(f"Failed to evaluate model: {e}")
        raise

def log_experiment(accuracy, report, model):
    try:
        mlflow.set_experiment("Fraud Detection Experiment")
        with mlflow.start_run():
            mlflow.log_metric("accuracy", accuracy)
            mlflow.log_text(report, "classification_report.txt")
            mlflow.sklearn.log_model(model, "model")
        logger.info("Experiment logged successfully.")
    except Exception as e:
        logger.error(f"Failed to log experiment: {e}")
        raise

def main(file_path):
    try:
        data = load_data(file_path)
        X, y = preprocess_data(data)
        X, y = shuffle(X, y, random_state=42)
        X_train, X_test, y_train, y_test = split_data(X, y)
        model = train_model(X_train, y_train)
        accuracy, report = evaluate_model(model, X_test, y_test)
        log_experiment(accuracy, report, model)
    except Exception as e:
        logger.error(f"Pipeline failed: {e}")

if __name__ == "__main__":
    main("creditcard.csv")
```

This script assumes that your dataset is named `creditcard.csv` and located in the same directory as the script. It handles exceptions at each stage and logs informative messages to facilitate debugging.
