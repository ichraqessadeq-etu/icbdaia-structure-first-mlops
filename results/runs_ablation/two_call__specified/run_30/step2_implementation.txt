Below is a complete Python pipeline script for training, evaluating, and logging a scikit-learn `RandomForestClassifier` model to MLflow. This script handles data loading, preprocessing, model training, and evaluation. It also includes error handling.

```python
import pandas as pd
import numpy as np
from sklearn.ensemble import RandomForestClassifier
from sklearn.model_selection import train_test_split
from sklearn.metrics import classification_report, confusion_matrix
from sklearn.utils import resample
import mlflow
import mlflow.sklearn
import logging
import joblib

# Configure logging
logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)

def load_data(file_path):
    """Load the dataset from a CSV file."""
    try:
        data = pd.read_csv(file_path)
        logger.info("Data loaded successfully.")
        return data
    except Exception as e:
        logger.error(f"Error loading data: {e}")
        raise

def preprocess_data(data):
    """Preprocess the dataset."""
    try:
        # Separate majority and minority classes
        df_majority = data[data.Class == 0]
        df_minority = data[data.Class == 1]

        # Upsample minority class
        df_minority_upsampled = resample(df_minority, 
                                         replace=True,     # sample with replacement
                                         n_samples=len(df_majority),    # to match majority class
                                         random_state=123) # reproducible results

        # Combine majority class with upsampled minority class
        df_upsampled = pd.concat([df_majority, df_minority_upsampled])

        # Shuffle the data
        df_upsampled = df_upsampled.sample(frac=1).reset_index(drop=True)

        X = df_upsampled.drop('Class', axis=1)
        y = df_upsampled['Class']
        
        logger.info("Data preprocessing completed.")
        return X, y
    except Exception as e:
        logger.error(f"Error during data preprocessing: {e}")
        raise

def train_model(X_train, y_train):
    """Train the RandomForest model."""
    try:
        model = RandomForestClassifier(n_estimators=100, random_state=42)
        model.fit(X_train, y_train)
        logger.info("Model training completed.")
        return model
    except Exception as e:
        logger.error(f"Error during model training: {e}")
        raise

def evaluate_model(model, X_test, y_test):
    """Evaluate the model."""
    try:
        predictions = model.predict(X_test)
        report = classification_report(y_test, predictions)
        matrix = confusion_matrix(y_test, predictions)
        
        logger.info("Model evaluation completed.")
        logger.info(f"Classification Report:\n {report}")
        logger.info(f"Confusion Matrix:\n {matrix}")
        
        return report, matrix
    except Exception as e:
        logger.error(f"Error during model evaluation: {e}")
        raise

def log_to_mlflow(model, X_test, y_test):
    """Log the model and metrics to MLflow."""
    try:
        with mlflow.start_run():
            mlflow.sklearn.log_model(model, "model")
            mlflow.log_params(model.get_params())
            
            predictions = model.predict(X_test)
            accuracy = np.mean(predictions == y_test)
            mlflow.log_metric("accuracy", accuracy)
            
            # Save the model locally
            joblib.dump(model, 'random_forest_model.pkl')
            
            logger.info("Model and metrics logged to MLflow.")
    except Exception as e:
        logger.error(f"Error during logging to MLflow: {e}")
        raise

def main():
    file_path = 'creditcard.csv'
    
    # Load and preprocess data
    data = load_data(file_path)
    X, y = preprocess_data(data)
    
    # Split data into train and test sets
    X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.3, random_state=42)
    
    # Train and evaluate model
    model = train_model(X_train, y_train)
    evaluate_model(model, X_test, y_test)
    
    # Log to MLflow
    log_to_mlflow(model, X_test, y_test)

if __name__ == "__main__":
    try:
        main()
    except Exception as e:
        logger.error(f"Pipeline execution failed: {e}")
```

This script includes error handling, logging, and model persistence. It preprocesses the data by upsampling the minority class to handle class imbalance, trains a `RandomForestClassifier`, evaluates it, and logs the results to MLflow.
