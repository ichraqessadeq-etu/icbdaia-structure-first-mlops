import pytest
import pandas as pd
from sklearn.ensemble import RandomForestClassifier
from pipeline_module import load_data, preprocess_data, train_model, evaluate_model, save_model, load_model

# Test data path
TEST_DATA_PATH = "test_creditcard.csv"  # Use a small test dataset for testing purposes

# Mock data for testing
@pytest.fixture
def mock_data():
    data = {
        'V1': [0.1, -0.1, 0.2, -0.2, 0.3],
        'V2': [0.1, -0.1, 0.2, -0.2, 0.3],
        'V3': [0.1, -0.1, 0.2, -0.2, 0.3],
        'Class': [0, 1, 0, 1, 0]
    }
    return pd.DataFrame(data)

def test_load_data():
    data = load_data(TEST_DATA_PATH)
    assert isinstance(data, pd.DataFrame)

def test_preprocess_data(mock_data):
    X_train, X_test, y_train, y_test = preprocess_data(mock_data)
    assert len(X_train) > 0
    assert len(X_test) > 0
    assert len(y_train) > 0
    assert len(y_test) > 0

def test_train_model(mock_data):
    X_train, _, y_train, _ = preprocess_data(mock_data)
    model = train_model(X_train, y_train)
    assert isinstance(model, RandomForestClassifier)

def test_evaluate_model(mock_data):
    X_train, X_test, y_train, y_test = preprocess_data(mock_data)
    model = train_model(X_train, y_train)
    roc_auc, report = evaluate_model(model, X_test, y_test)
    assert roc_auc >= 0.0  # ROC AUC should be between 0 and 1
    assert isinstance(report, str)

def test_save_and_load_model(mock_data, tmp_path):
    X_train, _, y_train, _ = preprocess_data(mock_data)
    model = train_model(X_train, y_train)
    model_path = tmp_path / "test_model.joblib"
    save_model(model, model_path)
    loaded_model = load_model(model_path)
    assert isinstance(loaded_model, RandomForestClassifier)
