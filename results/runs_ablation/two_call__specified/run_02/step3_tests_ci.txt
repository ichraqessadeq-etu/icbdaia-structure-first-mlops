import pytest
import pandas as pd
from sklearn.ensemble import RandomForestClassifier
from pipeline_module import load_data, preprocess_data, train_model, evaluate_model  # Adjust the import statement as needed

def test_load_data():
    # Test that the data loads correctly
    data = load_data("tests/test_data.csv")
    assert isinstance(data, pd.DataFrame)
    assert not data.empty

def test_preprocess_data():
    # Test that data is split into train and test sets
    data = pd.DataFrame({
        'feature1': [1, 2, 3, 4],
        'feature2': [5, 6, 7, 8],
        'Class': [0, 1, 0, 1]
    })
    X_train, X_test, y_train, y_test = preprocess_data(data)
    assert len(X_train) > 0
    assert len(X_test) > 0
    assert len(y_train) > 0
    assert len(y_test) > 0

def test_train_model():
    # Test model training
    X_train = pd.DataFrame({'feature1': [1, 2, 3], 'feature2': [4, 5, 6]})
    y_train = pd.Series([0, 1, 0])
    model = train_model(X_train, y_train)
    assert isinstance(model, RandomForestClassifier)

def test_evaluate_model():
    # Test model evaluation
    X_test = pd.DataFrame({'feature1': [1, 2], 'feature2': [4, 5]})
    y_test = pd.Series([0, 1])
    model = RandomForestClassifier(random_state=42)

### (b) GitHub Actions Workflow

```yaml
name: Python Package

on:
  push:
    branches: [ main ]
  pull_request:
    branches: [ main ]

jobs:
  build:

    runs-on: ubuntu-latest

    steps:
    - uses: actions/checkout@v2

    - name: Set up Python 3.8
      uses: actions/setup-python@v2
      with:
        python-version: '3.8'
    
    - name: Install dependencies
      run: |
        python -m pip install --upgrade pip
        pip install pytest
        pip install scikit-learn
        pip install pandas
        pip install numpy
        pip install mlflow
    
    - name: Test with pytest
      run: |
        pytest tests/
```
