Below is a complete Python pipeline script that covers data loading, model training, evaluation, and logging to MLflow. This script includes error handling and is designed to be production-ready.

```python
import numpy as np
import pandas as pd
from sklearn.ensemble import RandomForestClassifier
from sklearn.model_selection import train_test_split
from sklearn.metrics import classification_report, roc_auc_score
from sklearn.utils import resample
import mlflow
import mlflow.sklearn
import logging
from mlflow.exceptions import MlflowException

# Configure logging
logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)


def load_data(file_path):
    """Load the dataset from a CSV file."""
    try:
        data = pd.read_csv(file_path)
        logger.info("Data loaded successfully.")
        return data
    except Exception as e:
        logger.error(f"Failed to load data: {str(e)}")
        raise


def preprocess_data(df):
    """Preprocess the data: Handle class imbalance and split into features and target."""
    try:
        # Separate input features and target
        X = df.drop('Class', axis=1)
        y = df['Class']
        
        # Address class imbalance by downsampling majority class
        df_majority = df[df.Class == 0]
        df_minority = df[df.Class == 1]
        
        df_majority_downsampled = resample(df_majority,
                                           replace=False,
                                           n_samples=len(df_minority),
                                           random_state=42)
        
        df_downsampled = pd.concat([df_majority_downsampled, df_minority])
        
        X_balanced = df_downsampled.drop('Class', axis=1)
        y_balanced = df_downsampled['Class']
        
        logger.info("Data preprocessed successfully.")
        return train_test_split(X_balanced, y_balanced, test_size=0.2, random_state=42)
    except Exception as e:
        logger.error(f"Failed to preprocess data: {str(e)}")
        raise


def train_model(X_train, y_train):
    """Train the RandomForestClassifier model."""
    try:
        model = RandomForestClassifier(n_estimators=100, random_state=42)
        model.fit(X_train, y_train)
        logger.info("Model trained successfully.")
        return model
    except Exception as e:
        logger.error(f"Failed to train model: {str(e)}")
        raise


def evaluate_model(model, X_test, y_test):
    """Evaluate the model and return performance metrics."""
    try:
        y_pred = model.predict(X_test)
        y_proba = model.predict_proba(X_test)[:, 1]
        report = classification_report(y_test, y_pred)
        roc_auc = roc_auc_score(y_test, y_proba)
        logger.info("Model evaluation completed successfully.")
        return report, roc_auc
    except Exception as e:
        logger.error(f"Failed to evaluate model: {str(e)}")
        raise


def log_experiment(model, report, roc_auc):
    """Log the experiment details to MLflow."""
    try:
        mlflow.set_experiment("Fraud Detection Experiment")
        with mlflow.start_run():
            mlflow.log_param("model_type", "RandomForestClassifier")
            mlflow.log_metric("roc_auc", roc_auc)
            mlflow.log_text(report, "classification_report.txt")
            mlflow.sklearn.log_model(model, "random_forest_model")
            logger.info("Experiment logged successfully in MLflow.")
    except MlflowException as e:
        logger.error(f"Failed to log experiment in MLflow: {str(e)}")
        raise


def main(file_path):
    """Main function to orchestrate the pipeline."""
    try:
        data = load_data(file_path)
        X_train, X_test, y_train, y_test = preprocess_data(data)
        model = train_model(X_train, y_train)
        report, roc_auc = evaluate_model(model, X_test, y_test)
        log_experiment(model, report, roc_auc)
    except Exception as e:
        logger.error(f"Pipeline execution failed: {str(e)}")


if __name__ == "__main__":
    # Example file path, replace with actual path
    file_path = "creditcard.csv"
    main(file_path)
```

This script assumes the dataset is in a CSV file named `creditcard.csv` with a `Class` column indicating fraud. It includes error handling through exception catching and logging, making it robust for production use.
