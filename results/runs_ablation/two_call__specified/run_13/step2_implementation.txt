Below is a complete Python pipeline script for a fraud detection model using scikit-learn's `RandomForestClassifier`, with MLflow for experiment tracking. This script includes data loading, model training, evaluation, and logging to MLflow.

```python
import os
import logging
import pandas as pd
import numpy as np
from sklearn.model_selection import train_test_split
from sklearn.ensemble import RandomForestClassifier
from sklearn.metrics import precision_score, recall_score, f1_score, roc_auc_score
from sklearn.utils import shuffle
import mlflow
import mlflow.sklearn
from datetime import datetime
from scipy.stats import ks_2samp

# Configure logging
logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)

# Constants
DATA_URL = "https://example.com/creditcard.csv"  # replace with actual dataset URL
RANDOM_STATE = 42
THRESHOLD_DRIFT = 0.1  # Threshold for concept drift detection

def load_data(url):
    try:
        logger.info("Loading data...")
        data = pd.read_csv(url)
        logger.info("Data loaded successfully.")
        return data
    except Exception as e:
        logger.error(f"Error loading data: {e}")
        raise

def preprocess_data(data):
    logger.info("Preprocessing data...")
    data = shuffle(data, random_state=RANDOM_STATE)
    X = data.drop('Class', axis=1)
    y = data['Class']
    X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.2, random_state=RANDOM_STATE, stratify=y)
    logger.info("Data preprocessed successfully.")
    return X_train, X_test, y_train, y_test

def train_model(X_train, y_train):
    try:
        logger.info("Training model...")
        model = RandomForestClassifier(random_state=RANDOM_STATE)
        model.fit(X_train, y_train)
        logger.info("Model trained successfully.")
        return model
    except Exception as e:
        logger.error(f"Error training model: {e}")
        raise

def evaluate_model(model, X_test, y_test):
    try:
        logger.info("Evaluating model...")
        y_pred = model.predict(X_test)
        precision = precision_score(y_test, y_pred)
        recall = recall_score(y_test, y_pred)
        f1 = f1_score(y_test, y_pred)
        roc_auc = roc_auc_score(y_test, model.predict_proba(X_test)[:, 1])
        logger.info(f"Model evaluation: Precision={precision}, Recall={recall}, F1={f1}, ROC AUC={roc_auc}")
        return precision, recall, f1, roc_auc
    except Exception as e:
        logger.error(f"Error evaluating model: {e}")
        raise

def detect_concept_drift(X_train, X_test):
    logger.info("Detecting concept drift...")
    drift_statistic, drift_pvalue = ks_2samp(X_train.mean(axis=0), X_test.mean(axis=0))
    drift_detected = drift_pvalue < THRESHOLD_DRIFT
    logger.info(f"Concept drift detected: {drift_detected} (p-value: {drift_pvalue})")
    return drift_detected

def log_to_mlflow(model, precision, recall, f1, roc_auc):
    try:
        mlflow.set_experiment("Fraud Detection")
        with mlflow.start_run():
            mlflow.log_param("model_type", "RandomForestClassifier")
            mlflow.log_metric("precision", precision)
            mlflow.log_metric("recall", recall)
            mlflow.log_metric("f1", f1)
            mlflow.log_metric("roc_auc", roc_auc)
            mlflow.sklearn.log_model(model, "model")
            logger.info("Model logged to MLflow successfully.")
    except Exception as e:
        logger.error(f"Error logging to MLflow: {e}")
        raise

def main():
    try:
        data = load_data(DATA_URL)
        X_train, X_test, y_train, y_test = preprocess_data(data)
        drift_detected = detect_concept_drift(X_train, X_test)
        if drift_detected:
            model = train_model(X_train, y_train)
            precision, recall, f1, roc_auc = evaluate_model(model, X_test, y_test)
            log_to_mlflow(model, precision, recall, f1, roc_auc)
        else:
            logger.info("No significant concept drift detected. Skipping retraining.")
    except Exception as e:
        logger.error(f"Pipeline execution failed: {e}")

if __name__ == "__main__":
    main()
```

This script performs the following steps:
1. Loads the dataset from a specified URL.
2. Preprocesses the data by shuffling and splitting it into training and testing sets.
3. Detects concept drift using the KS test.
4. Trains a `RandomForestClassifier` model if concept drift is detected.
5. Evaluates the model using precision, recall, F1-score, and ROC AUC.
6. Logs the model and metrics to MLflow for experiment tracking.

Adjust the `DATA_URL` to point to your dataset's location.
