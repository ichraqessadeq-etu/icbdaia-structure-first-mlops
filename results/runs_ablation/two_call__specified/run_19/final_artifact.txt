```python
import os
import sys
import logging
import pandas as pd
import numpy as np
from sklearn.model_selection import train_test_split
from sklearn.ensemble import RandomForestClassifier
from sklearn.metrics import classification_report, roc_auc_score
from sklearn.utils import resample
import mlflow
import mlflow.sklearn

# Configure logging
logging.basicConfig(level=logging.INFO, format='%(asctime)s - %(levelname)s - %(message)s')

def load_data(file_path):
    """Load the dataset from a CSV file."""
    try:
        data = pd.read_csv(file_path)
        logging.info(f"Data loaded successfully from {file_path}")
        return data
    except Exception as e:
        logging.error(f"Failed to load data: {e}")
        sys.exit(1)

def preprocess_data(data):
    """Preprocess the data, including handling class imbalance."""
    try:
        # Separate majority and minority classes
        df_majority = data[data.Class == 0]
        df_minority = data[data.Class == 1]

        # Upsample minority class
        df_minority_upsampled = resample(df_minority, 
                                         replace=True,     # sample with replacement
                                         n_samples=len(df_majority),    # to match majority class
                                         random_state=123) # reproducible results

        # Combine majority class with upsampled minority class
        data_upsampled = pd.concat([df_majority, df_minority_upsampled])

        # Split features and target
        X = data_upsampled.drop('Class', axis=1)
        y = data_upsampled['Class']
        logging.info("Data preprocessing completed")
        return X, y
    except Exception as e:
        logging.error(f"Data preprocessing failed: {e}")
        sys.exit(1)

def train_model(X_train, y_train):
    """Train a Random Forest Classifier."""
    try:
        model = RandomForestClassifier(n_estimators=100, random_state=42)
        model.fit(X_train, y_train)
        logging.info("Model training completed")
        return model
    except Exception as e:
        logging.error(f"Model training failed: {e}")
        sys.exit(1)

def evaluate_model(model, X_test, y_test):
    """Evaluate the model and return the evaluation metrics."""
    try:
        y_pred = model.predict(X_test)
        y_prob = model.predict_proba(X_test)[:, 1]
        report = classification_report(y_test, y_pred, output_dict=True)
        auc_score = roc_auc_score(y_test, y_prob)
        logging.info("Model evaluation completed")
        return report, auc_score
    except Exception as e:
        logging.error(f"Model evaluation failed: {e}")
        sys.exit(1)

def log_experiment(model, X_test, y_test, report, auc_score):
    """Log the model and metrics to MLflow."""
    try:
        mlflow.set_experiment("Fraud Detection Model")
        with mlflow.start_run():
            mlflow.sklearn.log_model(model, "random_forest_model")
            mlflow.log_metrics({
                "precision": report['1']['precision'],
                "recall": report['1']['recall'],
                "f1-score": report['1']['f1-score'],
                "roc_auc": auc_score
            })
            logging.info("Experiment logged to MLflow")
    except Exception as e:
        logging.error(f"Logging to MLflow failed: {e}")
        sys.exit(1)

def main():
    FILE_PATH = "creditcard.csv"
    
    # Load and preprocess data
    data = load_data(FILE_PATH)
    X, y = preprocess_data(data)

    # Split the data
    X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.3, random_state=42)

    # Train the model
    model = train_model(X_train, y_train)

    # Evaluate the model
    report, auc_score = evaluate_model(model, X_test, y_test)
    logging.info(f"Classification Report: {report}")
    logging.info(f"ROC AUC Score: {auc_score}")

    # Log the experiment
    log_experiment(model, X_test, y_test, report, auc_score)

if __name__ == "__main__":
    main()
```

This script is a full production-ready pipeline for a fraud detection model using a RandomForestClassifier with MLflow tracking, including error handling and logging.
