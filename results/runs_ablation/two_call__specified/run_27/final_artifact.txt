```python
import pandas as pd
import numpy as np
import mlflow
import mlflow.sklearn
from sklearn.model_selection import train_test_split
from sklearn.ensemble import RandomForestClassifier
from sklearn.metrics import classification_report, roc_auc_score
from sklearn.utils import resample

def load_data(file_path):
    try:
        data = pd.read_csv(file_path)
        return data
    except Exception as e:
        raise RuntimeError(f"Error loading data from {file_path}: {e}")

def preprocess_data(data):
    try:
        # Assume 'Class' is the target variable and others are features
        features = data.drop('Class', axis=1)
        target = data['Class']
        return features, target
    except KeyError:
        raise RuntimeError("Target column 'Class' not found in data")
    except Exception as e:
        raise RuntimeError(f"Error during data preprocessing: {e}")

def handle_imbalance(X, y):
    # Upsample minority class
    try:
        X['Class'] = y
        fraud = X[X.Class == 1]
        non_fraud = X[X.Class == 0]
        
        fraud_upsampled = resample(fraud, 
                                   replace=True, 
                                   n_samples=len(non_fraud), 
                                   random_state=42)
        upsampled = pd.concat([non_fraud, fraud_upsampled])
        y_upsampled = upsampled.Class
        X_upsampled = upsampled.drop('Class', axis=1)
        return X_upsampled, y_upsampled
    except Exception as e:
        raise RuntimeError(f"Error handling class imbalance: {e}")

def train_model(X_train, y_train):
    try:
        model = RandomForestClassifier(random_state=42)
        model.fit(X_train, y_train)
        return model
    except Exception as e:
        raise RuntimeError(f"Error training model: {e}")

def evaluate_model(model, X_test, y_test):
    try:
        predictions = model.predict(X_test)
        roc_auc = roc_auc_score(y_test, predictions)
        report = classification_report(y_test, predictions)
        return roc_auc, report
    except Exception as e:
        raise RuntimeError(f"Error evaluating model: {e}")

def log_experiment(model, roc_auc, report):
    try:
        mlflow.log_metric("roc_auc", roc_auc)
        mlflow.log_param("model_type", "RandomForestClassifier")
        mlflow.sklearn.log_model(model, "model")
        print("Classification Report:\n", report)
    except Exception as e:
        raise RuntimeError(f"Error logging to MLflow: {e}")

if __name__ == "__main__":
    FILE_PATH = "creditcard.csv"
    mlflow.set_experiment("Fraud Detection Experiment")

    with mlflow.start_run():
        data = load_data(FILE_PATH)
        features, target = preprocess_data(data)
        X_upsampled, y_upsampled = handle_imbalance(features, target)
        X_train, X_test, y_train, y_test = train_test_split(X_upsampled, y_upsampled, test_size=0.2, random_state=42)
        
        model = train_model(X_train, y_train)
        roc_auc, report = evaluate_model(model, X_test, y_test)
        log_experiment(model, roc_auc, report)
```

This script includes:

- Data loading with error handling for file reading issues.
- Preprocessing with checks for the presence of the target column.
- Handling class imbalance through upsampling.
- Model training and evaluation with error handling.
- Logging metrics and parameters to MLflow.

import pytest
import pandas as pd
from sklearn.ensemble import RandomForestClassifier

def test_load_data():

    data = load_data('creditcard.csv')

def test_preprocess_data():
    data = pd.DataFrame({

        'Feature2': [4, 5, 6],
        'Class': [0, 1, 0]

    y = pd.Series([0, 1, 0, 1, 0, 0, 0, 0])
    X_balanced, y_balanced = handle_imbalance(X, y)
    assert len(X_balanced) == 2 * len(X[y == 0])
    
    # Ensure balanced dataset has equal class distribution
