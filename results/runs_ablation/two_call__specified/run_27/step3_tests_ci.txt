# test_pipeline.py

import pytest
import pandas as pd
from sklearn.ensemble import RandomForestClassifier
from sklearn.model_selection import train_test_split
from sklearn.metrics import roc_auc_score, classification_report
from pipeline import load_data, preprocess_data, handle_imbalance, train_model, evaluate_model

def test_load_data():
    # Test loading data with a valid file path
    data = load_data('creditcard.csv')
    assert isinstance(data, pd.DataFrame)
    
    # Test loading data with an invalid file path
    with pytest.raises(RuntimeError):
        load_data('invalid_path.csv')

def test_preprocess_data():
    data = pd.DataFrame({
        'Feature1': [1, 2, 3],
        'Feature2': [4, 5, 6],
        'Class': [0, 1, 0]
    })
    features, target = preprocess_data(data)
    assert 'Class' not in features.columns
    assert list(target) == [0, 1, 0]
    
    # Test when the 'Class' column is missing
    with pytest.raises(RuntimeError):
        preprocess_data(data.drop('Class', axis=1))

def test_handle_imbalance():
    X = pd.DataFrame({'Feature': [1, 2, 3, 4, 5, 6, 7, 8]})
    y = pd.Series([0, 1, 0, 1, 0, 0, 0, 0])
    X_balanced, y_balanced = handle_imbalance(X, y)
    assert len(X_balanced) == 2 * len(X[y == 0])
    
    # Ensure balanced dataset has equal class distribution
    assert y_balanced.value_counts()[0] == y_balanced.value_counts()[1]

def test_train_model():
    X = pd.DataFrame({'Feature': [1, 2, 3, 4]})
    y = pd.Series([0, 1, 0, 1])
    model = train_model(X, y)
    assert isinstance(model, RandomForestClassifier)

def test_evaluate_model():
    X_train = pd.DataFrame({'Feature': [1, 2, 3, 4]})
    y_train = pd.Series([0, 1, 0, 1])
    X_test = pd.DataFrame({'Feature': [5, 6]})
    y_test = pd.Series([0, 1])
    model = train_model(X_train, y_train)
    roc_auc, report = evaluate_model(model, X_test, y_test)
    assert isinstance(roc_auc, float)
    assert isinstance(report, str)
