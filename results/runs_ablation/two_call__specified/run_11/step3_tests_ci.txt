# test_pipeline.py

import pytest
import pandas as pd
from pipeline_module import load_data, preprocess_data, train_model, evaluate_model

@pytest.fixture
def sample_data():
    """Fixture to create a sample dataframe similar to the credit card fraud data."""
    data = {
        'V1': [0.1, -0.2, 0.3, -0.4],
        'V2': [-0.1, 0.2, -0.3, 0.4],
        'V3': [0.3, -0.1, 0.4, -0.5],
        'Class': [0, 1, 0, 1]
    }
    return pd.DataFrame(data)

def test_load_data():
    """Test the load_data function with a non-existent file."""
    with pytest.raises(FileNotFoundError):
        load_data("non_existent_file.csv")

def test_preprocess_data(sample_data):
    """Test the preprocess_data function with sample data."""
    X, y = preprocess_data(sample_data)
    assert len(X) == len(y)
    assert set(y) == {0, 1}

def test_train_model(sample_data):
    """Test the train_model function with sample data."""
    X, y = preprocess_data(sample_data)
    model = train_model(X, y)
    assert model is not None
    assert hasattr(model, "predict")

def test_evaluate_model(sample_data):
    """Test the evaluate_model function with sample data."""
    X, y = preprocess_data(sample_data)
    model = train_model(X, y)
    report = evaluate_model(model, X, y)
    assert isinstance(report, dict)
    assert '1' in report
    assert 'precision' in report['1']

### (b) GitHub Actions Workflow

This GitHub Actions workflow is configured to run `pytest` whenever there's a push or pull request to the repository, ensuring the code is tested automatically.

```yaml
# .github/workflows/test.yml

name: Python application

on:
  push:
    branches: [ main ]
  pull_request:
    branches: [ main ]

jobs:
  test:
    runs-on: ubuntu-latest

    steps:
    - uses: actions/checkout@v2

    - name: Set up Python
      uses: actions/setup-python@v2
      with:
        python-version: '3.8'

    - name: Install dependencies
      run: |
        python -m pip install --upgrade pip
        pip install pandas scikit-learn pytest mlflow

    - name: Run tests
      run: |
        pytest test_pipeline.py
```
