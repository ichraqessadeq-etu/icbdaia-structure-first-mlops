### 1. Python Pipeline Script

```python
import numpy as np
import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.ensemble import RandomForestClassifier
from sklearn.metrics import classification_report, confusion_matrix
from sklearn.utils import resample
import mlflow
import mlflow.sklearn
import logging

# Setup logging
logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)

RANDOM_SEED = 42

def load_data():
    try:
        # Assuming the dataset is available at a particular path
        logger.info("Loading data...")
        df = pd.read_csv('credit_card_fraud.csv')
        return df
    except Exception as e:
        logger.error(f"Failed to load data: {e}")
        raise

def preprocess_data(df):
    try:
        # Separate majority and minority classes
        df_majority = df[df.Class == 0]
        df_minority = df[df.Class == 1]

        # Downsample majority class
        df_majority_downsampled = resample(df_majority, 
                                           replace=False,    
                                           n_samples=len(df_minority),     
                                           random_state=RANDOM_SEED)

        # Combine minority class with downsampled majority class
        df_resampled = pd.concat([df_majority_downsampled, df_minority])

        # Split data
        X = df_resampled.drop('Class', axis=1)
        y = df_resampled['Class']
        
        X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.3, random_state=RANDOM_SEED)
        logger.info("Data preprocessing completed.")
        return X_train, X_test, y_train, y_test
    except Exception as e:
        logger.error(f"Failed to preprocess data: {e}")
        raise

def train_model(X_train, y_train):
    try:
        logger.info("Training model...")
        model = RandomForestClassifier(random_state=RANDOM_SEED)
        model.fit(X_train, y_train)
        logger.info("Model training completed.")
        return model
    except Exception as e:
        logger.error(f"Failed to train model: {e}")
        raise

def evaluate_model(model, X_test, y_test):
    try:
        logger.info("Evaluating model...")
        y_pred = model.predict(X_test)
        report = classification_report(y_test, y_pred, output_dict=True)
        cm = confusion_matrix(y_test, y_pred)
        logger.info("Model evaluation completed.")
        return report, cm
    except Exception as e:
        logger.error(f"Failed to evaluate model: {e}")
        raise

def log_experiment(model, report, cm):
    try:
        logger.info("Logging experiment with MLflow...")
        mlflow.start_run()
        mlflow.log_params(model.get_params())
        mlflow.log_metrics({'precision': report['1']['precision'], 
                            'recall': report['1']['recall'], 
                            'f1-score': report['1']['f1-score']})
        mlflow.log_artifact('credit_card_fraud.csv')
        mlflow.sklearn.log_model(model, "model")
        logger.info("Experiment logged successfully.")
    except Exception as e:
        logger.error(f"Failed to log experiment: {e}")
        raise
    finally:
        mlflow.end_run()

def main():
    df = load_data()
    X_train, X_test, y_train, y_test = preprocess_data(df)
    model = train_model(X_train, y_train)
    report, cm = evaluate_model(model, X_test, y_test)
    log_experiment(model, report, cm)

if __name__ == "__main__":
    main()
```
