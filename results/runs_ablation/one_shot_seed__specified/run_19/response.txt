Below is a complete solution for setting up a CI/CD pipeline for a fraud detection model using `scikit-learn`, `MLflow`, `pytest`, and GitHub Actions. The pipeline will handle data loading, model training, evaluation, and logging, with tests and workflow configurations included.

### 1. Python Pipeline Script

```python
# fraud_detection_pipeline.py

import numpy as np
import pandas as pd
from sklearn.ensemble import RandomForestClassifier
from sklearn.metrics import classification_report
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler
import mlflow
import mlflow.sklearn
import logging

# Configure logging
logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)

# Reproducibility
RANDOM_SEED = 42
np.random.seed(RANDOM_SEED)

def load_data():
    try:
        # Assuming dataset is a CSV file. Adjust the path as needed.
        data = pd.read_csv('creditcard.csv')
        logger.info("Data loaded successfully.")
        return data
    except Exception as e:
        logger.error(f"Error loading data: {e}")
        raise

def preprocess_data(data):
    try:
        X = data.drop('Class', axis=1)
        y = data['Class']
        
        # Standardize features by removing the mean and scaling to unit variance
        scaler = StandardScaler()
        X_scaled = scaler.fit_transform(X)
        
        X_train, X_test, y_train, y_test = train_test_split(X_scaled, y, test_size=0.3, random_state=RANDOM_SEED, stratify=y)
        logger.info("Data preprocessed successfully.")
        return X_train, X_test, y_train, y_test
    except Exception as e:
        logger.error(f"Error in data preprocessing: {e}")
        raise

def train_model(X_train, y_train):
    try:
        model = RandomForestClassifier(random_state=RANDOM_SEED)
        model.fit(X_train, y_train)
        logger.info("Model trained successfully.")
        return model
    except Exception as e:
        logger.error(f"Error in model training: {e}")
        raise

def evaluate_model(model, X_test, y_test):
    try:
        predictions = model.predict(X_test)
        report = classification_report(y_test, predictions, output_dict=True)
        logger.info("Model evaluated successfully.")
        return report
    except Exception as e:
        logger.error(f"Error in model evaluation: {e}")
        raise

def log_experiment(model, report):
    try:
        mlflow.set_experiment('Fraud Detection')
        with mlflow.start_run():
            mlflow.sklearn.log_model(model, "model")
            mlflow.log_metrics({
                'precision': report['1']['precision'],
                'recall': report['1']['recall'],
                'f1-score': report['1']['f1-score']
            })
            logger.info("Experiment logged successfully.")
    except Exception as e:
        logger.error(f"Error logging experiment: {e}")
        raise

def main():
    data = load_data()
    X_train, X_test, y_train, y_test = preprocess_data(data)
    model = train_model(X_train, y_train)
    report = evaluate_model(model, X_test, y_test)
    log_experiment(model, report)

if __name__ == "__main__":
    main()
```

### 2. Pytest Test File

```python
# test_fraud_detection_pipeline.py

import pytest
import numpy as np
from sklearn.ensemble import RandomForestClassifier
from sklearn.exceptions import NotFittedError
from fraud_detection_pipeline import load_data, preprocess_data, train_model, evaluate_model

RANDOM_SEED = 42
np.random.seed(RANDOM_SEED)

def test_load_data():
    data = load_data()
    assert data is not None, "Data should not be None"
    assert 'Class' in data.columns, "Data should contain 'Class' column"

def test_preprocess_data():
    data = load_data()
    X_train, X_test, y_train, y_test = preprocess_data(data)
    assert X_train.shape[0] > 0, "X_train should not be empty"
    assert X_test.shape[0] > 0, "X_test should not be empty"
    assert y_train.shape[0] > 0, "y_train should not be empty"
    assert y_test.shape[0] > 0, "y_test should not be empty"

def test_train_model():
    data = load_data()
    X_train, X_test, y_train, y_test = preprocess_data(data)
    model = train_model(X_train, y_train)
    assert isinstance(model, RandomForestClassifier), "Model should be a RandomForestClassifier"
    # Check that model raises NotFittedError when predicting before training
    with pytest.raises(NotFittedError):
        RandomForestClassifier().predict(X_test)
    # Ensure that the model is fitted
    assert len(model.predict(X_test)) == len(y_test), "Predictions should be equal to test labels length"
```

### 3. GitHub Actions YAML Workflow

```yaml
# .github/workflows/ci-cd.yml

name: CI/CD Pipeline

on:
  push:
    branches:
      - main
  pull_request:
    branches:
      - main
  schedule:
    - cron: '0 0 * * 0'  # Run every Sunday at midnight

jobs:
  build:
    runs-on: ubuntu-latest
    
    steps:
    - name: Checkout
      uses: actions/checkout@v2
    
    - name: Set up Python
      uses: actions/setup-python@v2
      with:
        python-version: '3.8'
    
    - name: Install dependencies
      run: |
        python -m pip install --upgrade pip
        pip install -r requirements.txt
    
    - name: Run tests
      run: |
        pytest --maxfail=1 --disable-warnings
    
    - name: Run pipeline
      run: |
        python fraud_detection_pipeline.py
```

### Notes:
- Ensure you have a `requirements.txt` file listing the necessary Python packages, such as `scikit-learn`, `pandas`, `numpy`, `mlflow`, `pytest`, etc.
- The GitHub Actions workflow is triggered on every push and pull request to the main branch, as well as on a weekly schedule.
- The script includes error handling to catch exceptions during data loading, preprocessing, training, evaluation, and experiment logging.
- The tests ensure data loading, preprocessing, and model training behave as expected.
- Adjust the data loading path and dependencies based on your specific environment and dataset location.
