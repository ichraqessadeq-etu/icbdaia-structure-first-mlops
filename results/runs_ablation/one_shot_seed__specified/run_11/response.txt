Here's a complete solution for a CI/CD pipeline for a fraud detection model using scikit-learn, MLflow, pytest, and GitHub Actions.

### 1. Python Pipeline Script

```python
# pipeline.py
import mlflow
from sklearn.ensemble import RandomForestClassifier
from sklearn.metrics import classification_report
from sklearn.model_selection import train_test_split
from sklearn.utils import shuffle
from sklearn.datasets import make_classification
import numpy as np
import pandas as pd
import joblib
import logging

# Set up logging
logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)

# Seed for reproducibility
SEED = 42
np.random.seed(SEED)

def load_data():
    try:
        # Simulating data loading; replace with actual data loading
        X, y = make_classification(n_samples=10000, n_features=20, n_classes=2, weights=[0.9983, 0.0017], random_state=SEED)
        logger.info("Data loaded successfully.")
        return pd.DataFrame(X), pd.Series(y)
    except Exception as e:
        logger.error("Error loading data: %s", e)
        raise

def preprocess_data(X, y):
    try:
        X, y = shuffle(X, y, random_state=SEED)
        logger.info("Data shuffled successfully.")
    except Exception as e:
        logger.error("Error preprocessing data: %s", e)
        raise
    return X, y

def train_model(X_train, y_train):
    try:
        model = RandomForestClassifier(random_state=SEED)
        model.fit(X_train, y_train)
        logger.info("Model trained successfully.")
        return model
    except Exception as e:
        logger.error("Error training model: %s", e)
        raise

def evaluate_model(model, X_test, y_test):
    try:
        predictions = model.predict(X_test)
        report = classification_report(y_test, predictions, output_dict=True)
        logger.info("Model evaluated successfully.")
        return report
    except Exception as e:
        logger.error("Error evaluating model: %s", e)
        raise

def log_experiment(model, report):
    try:
        mlflow.set_experiment("FraudDetection")
        with mlflow.start_run():
            mlflow.log_params(model.get_params())
            mlflow.log_metrics({f'precision_{k}': v['precision'] for k, v in report.items() if isinstance(v, dict)})
            mlflow.log_metrics({f'recall_{k}': v['recall'] for k, v in report.items() if isinstance(v, dict)})
            mlflow.log_metrics({f'f1-score_{k}': v['f1-score'] for k, v in report.items() if isinstance(v, dict)})
            joblib.dump(model, "model.pkl")
            mlflow.log_artifact("model.pkl")
        logger.info("Experiment logged successfully.")
    except Exception as e:
        logger.error("Error logging experiment: %s", e)
        raise

def main():
    X, y = load_data()
    X, y = preprocess_data(X, y)
    X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.2, random_state=SEED)
    model = train_model(X_train, y_train)
    report = evaluate_model(model, X_test, y_test)
    log_experiment(model, report)

if __name__ == "__main__":
    main()
```

### 2. Pytest Test File

```python
# test_pipeline.py
import pytest
from sklearn.ensemble import RandomForestClassifier
from sklearn.datasets import make_classification
from pipeline import train_model, evaluate_model, preprocess_data

# Seed for reproducibility
SEED = 42

@pytest.fixture
def synthetic_data():
    X, y = make_classification(n_samples=1000, n_features=20, n_classes=2, weights=[0.9983, 0.0017], random_state=SEED)
    return preprocess_data(X, y)

def test_train_model(synthetic_data):
    X_train, y_train = synthetic_data
    model = train_model(X_train, y_train)
    assert isinstance(model, RandomForestClassifier), "Model should be a RandomForestClassifier."

def test_evaluate_model(synthetic_data):
    X_train, y_train = synthetic_data
    model = train_model(X_train, y_train)
    report = evaluate_model(model, X_train, y_train)
    assert '0' in report and '1' in report, "Classification report should contain keys for both classes."

def test_preprocess_data(synthetic_data):
    X, y = synthetic_data
    assert X.shape[0] == y.shape[0], "Features and labels should have the same number of samples."
```

### 3. GitHub Actions Workflow

```yaml
# .github/workflows/ci-cd.yaml
name: CI/CD Pipeline

on:
  push:
    branches:
      - main
  pull_request:
    branches:
      - main

jobs:
  test:
    runs-on: ubuntu-latest

    steps:
    - name: Checkout code
      uses: actions/checkout@v2

    - name: Set up Python
      uses: actions/setup-python@v2
      with:
        python-version: '3.8'

    - name: Install dependencies
      run: |
        python -m pip install --upgrade pip
        pip install -r requirements.txt

    - name: Run tests
      run: |
        pytest test_pipeline.py

  deploy:
    runs-on: ubuntu-latest
    needs: test

    steps:
    - name: Checkout code
      uses: actions/checkout@v2

    - name: Set up Python
      uses: actions/setup-python@v2
      with:
        python-version: '3.8'

    - name: Install dependencies
      run: |
        python -m pip install --upgrade pip
        pip install -r requirements.txt

    - name: Run pipeline
      env:
        MLFLOW_TRACKING_URI: <your_mlflow_tracking_uri>
        MLFLOW_TRACKING_USERNAME: <your_mlflow_username>
        MLFLOW_TRACKING_PASSWORD: <your_mlflow_password>
      run: |
        python pipeline.py
```

**Note:** Make sure to replace `<your_mlflow_tracking_uri>`, `<your_mlflow_username>`, and `<your_mlflow_password>` with your actual MLflow tracking server details. Also, ensure your `requirements.txt` includes all necessary dependencies such as `scikit-learn`, `mlflow`, `joblib`, and `pytest`.
