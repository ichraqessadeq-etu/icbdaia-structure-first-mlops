Sure, below is a complete CI/CD setup for a fraud detection model using scikit-learn's RandomForestClassifier. This setup includes a Python pipeline script for data loading, training, evaluation, and logging to MLflow, a pytest test file, and a GitHub Actions workflow.

### Python Pipeline Script

```python
# pipeline.py

import pandas as pd
import numpy as np
import mlflow
import mlflow.sklearn
from sklearn.model_selection import train_test_split
from sklearn.ensemble import RandomForestClassifier
from sklearn.metrics import classification_report, roc_auc_score, confusion_matrix
from sklearn.utils import shuffle
from sklearn.preprocessing import StandardScaler
import logging

# Set up logging
logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)

RANDOM_SEED = 42

def load_data():
    # Simulate loading data
    logger.info("Loading data...")
    # Here you would typically load data from a database or a file
    # For demonstration, create synthetic data
    np.random.seed(RANDOM_SEED)
    n_samples = 10000
    n_fraud = int(n_samples * 0.0017)
    n_non_fraud = n_samples - n_fraud

    fraud_data = np.random.rand(n_fraud, 30)
    non_fraud_data = np.random.rand(n_non_fraud, 30)

    X = np.vstack((fraud_data, non_fraud_data))
    y = np.hstack((np.ones(n_fraud), np.zeros(n_non_fraud)))

    X, y = shuffle(X, y, random_state=RANDOM_SEED)
    
    return X, y

def preprocess_data(X):
    logger.info("Preprocessing data...")
    scaler = StandardScaler()
    X_scaled = scaler.fit_transform(X)
    return X_scaled

def train_model(X_train, y_train):
    logger.info("Training model...")
    model = RandomForestClassifier(random_state=RANDOM_SEED)
    model.fit(X_train, y_train)
    return model

def evaluate_model(model, X_test, y_test):
    logger.info("Evaluating model...")
    y_pred = model.predict(X_test)
    y_proba = model.predict_proba(X_test)[:, 1]
    
    report = classification_report(y_test, y_pred)
    roc_auc = roc_auc_score(y_test, y_proba)
    cm = confusion_matrix(y_test, y_pred)

    logger.info(f"Classification Report:\n{report}")
    logger.info(f"ROC AUC Score: {roc_auc}")
    logger.info(f"Confusion Matrix:\n{cm}")

    return report, roc_auc, cm

def log_to_mlflow(model, report, roc_auc, cm):
    logger.info("Logging to MLflow...")
    with mlflow.start_run():
        mlflow.log_param("random_state", RANDOM_SEED)
        mlflow.log_metric("roc_auc", roc_auc)
        mlflow.sklearn.log_model(model, "model")
        mlflow.log_artifact("classification_report.txt")
        mlflow.log_artifact("confusion_matrix.txt")

def main():
    try:
        X, y = load_data()
        X = preprocess_data(X)
        X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.2, random_state=RANDOM_SEED)
        
        model = train_model(X_train, y_train)
        report, roc_auc, cm = evaluate_model(model, X_test, y_test)
        
        log_to_mlflow(model, report, roc_auc, cm)
    except Exception as e:
        logger.error(f"An error occurred: {e}")

if __name__ == "__main__":
    main()
```

### Pytest Test File

```python
# test_pipeline.py

import pytest
import numpy as np
from pipeline import load_data, preprocess_data, train_model, evaluate_model

RANDOM_SEED = 42

def test_load_data():
    X, y = load_data()
    assert X.shape[0] == 10000, "Number of samples should be 10000"
    assert X.shape[1] == 30, "Number of features should be 30"
    assert np.sum(y) == pytest.approx(17, abs=0.1), "Number of fraud samples should be approximately 17"

def test_preprocess_data():
    X, _ = load_data()
    X_scaled = preprocess_data(X)
    assert np.allclose(X_scaled.mean(axis=0), 0, atol=1e-7), "Mean of scaled data should be approximately 0"
    assert np.allclose(X_scaled.std(axis=0), 1, atol=1e-7), "Std of scaled data should be approximately 1"

def test_train_model():
    X, y = load_data()
    X = preprocess_data(X)
    X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.2, random_state=RANDOM_SEED)
    
    model = train_model(X_train, y_train)
    assert hasattr(model, "predict"), "Model should have a predict method"

    report, roc_auc, cm = evaluate_model(model, X_test, y_test)
    assert roc_auc > 0.5, "ROC AUC should be greater than 0.5"

```

### GitHub Actions Workflow

```yaml
# .github/workflows/main.yml

name: CI/CD Pipeline

on:
  push:
    branches:
      - main
  pull_request:
    branches:
      - main

jobs:
  build:
    runs-on: ubuntu-latest

    steps:
    - uses: actions/checkout@v2

    - name: Set up Python
      uses: actions/setup-python@v2
      with:
        python-version: '3.8'

    - name: Install dependencies
      run: |
        python -m pip install --upgrade pip
        pip install scikit-learn pandas numpy mlflow pytest

    - name: Run tests
      run: |
        pytest test_pipeline.py

    - name: Run pipeline
      run: |
        python pipeline.py

```

Each part of this setup is designed to be modular and robust, ensuring that the fraud detection model can be trained, evaluated, and logged using MLflow while maintaining code quality with pytest. The GitHub Actions workflow automates the process of testing and deploying the pipeline, ensuring continuous integration and deployment.
