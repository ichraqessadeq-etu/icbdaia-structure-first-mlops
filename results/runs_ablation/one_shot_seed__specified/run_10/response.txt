### 1. Python Pipeline Script

```python
import numpy as np
import pandas as pd
import mlflow
import mlflow.sklearn
from sklearn.ensemble import RandomForestClassifier
from sklearn.model_selection import train_test_split
from sklearn.metrics import classification_report, roc_auc_score
from sklearn.utils import resample

# Set random seed for reproducibility
RANDOM_SEED = 42

def load_data():
    try:
        # Load the credit card fraud dataset
        data = pd.read_csv('creditcard.csv')
        return data
    except Exception as e:
        raise FileNotFoundError("Dataset not found or could not be loaded. Error: {}".format(e))

def preprocess_data(data):
    # Handle class imbalance using upsampling
    fraud = data[data.Class == 1]
    non_fraud = data[data.Class == 0]

    fraud_upsampled = resample(fraud, 
                               replace=True,
                               n_samples=len(non_fraud),
                               random_state=RANDOM_SEED)

    upsampled_data = pd.concat([non_fraud, fraud_upsampled])
    X = upsampled_data.drop('Class', axis=1)
    y = upsampled_data['Class']

    return X, y

def train_model(X_train, y_train):
    model = RandomForestClassifier(random_state=RANDOM_SEED)
    model.fit(X_train, y_train)
    return model

def evaluate_model(model, X_test, y_test):
    predictions = model.predict(X_test)
    auc_score = roc_auc_score(y_test, predictions)
    print(classification_report(y_test, predictions))
    return auc_score

def log_experiment(model, auc_score):
    try:
        mlflow.start_run()
        mlflow.log_param("random_state", RANDOM_SEED)
        mlflow.log_metric("AUC", auc_score)
        mlflow.sklearn.log_model(model, "model")
        mlflow.end_run()
    except Exception as e:
        raise RuntimeError("Failed to log experiment. Error: {}".format(e))

def main():
    data = load_data()
    X, y = preprocess_data(data)
    X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.2, random_state=RANDOM_SEED)
    
    model = train_model(X_train, y_train)
    auc_score = evaluate_model(model, X_test, y_test)
    
    log_experiment(model, auc_score)

if __name__ == '__main__':
    main()
```

### 2. Pytest Test File

```python
import pytest
import pandas as pd
import numpy as np
from sklearn.ensemble import RandomForestClassifier
from pipeline import load_data, preprocess_data, train_model, evaluate_model

RANDOM_SEED = 42

def test_load_data():
    data = load_data()
    assert not data.empty, "Data should not be empty"
    assert 'Class' in data.columns, "Data should contain 'Class' column"

def test_preprocess_data():
    data = load_data()
    X, y = preprocess_data(data)
    assert len(X) == len(y), "Features and labels should have the same length"
    assert set(y.unique()) == {0, 1}, "Labels should be binary (0 and 1)"

def test_train_model():
    data = load_data()
    X, y = preprocess_data(data)
    X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.2, random_state=RANDOM_SEED)
    
    model = train_model(X_train, y_train)
    assert isinstance(model, RandomForestClassifier), "Model should be a RandomForestClassifier"
    assert model.n_classes_ == 2, "Model should be trained for binary classification"

def test_evaluate_model():
    data = load_data()
    X, y = preprocess_data(data)
    X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.2, random_state=RANDOM_SEED)
    
    model = train_model(X_train, y_train)
    auc_score = evaluate_model(model, X_test, y_test)
    assert 0 <= auc_score <= 1, "AUC score should be between 0 and 1"
```
