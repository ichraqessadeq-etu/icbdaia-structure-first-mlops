Below is a complete setup for a CI/CD pipeline for a fraud detection model using scikit-learn's RandomForestClassifier. The setup includes a Python pipeline script, a pytest test file, and a GitHub Actions YAML configuration.

### Python Pipeline Script

```python
import numpy as np
import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.ensemble import RandomForestClassifier
from sklearn.metrics import classification_report, roc_auc_score
from sklearn.utils import resample
import mlflow
import mlflow.sklearn
import logging

# Set up logging
logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)

# Set random seed for reproducibility
RANDOM_SEED = 42

def load_data(file_path):
    """Load the credit card fraud dataset."""
    try:
        data = pd.read_csv(file_path)
        logger.info("Data loaded successfully")
        return data
    except Exception as e:
        logger.error(f"Error loading data: {e}")
        raise

def preprocess_data(data):
    """Balance the dataset using undersampling."""
    try:
        # Separate majority and minority classes
        df_majority = data[data.Class == 0]
        df_minority = data[data.Class == 1]
        
        # Downsample majority class
        df_majority_downsampled = resample(df_majority,
                                           replace=False,
                                           n_samples=len(df_minority),
                                           random_state=RANDOM_SEED)
        
        # Combine minority class with downsampled majority class
        df_balanced = pd.concat([df_minority, df_majority_downsampled])
        
        logger.info("Data preprocessing completed")
        return df_balanced
    except Exception as e:
        logger.error(f"Error in preprocessing data: {e}")
        raise

def train_model(data):
    """Train a RandomForestClassifier."""
    try:
        X = data.drop('Class', axis=1)
        y = data['Class']
        X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.2, random_state=RANDOM_SEED)
        
        model = RandomForestClassifier(random_state=RANDOM_SEED)
        model.fit(X_train, y_train)
        
        y_pred = model.predict(X_test)
        y_pred_proba = model.predict_proba(X_test)[:, 1]
        
        # Evaluation
        roc_auc = roc_auc_score(y_test, y_pred_proba)
        report = classification_report(y_test, y_pred, output_dict=True)
        
        logger.info("Model training completed")
        return model, roc_auc, report
    except Exception as e:
        logger.error(f"Error training model: {e}")
        raise

def log_to_mlflow(model, roc_auc, report):
    """Log model and metrics to MLflow."""
    try:
        mlflow.start_run()
        mlflow.log_metric("roc_auc", roc_auc)
        mlflow.log_metrics({f"precision_{k}": v['precision'] for k, v in report.items() if isinstance(v, dict)})
        mlflow.log_metrics({f"recall_{k}": v['recall'] for k, v in report.items() if isinstance(v, dict)})
        
        mlflow.sklearn.log_model(model, "random_forest_model")
        mlflow.end_run()
        
        logger.info("Model and metrics logged to MLflow")
    except Exception as e:
        logger.error(f"Error logging to MLflow: {e}")
        raise

def main():
    try:
        data = load_data('creditcard.csv')
        balanced_data = preprocess_data(data)
        model, roc_auc, report = train_model(balanced_data)
        log_to_mlflow(model, roc_auc, report)
    except Exception as e:
        logger.error(f"Error in main execution: {e}")

if __name__ == "__main__":
    main()
```

### GitHub Actions YAML Workflow

```yaml
name: CI/CD Pipeline

on:
  push:
    branches: [ main ]
  pull_request:
    branches: [ main ]
  schedule:
    - cron: '0 0 * * 0' # Every Sunday at midnight
  workflow_dispatch:

jobs:
  build:
    runs-on: ubuntu-latest
    
    steps:
      - name: Checkout repository
        uses: actions/checkout@v2

      - name: Set up Python
        uses: actions/setup-python@v2
        with:
          python-version: '3.8'

      - name: Install dependencies
        run: |
          python -m pip install --upgrade pip
          pip install -r requirements.txt

      - name: Run tests
        run: |
          pytest --maxfail=1 --disable-warnings

      - name: Run pipeline
        run: |
          python pipeline.py

      - name: Detect concept drift
        run: |
          # Add your concept drift detection logic here
          echo "Detecting concept drift..."
```
