To create a complete CI/CD pipeline for a fraud detection model using `scikit-learn` and `MLflow`, we will create three main components:

1. A Python script for the pipeline which includes data loading, model training, evaluation, and logging to MLflow.
2. A `pytest` test file to ensure the pipeline's correctness.
3. A GitHub Actions YAML file to automate the CI/CD process, including retraining upon detecting concept drift.

Here's how you can implement each component:

### 1. Python Pipeline Script

```python
# pipeline.py
import numpy as np
import pandas as pd
from sklearn.ensemble import RandomForestClassifier
from sklearn.metrics import classification_report, accuracy_score
from sklearn.model_selection import train_test_split
from sklearn.utils import resample
import mlflow
import mlflow.sklearn
import logging

# Set random seed for reproducibility
RANDOM_SEED = 42
np.random.seed(RANDOM_SEED)

# Configure logging
logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)

def load_data():
    # Simulate loading the credit card fraud dataset
    # In practice, replace this with actual data loading code
    # For example, pd.read_csv('creditcard.csv')
    logger.info("Loading data...")
    n_samples = 10000
    n_fraud = int(n_samples * 0.0017)
    n_normal = n_samples - n_fraud

    fraud_data = np.random.randn(n_fraud, 30) + 2
    normal_data = np.random.randn(n_normal, 30)

    X = np.vstack((normal_data, fraud_data))
    y = np.array([0] * n_normal + [1] * n_fraud)
    
    return X, y

def train_model(X_train, y_train):
    logger.info("Training model...")
    model = RandomForestClassifier(random_state=RANDOM_SEED)
    model.fit(X_train, y_train)
    return model

def evaluate_model(model, X_test, y_test):
    logger.info("Evaluating model...")
    y_pred = model.predict(X_test)
    accuracy = accuracy_score(y_test, y_pred)
    report = classification_report(y_test, y_pred)
    logger.info(f"Accuracy: {accuracy}")
    logger.info(f"Classification Report:\n{report}")
    return accuracy, report

def log_to_mlflow(model, accuracy, report):
    logger.info("Logging to MLflow...")
    mlflow.set_experiment("fraud_detection")
    with mlflow.start_run():
        mlflow.sklearn.log_model(model, "model")
        mlflow.log_metric("accuracy", accuracy)
        mlflow.log_text(report, "classification_report.txt")

def main():
    X, y = load_data()
    X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.2, random_state=RANDOM_SEED)
    
    model = train_model(X_train, y_train)
    accuracy, report = evaluate_model(model, X_test, y_test)
    log_to_mlflow(model, accuracy, report)

if __name__ == "__main__":
    main()
```

### 2. Pytest Test File

```python
# test_pipeline.py
import pytest
from sklearn.ensemble import RandomForestClassifier
import numpy as np
from pipeline import load_data, train_model, evaluate_model

RANDOM_SEED = 42

def test_load_data():
    X, y = load_data()
    assert X.shape[0] == len(y), "Mismatch between number of samples and labels"
    assert set(y) == {0, 1}, "Labels are not binary"

def test_train_model():
    X, y = load_data()
    X_train, _, y_train, _ = train_test_split(X, y, test_size=0.2, random_state=RANDOM_SEED)
    model = train_model(X_train, y_train)
    assert isinstance(model, RandomForestClassifier), "Model is not a RandomForestClassifier"
    assert hasattr(model, "predict"), "Model does not have a predict method"

def test_evaluate_model():
    X, y = load_data()
    X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.2, random_state=RANDOM_SEED)
    model = train_model(X_train, y_train)
    accuracy, report = evaluate_model(model, X_test, y_test)
    assert isinstance(accuracy, float), "Accuracy is not a float"
    assert "precision" in report, "Classification report does not contain precision"

```
