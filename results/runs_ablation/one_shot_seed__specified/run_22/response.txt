Below are the complete files and configurations for a CI/CD pipeline for a fraud detection model using a RandomForestClassifier, with MLflow for experiment tracking and pytest for testing. The pipeline includes data loading, training, evaluation, and logging to MLflow. The GitHub Actions workflow is set to trigger automatically when changes are pushed to the repository.

### 1. Python Pipeline Script (`pipeline.py`)

```python
import pandas as pd
import numpy as np
import mlflow
import mlflow.sklearn
import joblib
from sklearn.model_selection import train_test_split
from sklearn.ensemble import RandomForestClassifier
from sklearn.metrics import classification_report, roc_auc_score
from sklearn.utils import resample
import logging

RANDOM_SEED = 42
logging.basicConfig(level=logging.INFO)


def load_data(file_path):
    try:
        data = pd.read_csv(file_path)
        logging.info(f"Data loaded successfully with shape {data.shape}")
        return data
    except Exception as e:
        logging.error(f"Failed to load data: {e}")
        raise


def preprocess_data(data):
    try:
        # Assuming the class label is 'Class' and features are all other columns
        X = data.drop('Class', axis=1)
        y = data['Class']
        X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.2, random_state=RANDOM_SEED)
        
        # Upsample the minority class in the training set
        X_train_majority = X_train[y_train == 0]
        X_train_minority = X_train[y_train == 1]
        
        X_train_minority_upsampled = resample(X_train_minority,
                                              replace=True,
                                              n_samples=X_train_majority.shape[0],
                                              random_state=RANDOM_SEED)
        y_train_minority_upsampled = np.ones(X_train_minority_upsampled.shape[0], dtype=int)
        
        X_train_upsampled = pd.concat([X_train_majority, X_train_minority_upsampled])
        y_train_upsampled = np.concatenate([y_train[y_train == 0], y_train_minority_upsampled])
        
        logging.info("Data preprocessing completed successfully.")
        return X_train_upsampled, X_test, y_train_upsampled, y_test
    except Exception as e:
        logging.error(f"Error in data preprocessing: {e}")
        raise


def train_model(X_train, y_train):
    try:
        model = RandomForestClassifier(random_state=RANDOM_SEED)
        model.fit(X_train, y_train)
        logging.info("Model training completed successfully.")
        return model
    except Exception as e:
        logging.error(f"Error in model training: {e}")
        raise


def evaluate_model(model, X_test, y_test):
    try:
        predictions = model.predict(X_test)
        auc_score = roc_auc_score(y_test, predictions)
        report = classification_report(y_test, predictions, output_dict=True)
        logging.info(f"Model evaluation completed. AUC: {auc_score}")
        return auc_score, report
    except Exception as e:
        logging.error(f"Error in model evaluation: {e}")
        raise


def log_experiment(params, metrics, model):
    try:
        mlflow.set_experiment('Fraud Detection Experiment')
        with mlflow.start_run():
            mlflow.log_params(params)
            mlflow.log_metrics(metrics)
            mlflow.sklearn.log_model(model, "model")
        logging.info("Experiment logged to MLflow successfully.")
    except Exception as e:
        logging.error(f"Error in logging experiment to MLflow: {e}")
        raise


def main(data_path):
    data = load_data(data_path)
    X_train, X_test, y_train, y_test = preprocess_data(data)
    model = train_model(X_train, y_train)
    auc_score, report = evaluate_model(model, X_test, y_test)
    params = model.get_params()
    metrics = {'AUC': auc_score, 'precision': report['1']['precision'], 'recall': report['1']['recall'], 'f1-score': report['1']['f1-score']}
    log_experiment(params, metrics, model)


if __name__ == "__main__":
    data_path = "creditcard.csv"
    main(data_path)
```

### 2. Pytest Test File (`test_pipeline.py`)

```python
import pytest
import pandas as pd
from sklearn.ensemble import RandomForestClassifier
from pipeline import load_data, preprocess_data, train_model, evaluate_model

RANDOM_SEED = 42

@pytest.fixture
def dummy_data():
    data = {
        'feature1': [0.1, 0.2, 0.1, 0.3],
        'feature2': [1.0, 0.9, 0.8, 1.1],
        'Class': [0, 0, 1, 1]
    }
    return pd.DataFrame(data)


def test_load_data(dummy_data, monkeypatch):
    def mock_load_data(file_path):
        return dummy_data

    monkeypatch.setattr('pipeline.load_data', mock_load_data)
    data = load_data("dummy_path")
    assert not data.empty, "Data should not be empty"


def test_preprocess_data(dummy_data):
    X_train, X_test, y_train, y_test = preprocess_data(dummy_data)
    assert len(X_train) == len(y_train), "Mismatch in train feature and label size"
    assert len(X_test) == len(y_test), "Mismatch in test feature and label size"


def test_train_model(dummy_data):
    X_train, _, y_train, _ = preprocess_data(dummy_data)
    model = train_model(X_train, y_train)
    assert isinstance(model, RandomForestClassifier), "Model should be an instance of RandomForestClassifier"
```

### 3. GitHub Actions Workflow (`.github/workflows/ci-cd.yml`)

```yaml
name: CI/CD Pipeline

on:
  push:
    branches:
      - main
  workflow_dispatch:

jobs:
  build:
    runs-on: ubuntu-latest

    steps:
    - name: Checkout Code
      uses: actions/checkout@v2

    - name: Set up Python
      uses: actions/setup-python@v2
      with:
        python-version: '3.8'

    - name: Install Dependencies
      run: |
        python -m pip install --upgrade pip
        pip install -r requirements.txt

    - name: Run Tests
      run: |
        pytest --maxfail=1 --disable-warnings

    - name: Run Pipeline
      run: |
        python pipeline.py
```

### Additional Notes:
