### 1. Python Pipeline Script

This script includes data loading, model training, evaluation, and logging to MLflow. It also checks for concept drift to trigger retraining.

```python
# fraud_detection_pipeline.py

import numpy as np
import pandas as pd
from sklearn.ensemble import RandomForestClassifier
from sklearn.metrics import classification_report, roc_auc_score
from sklearn.model_selection import train_test_split
from sklearn.utils import resample
import mlflow
import mlflow.sklearn
import logging
import os

# Set logging
logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)

# Seed for reproducibility
SEED = 42

def load_data():
    """Load and return the dataset."""
    try:
        data = pd.read_csv('creditcard.csv')
        return data
    except Exception as e:
        logger.error(f"Error loading data: {e}")
        raise

def preprocess_data(df):
    """Preprocess the data, handle imbalance."""
    df_majority = df[df.Class == 0]
    df_minority = df[df.Class == 1]
    
    df_minority_upsampled = resample(df_minority, 
                                     replace=True,
                                     n_samples=int(len(df_majority) * 0.1), 
                                     random_state=SEED)
    
    df_upsampled = pd.concat([df_majority, df_minority_upsampled])
    
    X = df_upsampled.drop('Class', axis=1)
    y = df_upsampled['Class']
    
    return train_test_split(X, y, test_size=0.2, random_state=SEED)

def train_model(X_train, y_train):
    """Train the RandomForest model."""
    model = RandomForestClassifier(random_state=SEED)
    model.fit(X_train, y_train)
    return model

def evaluate_model(model, X_test, y_test):
    """Evaluate the model."""
    y_pred = model.predict(X_test)
    y_prob = model.predict_proba(X_test)[:, 1]
    report = classification_report(y_test, y_pred)
    roc_auc = roc_auc_score(y_test, y_prob)
    logger.info(f"Classification Report:\n{report}")
    logger.info(f"ROC-AUC Score: {roc_auc}")
    return roc_auc

def log_experiment(roc_auc, model):
    """Log experiment to MLflow."""
    try:
        mlflow.start_run()
        mlflow.log_metric("roc_auc", roc_auc)
        mlflow.sklearn.log_model(model, "model")
        mlflow.end_run()
    except Exception as e:
        logger.error(f"Error logging to MLflow: {e}")
        raise

def check_concept_drift(prev_roc_auc, current_roc_auc, threshold=0.05):
    """Check for concept drift."""
    return abs(prev_roc_auc - current_roc_auc) > threshold

def main():
    try:
        data = load_data()
        X_train, X_test, y_train, y_test = preprocess_data(data)
        model = train_model(X_train, y_train)
        roc_auc = evaluate_model(model, X_test, y_test)
        prev_roc_auc = 0.90  # Example previous ROC-AUC score for drift detection
        if check_concept_drift(prev_roc_auc, roc_auc):
            logger.info("Concept drift detected. Retraining the model.")
            # Retrain logic or further actions can be added here
        log_experiment(roc_auc, model)
    except Exception as e:
        logger.error(f"Pipeline execution failed: {e}")

if __name__ == "__main__":
    main()
```

### 3. GitHub Actions Workflow

This workflow automates testing and deployment when changes are pushed to the main branch.

```yaml
# .github/workflows/ci_cd_pipeline.yml

name: CI/CD Pipeline

on:
  push:
    branches:
      - main
  pull_request:
    branches:
      - main

jobs:
  build:
    runs-on: ubuntu-latest

    steps:
    - name: Checkout code
      uses: actions/checkout@v2

    - name: Set up Python
      uses: actions/setup-python@v2
      with:
        python-version: '3.8'

    - name: Install dependencies
      run: |
        python -m pip install --upgrade pip
        pip install -r requirements.txt

    - name: Run tests
      run: |
        pytest

    - name: Run pipeline
      env:
        MLFLOW_TRACKING_URI: 'http://your_mlflow_server:5000'  # Set your MLflow server
      run: |
        python fraud_detection_pipeline.py
```
