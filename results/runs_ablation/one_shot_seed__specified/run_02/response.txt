Here's a complete CI/CD pipeline configuration for a fraud detection model using scikit-learn's RandomForestClassifier, MLflow for tracking, pytest for testing, and GitHub Actions for CI/CD. The pipeline is designed to automatically retrain the model when concept drift is detected.

### 1. Python Pipeline Script

```python
# pipeline.py

import numpy as np
import pandas as pd
from sklearn.ensemble import RandomForestClassifier
from sklearn.metrics import classification_report, roc_auc_score
from sklearn.model_selection import train_test_split
from sklearn.utils import shuffle
import mlflow
import mlflow.sklearn
import logging

SEED = 42
np.random.seed(SEED)

# Set up logging
logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)

def load_data(file_path='creditcard.csv'):
    try:
        data = pd.read_csv(file_path)
        return data
    except Exception as e:
        logger.error(f"Error loading data: {e}")
        raise

def preprocess_data(data):
    try:
        X = data.drop('Class', axis=1)
        y = data['Class']
        return X, y
    except Exception as e:
        logger.error(f"Error preprocessing data: {e}")
        raise

def train_model(X_train, y_train):
    try:
        model = RandomForestClassifier(random_state=SEED)
        model.fit(X_train, y_train)
        return model
    except Exception as e:
        logger.error(f"Error training model: {e}")
        raise

def evaluate_model(model, X_test, y_test):
    try:
        y_pred = model.predict(X_test)
        report = classification_report(y_test, y_pred)
        auc_score = roc_auc_score(y_test, y_pred)
        logger.info(f"Classification Report:\n{report}")
        logger.info(f"AUC Score: {auc_score}")
        return auc_score
    except Exception as e:
        logger.error(f"Error evaluating model: {e}")
        raise

def log_experiment(auc_score, model):
    try:
        mlflow.log_metric("AUC Score", auc_score)
        mlflow.sklearn.log_model(model, "model")
    except Exception as e:
        logger.error(f"Error logging to MLflow: {e}")
        raise

def detect_concept_drift(auc_score, threshold=0.8):
    # Placeholder for a real drift detection method
    return auc_score < threshold

def main():
    with mlflow.start_run():
        data = load_data()
        X, y = preprocess_data(data)
        X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.2, random_state=SEED, stratify=y)
        
        model = train_model(X_train, y_train)
        auc_score = evaluate_model(model, X_test, y_test)
        
        log_experiment(auc_score, model)

        if detect_concept_drift(auc_score):
            logger.info("Concept drift detected. Retraining the model...")
            model = train_model(X_train, y_train)
            auc_score = evaluate_model(model, X_test, y_test)
            log_experiment(auc_score, model)

if __name__ == '__main__':
    main()
```

### 2. Pytest Test File

```python
# test_pipeline.py

import pytest
from pipeline import load_data, preprocess_data, train_model, evaluate_model

SEED = 42

def test_load_data():
    data = load_data()
    assert not data.empty, "Data should not be empty"

def test_preprocess_data():
    data = load_data()
    X, y = preprocess_data(data)
    assert X.shape[0] == y.shape[0], "Features and target should have the same number of samples"

def test_train_model():
    data = load_data()
    X, y = preprocess_data(data)
    X_train, _, y_train, _ = train_test_split(X, y, test_size=0.2, random_state=SEED, stratify=y)
    model = train_model(X_train, y_train)
    assert hasattr(model, "predict"), "Model should have a predict method"
```

### 3. GitHub Actions YAML Workflow

```yaml
# .github/workflows/ci-cd.yml

name: CI/CD Pipeline

on:
  push:
    branches:
      - main
  pull_request:
    branches:
      - main
  workflow_dispatch:

jobs:
  build:
    runs-on: ubuntu-latest

    steps:
    - name: Checkout code
      uses: actions/checkout@v2

    - name: Set up Python
      uses: actions/setup-python@v2
      with:
        python-version: '3.8'

    - name: Install dependencies
      run: |
        python -m pip install --upgrade pip
        pip install -r requirements.txt

    - name: Run tests
      run: |
        pytest --tb=short --disable-warnings

    - name: Run pipeline
      run: |
        python pipeline.py
```

### Notes
- Ensure `requirements.txt` includes necessary libraries: `pandas`, `scikit-learn`, `mlflow`, `pytest`, and any other dependencies used.
- The pipeline includes a placeholder function `detect_concept_drift` that currently uses a simple threshold for demonstration. In production, consider using more sophisticated drift detection techniques.
- Make sure the CSV file for the dataset is in the right path or update the `load_data` function with the correct path.
- This setup assumes your GitHub repository is connected to an MLflow server for logging experiments. Adjust the MLflow configuration as needed.
