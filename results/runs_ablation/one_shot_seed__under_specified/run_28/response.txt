### Python Pipeline Module (`pipeline.py`)

This module will include a simple fraud detection model using scikit-learn. We'll use a logistic regression model as an example.

```python
# pipeline.py

import numpy as np
import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.linear_model import LogisticRegression
from sklearn.metrics import accuracy_score, precision_score, recall_score, f1_score
import joblib

RANDOM_SEED = 42

def generate_synthetic_data(n_samples=1000, random_state=RANDOM_SEED):
    np.random.seed(random_state)
    X = np.random.rand(n_samples, 10)
    y = np.random.randint(0, 2, n_samples)
    return pd.DataFrame(X, columns=[f'feature_{i}' for i in range(X.shape[1])]), pd.Series(y)

def train_model(X, y, random_state=RANDOM_SEED):
    X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.2, random_state=random_state)
    model = LogisticRegression(random_state=random_state)
    model.fit(X_train, y_train)
    y_pred = model.predict(X_test)
    metrics = {
        'accuracy': accuracy_score(y_test, y_pred),
        'precision': precision_score(y_test, y_pred),
        'recall': recall_score(y_test, y_pred),
        'f1_score': f1_score(y_test, y_pred)
    }
    return model, metrics

def save_model(model, filename='fraud_detection_model.pkl'):
    joblib.dump(model, filename)

if __name__ == '__main__':
    X, y = generate_synthetic_data()
    model, metrics = train_model(X, y)
    save_model(model)
    print(f"Model metrics: {metrics}")
```

### Pytest Test File (`test_pipeline.py`)

This test file will ensure the pipeline works as expected and the model generates reproducible results.

```python
# test_pipeline.py

import pytest
from pipeline import generate_synthetic_data, train_model, RANDOM_SEED

def test_generate_synthetic_data():
    X, y = generate_synthetic_data()
    assert X.shape == (1000, 10)
    assert len(y) == 1000

def test_train_model_reproducibility():
    X, y = generate_synthetic_data()
    model1, metrics1 = train_model(X, y)
    model2, metrics2 = train_model(X, y)
    
    assert metrics1 == metrics2
    assert model1.coef_.all() == model2.coef_.all()

if __name__ == '__main__':
    pytest.main()
```
