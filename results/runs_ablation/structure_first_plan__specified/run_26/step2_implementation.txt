def train_model(data):
    """Train a RandomForestClassifier model on the given data."""
    X, y = data
    X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.3, random_state=42)

    model.fit(X_train, y_train)

def evaluate_model(model, X_test, y_test):
    """Evaluate the trained model and return evaluation metrics."""
    y_pred = model.predict(X_test)
    metrics = {
        'accuracy': accuracy_score(y_test, y_pred),
        'precision': precision_score(y_test, y_pred),
        'recall': recall_score(y_test, y_pred),
        'f1_score': f1_score(y_test, y_pred)
    }
    return metrics

def log_to_mlflow(model, metrics):
    """Log model and metrics to MLflow."""
    with mlflow.start_run():
        # Log model parameters
        mlflow.log_param("n_estimators", model.get_params()['n_estimators'])
        mlflow.log_param("random_state", model.get_params()['random_state'])

        # Log evaluation metrics
        for key, value in metrics.items():
            mlflow.log_metric(key, value)

        # Log the model itself
        mlflow.sklearn.log_model(model, "random_forest_model")
        print("Model and metrics logged to MLflow.")
