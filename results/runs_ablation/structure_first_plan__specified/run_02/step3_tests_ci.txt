### (b) `.github/workflows/ci_cd_pipeline.yml`
```yaml
name: CI/CD Pipeline

on:
  push:
    branches:
      - main
  pull_request:
    branches:
      - main
  schedule:
    - cron: '0 0 * * 0' # Weekly schedule
  workflow_dispatch: # Manual trigger

jobs:
  check-drift:
    runs-on: ubuntu-latest
    outputs:
      drift_detected: ${{ steps.detect_drift.outputs.drift_detected }}
    steps:
      - name: Checkout repository
        uses: actions/checkout@v2
      
      - name: Set up Python
        uses: actions/setup-python@v2
        with:
          python-version: '3.8'
      
      - name: Install dependencies
        run: pip install -r requirements.txt
      
      - name: Detect Concept Drift
        id: detect_drift
        run: |
          python -c "
          from fraud_detection.pipeline.data_loader import load_data, detect_concept_drift
          data = load_data('creditcard.csv')
          drift_detected = detect_concept_drift(data)
          print(f'::set-output name=drift_detected::{drift_detected}')"
      
  train-and-eval:
    needs: check-drift
    if: ${{ needs.check-drift.outputs.drift_detected == 'true' }}
    runs-on: ubuntu-latest
    steps:
      - name: Checkout repository
        uses: actions/checkout@v2
      
      - name: Set up Python
        uses: actions/setup-python@v2
        with:
          python-version: '3.8'
      
      - name: Install dependencies
        run: pip install -r requirements.txt
      
      - name: Run Training and Evaluation
        run: |
          python -c "
          from fraud_detection.pipeline.data_loader import load_data
          from fraud_detection.pipeline.model_trainer import train_model
          from fraud_detection.pipeline.model_evaluator import evaluate_model
          from fraud_detection.pipeline.mlflow_logger import log_experiment
          data = load_data('creditcard.csv')
          model = train_model(data)
          metrics = evaluate_model(model, data)
          log_experiment(metrics, model)"
      
  test:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout repository
        uses: actions/checkout@v2
      
      - name: Set up Python
        uses: actions/setup-python@v2
        with:
          python-version: '3.8'
      
      - name: Install dependencies
        run: pip install -r requirements.txt
      
      - name: Run Tests
        run: pytest tests/
```
