### .github/workflows/ci_cd_pipeline.yml
```yaml
name: CI/CD Pipeline

on:
  push:
    branches:
      - main
  pull_request:
    branches:
      - main
  schedule:
    - cron: '0 0 * * 0' # Weekly schedule
  workflow_dispatch: # Manual trigger

jobs:
  check-drift:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout repository
        uses: actions/checkout@v2
      
      - name: Set up Python
        uses: actions/setup-python@v2
        with:
          python-version: '3.8'
      
      - name: Install dependencies
        run: pip install -r requirements.txt
      
      - name: Detect Concept Drift
        run: python -m pipeline.data_loader

  train-and-eval:
    needs: check-drift
    if: ${{ needs.check-drift.outputs.drift_detected == 'true' }}
    runs-on: ubuntu-latest
    steps:
      - name: Checkout repository
        uses: actions/checkout@v2
      
      - name: Set up Python
        uses: actions/setup-python@v2
        with:
          python-version: '3.8'
      
      - name: Install dependencies
        run: pip install -r requirements.txt
      
      - name: Run Training and Evaluation
        run: python -m pipeline.model_trainer

  test:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout repository
        uses: actions/checkout@v2
      
      - name: Set up Python
        uses: actions/setup-python@v2
        with:
          python-version: '3.8'
      
      - name: Install dependencies
        run: pip install -r requirements.txt
      
      - name: Run Tests
        run: pytest tests/
```

This structure provides a clear separation of concerns and modularity, with each component responsible for a specific part of the pipeline. The CI/CD workflow is organized to detect concept drift, train and evaluate the model, and run tests on GitHub Actions.
