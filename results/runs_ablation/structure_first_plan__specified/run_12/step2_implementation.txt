```python
import pandas as pd
from sklearn.ensemble import RandomForestClassifier
from sklearn.model_selection import train_test_split
from sklearn.metrics import accuracy_score, precision_score, recall_score, f1_score
import mlflow
import mlflow.sklearn

def load_data(file_path: str) -> pd.DataFrame:
    """Load the credit card fraud dataset from a CSV file."""
    data = pd.read_csv(file_path)
    return data

def preprocess_data(data: pd.DataFrame) -> (pd.DataFrame, pd.Series):
    """Preprocess the dataset and return features and target variable."""
    # Assuming the last column is the target variable
    X = data.iloc[:, :-1]
    y = data.iloc[:, -1]
    # Add preprocessing steps as needed
    return X, y

def split_data(features: pd.DataFrame, target: pd.Series) -> (pd.DataFrame, pd.DataFrame, pd.Series, pd.Series):
    """Split the dataset into training and test sets."""
    X_train, X_test, y_train, y_test = train_test_split(features, target, test_size=0.2, random_state=42)
    return X_train, X_test, y_train, y_test

def train_model(X_train: pd.DataFrame, y_train: pd.Series) -> RandomForestClassifier:
    """Train a RandomForestClassifier model."""
    model = RandomForestClassifier(n_estimators=100, random_state=42)
    model.fit(X_train, y_train)
    return model

def evaluate_model(model: RandomForestClassifier, X_test: pd.DataFrame, y_test: pd.Series) -> dict:
    """Evaluate the model and return performance metrics."""
    predictions = model.predict(X_test)
    metrics = {
        'accuracy': accuracy_score(y_test, predictions),
        'precision': precision_score(y_test, predictions),
        'recall': recall_score(y_test, predictions),
        'f1_score': f1_score(y_test, predictions)
    }
    return metrics

def log_to_mlflow(model: RandomForestClassifier, metrics: dict) -> None:
    """Log the model and metrics to MLflow."""
    with mlflow.start_run():
        mlflow.log_params(model.get_params())
        mlflow.log_metrics(metrics)
        mlflow.sklearn.log_model(model, "random_forest_model")

def check_concept_drift() -> bool:
    """Check for concept drift and return whether retraining is needed."""
    # Placeholder implementation
    # Concept drift detection logic would be added here
    return False

def main(file_path: str) -> None:
    """Execute the complete fraud detection pipeline."""
    data = load_data(file_path)
    X, y = preprocess_data(data)
    X_train, X_test, y_train, y_test = split_data(X, y)
    model = train_model(X_train, y_train)
    metrics = evaluate_model(model, X_test, y_test)
    log_to_mlflow(model, metrics)
    concept_drift = check_concept_drift()
    if concept_drift:
        print("Concept drift detected. Consider retraining the model.")
    else:
        print("No concept drift detected. Model is up-to-date.")

if __name__ == "__main__":
    import argparse
    parser = argparse.ArgumentParser(description="Fraud Detection Pipeline")
    parser.add_argument('--file_path', type=str, required=True, help='Path to the credit card fraud dataset CSV file')
    args = parser.parse_args()
    main(args.file_path)
```
