on:
  push:
    branches:
      - main
  pull_request:
    branches:
      - main

    - cron: '0 0 * * *'  # Daily at midnight

  build:
    runs-on: ubuntu-latest

      - name: Checkout repository

      - name: Set up Python

        with:
          python-version: '3.8'

      - name: Install dependencies
        run: |
          python -m pip install --upgrade pip

      - name: Run tests
        run: |
          pytest tests/

  deploy:
    runs-on: ubuntu-latest
    needs: build
    steps:
      - name: Checkout repository
        uses: actions/checkout@v2

      - name: Deploy model
        run: echo "Deploy the model to the production environment"
