### GitHub Actions Workflow: `.github/workflows/ci_cd_pipeline.yml`

```yaml
name: CI/CD Pipeline

on:
  push:
    branches:
      - main
  pull_request:
    branches:
      - main
  schedule:
    - cron: '0 0 * * 0' # Weekly schedule
  workflow_dispatch:

jobs:
  setup:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout code
        uses: actions/checkout@v2

      - name: Set up Python
        uses: actions/setup-python@v2
        with:
          python-version: '3.8'

  install_dependencies:
    runs-on: ubuntu-latest
    steps:
      - name: Install dependencies
        run: |
          python -m pip install --upgrade pip
          pip install -r requirements.txt

  test:
    runs-on: ubuntu-latest
    steps:
      - name: Run pytest
        run: |
          pytest tests/

  run_pipeline:
    runs-on: ubuntu-latest
    steps:
      - name: Execute pipeline
        run: |
          python fraud_detection_pipeline/pipeline.py --file_path data/creditcard.csv

  notify:
    runs-on: ubuntu-latest
    steps:
      - name: Notify stakeholders
        run: echo "Pipeline execution complete."
```

This structure includes the necessary elements to execute a CI/CD pipeline for a fraud detection model using scikit-learn, MLflow, and pytest, with GitHub Actions handling the automation.
