from scipy.stats import ks_2samp

def detect_concept_drift(current_data, reference_data):
    """Detect concept drift between current and reference datasets."""
    drift_detected = False
    for column in current_data.columns:
        stat, p_value = ks_2samp(current_data[column], reference_data[column])
        if p_value < 0.05:
            drift_detected = True
            print(f"Concept drift detected in column: {column}")
    return drift_detected
