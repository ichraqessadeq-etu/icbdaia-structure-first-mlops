# fraud_detection_pipeline/tests/test_pipeline.py
import pytest
from pipeline.data_loader import load_data, preprocess_data
from pipeline.model_trainer import train_model
from pipeline.model_evaluator import evaluate_model
import pandas as pd
from sklearn.datasets import make_classification
from sklearn.ensemble import RandomForestClassifier

@pytest.fixture
def sample_data():
    """Fixture to provide sample data for testing."""
    X, y = make_classification(n_samples=100, n_features=20, n_informative=2, n_redundant=10, random_state=42)
    data = pd.DataFrame(X, columns=[f'feature_{i}' for i in range(20)])
    data['target'] = y
    return data

def test_load_data(tmp_path):
    """Test that data is loaded correctly."""
    df = pd.DataFrame({'feature_1': [1, 2], 'feature_2': [3, 4], 'target': [0, 1]})
    file_path = tmp_path / "test.csv"
    df.to_csv(file_path, index=False)

    loaded_data = load_data(str(file_path))
    pd.testing.assert_frame_equal(loaded_data, df)

def test_preprocess_data(sample_data):
    """Test that preprocessing is performed correctly."""
    X_train, X_test, y_train, y_test = preprocess_data(sample_data)
    assert X_train.shape[0] > 0
    assert X_test.shape[0] > 0
    assert X_train.shape[1] == sample_data.shape[1] - 1  # Minus target column

def test_train_model(sample_data):
    """Test that the model trains without errors."""
    X_train, _, y_train, _ = preprocess_data(sample_data)
    model = train_model(X_train, y_train)
    assert isinstance(model, RandomForestClassifier)

def test_evaluate_model(sample_data):
    """Test that the evaluation returns expected metrics format."""
    X_train, X_test, y_train, y_test = preprocess_data(sample_data)
    model = train_model(X_train, y_train)
    metrics = evaluate_model(model, X_test, y_test)
    assert 'accuracy' in metrics
    assert 'report' in metrics
    assert isinstance(metrics['accuracy'], float)
    assert isinstance(metrics['report'], str)
