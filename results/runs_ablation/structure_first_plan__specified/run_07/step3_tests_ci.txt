from project.pipeline.data import load_data, check_imbalance

def test_check_imbalance(sample_data):
    """Test if the imbalance check function works correctly."""

    assert isinstance(is_imbalanced, bool)

def test_train_model(sample_data):
    """Test if the model is trained successfully."""
    model = train_model(sample_data)
    assert model is not None
    assert hasattr(model, "predict")

def test_evaluate_model(sample_data):
    """Test if the model evaluation metrics are correct."""
    model = train_model(sample_data)
    metrics = evaluate_model(model, sample_data)
    assert isinstance(metrics, dict)
    assert "accuracy" in metrics
    assert "precision" in metrics
