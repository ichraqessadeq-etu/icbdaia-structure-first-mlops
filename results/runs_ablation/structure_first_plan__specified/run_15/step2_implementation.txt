```python
import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.ensemble import RandomForestClassifier
from sklearn.metrics import classification_report, accuracy_score
from imblearn.over_sampling import SMOTE
import mlflow
import numpy as np

def load_data(file_path: str) -> tuple:
    """Load the dataset from the given file path."""
    data = pd.read_csv(file_path)
    X = data.drop('Class', axis=1)
    y = data['Class']
    return X, y

def preprocess_data(X, y) -> tuple:
    """Preprocess the data, handle imbalanced classes."""
    smote = SMOTE(random_state=42)
    X_resampled, y_resampled = smote.fit_resample(X, y)
    X_train, X_test, y_train, y_test = train_test_split(X_resampled, y_resampled, test_size=0.2, random_state=42)
    return X_train, X_test, y_train, y_test

def train_model(X_train, y_train) -> RandomForestClassifier:
    """Train the RandomForestClassifier model."""
    model = RandomForestClassifier(n_estimators=100, random_state=42)
    model.fit(X_train, y_train)
    return model

def evaluate_model(model, X_test, y_test) -> dict:
    """Evaluate the model and return evaluation metrics."""
    y_pred = model.predict(X_test)
    accuracy = accuracy_score(y_test, y_pred)
    report = classification_report(y_test, y_pred, output_dict=True)
    return {'accuracy': accuracy, 'report': report}

def log_to_mlflow(metrics: dict, model: RandomForestClassifier, params: dict) -> None:
    """Log the model and metrics to MLflow."""
    mlflow.set_experiment("Fraud Detection")
    with mlflow.start_run():
        mlflow.log_params(params)
        mlflow.log_metrics(metrics)
        mlflow.sklearn.log_model(model, "model")

def detect_concept_drift(reference_data, current_data) -> bool:
    """Detect concept drift between reference and current data."""
    reference_mean = np.mean(reference_data, axis=0)
    current_mean = np.mean(current_data, axis=0)
    drift_detected = np.linalg.norm(reference_mean - current_mean) > 0.1  # arbitrary threshold
    return drift_detected

def main(file_path: str) -> None:
    """Main function to execute the pipeline."""
    # Load and preprocess data
    X, y = load_data(file_path)
    X_train, X_test, y_train, y_test = preprocess_data(X, y)

    # Train model
    model = train_model(X_train, y_train)

    # Evaluate model
    metrics = evaluate_model(model, X_test, y_test)
    print("Evaluation Metrics:", metrics)

    # Log to MLflow
    params = {'n_estimators': 100, 'random_state': 42}
    log_to_mlflow(metrics, model, params)

    # Detect concept drift (example use)
    drift_detected = detect_concept_drift(X_train, X_test)
    print("Concept Drift Detected:", drift_detected)

if __name__ == "__main__":
    import argparse
    parser = argparse.ArgumentParser(description="Fraud Detection Pipeline")
    parser.add_argument('--file-path', type=str, required=True, help='Path to the dataset file')
    args = parser.parse_args()
    main(args.file_path)
```
