# src/data_loader.py
import pandas as pd
from sklearn.model_selection import train_test_split

def load_data() -> tuple:
    """Load and preprocess the credit card fraud dataset."""
    # Load the dataset
    df = pd.read_csv('creditcard.csv')

    # Preprocess the dataset (e.g., scaling, removing nulls)
    df.fillna(0, inplace=True)
    X = df.drop('Class', axis=1)
    y = df['Class']

    # Split the dataset into training and testing sets
    X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.3, random_state=42)

    return X_train, X_test, y_train, y_test

# src/model_trainer.py
from sklearn.ensemble import RandomForestClassifier

def train_model(X_train, y_train) -> object:
    """Train a RandomForestClassifier model."""
    model = RandomForestClassifier(n_estimators=100, random_state=42)
    model.fit(X_train, y_train)
    return model

# src/model_evaluator.py
from sklearn.metrics import accuracy_score, precision_score, recall_score, f1_score

def evaluate_model(model, X_test, y_test) -> dict:
    """Evaluate the trained model and return performance metrics."""
    y_pred = model.predict(X_test)

    metrics = {
        'accuracy': accuracy_score(y_test, y_pred),
        'precision': precision_score(y_test, y_pred),
        'recall': recall_score(y_test, y_pred),
        'f1_score': f1_score(y_test, y_pred),
    }

    return metrics

# src/mlflow_logger.py
import mlflow

def log_to_mlflow(params: dict, metrics: dict, model: object) -> None:
    """Log parameters, metrics, and model to MLflow."""
    with mlflow.start_run():
        mlflow.log_params(params)
        mlflow.log_metrics(metrics)

        # Log the model
        mlflow.sklearn.log_model(model, "model")
