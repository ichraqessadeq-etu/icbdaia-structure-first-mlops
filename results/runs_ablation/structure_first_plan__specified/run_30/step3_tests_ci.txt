### (b) GitHub Actions Workflow

```yaml
# .github/workflows/ci_cd_pipeline.yml

name: CI/CD Pipeline

on:
  push:
    branches:
      - main
  pull_request:
    branches:
      - main
  schedule:
    - cron: '0 0 * * SUN'  # Weekly scheduled run

jobs:
  setup:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout code
        uses: actions/checkout@v2

      - name: Set up Python
        uses: actions/setup-python@v2
        with:
          python-version: '3.x'

      - name: Install dependencies
        run: |
          python -m pip install --upgrade pip
          pip install -r requirements.txt

  test:
    runs-on: ubuntu-latest
    needs: setup
    steps:
      - name: Run tests
        run: |
          pytest tests/

  detect_concept_drift:
    runs-on: ubuntu-latest
    needs: test
    steps:
      - name: Check for concept drift
        run: |
          python detect_concept_drift.py

  retrain_model:
    runs-on: ubuntu-latest
    if: success() && needs.detect_concept_drift.outputs.drift_detected == 'true'
    steps:
      - name: Checkout code
        uses: actions/checkout@v2

      - name: Retrain model
        run: |
          python -m fraud_detection.pipeline
```
