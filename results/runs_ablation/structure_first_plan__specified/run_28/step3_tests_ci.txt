import pytest
import pandas as pd
from fraud_detection_pipeline.pipeline import load_data, preprocess_data, train_model

def test_load_data():
    """Test the data loading function."""
    # Assuming there is a sample CSV file for testing
    sample_data_path = 'sample_data.csv'
    data = load_data(sample_data_path)
    assert isinstance(data, pd.DataFrame)
    assert not data.empty

def test_preprocess_data():
    """Test the data preprocessing function."""
    # Sample data creation
    sample_data = pd.DataFrame({
        'feature1': [1, 2, 3, None],
        'feature2': ['A', 'B', 'A', 'C'],
        'target': [0, 1, 0, 1]
    })
    preprocessed_data = preprocess_data(sample_data)
    assert not preprocessed_data.isnull().values.any()
    assert 'feature2_A' in preprocessed_data.columns
    assert 'feature2_B' in preprocessed_data.columns
    assert 'feature2_C' in preprocessed_data.columns

def test_train_model():
    """Test the model training function."""
    # Sample preprocessed data with target
    sample_data = pd.DataFrame({
        'feature1': [1, 2, 3, 4],
        'feature2_A': [1, 0, 1, 0],
        'feature2_B': [0, 1, 0, 0],
        'feature2_C': [0, 0, 0, 1],
        'target': [0, 1, 0, 1]

### .github/workflows/ci_cd.yml

```yaml
name: CI/CD Pipeline

on:
  push:
    branches:
      - main
  pull_request:
    branches:
      - main

jobs:
  test:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout repository
        uses: actions/checkout@v2
      - name: Set up Python
        uses: actions/setup-python@v2
        with:
          python-version: '3.8'
      - name: Install dependencies
        run: pip install -r requirements.txt
      - name: Run tests
        run: pytest fraud_detection_pipeline/test_pipeline.py

  deploy:
    runs-on: ubuntu-latest
    needs: test
    steps:
      - name: Checkout repository
        uses: actions/checkout@v2
      - name: Set up Python
        uses: actions/setup-python@v2
        with:
          python-version: '3.8'
      - name: Install dependencies
        run: pip install -r requirements.txt
      - name: Run pipeline
        run: python fraud_detection_pipeline/pipeline.py
```

Ensure that the `requirements.txt` file contains all necessary dependencies, such as `pandas`, `scikit-learn`, `mlflow`, and `pytest`, so that they are installed correctly during the CI/CD process. Also, replace `'path/to/your/data.csv'` with the actual path to your dataset in `pipeline.py` for running the pipeline.
