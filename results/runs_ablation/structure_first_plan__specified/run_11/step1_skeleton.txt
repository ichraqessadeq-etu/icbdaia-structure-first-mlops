### GitHub Actions Workflow

#### `.github/workflows/ci-cd.yml`

```yaml
name: CI/CD Pipeline

on:
  push:
    branches:
      - main
  pull_request:
    branches:
      - main

jobs:
  setup:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout code
        uses: actions/checkout@v2
      
      - name: Set up Python
        uses: actions/setup-python@v2
        with:
          python-version: '3.8'
      
      - name: Install dependencies
        run: pip install -r requirements.txt

  test:
    runs-on: ubuntu-latest
    needs: setup
    steps:
      - name: Run tests
        run: pytest tests/

  build-and-deploy:
    runs-on: ubuntu-latest
    needs: test
    steps:
      - name: Run pipeline script
        run: python pipeline/main.py
      
      - name: Deploy model
        run: echo "Deploying model..."
```
