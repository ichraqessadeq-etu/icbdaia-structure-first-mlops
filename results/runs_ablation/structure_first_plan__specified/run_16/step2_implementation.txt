import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.ensemble import RandomForestClassifier
from sklearn.metrics import accuracy_score, precision_score, recall_score, f1_score
from sklearn.utils import resample
import mlflow
import mlflow.sklearn
import numpy as np

def load_data(file_path: str) -> tuple:
    """Load the credit card fraud dataset."""
    data = pd.read_csv(file_path)
    X = data.drop('Class', axis=1)
    y = data['Class']
    return X, y

def preprocess_data(data: tuple) -> tuple:
    """Preprocess the data to handle imbalance and prepare for training."""
    X, y = data
    # Handle class imbalance using resampling
    X_minority = X[y == 1]
    y_minority = y[y == 1]
    X_majority = X[y == 0]
    y_majority = y[y == 0]

    X_minority_upsampled, y_minority_upsampled = resample(
        X_minority, y_minority,
        replace=True,  # sample with replacement
        n_samples=len(y_majority),  # to match majority class
        random_state=42  # reproducible results
    )

    X_balanced = pd.concat([X_majority, X_minority_upsampled])
    y_balanced = pd.concat([y_majority, y_minority_upsampled])

    X_train, X_test, y_train, y_test = train_test_split(X_balanced, y_balanced, test_size=0.2, random_state=42)
    return X_train, X_test, y_train, y_test

def train_model(X_train: any, y_train: any) -> any:
    """Train a RandomForestClassifier model."""
    model = RandomForestClassifier(n_estimators=100, random_state=42)
    model.fit(X_train, y_train)
    return model

def evaluate_model(model: any, X_test: any, y_test: any) -> dict:
    """Evaluate the model and return performance metrics."""
    y_pred = model.predict(X_test)
    metrics = {
        'accuracy': accuracy_score(y_test, y_pred),
        'precision': precision_score(y_test, y_pred),
        'recall': recall_score(y_test, y_pred),
        'f1_score': f1_score(y_test, y_pred)
    }
    return metrics

def log_to_mlflow(metrics: dict, model: any):
    """Log model and metrics to MLflow."""
    mlflow.set_experiment("Fraud Detection")
    with mlflow.start_run():
        mlflow.log_metrics(metrics)
        mlflow.sklearn.log_model(model, "model")
