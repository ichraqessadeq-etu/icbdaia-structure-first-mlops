import pandas as pd
import numpy as np
from sklearn.model_selection import train_test_split
from sklearn.ensemble import RandomForestClassifier
from sklearn.metrics import accuracy_score, precision_score, recall_score
from sklearn.preprocessing import StandardScaler
import mlflow
import mlflow.sklearn

def load_data(file_path: str) -> tuple:
    """Load dataset from the specified file path."""
    data = pd.read_csv(file_path)
    X = data.drop(columns=['label'])
    y = data['label']
    return X, y

def preprocess_data(data: tuple) -> tuple:
    """Perform preprocessing on the data."""
    X, y = data
    scaler = StandardScaler()
    X_scaled = scaler.fit_transform(X)
    return X_scaled, y

def check_concept_drift(new_data: tuple, reference_data: tuple) -> bool:
    """Check for concept drift between new data and reference data."""
    new_X, _ = new_data
    ref_X, _ = reference_data
    
    # This is a placeholder for concept drift detection logic.
    # In practice, you might use statistical tests or drift detection packages.
    drift_detected = np.random.rand() > 0.5  # Randomly decide drift for illustration
    return drift_detected

def train_model(X_train, y_train) -> RandomForestClassifier:
    """Train a RandomForestClassifier model."""
    model = RandomForestClassifier(n_estimators=100, random_state=42)
    model.fit(X_train, y_train)
    return model

def evaluate_model(model, X_test, y_test) -> dict:
    """Evaluate the model and return evaluation metrics."""
    y_pred = model.predict(X_test)
    accuracy = accuracy_score(y_test, y_pred)
    precision = precision_score(y_test, y_pred)
    recall = recall_score(y_test, y_pred)
    
    metrics = {
        'accuracy': accuracy,
        'precision': precision,
        'recall': recall
    }
    return metrics

def log_to_mlflow(model, metrics: dict):
    """Log the model and metrics to MLflow."""
    with mlflow.start_run():
        mlflow.sklearn.log_model(model, "model")
        mlflow.log_metrics(metrics)

def main(file_path: str):
    """Run the complete pipeline."""
    # Load and preprocess data
    X, y = load_data(file_path)
    X_scaled, y = preprocess_data((X, y))
    
    # Split data into training and test sets
    X_train, X_test, y_train, y_test = train_test_split(X_scaled, y, test_size=0.2, random_state=42)
    
    # Optional: Check for concept drift
    drift_detected = check_concept_drift((X_test, y_test), (X_train, y_train))
    if drift_detected:
        print("Concept drift detected. Consider retraining the model.")
    
    # Train and evaluate model
    model = train_model(X_train, y_train)
    metrics = evaluate_model(model, X_test, y_test)
    
    # Log model and metrics
    log_to_mlflow(model, metrics)

    print("Pipeline execution completed.")

if __name__ == "__main__":
    main("path/to/your/data.csv")
