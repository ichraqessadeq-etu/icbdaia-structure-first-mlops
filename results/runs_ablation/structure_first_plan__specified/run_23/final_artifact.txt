import pandas as pd

def load_data(filepath: str) -> tuple:
    """Load dataset from a CSV file."""
    data = pd.read_csv(filepath)
    X = data.drop('target', axis=1)
    y = data['target']
    return X, y

# fraud_detection/pipeline/model_trainer.py

from sklearn.ensemble import RandomForestClassifier
from sklearn.model_selection import train_test_split
import numpy as np

def train_model(X_train, y_train) -> object:
    """Train the RandomForestClassifier model."""
    model = RandomForestClassifier(n_estimators=100, random_state=42)
    model.fit(X_train, y_train)
    return model

def detect_concept_drift(new_data: tuple, reference_stats: dict) -> bool:
    """Detect concept drift based on statistical comparison."""
    X_new, _ = new_data
    new_stats = X_new.mean().to_dict()
    drift_detected = any(abs(new_stats[k] - reference_stats.get(k, 0)) > 0.1 for k in new_stats)
    return drift_detected

# fraud_detection/pipeline/model_evaluator.py

from sklearn.metrics import accuracy_score, precision_score, recall_score, f1_score

def evaluate_model(model, X_test, y_test) -> dict:
    """Evaluate the model and return performance metrics."""
    predictions = model.predict(X_test)
    metrics = {
        'accuracy': accuracy_score(y_test, predictions),
        'precision': precision_score(y_test, predictions),
        'recall': recall_score(y_test, predictions),
        'f1_score': f1_score(y_test, predictions)
    }
    return metrics

import mlflow

def log_to_mlflow(params: dict, metrics: dict, model):

    with mlflow.start_run():

        mlflow.log_metrics(metrics)
        mlflow.sklearn.log_model(model, "model")
