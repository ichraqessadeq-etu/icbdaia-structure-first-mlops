def detect_concept_drift(new_data: tuple, reference_stats: dict) -> bool:
    """Detect concept drift based on statistical comparison."""
    X_new, _ = new_data
    new_stats = X_new.mean().to_dict()
    drift_detected = any(abs(new_stats[k] - reference_stats.get(k, 0)) > 0.1 for k in new_stats)
    return drift_detected
