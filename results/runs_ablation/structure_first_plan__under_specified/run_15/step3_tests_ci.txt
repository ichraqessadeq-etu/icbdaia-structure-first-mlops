Below are the complete test files using `pytest` for the fraud detection pipeline and a GitHub Actions workflow configuration file. These implementations are designed to ensure that each component of the pipeline functions correctly and can be integrated into a CI/CD pipeline.

### Test Files

**`tests/test_data_preprocessing.py`**
```python
import pytest
import pandas as pd
from fraud_detection.pipeline.data_preprocessing import load_data, clean_data, split_data

def test_load_data():
    """Test loading data from a file."""
    test_file_path = "tests/test_data.csv"
    data = load_data(test_file_path)
    assert isinstance(data, pd.DataFrame), "Data should be loaded into a DataFrame"

def test_clean_data():
    """Test the data cleaning process."""
    data = pd.DataFrame({
        'A': [1, 2, None, 4],
        'B': [None, 2, 3, 4],
        'C': [1, 2, 2, 4]
    })
    cleaned_data = clean_data(data)
    assert cleaned_data.isnull().sum().sum() == 0, "Data should not have missing values"
    assert cleaned_data.duplicated().sum() == 0, "Data should not have duplicates"

def test_split_data():
    """Test splitting data into training and test sets."""
    data = pd.DataFrame({
        'feature1': range(100),
        'label': [0] * 50 + [1] * 50
    })
    train_data, test_data = split_data(data, test_size=0.2)
    assert len(train_data) == 80, "Training data should contain 80 samples"
    assert len(test_data) == 20, "Test data should contain 20 samples"
```

**`tests/test_model_training.py`**
```python
import pytest
from sklearn.ensemble import RandomForestClassifier
from fraud_detection.pipeline.model_training import train_model, save_model

def test_train_model():
    """Test training the model."""
    data = {
        'feature1': range(10),
        'label': [0, 1]*5
    }
    parameters = {'n_estimators': 10, 'random_state': 42}
    model = train_model(data, parameters)
    assert isinstance(model, RandomForestClassifier), "Model should be an instance of RandomForestClassifier"

def test_save_model(tmp_path):
    """Test saving the trained model."""
    model = RandomForestClassifier()
    file_path = tmp_path / "model.joblib"
    save_model(model, file_path)
    assert file_path.exists(), "Model file should be saved"
```

**`tests/test_model_evaluation.py`**
```python
import pytest
from fraud_detection.pipeline.model_evaluation import evaluate_model, log_metrics
from sklearn.ensemble import RandomForestClassifier

def test_evaluate_model():
    """Test evaluating the model."""
    model = RandomForestClassifier()
    data = {
        'feature1': range(10),
        'label': [0, 1]*5
    }
    metrics = evaluate_model(model, data)
    assert 'accuracy' in metrics, "Metrics should include accuracy"
    assert 'precision' in metrics, "Metrics should include precision"
    assert 'recall' in metrics, "Metrics should include recall"

def test_log_metrics(capfd):
    """Test logging of the evaluation metrics."""
    metrics = {"accuracy": 0.9, "precision": 0.8, "recall": 0.7}
    log_metrics(metrics)
    captured = capfd.readouterr()
    assert "accuracy: 0.9" in captured.out, "Output should log accuracy"
    assert "precision: 0.8" in captured.out, "Output should log precision"
    assert "recall: 0.7" in captured.out, "Output should log recall"
```

### GitHub Actions Workflow

**`.github/workflows/ci_cd_pipeline.yml`**
```yaml
name: CI/CD Pipeline

on:
  push:
    branches:
      - main
  pull_request:
    branches:
      - main

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout code
        uses: actions/checkout@v2

      - name: Set up Python
        uses: actions/setup-python@v2
        with:
          python-version: '3.x'

      - name: Install dependencies
        run: |
          python -m pip install --upgrade pip
          pip install -r requirements.txt
```
