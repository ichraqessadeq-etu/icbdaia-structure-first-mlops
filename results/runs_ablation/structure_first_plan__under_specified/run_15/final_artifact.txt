def deploy_model(model_path: str, endpoint_url: str) -> None:
    """Deploy the model to a specified endpoint."""
    import requests
    try:
        with open(model_path, 'rb') as model_file:
            response = requests.post(endpoint_url, files={'model': model_file})
        if response.status_code != 200:
            raise Exception(f"Failed to deploy model, status code: {response.status_code}")
    except Exception as e:
        raise Exception(f"Error deploying model: {e}")

def check_endpoint_status(endpoint_url: str) -> bool:
    """Check the status of the deployed model endpoint."""
    import requests
    try:
        response = requests.get(endpoint_url)
        return response.status_code == 200
    except Exception as e:
        raise Exception(f"Error checking endpoint status: {e}")
