from sklearn.ensemble import RandomForestClassifier
from fraud_detection.pipeline.model_training import split_data, train_model, save_model
import joblib
import os

def test_split_data():
    """Test the data splitting function."""
    df = pd.DataFrame({'feature1': [1, 2, 3, 4], 'fraud': [0, 1, 0, 1]})
    X_train, X_test, y_train, y_test = split_data(df, test_size=0.5)

    assert len(X_test) == 2
    assert len(y_train) == 2
    assert len(y_test) == 2

    """Test the model training function."""
    X_train = pd.DataFrame({'feature1': [1, 2], 'feature2': [3, 4]})
    y_train = pd.Series([0, 1])
    model = train_model(X_train, y_train)
    assert isinstance(model, RandomForestClassifier)
    assert hasattr(model, 'predict')

    """Test the model saving function."""
    model = RandomForestClassifier()
    file_path = 'test_model.joblib'
    save_model(model, file_path)
