from typing import Tuple, Any
from sklearn.model_selection import train_test_split
from sklearn.ensemble import RandomForestClassifier
import joblib

def split_data(df: pd.DataFrame, test_size: float) -> Tuple[pd.DataFrame, pd.DataFrame]:
    """Split the data into training and testing sets."""
    X = df.drop('fraud', axis=1)
    y = df['fraud']
    return train_test_split(X, y, test_size=test_size, random_state=42)

def train_model(X_train: pd.DataFrame, y_train: pd.Series) -> Any:
    """Train the model using the training data."""
    model = RandomForestClassifier(n_estimators=100, random_state=42)
    model.fit(X_train, y_train)
    return model

def save_model(model: Any, file_path: str) -> None:
    """Save the trained model to a specified file path."""
    joblib.dump(model, file_path)


# fraud_detection/pipeline/model_evaluation.py

from typing import Dict
from sklearn.metrics import accuracy_score, precision_score, recall_score, f1_score
import matplotlib.pyplot as plt

def evaluate_model(model: Any, X_test: pd.DataFrame, y_test: pd.Series) -> Dict[str, float]:
    """Evaluate the model's performance on the test data."""
    y_pred = model.predict(X_test)
    metrics = {
        'accuracy': accuracy_score(y_test, y_pred),
        'precision': precision_score(y_test, y_pred),
        'recall': recall_score(y_test, y_pred),
        'f1': f1_score(y_test, y_pred)
    }
    return metrics

def plot_evaluation_metrics(metrics: Dict[str, float]) -> None:
    """Plot evaluation metrics for the model."""
    plt.bar(metrics.keys(), metrics.values())
    plt.title('Model Evaluation Metrics')
    plt.xlabel('Metric')
    plt.ylabel('Score')
    plt.ylim(0, 1)
    plt.show()
