```python
# fraud_detection/pipeline/main.py

import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.ensemble import RandomForestClassifier
from sklearn.metrics import accuracy_score, precision_score, recall_score, f1_score
import joblib
from typing import Any, Dict

def load_data(data_path: str) -> pd.DataFrame:
    """Load data from the specified path."""
    try:
        df = pd.read_csv(data_path)
        return df
    except Exception as e:
        raise IOError(f"Error loading data from {data_path}: {e}")

def preprocess_data(df: pd.DataFrame) -> pd.DataFrame:
    """Preprocess the raw data for model training."""
    # Example preprocessing steps: fill missing values, encode categorical variables, etc.
    df.fillna(0, inplace=True)
    # Assuming 'label' is the target column and the rest are features
    features = df.drop(columns=['label'])
    labels = df['label']
    return features, labels

def train_model(df: pd.DataFrame) -> Any:
    """Train the fraud detection model."""
    features, labels = preprocess_data(df)
    X_train, X_test, y_train, y_test = train_test_split(features, labels, test_size=0.2, random_state=42)
    model = RandomForestClassifier(n_estimators=100, random_state=42)
    model.fit(X_train, y_train)
    return model, X_test, y_test

def evaluate_model(model: Any, X_test: pd.DataFrame, y_test: pd.Series) -> Dict[str, float]:
    """Evaluate the model's performance."""
    y_pred = model.predict(X_test)
    metrics = {
        'accuracy': accuracy_score(y_test, y_pred),
        'precision': precision_score(y_test, y_pred),
        'recall': recall_score(y_test, y_pred),
        'f1_score': f1_score(y_test, y_pred)
    }
    return metrics

def save_model(model: Any, file_path: str) -> None:
    """Save the trained model to the specified file path."""
    try:
        joblib.dump(model, file_path)
        print(f"Model saved to {file_path}")
    except Exception as e:
        raise IOError(f"Error saving model to {file_path}: {e}")

def main(data_path: str, model_path: str):
    df = load_data(data_path)
    model, X_test, y_test = train_model(df)
    metrics = evaluate_model(model, X_test, y_test)
    print("Model evaluation metrics:", metrics)
    save_model(model, model_path)

if __name__ == "__main__":
    # Example usage
    main("data/fraud_data.csv", "models/fraud_detection_model.pkl")
```

This implementation covers the entire pipeline from loading data, preprocessing, training, evaluating, and saving the model. You can adjust the specifics of preprocessing and model training to better suit your dataset and requirements.
