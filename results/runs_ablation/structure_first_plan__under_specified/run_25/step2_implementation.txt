# src/data_preprocessing.py

import pandas as pd
from sklearn.model_selection import train_test_split

def load_data(file_path: str) -> pd.DataFrame:
    """Load data from a given file path."""
    return pd.read_csv(file_path)

def clean_data(raw_data: pd.DataFrame) -> pd.DataFrame:
    """Clean and preprocess raw data."""
    # Example preprocessing steps
    processed_data = raw_data.dropna()  # Drop missing values
    processed_data = pd.get_dummies(processed_data)  # One-hot encode categorical variables
    return processed_data

def split_data(processed_data: pd.DataFrame, test_size: float = 0.2) -> tuple:
    """Split data into training and testing sets."""
    X = processed_data.drop('target', axis=1)
    y = processed_data['target']
    X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=test_size, random_state=42)
    return X_train, X_test, y_train, y_test


# src/model_training.py

from sklearn.ensemble import RandomForestClassifier
import joblib

def train_model(training_data: tuple, parameters: dict = None) -> RandomForestClassifier:
    """Train a model using the training data and parameters."""
    X_train, y_train = training_data
    model = RandomForestClassifier(**parameters) if parameters else RandomForestClassifier()
    model.fit(X_train, y_train)
    return model

def save_model(model: RandomForestClassifier, file_path: str) -> None:
    """Save the trained model to a file."""
    joblib.dump(model, file_path)


# src/model_evaluation.py

from sklearn.metrics import classification_report

def evaluate_model(model: RandomForestClassifier, test_data: tuple) -> dict:
    """Evaluate the model on test data and return metrics."""
    X_test, y_test = test_data
    predictions = model.predict(X_test)
    metrics = classification_report(y_test, predictions, output_dict=True)
    return metrics
