import pandas as pd
from sklearn.model_selection import train_test_split

def load_data(file_path: str) -> pd.DataFrame:
    """Load the dataset from a given file path."""
    return pd.read_csv(file_path)

def clean_data(data: pd.DataFrame) -> pd.DataFrame:
    """Clean the dataset by handling missing values and outliers."""
    # Example: Fill missing values with median
    data.fillna(data.median(), inplace=True)
    return data

def feature_engineering(data: pd.DataFrame) -> pd.DataFrame:
    """Perform feature engineering on the dataset."""
    # Example: Create new features or transform existing ones
    data['new_feature'] = data['existing_feature'] ** 2
    return data

def split_data(data: pd.DataFrame, test_size: float) -> tuple:
    """Split the data into training and testing sets."""
    X = data.drop('target', axis=1)
    y = data['target']
    return train_test_split(X, y, test_size=test_size, random_state=42)


# fraud_detection/pipeline/model_training.py

from sklearn.ensemble import RandomForestClassifier
import joblib

def train_model(X_train: pd.DataFrame, y_train: pd.Series) -> RandomForestClassifier:
    """Train the machine learning model using training data."""
    model = RandomForestClassifier(random_state=42)
    model.fit(X_train, y_train)
    return model

def save_model(model: RandomForestClassifier, file_path: str) -> None:
    """Save the trained model to a file."""
    joblib.dump(model, file_path)


# fraud_detection/pipeline/model_evaluation.py

import joblib
from sklearn.metrics import accuracy_score, precision_score, recall_score, f1_score

def load_model(file_path: str) -> RandomForestClassifier:
    """Load the trained model from a file."""
    return joblib.load(file_path)

def evaluate_model(model: RandomForestClassifier, X_test: pd.DataFrame, y_test: pd.Series) -> dict:
    """Evaluate the model's performance on the test set."""
    predictions = model.predict(X_test)
    return {
        'accuracy': accuracy_score(y_test, predictions),
        'precision': precision_score(y_test, predictions),
        'recall': recall_score(y_test, predictions),
        'f1_score': f1_score(y_test, predictions)
    }
