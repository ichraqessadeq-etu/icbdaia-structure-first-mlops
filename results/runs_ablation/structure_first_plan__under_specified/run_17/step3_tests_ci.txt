import pytest
import pandas as pd
from fraud_detection.pipeline.data_preprocessing import load_data, clean_data, feature_engineering, split_data
from fraud_detection.pipeline.model_training import train_model, save_model
from fraud_detection.pipeline.model_evaluation import load_model, evaluate_model
from fraud_detection.pipeline.model_deployment import deploy_model
from sklearn.ensemble import RandomForestClassifier
import joblib
import os

# Mock data for testing
@pytest.fixture
def mock_data():
    return pd.DataFrame({
        'existing_feature': [1, 2, 3, 4, 5],
        'target': [0, 1, 0, 1, 0]
    })

def test_load_data(mock_data, tmpdir):
    """Test loading of the dataset."""
    file_path = tmpdir.join("test_data.csv")
    mock_data.to_csv(file_path, index=False)
    loaded_data = load_data(file_path)
    pd.testing.assert_frame_equal(loaded_data, mock_data)

def test_clean_data(mock_data):
    """Test cleaning of the dataset."""
    mock_data.loc[0, 'existing_feature'] = None  # Introduce missing value
    cleaned_data = clean_data(mock_data)
    assert cleaned_data.isnull().sum().sum() == 0  # All missing values should be filled

def test_feature_engineering(mock_data):
    """Test feature engineering on the dataset."""
    engineered_data = feature_engineering(mock_data)
    assert 'new_feature' in engineered_data.columns
    assert (engineered_data['new_feature'] == mock_data['existing_feature'] ** 2).all()

def test_split_data(mock_data):
    """Test splitting of data into training and testing sets."""
    X_train, X_test, y_train, y_test = split_data(mock_data, test_size=0.2)
    assert len(X_train) + len(X_test) == len(mock_data)
    assert len(y_train) + len(y_test) == len(mock_data)

def test_train_model(mock_data):
    """Test training of the machine learning model."""
    X_train, _, y_train, _ = split_data(mock_data, test_size=0.2)
    model = train_model(X_train, y_train)
    assert isinstance(model, RandomForestClassifier)

def test_save_model(mock_data, tmpdir):
    """Test saving of the trained model to a file."""
    X_train, _, y_train, _ = split_data(mock_data, test_size=0.2)
    model = train_model(X_train, y_train)
    file_path = tmpdir.join("model.joblib")
    save_model(model, file_path)

def test_load_model(mock_data, tmpdir):
    """Test loading of the trained model from a file."""

    save_model(model, file_path)

def test_deploy_model(mock_data, tmpdir):
