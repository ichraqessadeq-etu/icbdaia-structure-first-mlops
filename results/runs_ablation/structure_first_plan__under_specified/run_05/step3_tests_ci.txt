import pytest
import pandas as pd
from sklearn.ensemble import RandomForestClassifier
from fraud_detection.pipeline.model_evaluation import load_model, evaluate_model, generate_report
import joblib

@pytest.fixture
def sample_data():
    data = {
        'feature1': [1, 2, 3, 4],
        'feature2': [5, 6, 7, 8],
        'is_fraud': [0, 1, 0, 1]
    }
    return pd.DataFrame(data)

@pytest.fixture
def trained_model(tmpdir, sample_data):
    X_train, _, y_train, _ = split_data(sample_data, test_size=0.25)
    model = train_model(X_train, y_train)
    file_path = tmpdir.join("model.joblib")
    joblib.dump(model, str(file_path))
    return str(file_path)

def test_load_model(trained_model):
    model = load_model(trained_model)
    assert isinstance(model, RandomForestClassifier)

def test_evaluate_model(sample_data, trained_model):
    model = load_model(trained_model)
    _, X_test, _, y_test = split_data(sample_data, test_size=0.25)
    accuracy = evaluate_model(model, X_test, y_test)
    assert 0 <= accuracy <= 1

def test_generate_report(capsys):
    generate_report(0.95)
    captured = capsys.readouterr()
    assert "Model Accuracy: 95.00%" in captured.out

### GitHub Actions Workflow

#### `.github/workflows/ci_cd_pipeline.yml`
```yaml
name: CI/CD Pipeline

on:
  push:
    branches:
      - main
  pull_request:
    branches:
      - main

jobs:
  setup:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout code
        uses: actions/checkout@v3

      - name: Set up Python
        uses: actions/setup-python@v4
        with:
          python-version: '3.x'

      - name: Install dependencies
        run: pip install -r requirements.txt

  test:
    runs-on: ubuntu-latest
    needs: setup
    steps:
      - name: Checkout code
        uses: actions/checkout@v3

      - name: Set up Python
        uses: actions/setup-python@v4
        with:
          python-version: '3.x'

      - name: Install dependencies
        run: pip install -r requirements.txt
        
      - name: Run Pytest
        run: pytest tests/

  build:
    runs-on: ubuntu-latest
    needs: test
    steps:
      - name: Checkout code
        uses: actions/checkout@v3

      - name: Set up Python
        uses: actions/setup-python@v4
        with:
          python-version: '3.x'

      - name: Install dependencies
        run: pip install -r requirements.txt

      - name: Build model
        run: python -m fraud_detection.pipeline.model_training

  deploy:
    runs-on: ubuntu-latest
    needs: build
    steps:
      - name: Deploy model
        run: echo "Deploying the model..."
```
