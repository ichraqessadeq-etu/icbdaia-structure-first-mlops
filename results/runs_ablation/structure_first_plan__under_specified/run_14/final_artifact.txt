```python
import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.ensemble import RandomForestClassifier
from sklearn.metrics import accuracy_score
import joblib


def load_data(source: str) -> pd.DataFrame:
    """Load data from the specified source."""
    try:
        data = pd.read_csv(source)
        print(f"Data loaded successfully from {source}")
        return data
    except Exception as e:
        print(f"Error loading data from {source}: {e}")
        raise


def preprocess_data(data: pd.DataFrame) -> pd.DataFrame:
    """Preprocess the data for model training."""
    try:
        # Dummy pre-processing: dropping NA and splitting features/labels
        data = data.dropna()
        features = data.drop(columns='label')
        labels = data['label']
        print("Data preprocessed successfully")
        return features, labels
    except Exception as e:
        print(f"Error during data preprocessing: {e}")
        raise


def train_model(preprocessed_data: tuple) -> RandomForestClassifier:
    """Train the fraud detection model."""
    try:
        features, labels = preprocessed_data
        X_train, X_test, y_train, y_test = train_test_split(features, labels, test_size=0.2, random_state=42)
        
        model = RandomForestClassifier(n_estimators=100, random_state=42)
        model.fit(X_train, y_train)
        
        print("Model training completed successfully")
        return model
    except Exception as e:
        print(f"Error during model training: {e}")
        raise


def evaluate_model(model: RandomForestClassifier, test_data: tuple) -> float:
    """Evaluate the model performance."""
    try:
        X_test, y_test = test_data
        predictions = model.predict(X_test)
        accuracy = accuracy_score(y_test, predictions)
        
        print(f"Model evaluation completed with accuracy: {accuracy}")
        return accuracy
    except Exception as e:
        print(f"Error during model evaluation: {e}")
        raise


def save_model(model: RandomForestClassifier, destination: str) -> None:
    """Save the trained model to the specified destination."""
    try:
        joblib.dump(model, destination)
        print(f"Model saved successfully to {destination}")
    except Exception as e:
        print(f"Error saving model to {destination}: {e}")
        raise
```
