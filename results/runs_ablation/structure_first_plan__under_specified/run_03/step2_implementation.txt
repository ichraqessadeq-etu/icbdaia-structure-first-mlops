from sklearn.model_selection import train_test_split
from sklearn.ensemble import RandomForestClassifier
import joblib

def split_data(data: pd.DataFrame, test_size: float) -> tuple:
    """Split the data into training and testing sets."""
    try:
        X = data.drop('target', axis=1)
        y = data['target']
        X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=test_size, random_state=42)
        return X_train, X_test, y_train, y_test
    except Exception as e:
        raise ValueError(f"Failed to split data: {e}")

def train_model(train_data: tuple) -> RandomForestClassifier:
    """Train the fraud detection model."""
    try:
        X_train, _, y_train, _ = train_data
        model = RandomForestClassifier(n_estimators=100, random_state=42)
        model.fit(X_train, y_train)
        return model
    except Exception as e:
        raise ValueError(f"Failed to train model: {e}")

def save_model(model: RandomForestClassifier, model_path: str) -> None:
    """Save the trained model to the specified path."""
    try:
        joblib.dump(model, model_path)
    except Exception as e:
        raise ValueError(f"Failed to save model: {e}")

from sklearn.metrics import classification_report
import joblib

def load_model(model_path: str) -> RandomForestClassifier:
    """Load a trained model from the specified path."""
    try:
        model = joblib.load(model_path)
        return model

def evaluate_model(model: RandomForestClassifier, test_data: tuple) -> dict:
    """Evaluate the model on the test data and return metrics."""
    try:
        _, X_test, _, y_test = test_data
        predictions = model.predict(X_test)
