# fraud_detection/pipeline/model_training.py

from typing import Tuple, Any
import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.linear_model import LogisticRegression
import joblib

def split_data(data: pd.DataFrame, test_size: float) -> Tuple[pd.DataFrame, pd.DataFrame, pd.Series, pd.Series]:
    """Split the data into training and test datasets."""
    X = data.drop('target', axis=1)
    y = data['target']
    return train_test_split(X, y, test_size=test_size, random_state=42)

def train_model(X_train: pd.DataFrame, y_train: pd.Series) -> Any:
    """Train the model using training data."""
    model = LogisticRegression()
    model.fit(X_train, y_train)
    return model

def save_model(model: Any, file_path: str):
    """Save the trained model to a file."""
    joblib.dump(model, file_path)


# fraud_detection/pipeline/model_evaluation.py

from typing import Dict
import pandas as pd
from sklearn.metrics import accuracy_score, precision_score, recall_score
import joblib

def load_model(file_path: str) -> Any:
    """Load a model from a file."""
    return joblib.load(file_path)

def evaluate_model(model: Any, X_test: pd.DataFrame, y_test: pd.Series) -> Dict[str, float]:
    """Evaluate the model on test data."""
    y_pred = model.predict(X_test)
    return {
        'accuracy': accuracy_score(y_test, y_pred),
        'precision': precision_score(y_test, y_pred),
        'recall': recall_score(y_test, y_pred)
    }

import requests

def deploy_model(model: Any, endpoint_url: str):
    """Deploy the model to a specified endpoint."""

    model_data = joblib.dump(model, 'model.pkl')
