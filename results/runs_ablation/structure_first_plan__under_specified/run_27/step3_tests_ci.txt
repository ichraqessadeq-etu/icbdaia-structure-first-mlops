```python
import pytest
import pandas as pd
from sklearn.linear_model import LogisticRegression
from pipeline.data_preprocessing import load_data, preprocess_data
from pipeline.model_training import split_data, train_model
from pipeline.model_evaluation import evaluate_model
from pipeline.model_deployment import deploy_model
import os

# Sample data for testing
TEST_DATA_PATH = 'test_data.csv'
TARGET_COLUMN = 'target'

@pytest.fixture
def sample_data():
    # Create a small DataFrame for testing
    data = pd.DataFrame({
        'feature1': [1, 2, 3, 4, 5],
        'feature2': [5, 4, 3, 2, 1],
        'target': [0, 1, 0, 1, 0]
    })
    data.to_csv(TEST_DATA_PATH, index=False)
    yield data
    os.remove(TEST_DATA_PATH)

def test_load_data(sample_data):
    """Test loading data from a file."""
    data = load_data(TEST_DATA_PATH)
    pd.testing.assert_frame_equal(data, sample_data)

def test_preprocess_data(sample_data):
    """Test the data preprocessing step."""
    processed_data = preprocess_data(sample_data.drop(columns=[TARGET_COLUMN]))
    assert processed_data.shape == (5, 2)  # Check shape
    assert processed_data.isnull().sum().sum() == 0  # Check no missing values

def test_split_data(sample_data):
    """Test splitting the data into training and testing sets."""
    X_train, X_test, y_train, y_test = split_data(sample_data, TARGET_COLUMN)
    assert len(X_train) + len(X_test) == len(sample_data)
    assert len(y_train) + len(y_test) == len(sample_data)

def test_train_model(sample_data):
    """Test the model training process."""
    X_train, X_test, y_train, y_test = split_data(sample_data, TARGET_COLUMN)
    model = train_model(X_train, y_train)
    assert isinstance(model, LogisticRegression)

def test_evaluate_model(sample_data):
    """Test the model evaluation metrics."""
    X_train, X_test, y_train, y_test = split_data(sample_data, TARGET_COLUMN)
    model = train_model(X_train, y_train)
    metrics = evaluate_model(model, X_test, y_test)
    assert 'accuracy' in metrics and 'precision' in metrics
    assert 'recall' in metrics and 'f1_score' in metrics

def test_deploy_model(sample_data):
    """Test deploying the model to production."""
    X_train, X_test, y_train, y_test = split_data(sample_data, TARGET_COLUMN)
    model = train_model(X_train, y_train)
    deploy_model(model, 'test_deployed_model.pkl')
    assert os.path.isfile('test_deployed_model.pkl')
    os.remove('test_deployed_model.pkl')
```

### CI Workflow

#### `ci_cd_pipeline.yml`

```yaml
name: CI/CD Pipeline

on:
  push:
    branches:
      - main
  pull_request:
    branches:
      - main

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout code
        uses: actions/checkout@v2

      - name: Set up Python
        uses: actions/setup-python@v2
        with:
          python-version: '3.x'

      - name: Install dependencies
        run: pip install -r requirements.txt

      - name: Run tests
        run: pytest tests/

  deploy:
    runs-on: ubuntu-latest
    needs: build
    if: github.ref == 'refs/heads/main' && success()
    steps:
      - name: Checkout code
        uses: actions/checkout@v2

      - name: Set up Python
        uses: actions/setup-python@v2
        with:
          python-version: '3.x'

      - name: Install dependencies
        run: pip install -r requirements.txt

      - name: Deploy model
        run: python -m pipeline.model_deployment
```

This setup ensures that your pipeline is thoroughly tested with `pytest`, and the CI/CD workflow is automated using GitHub Actions to build, test, and deploy your fraud detection model.
