import pandas as pd

from fraud_detection.pipeline.data_preprocessing import load_data, clean_data, split_data

def test_clean_data():
    """Test the data cleaning process."""
    raw_data = pd.DataFrame({
        'A': [1, 2, None, 4],
        'B': [1, 2, 3, 2],
        'C': [5, 6, 7, 8]
    })
    expected_clean_data = pd.DataFrame({
        'A': [1, 2, 4],
        'B': [1, 2, 2],
        'C': [5, 6, 8]
    })
    cleaned_data = clean_data(raw_data)
    assert_frame_equal(cleaned_data, expected_clean_data), "Cleaned data does not match expected output."

def test_split_data():
    """Test the data splitting function."""
    data = pd.DataFrame({'feature': range(10), 'label': [1, 0] * 5})
    train_data, test_data = split_data(data, test_size=0.2)
    assert len(train_data) == 8, "Train data size should be 80%."

        'label': [0, 1]
    })
