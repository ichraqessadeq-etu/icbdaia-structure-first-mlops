from sklearn.ensemble import RandomForestClassifier
import joblib

def train_model(train_data: pd.DataFrame, algorithm: str) -> object:
    """Train the model using the specified algorithm."""
    try:
        X_train = train_data.drop('target', axis=1)
        y_train = train_data['target']
        
        if algorithm == 'random_forest':
            model = RandomForestClassifier()
        else:
            raise ValueError(f"Algorithm {algorithm} is not supported.")
        
        model.fit(X_train, y_train)
        return model
    except Exception as e:
        raise RuntimeError(f"Failed to train model: {e}")

def save_model(model: object, file_path: str) -> None:
    """Save the trained model to a file."""
    try:
        joblib.dump(model, file_path)
    except Exception as e:
        raise RuntimeError(f"Failed to save model: {e}")

from sklearn.metrics import accuracy_score, classification_report

def evaluate_model(model: object, test_data: pd.DataFrame) -> dict:
    """Evaluate the model's performance on test data."""
    try:
        X_test = test_data.drop('target', axis=1)
        y_test = test_data['target']
        predictions = model.predict(X_test)
        
        accuracy = accuracy_score(y_test, predictions)
        report = classification_report(y_test, predictions)
        
        return {'accuracy': accuracy, 'report': report}
    except Exception as e:
        raise RuntimeError(f"Failed to evaluate model: {e}")
