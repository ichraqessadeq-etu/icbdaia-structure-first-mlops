```python
import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.ensemble import RandomForestClassifier
from sklearn.metrics import accuracy_score, precision_score, recall_score
import joblib

def load_data(file_path: str) -> pd.DataFrame:
    """Load data from the specified file path."""
    data = pd.read_csv(file_path)
    return data

def preprocess_data(data: pd.DataFrame) -> pd.DataFrame:
    """Preprocess the data for model training."""
    # Assuming that the last column is the target variable
    data = data.dropna()  # Example preprocessing step: remove missing values
    return data

def split_data(data: pd.DataFrame, test_size: float) -> tuple:
    """Split the data into training and test sets."""
    X = data.iloc[:, :-1]  # Features
    y = data.iloc[:, -1]   # Target
    X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=test_size, random_state=42)
    return X_train, X_test, y_train, y_test

def train_model(train_data: tuple) -> RandomForestClassifier:
    """Train a fraud detection model using the training data."""
    X_train, y_train = train_data
    model = RandomForestClassifier(n_estimators=100, random_state=42)
    model.fit(X_train, y_train)
    return model

def evaluate_model(model: RandomForestClassifier, test_data: tuple) -> dict:
    """Evaluate the model using the test data."""
    X_test, y_test = test_data
    y_pred = model.predict(X_test)
    evaluation_metrics = {
        'accuracy': accuracy_score(y_test, y_pred),
        'precision': precision_score(y_test, y_pred, average='binary'),
        'recall': recall_score(y_test, y_pred, average='binary')
    }
    return evaluation_metrics

def save_model(model: RandomForestClassifier, file_path: str) -> None:
    """Save the trained model to the specified file path."""
    joblib.dump(model, file_path)
```

### Explanation:

- **load_data**: This function loads data from a CSV file into a pandas DataFrame.
- **preprocess_data**: A simple preprocessing step is implemented here, which removes missing values.
- **split_data**: This function splits the DataFrame into feature (`X`) and target (`y`) variables and then into training and testing sets.
- **train_model**: Trains a RandomForestClassifier using the training data.
- **evaluate_model**: Evaluates the model's performance using accuracy, precision, and recall metrics.
- **save_model**: Saves the trained model to a specified file path using `joblib`.
