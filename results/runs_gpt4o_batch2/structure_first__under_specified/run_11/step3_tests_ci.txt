# test_fraud_detection_pipeline.py

import pytest
import pandas as pd
from fraud_detection_pipeline.pipeline.data_processing import load_data, preprocess_data, split_data

from fraud_detection_pipeline.pipeline.model_evaluation import evaluate_model
from sklearn.datasets import make_classification
from sklearn.ensemble import RandomForestClassifier

def test_load_data(tmp_path):
    # Create a temporary CSV file
    data = pd.DataFrame({'A': [1, 2, 3], 'B': [4, 5, 6]})
    file_path = tmp_path / "test_data.csv"
    data.to_csv(file_path, index=False)

    # Test loading the data
    loaded_data = load_data(file_path)
    pd.testing.assert_frame_equal(data, loaded_data)

def test_preprocess_data():
    # Sample data with missing values and categorical variables
    data = pd.DataFrame({
        'A': [1, None, 3],
        'B': ['x', 'y', 'z']

    preprocessed_data = preprocess_data(data)

    # Check if missing values are filled and categorical variables are encoded
    assert preprocessed_data.isnull().sum().sum() == 0
    assert 'B_x' in preprocessed_data.columns
    assert 'B_y' in preprocessed_data.columns
    assert 'B_z' in preprocessed_data.columns

def test_split_data():
    # Create sample data
    data = pd.DataFrame({'A': range(10), 'B': range(10, 20)})
    train_data, test_data = split_data(data, test_size=0.2)

def test_train_model():
    # Generate a classification dataset
    X, y = make_classification(n_samples=100, n_features=4, random_state=42)
    X_train, y_train = pd.DataFrame(X), pd.Series(y)

    model = train_model(X_train, y_train)

    # Check if the model is a RandomForestClassifier

    X_train, X_test, y_train, y_test = split_data(pd.DataFrame(X), test_size=0.2)

### (b) GitHub Actions Workflow

Below is a GitHub Actions workflow file that automates the testing process using pytest. This workflow is triggered on every push and pull request to the repository, ensuring that the code is continuously tested.

```yaml
# .github/workflows/test_pipeline.yml

name: Test Fraud Detection Pipeline

on:
  push:
    branches: [main]
  pull_request:
    branches: [main]

jobs:
  build:
    runs-on: ubuntu-latest

    steps:
    - name: Checkout code
      uses: actions/checkout@v2

    - name: Set up Python
      uses: actions/setup-python@v2
      with:
        python-version: '3.8'

    - name: Install dependencies
      run: |
        python -m pip install --upgrade pip
        pip install pytest pandas scikit-learn

    - name: Run tests
      run: |
        pytest --maxfail=1 --disable-warnings
```

This setup ensures that all aspects of the pipeline from data loading to model evaluation are tested, and any issues are quickly identified through CI/CD integration with GitHub Actions.
