### GitHub Actions Workflow (`ci_cd_pipeline.yml`)
```yaml
name: CI/CD Pipeline

on:
  push:
    branches:
      - main
  pull_request:
    branches:
      - main

jobs:
  setup:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout code
        uses: actions/checkout@v2
      - name: Set up Python
        uses: actions/setup-python@v2
        with:
          python-version: '3.x'
      - name: Install dependencies
        run: pip install -r requirements.txt

  test:
    runs-on: ubuntu-latest
    needs: setup
    steps:
      - name: Run tests
        run: pytest tests/

  build:
    runs-on: ubuntu-latest
    needs: test
    steps:
      - name: Build package
        run: python setup.py sdist bdist_wheel

  deploy:
    runs-on: ubuntu-latest
    needs: build
    steps:
      - name: Deploy model
        run: python -m pipeline.model_deployment
```
