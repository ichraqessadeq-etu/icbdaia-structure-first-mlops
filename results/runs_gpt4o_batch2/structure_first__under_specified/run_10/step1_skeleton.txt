### GitHub Actions Workflow: `ci_cd_pipeline.yml`

```yaml
# fraud_detection_pipeline/.github/workflows/ci_cd_pipeline.yml

name: CI/CD Pipeline

on:
  push:
    branches:
      - main
  pull_request:
    branches:
      - main

jobs:
  setup-environment:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout code
        uses: actions/checkout@v2

      - name: Set up Python
        uses: actions/setup-python@v2
        with:
          python-version: '3.x'

  install-dependencies:
    runs-on: ubuntu-latest
    needs: setup-environment
    steps:
      - name: Install dependencies
        run: pip install -r requirements.txt

  run-tests:
    runs-on: ubuntu-latest
    needs: install-dependencies
    steps:
      - name: Run tests
        run: pytest tests

  deploy:
    runs-on: ubuntu-latest
    needs: run-tests
    steps:
      - name: Deploy Model
        run: echo "Deployment step would go here"
```
