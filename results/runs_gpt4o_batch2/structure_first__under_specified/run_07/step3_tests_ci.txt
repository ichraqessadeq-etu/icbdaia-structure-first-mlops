# tests/test_pipeline.py

import pytest
import pandas as pd
from fraud_detection.pipeline.data_preprocessing import load_data, clean_data, feature_engineering
from fraud_detection.pipeline.model_training import split_data, train_model
from fraud_detection.pipeline.model_evaluation import evaluate_model
from fraud_detection.pipeline.model_deployment import save_model, load_model
import os

@pytest.fixture
def sample_data():
    data = {
        'amount': [100, 200, 300, None, 500],
        'transaction_count': [1, 2, 3, 4, None],
        'target': [0, 1, 0, 1, 0]
    }
    return pd.DataFrame(data)

def test_load_data(tmp_path):
    # Create a temporary CSV file
    file_path = tmp_path / "data.csv"
    sample_data().to_csv(file_path, index=False)
    
    # Test loading data
    data = load_data(str(file_path))
    assert not data.empty

def test_clean_data(sample_data):
    # Test cleaning data
    cleaned_data = clean_data(sample_data)
    assert cleaned_data.isna().sum().sum() == 0
    assert cleaned_data.duplicated().sum() == 0

def test_feature_engineering(sample_data):
    # Test feature engineering
    engineered_data = feature_engineering(sample_data)
    assert 'new_feature' in engineered_data.columns
    assert not engineered_data['new_feature'].isna().any()

def test_split_data(sample_data):
    # Test splitting data
    cleaned_data = clean_data(sample_data)
    X_train, X_test, y_train, y_test = split_data(cleaned_data, test_size=0.2)
    assert len(X_train) == 3
    assert len(X_test) == 1
    assert len(y_train) == 3
    assert len(y_test) == 1

def test_train_model(sample_data):
    # Test model training
    cleaned_data = clean_data(sample_data)
    X_train, X_test, y_train, y_test = split_data(cleaned_data, test_size=0.2)
    model = train_model(X_train, y_train)
    assert model is not None

def test_evaluate_model(sample_data):
    # Test model evaluation
    cleaned_data = clean_data(sample_data)
    X_train, X_test, y_train, y_test = split_data(cleaned_data, test_size=0.2)
    model = train_model(X_train, y_train)
    evaluation = evaluate_model(model, X_test, y_test)
    assert 'accuracy' in evaluation
    assert 'precision' in evaluation
    assert 'recall' in evaluation

def test_save_and_load_model(sample_data, tmp_path):
    # Test saving and loading model
    cleaned_data = clean_data(sample_data)
    X_train, X_test, y_train, y_test = split_data(cleaned_data, test_size=0.2)
    model = train_model(X_train, y_train)
    
    model_path = tmp_path / "model.joblib"
    save_model(model, str(model_path))
    
    loaded_model = load_model(str(model_path))
    assert loaded_model is not None

### (b) GitHub Actions Workflow

```yaml
# .github/workflows/python-package.yml

name: Python package

on:
  push:
    branches: [ "main" ]
  pull_request:
    branches: [ "main" ]

jobs:
  build:

    runs-on: ubuntu-latest

    strategy:
      matrix:
        python-version: [3.8, 3.9, 3.10]

    steps:
    - uses: actions/checkout@v2
    - name: Set up Python ${{ matrix.python-version }}
      uses: actions/setup-python@v2
      with:
        python-version: ${{ matrix.python-version }}
    - name: Install dependencies
      run: |
        python -m pip install --upgrade pip
        pip install pytest pandas scikit-learn joblib
    - name: Test with pytest
      run: |
        pytest tests
```

This setup ensures that the pipeline module is thoroughly tested using pytest, and the GitHub Actions workflow will automatically run these tests whenever there is a push or pull request to the main branch. The workflow tests the code across multiple Python versions for better compatibility.
