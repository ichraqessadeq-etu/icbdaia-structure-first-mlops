```python
# test_pipeline.py

import pytest
import pandas as pd
from fraud_detection.pipeline.data_preprocessing import load_data, clean_data, split_data
from fraud_detection.pipeline.feature_engineering import generate_features, select_features
from fraud_detection.pipeline.model_training import train_model, save_model
from fraud_detection.pipeline.model_evaluation import evaluate_model, generate_report
from sklearn.ensemble import RandomForestClassifier
from sklearn.datasets import make_classification
import joblib
import os

@pytest.fixture
def sample_data():
    X, y = make_classification(n_samples=100, n_features=15, n_informative=10, random_state=42)
    data = pd.DataFrame(X, columns=[f'feature_{i}' for i in range(15)])
    data['label'] = y
    return data

def test_load_data(sample_data, tmpdir):
    file_path = tmpdir.join("sample_data.csv")
    sample_data.to_csv(file_path, index=False)
    loaded_data = load_data(str(file_path))
    pd.testing.assert_frame_equal(loaded_data, sample_data)
    
def test_clean_data(sample_data):
    sample_data.iloc[0, 0] = None
    cleaned_data = clean_data(sample_data)
    assert cleaned_data.isnull().sum().sum() == 0

def test_split_data(sample_data):
    train, test = split_data(sample_data, test_size=0.2)
    assert len(train) + len(test) == len(sample_data)

def test_generate_features(sample_data):
    features = generate_features(sample_data)
    assert features.shape[1] == sample_data.shape[1] - 1

def test_select_features(sample_data):
    selected_features = select_features(sample_data)
    assert selected_features.shape[1] == 10

def test_train_model(sample_data):
    features = sample_data.iloc[:, :-1]
    labels = sample_data['label']
    model = train_model(features, labels)
    assert isinstance(model, RandomForestClassifier)

def test_save_model(sample_data, tmpdir):
    features = sample_data.iloc[:, :-1]
    labels = sample_data['label']
    model = train_model(features, labels)
    file_path = tmpdir.join("model.joblib")
    save_model(model, str(file_path))
    assert os.path.exists(file_path)

def test_evaluate_model(sample_data):
    features = sample_data.iloc[:, :-1]
    labels = sample_data['label']
    model = train_model(features, labels)
    evaluation_metrics = evaluate_model(model, features, labels)
    assert 'accuracy' in evaluation_metrics and 'report' in evaluation_metrics

def test_generate_report(sample_data, capsys):
    features = sample_data.iloc[:, :-1]
    labels = sample_data['label']
    model = train_model(features, labels)
    evaluation_metrics = evaluate_model(model, features, labels)
    generate_report(evaluation_metrics)
    captured = capsys.readouterr()
    assert "Evaluation Report" in captured.out
```

### (b) Complete GitHub Actions Workflow

```yaml
# .github/workflows/test_pipeline.yml

name: Test Pipeline

on:
  push:
    branches:
      - main
  pull_request:
    branches:
      - main

jobs:
  build:

    runs-on: ubuntu-latest

    steps:
    - name: Check out the repository
      uses: actions/checkout@v3

    - name: Set up Python
      uses: actions/setup-python@v4
      with:
        python-version: '3.8'

    - name: Install dependencies
      run: |
        python -m pip install --upgrade pip
        pip install pytest pandas scikit-learn joblib

    - name: Run tests
      run: |
        pytest test_pipeline.py
```

This setup ensures that your pipeline is thoroughly tested and that tests are automatically run on each push or pull request to the main branch. The workflow installs the necessary dependencies and executes the tests using `pytest`.
