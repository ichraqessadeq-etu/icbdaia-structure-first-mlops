# test_pipeline.py

import pytest
import pandas as pd
from sklearn.ensemble import RandomForestClassifier
from pipeline.data_preprocessing import load_data, preprocess_data
from pipeline.model_training import split_data, train_model
from pipeline.model_evaluation import evaluate_model
from pipeline.model_deployment import deploy_model
import os

@pytest.fixture
def sample_data():
    data = {
        'feature1': [1, 2, 3, 4, 5],
        'feature2': [6, 7, 8, 9, 10],
        'target': [0, 1, 0, 1, 0]
    }
    return pd.DataFrame(data)

def test_load_data():
    # Create a temporary CSV file to test loading
    temp_file = 'temp_data.csv'
    sample_df = pd.DataFrame({'col1': [1, 2], 'col2': [3, 4]})
    sample_df.to_csv(temp_file, index=False)
    
    loaded_data = load_data(temp_file)
    pd.testing.assert_frame_equal(loaded_data, sample_df)
    
    # Clean up
    os.remove(temp_file)

def test_preprocess_data(sample_data):
    preprocessed_data = preprocess_data(sample_data)
    assert not preprocessed_data.isnull().values.any()  # Check no missing values

def test_split_data(sample_data):
    X_train, X_test, y_train, y_test = split_data(sample_data, test_size=0.2)
    assert len(X_train) == 4
    assert len(X_test) == 1
    assert len(y_train) == 4
    assert len(y_test) == 1

def test_train_model(sample_data):
    X_train, X_test, y_train, y_test = split_data(sample_data, test_size=0.2)
    model = train_model(X_train, y_train)
    assert isinstance(model, RandomForestClassifier)

def test_evaluate_model(sample_data):
    X_train, X_test, y_train, y_test = split_data(sample_data, test_size=0.2)
    model = train_model(X_train, y_train)
    metrics = evaluate_model(model, X_test, y_test)
    assert 'accuracy' in metrics
    assert 'precision' in metrics
    assert 'recall' in metrics

def test_deploy_model(sample_data):
    X_train, X_test, y_train, y_test = split_data(sample_data, test_size=0.2)
    model = train_model(X_train, y_train)
    deployment_path = 'temp_model.pkl'
    deploy_model(model, deployment_path)
    assert os.path.exists(deployment_path)
    
    # Clean up
    os.remove(deployment_path)

### (b) GitHub Actions Workflow

```yaml
# .github/workflows/ci.yml

name: CI

on:
  push:
    branches:
      - main
  pull_request:
    branches:
      - main

jobs:
  test:
    runs-on: ubuntu-latest

    steps:
    - name: Checkout code
      uses: actions/checkout@v2

    - name: Set up Python
      uses: actions/setup-python@v2
      with:
        python-version: '3.8'

    - name: Install dependencies
      run: |
        python -m pip install --upgrade pip
        pip install pytest pandas scikit-learn joblib

    - name: Run tests
      run: |
        pytest --maxfail=1 --disable-warnings
```

This configuration will automatically run the tests every time there is a push or pull request to the `main` branch. It uses Python 3.8, installs necessary dependencies, and executes the tests using `pytest`.
