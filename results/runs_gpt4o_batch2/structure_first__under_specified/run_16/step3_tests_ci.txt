import pytest
from fraud_detection.src.data_preprocessing import load_data, clean_data, feature_engineering

import os

    assert not df.empty

    assert cleaned_df.isnull().sum().sum() == 0

    engineered_df = feature_engineering(df)

    assert (engineered_df['new_feature'] == df['feature1'] * df['feature2']).all()

    df = pd.DataFrame({'feature1': [1, 2, 3, 4], 'target': [0, 1, 0, 1]})

    assert len(X_train) == len(y_train)

### (b) GitHub Actions Workflow

Create a GitHub Actions workflow file named `python-app.yml` in the `.github/workflows` directory of your project:

```yaml
# .github/workflows/python-app.yml
name: Python application

on:
  push:
    branches: [main]
  pull_request:
    branches: [main]

jobs:
  build:
    runs-on: ubuntu-latest

    steps:
    - uses: actions/checkout@v2

    - name: Set up Python
      uses: actions/setup-python@v2
      with:
        python-version: '3.x'

    - name: Install dependencies
      run: |
        python -m pip install --upgrade pip
        pip install pytest
        pip install scikit-learn
        pip install joblib

    - name: Run tests
      run: |
        pytest tests/

```
