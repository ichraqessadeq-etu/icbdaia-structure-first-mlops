# tests/test_pipeline.py

import pytest
import pandas as pd
from sklearn.ensemble import RandomForestClassifier
from fraud_detection.pipeline.data_preprocessing import load_data, clean_data, feature_engineering
from fraud_detection.pipeline.model_training import split_data, train_model, save_model
from fraud_detection.pipeline.model_evaluation import load_model, evaluate_model

# Sample data for testing
@pytest.fixture
def sample_data():
    data = {
        'feature1': [1, 2, 3, 4, 5],
        'feature2': [5, 4, 3, 2, 1],
        'target': [0, 1, 0, 1, 0]
    }
    return pd.DataFrame(data)

def test_load_data(tmp_path):
    # Create a temporary CSV file
    file_path = tmp_path / "sample.csv"
    data = "feature1,feature2,target\n1,5,0\n2,4,1\n3,3,0\n4,2,1\n5,1,0"
    file_path.write_text(data)
    
    df = load_data(str(file_path))
    assert not df.empty
    assert list(df.columns) == ['feature1', 'feature2', 'target']

def test_clean_data(sample_data):
    df = sample_data.copy()
    df.loc[0, 'feature1'] = None  # Introduce a missing value
    cleaned_df = clean_data(df)
    assert cleaned_df.isnull().sum().sum() == 0

def test_feature_engineering(sample_data):
    df = feature_engineering(sample_data)
    assert df.mean().round(6).all() == 0  # Check if standardized

def test_split_data(sample_data):
    X_train, X_test, y_train, y_test = split_data(sample_data, test_size=0.2)
    assert len(X_train) == 4
    assert len(X_test) == 1

def test_train_model(sample_data):
    X_train, X_test, y_train, y_test = split_data(sample_data, test_size=0.2)
    model = train_model(X_train, y_train)
    assert isinstance(model, RandomForestClassifier)

def test_save_and_load_model(tmp_path, sample_data):
    X_train, X_test, y_train, y_test = split_data(sample_data, test_size=0.2)
    model = train_model(X_train, y_train)
    file_path = tmp_path / "model.joblib"
    save_model(model, str(file_path))
    loaded_model = load_model(str(file_path))
    assert isinstance(loaded_model, RandomForestClassifier)

def test_evaluate_model(sample_data):
    X_train, X_test, y_train, y_test = split_data(sample_data, test_size=0.2)
    model = train_model(X_train, y_train)
    metrics = evaluate_model(model, X_test, y_test)
    assert 'accuracy' in metrics
    assert 'precision' in metrics
    assert 'recall' in metrics
    assert 'f1_score' in metrics

### (b) Complete GitHub Actions workflow

```yaml
# .github/workflows/python-app.yml

name: Python application

on:
  push:
    branches:
      - main
  pull_request:
    branches:
      - main

jobs:
  build:
    runs-on: ubuntu-latest

    steps:
    - name: Check out the repository
      uses: actions/checkout@v2

    - name: Set up Python
      uses: actions/setup-python@v2
      with:
        python-version: '3.8'

    - name: Install dependencies
      run: |
        python -m pip install --upgrade pip
        pip install pytest scikit-learn joblib pandas

    - name: Run tests
      run: |
        pytest tests
```

This setup includes a pytest test file that checks each function in your pipeline for expected behavior. The GitHub Actions workflow is configured to run these tests on every push or pull request to the `main` branch. It sets up Python, installs necessary dependencies, and executes the tests using pytest.
