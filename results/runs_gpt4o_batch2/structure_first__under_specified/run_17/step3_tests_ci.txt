# tests/test_pipeline.py

import pytest
import pandas as pd
from pandas.testing import assert_frame_equal
from sklearn.ensemble import RandomForestClassifier
from pipeline.data_preprocessing import load_data, clean_data, split_data
from pipeline.feature_engineering import create_features, select_features
from pipeline.model_training import train_model, save_model
from pipeline.model_evaluation import evaluate_model, log_evaluation_metrics
from pipeline.model_deployment import deploy_model, monitor_model

# Sample data for testing
@pytest.fixture
def sample_data():
    data = {
        'transaction_time': ['2023-01-01 10:00:00', '2023-01-01 11:00:00', '2023-01-01 12:00:00'],
        'amount': [100, 200, -50],
        'label': [0, 1, 0]
    }
    return pd.DataFrame(data)

def test_load_data(tmp_path):
    file_path = tmp_path / "test.csv"
    data = {
        'transaction_time': ['2023-01-01 10:00:00'],
        'amount': [100],
        'label': [0]
    }
    pd.DataFrame(data).to_csv(file_path, index=False)
    
    loaded_data = load_data(str(file_path))
    expected_data = pd.DataFrame(data)
    assert_frame_equal(loaded_data, expected_data)

def test_clean_data(sample_data):
    cleaned_data = clean_data(sample_data)
    expected_data = sample_data.dropna()
    expected_data = expected_data[expected_data['amount'] > 0]
    assert_frame_equal(cleaned_data, expected_data)

def test_split_data(sample_data):
    train, test = split_data(sample_data, test_size=0.33)
    assert len(train) + len(test) == len(sample_data)
    assert len(test) == 1  # With 3 samples and 0.33 test size

def test_create_features(sample_data):
    features_data = create_features(sample_data)
    assert 'transaction_hour' in features_data.columns

def test_select_features(sample_data):
    features_data = create_features(sample_data)
    labels = features_data['label']
    selected_features = select_features(features_data.drop(columns=['label']), labels)
    assert not selected_features.empty

def test_train_model(sample_data):
    features_data = create_features(sample_data)
    labels = features_data['label']
    model = train_model(features_data.drop(columns=['label']), labels)
    assert isinstance(model, RandomForestClassifier)

def test_save_model(tmp_path):
    model = RandomForestClassifier()
    file_path = tmp_path / "model.joblib"
    save_model(model, str(file_path))
    assert file_path.exists()

def test_evaluate_model(sample_data):
    features_data = create_features(sample_data)
    labels = features_data['label']
    train_features, test_features = split_data(features_data.drop(columns=['label']), test_size=0.33)
    train_labels, test_labels = labels.iloc[train_features.index], labels.iloc[test_features.index]
    model = train_model(train_features, train_labels)
    accuracy = evaluate_model(model, test_features, test_labels)
    assert 0 <= accuracy <= 1

def test_log_evaluation_metrics(capfd):

    captured = capfd.readouterr()

### b) Complete GitHub Actions Workflow

```yaml
# .github/workflows/python-package.yml

name: Python Package

on:
  push:
    branches:
      - main
  pull_request:
    branches:
      - main

jobs:
  build:
    runs-on: ubuntu-latest

    steps:
    - name: Checkout code
      uses: actions/checkout@v2

    - name: Set up Python
      uses: actions/setup-python@v2
      with:
        python-version: '3.8'

    - name: Install dependencies
      run: |
        python -m pip install --upgrade pip
        pip install pytest pandas scikit-learn joblib

    - name: Run tests
      run: |
        pytest tests

    - name: Upload coverage to Codecov
      if: success()
      uses: codecov/codecov-action@v2
      with:
        token: ${{ secrets.CODECOV_TOKEN }}
```

This setup includes the necessary pytest tests for each function in your pipeline, and the GitHub Actions workflow to automate testing on every push or pull request to the `main` branch. Make sure to replace `CODECOV_TOKEN` with your actual Codecov token in the GitHub repository secrets if you want to upload coverage reports.
