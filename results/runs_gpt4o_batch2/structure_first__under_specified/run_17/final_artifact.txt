import pandas as pd
from sklearn.model_selection import train_test_split

def load_data(file_path: str) -> pd.DataFrame:
    """Load data from a specified file path."""
    data = pd.read_csv(file_path)
    return data

def clean_data(data: pd.DataFrame) -> pd.DataFrame:
    """Clean the loaded data."""
    # Example cleaning operations
    data.dropna(inplace=True)
    data = data[data['amount'] > 0]
    return data

def split_data(data: pd.DataFrame, test_size: float) -> (pd.DataFrame, pd.DataFrame):
    """Split the data into training and test sets."""
    train, test = train_test_split(data, test_size=test_size, random_state=42)
    return train, test


# pipeline/feature_engineering.py

import pandas as pd
from sklearn.feature_selection import SelectKBest, f_classif

def create_features(data: pd.DataFrame) -> pd.DataFrame:
    """Create features from the given data."""
    # Example feature creation
    data['transaction_hour'] = pd.to_datetime(data['transaction_time']).dt.hour
    return data

def select_features(data: pd.DataFrame, labels: pd.Series) -> pd.DataFrame:
    """Select important features for the model."""
    selector = SelectKBest(score_func=f_classif, k='all')
    selector.fit(data, labels)
    selected_features = data.columns[selector.get_support()]
    return data[selected_features]

import joblib
