```python
# tests/test_pipeline.py

import pytest
import pandas as pd
from sklearn.ensemble import RandomForestClassifier
from fraud_detection_pipeline.pipeline.data_preprocessing import load_data, clean_data, feature_engineering
from fraud_detection_pipeline.pipeline.model_training import split_data, train_model, save_model
from fraud_detection_pipeline.pipeline.model_evaluation import load_model, evaluate_model

@pytest.fixture
def sample_data():
    # Create a sample dataframe for testing
    data = {
        'feature1': [1, 2, 3, 4, 5],
        'feature2': [6, 7, 8, 9, 10],
        'target': [0, 1, 0, 1, 0]
    }
    return pd.DataFrame(data)

def test_load_data(tmp_path):
    # Create a temporary CSV file and test loading
    file_path = tmp_path / "temp_data.csv"
    df = pd.DataFrame({'col1': [1, 2], 'col2': [3, 4]})
    df.to_csv(file_path, index=False)
    loaded_df = load_data(file_path)
    pd.testing.assert_frame_equal(df, loaded_df)

def test_clean_data(sample_data):
    # Introduce NaN values and test cleaning
    sample_data.loc[1, 'feature1'] = None
    cleaned_data = clean_data(sample_data)
    assert cleaned_data.isnull().sum().sum() == 0

def test_feature_engineering(sample_data):
    # Test feature engineering process
    scaled_data = feature_engineering(sample_data.drop(columns='target'))
    assert scaled_data.mean().round().sum() == 0  # StandardScaler should center data

def test_split_data(sample_data):
    # Test data splitting
    X_train, X_test, y_train, y_test = split_data(sample_data)
    assert len(X_train) + len(X_test) == len(sample_data)

def test_train_model(sample_data):
    # Test model training
    X_train, X_test, y_train, y_test = split_data(sample_data)
    model = train_model(X_train, y_train)
    assert isinstance(model, RandomForestClassifier)

def test_save_and_load_model(sample_data, tmp_path):
    # Test saving and loading of the model
    X_train, X_test, y_train, y_test = split_data(sample_data)
    model = train_model(X_train, y_train)
    model_path = tmp_path / "model.joblib"
    save_model(model, model_path)
    loaded_model = load_model(model_path)
    assert isinstance(loaded_model, RandomForestClassifier)

def test_evaluate_model(sample_data):
    # Test model evaluation
    X_train, X_test, y_train, y_test = split_data(sample_data)
    model = train_model(X_train, y_train)
    metrics = evaluate_model(model, X_test, y_test)
    assert 'accuracy' in metrics and 'precision' in metrics and 'recall' in metrics
```

### b) GitHub Actions Workflow

```yaml
# .github/workflows/python-package.yml

name: Python Package

on:
  push:
    branches: [main]
  pull_request:
    branches: [main]

jobs:
  build:
    runs-on: ubuntu-latest

    steps:
    - name: Checkout code
      uses: actions/checkout@v3

    - name: Set up Python
      uses: actions/setup-python@v4
      with:
        python-version: '3.8'

    - name: Install dependencies
      run: |
        python -m pip install --upgrade pip
        pip install pandas scikit-learn joblib pytest

    - name: Run tests
      run: |
        pytest tests/
```

This workflow will trigger on any push or pull request to the `main` branch, setting up Python, installing dependencies, and running the tests. The test file covers loading, cleaning, feature engineering, splitting, training, saving/loading, and evaluating the model.
