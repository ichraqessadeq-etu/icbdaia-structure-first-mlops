### (a) Pytest Test File

We'll write a `test_pipeline.py` file that tests the various functions in your pipeline. This will involve creating mock data to simulate the pipeline behavior.

```python
# tests/test_pipeline.py

import pytest
import pandas as pd
from sklearn.linear_model import LogisticRegression
from fraud_detection_pipeline.pipeline.data_preprocessing import load_data, clean_data, feature_engineering
from fraud_detection_pipeline.pipeline.model_training import split_data, train_model, save_model
from fraud_detection_pipeline.pipeline.model_evaluation import evaluate_model, generate_classification_report
from fraud_detection_pipeline.pipeline.model_deployment import load_model

@pytest.fixture
def sample_data():
    """Fixture for creating a sample dataframe."""
    data = {
        'feature1': [1.0, 2.0, 3.0, 4.0],
        'feature2': [2.0, 3.0, 4.0, 5.0],
        'target': [0, 1, 0, 1]
    }
    return pd.DataFrame(data)

def test_clean_data(sample_data):
    """Test the clean_data function."""
    dirty_data = sample_data.copy()
    dirty_data.iloc[0, 0] = None
    cleaned_data = clean_data(dirty_data)
    assert cleaned_data.shape[0] == 3  # Should drop one row

def test_feature_engineering(sample_data):
    """Test the feature_engineering function."""
    processed_data = feature_engineering(sample_data)
    assert 'target' in processed_data.columns
    assert processed_data.shape[1] == sample_data.shape[1]

def test_split_data(sample_data):
    """Test the split_data function."""
    X_train, X_test, y_train, y_test = split_data(sample_data, test_size=0.5)
    assert len(X_train) == 2
    assert len(X_test) == 2

def test_train_model(sample_data):
    """Test the train_model function."""
    X_train, X_test, y_train, y_test = split_data(sample_data, test_size=0.5)
    model = train_model(X_train, y_train)
    assert isinstance(model, LogisticRegression)

def test_save_and_load_model(tmp_path, sample_data):
    """Test the save_model and load_model functions."""
    X_train, X_test, y_train, y_test = split_data(sample_data, test_size=0.5)
    model = train_model(X_train, y_train)
    model_path = tmp_path / "model.joblib"
    save_model(model, str(model_path))
    loaded_model = load_model(str(model_path))
    assert isinstance(loaded_model, LogisticRegression)

def test_evaluate_model(sample_data):
    """Test the evaluate_model function."""
    X_train, X_test, y_train, y_test = split_data(sample_data, test_size=0.5)
    model = train_model(X_train, y_train)
    accuracy = evaluate_model(model, X_test, y_test)
    assert 0.0 <= accuracy <= 1.0

def test_generate_classification_report(sample_data):
    """Test the generate_classification_report function."""
    X_train, X_test, y_train, y_test = split_data(sample_data, test_size=0.5)
    model = train_model(X_train, y_train)
    y_pred = model.predict(X_test)
    report = generate_classification_report(y_test, y_pred)
    assert isinstance(report, str)
    assert "precision" in report
```

### (b) GitHub Actions Workflow

Below is a YAML configuration for a GitHub Actions workflow to automatically run your tests whenever code is pushed to the repository.

```yaml
# .github/workflows/python-package.yml

name: Python package

on:
  push:
    branches: [ main ]
  pull_request:
    branches: [ main ]

jobs:
  build:
    runs-on: ubuntu-latest

    steps:
    - name: Check out repository code
      uses: actions/checkout@v2

    - name: Set up Python
      uses: actions/setup-python@v2
      with:
        python-version: '3.8'

    - name: Install dependencies
      run: |
        python -m pip install --upgrade pip
        pip install pandas scikit-learn joblib pytest

    - name: Run tests
      run: |
        pytest tests/
```

This setup ensures that every time you make changes to your repository, the tests will be executed automatically, helping maintain the integrity of your pipeline.
