# tests/test_pipeline.py

import pytest
import pandas as pd
from fraud_detection.pipeline.data_preprocessing import load_data, clean_data, feature_engineering
from fraud_detection.pipeline.model_training import split_data, train_model, save_model
from fraud_detection.pipeline.model_evaluation import load_model, evaluate_model

@pytest.fixture
def sample_data():
    """Fixture to provide sample data for testing."""
    data = {
        'existing_feature': [1, 2, 3, 4, 5],
        'label': [0, 1, 0, 1, 0]
    }
    return pd.DataFrame(data)

def test_load_data(tmp_path):
    """Test loading data from a file."""
    file_path = tmp_path / "test.csv"
    sample_data = "existing_feature,label\n1,0\n2,1\n3,0\n4,1\n5,0"
    file_path.write_text(sample_data)
    
    data = load_data(str(file_path))
    assert not data.empty
    assert 'existing_feature' in data.columns

def test_clean_data(sample_data):
    """Test cleaning data."""
    raw_data = sample_data.copy()
    raw_data.loc[0, 'existing_feature'] = None  # introduce a NaN value
    cleaned_data = clean_data(raw_data)
    assert cleaned_data.isna().sum().sum() == 0

def test_feature_engineering(sample_data):
    """Test feature engineering."""
    cleaned_data = clean_data(sample_data)
    engineered_data = feature_engineering(cleaned_data)
    assert 'new_feature' in engineered_data.columns
    assert (engineered_data['new_feature'] == engineered_data['existing_feature'] * 2).all()

def test_split_data(sample_data):
    """Test splitting data."""
    features = sample_data[['existing_feature']]
    labels = sample_data['label']
    X_train, X_test, y_train, y_test = split_data(features, labels)
    assert len(X_train) == 4
    assert len(X_test) == 1

def test_train_model(sample_data):
    """Test model training."""
    features = sample_data[['existing_feature']]
    labels = sample_data['label']
    X_train, X_test, y_train, y_test = split_data(features, labels)
    model = train_model((X_train, X_test, y_train, y_test))
    assert model is not None
    assert hasattr(model, 'predict')

def test_save_load_model(tmp_path):
    """Test saving and loading the model."""
    features = pd.DataFrame({'existing_feature': [1, 2, 3, 4]})
    labels = pd.Series([0, 1, 0, 1])
    X_train, X_test, y_train, y_test = split_data(features, labels)
    model = train_model((X_train, X_test, y_train, y_test))
    
    model_path = tmp_path / "model.joblib"
    save_model(model, str(model_path))
    
    loaded_model = load_model(str(model_path))
    assert loaded_model is not None
    assert hasattr(loaded_model, 'predict')

def test_evaluate_model(sample_data):
    """Test model evaluation."""
    features = sample_data[['existing_feature']]
    labels = sample_data['label']
    X_train, X_test, y_train, y_test = split_data(features, labels)
    model = train_model((X_train, X_test, y_train, y_test))
    
    evaluation_metrics = evaluate_model(model, (X_test, y_test))
    assert 'accuracy' in evaluation_metrics
    assert 'precision' in evaluation_metrics
    assert 'recall' in evaluation_metrics
    assert 'f1_score' in evaluation_metrics

### (b) GitHub Actions Workflow
```yaml
# .github/workflows/test_pipeline.yml

name: Python CI

on:
  push:
    branches: [main]
  pull_request:
    branches: [main]

jobs:
  build:

    runs-on: ubuntu-latest

    steps:
    - name: Check out repository
      uses: actions/checkout@v2

    - name: Set up Python
      uses: actions/setup-python@v2
      with:
        python-version: '3.8'

    - name: Install dependencies
      run: |
        python -m pip install --upgrade pip
        pip install pytest pandas scikit-learn joblib requests

    - name: Run tests
      run: |
        pytest tests/
```
