```python
# tests/test_pipeline.py

import pytest
import pandas as pd
from fraud_detection.pipeline.data_processing import load_data, preprocess_data
from fraud_detection.pipeline.model_training import split_data, train_model
from fraud_detection.pipeline.model_evaluation import evaluate_model
from fraud_detection.pipeline.model_deployment import deploy_model
import os
import joblib

@pytest.fixture
def sample_data():
    """Fixture to provide sample data for testing."""
    data = {
        'feature1': [1, 2, 3, 4, 5],
        'feature2': ['A', 'B', 'A', 'B', 'A'],
        'fraud_label': [0, 1, 0, 1, 0]
    }
    return pd.DataFrame(data)

def test_load_data(tmp_path):
    """Test loading data from a file."""
    file_path = tmp_path / "sample.csv"
    sample_data = "feature1,feature2,fraud_label\n1,A,0\n2,B,1\n3,A,0\n4,B,1\n5,A,0"
    file_path.write_text(sample_data)
    data = load_data(file_path)
    assert not data.empty

def test_preprocess_data(sample_data):
    """Test data preprocessing."""
    processed_data = preprocess_data(sample_data)
    assert 'feature2_B' in processed_data.columns
    assert 'feature2_A' not in processed_data.columns

def test_split_data(sample_data):
    """Test splitting data into train and test sets."""
    processed_data = preprocess_data(sample_data)
    X_train, X_test, y_train, y_test = split_data(processed_data)
    assert len(X_train) == 4
    assert len(X_test) == 1
    assert len(y_train) == 4
    assert len(y_test) == 1

def test_train_model(sample_data):
    """Test model training."""
    processed_data = preprocess_data(sample_data)
    split_data_tuple = split_data(processed_data)
    model = train_model(split_data_tuple)
    assert model is not None
    assert hasattr(model, "predict")

def test_evaluate_model(sample_data):
    """Test model evaluation."""
    processed_data = preprocess_data(sample_data)
    split_data_tuple = split_data(processed_data)
    model = train_model(split_data_tuple)
    evaluation_metrics = evaluate_model(model, split_data_tuple)
    assert 'accuracy' in evaluation_metrics
    assert 'report' in evaluation_metrics

def test_deploy_model(tmp_path, sample_data):
    """Test model deployment."""
    model_path = tmp_path / "deployed_model.joblib"
    config = {'model_path': str(model_path)}
    processed_data = preprocess_data(sample_data)
    split_data_tuple = split_data(processed_data)
    model = train_model(split_data_tuple)
    deploy_model(model, config)
    assert os.path.exists(model_path)
    loaded_model = joblib.load(model_path)
    assert loaded_model is not None
```

### b) GitHub Actions Workflow

```yaml
# .github/workflows/ci.yml

name: CI Pipeline

on:
  push:
    branches:
      - main
  pull_request:
    branches:
      - main

jobs:
  build:
    runs-on: ubuntu-latest

    steps:
    - name: Checkout source code
      uses: actions/checkout@v2

    - name: Set up Python
      uses: actions/setup-python@v2
      with:
        python-version: '3.8'

    - name: Install dependencies
      run: |
        python -m pip install --upgrade pip
        pip install pytest pandas scikit-learn joblib

    - name: Run tests
      run: |
        pytest tests/
```

This configuration includes the necessary steps to test the pipeline module using `pytest` in a CI environment via GitHub Actions. The workflow triggers on pushes and pull requests to the main branch, and it installs dependencies before running the tests.
