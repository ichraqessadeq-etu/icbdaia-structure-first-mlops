This implementation includes the main steps required for a fraud detection pipeline using a `RandomForestClassifier` from `scikit-learn`, with data loading, preprocessing, training, evaluation, concept drift checking, and logging to MLflow.

```python
import pandas as pd
import numpy as np
from sklearn.model_selection import train_test_split
from sklearn.ensemble import RandomForestClassifier
from sklearn.metrics import accuracy_score, precision_score, recall_score, f1_score
from sklearn.utils import resample
import mlflow
import mlflow.sklearn

def load_data(filepath: str) -> tuple:
    """Load data from the specified file path."""
    data = pd.read_csv(filepath)
    X = data.drop('target', axis=1)
    y = data['target']
    return X, y

def preprocess_data(data: tuple) -> tuple:
    """Preprocess the data and handle imbalances."""
    X, y = data
    # Handle imbalance by upsampling the minority class
    X_majority = X[y == 0]
    y_majority = y[y == 0]
    X_minority = X[y == 1]
    y_minority = y[y == 1]

    X_minority_upsampled, y_minority_upsampled = resample(
        X_minority, y_minority,
        replace=True,
        n_samples=len(y_majority),
        random_state=123
    )

    X_upsampled = pd.concat([X_majority, X_minority_upsampled])
    y_upsampled = pd.concat([y_majority, y_minority_upsampled])

    return train_test_split(X_upsampled, y_upsampled, test_size=0.2, random_state=42)

def train_model(X_train, y_train) -> "RandomForestClassifier":
    """Train a RandomForestClassifier model."""
    model = RandomForestClassifier(n_estimators=100, random_state=42)
    model.fit(X_train, y_train)
    return model

def evaluate_model(model: "RandomForestClassifier", X_test, y_test) -> dict:
    """Evaluate the model and return evaluation metrics."""
    y_pred = model.predict(X_test)
    metrics = {
        'accuracy': accuracy_score(y_test, y_pred),
        'precision': precision_score(y_test, y_pred),
        'recall': recall_score(y_test, y_pred),
        'f1_score': f1_score(y_test, y_pred)
    }
    return metrics

def check_concept_drift(old_data: tuple, new_data: tuple) -> bool:
    """Check for concept drift between old and new data."""
    # Simple concept drift check by comparing the mean of features
    old_X, _ = old_data
    new_X, _ = new_data
    drift_detected = not np.allclose(old_X.mean(), new_X.mean(), atol=1e-2)
    return drift_detected

def log_to_mlflow(metrics: dict, model: "RandomForestClassifier"):
    """Log metrics and model to MLflow."""
    mlflow.log_metrics(metrics)
    mlflow.sklearn.log_model(model, "random_forest_model")

def main():
    """Main function to execute the pipeline steps."""
    # Assume file paths and old data for the example
    data_filepath = "data/fraud_data.csv"
    old_data_filepath = "data/fraud_data_old.csv"

    X, y = load_data(data_filepath)
    old_X, old_y = load_data(old_data_filepath)

    X_train, X_test, y_train, y_test = preprocess_data((X, y))

    model = train_model(X_train, y_train)

    metrics = evaluate_model(model, X_test, y_test)

    if check_concept_drift((old_X, old_y), (X, y)):
        print("Concept drift detected!")

    log_to_mlflow(metrics, model)

    print("Pipeline execution completed.")

if __name__ == "__main__":
    main()
```

This module assumes that the data files `data/fraud_data.csv` and `data/fraud_data_old.csv` are available and contain a `target` column indicating the class label. The concept drift detection is a simple check for mean differences, and MLflow is used for logging model metrics and the trained model. Adjustments may be needed based on the actual data and requirements.
