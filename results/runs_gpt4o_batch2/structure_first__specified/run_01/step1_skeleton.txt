### GitHub Actions YAML Workflow

#### `.github/workflows/ci_cd_pipeline.yml`

```yaml
name: CI/CD Pipeline

on:
  push:
    branches:
      - main
  pull_request:
    branches:
      - main
  schedule:
    - cron: '0 0 * * 0' # Weekly schedule

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout Code
        uses: actions/checkout@v2

      - name: Set up Python
        uses: actions/setup-python@v2
        with:
          python-version: '3.8'

      - name: Install Dependencies
        run: |
          python -m pip install --upgrade pip
          pip install -r requirements.txt

      - name: Run Tests
        run: |
          pytest tests/

  drift_detection:
    runs-on: ubuntu-latest
    needs: build
    steps:
      - name: Checkout Code
        uses: actions/checkout@v2

      - name: Detect Concept Drift
        run: |
          python src/main_pipeline.py detect_drift

  retrain_model:
    runs-on: ubuntu-latest
    needs: drift_detection
    if: needs.drift_detection.outputs.drift_detected == 'true'
    steps:
      - name: Checkout Code
        uses: actions/checkout@v2

      - name: Retrain Model
        run: |
          python src/main_pipeline.py retrain_model
```
