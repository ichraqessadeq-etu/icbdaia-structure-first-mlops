import pytest
import pandas as pd
import numpy as np
from sklearn.ensemble import RandomForestClassifier
from fraud_detection.pipeline.data_loader import load_data, preprocess_data
from fraud_detection.pipeline.trainer import train_model, detect_concept_drift
from fraud_detection.pipeline.evaluator import evaluate_model

@pytest.fixture
def sample_data():
    """Fixture for creating a sample dataset."""
    data = pd.DataFrame({
        'feature1': np.random.rand(100),
        'feature2': np.random.rand(100),
        'label': np.random.randint(0, 2, 100)
    })
    return data

def test_load_data(tmp_path):
    """Test data loading function."""
    # Create a temporary CSV file
    file_path = tmp_path / "sample_data.csv"
    sample_data = pd.DataFrame({
        'feature1': [0.1, 0.2],
        'feature2': [0.3, 0.4],
        'label': [1, 0]
    })
    sample_data.to_csv(file_path, index=False)
    
    # Test loading data
    data = load_data(file_path)
    assert isinstance(data, pd.DataFrame)
    assert not data.empty

def test_preprocess_data(sample_data):
    """Test data preprocessing function."""
    X_train, X_test, y_train, y_test = preprocess_data(sample_data)
    assert X_train.shape[0] > 0
    assert X_test.shape[0] > 0
    assert y_train.shape[0] > 0
    assert y_test.shape[0] > 0

def test_train_model(sample_data):
    """Test model training function."""
    X_train, _, y_train, _ = preprocess_data(sample_data)
    model = train_model(X_train, y_train)
    assert isinstance(model, RandomForestClassifier)
    assert hasattr(model, 'predict')

def test_detect_concept_drift(sample_data):
    """Test concept drift detection function."""
    X_train, _, y_train, _ = preprocess_data(sample_data)
    drift_detected = detect_concept_drift(X_train, y_train)
    assert isinstance(drift_detected, bool)

def test_evaluate_model(sample_data):
    """Test model evaluation function."""
    X_train, X_test, y_train, y_test = preprocess_data(sample_data)
    model = train_model(X_train, y_train)
    metrics = evaluate_model(model, X_test, y_test)

    assert 'f1_score' in metrics

name: Python application

on:
  push:
    branches: [main]
  pull_request:
    branches: [main]

jobs:
  build:

    runs-on: ubuntu-latest

    steps:
    - uses: actions/checkout@v2

    - name: Set up Python
      uses: actions/setup-python@v2
      with:
        python-version: '3.8'

    - name: Install dependencies
      run: |
        python -m pip install --upgrade pip
        pip install pytest scikit-learn pandas mlflow

    - name: Run tests
      run: |
        pytest tests
