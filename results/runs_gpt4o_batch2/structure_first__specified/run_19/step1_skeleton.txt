### GitHub Actions Workflow

**`ci_cd.yml`**
```yaml
name: CI/CD Pipeline

on:
  push:
    branches:
      - main
  pull_request:
    branches:
      - main
  schedule:
    - cron: '0 0 * * 0'  # Every Sunday at midnight

jobs:
  setup:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout code
        uses: actions/checkout@v2
    
  install-dependencies:
    runs-on: ubuntu-latest
    needs: setup
    steps:
      - name: Set up Python
        uses: actions/setup-python@v2
        with:
          python-version: '3.x'
      - name: Install dependencies
        run: |
          python -m pip install --upgrade pip
          pip install -r requirements.txt

  test:
    runs-on: ubuntu-latest
    needs: install-dependencies
    steps:
      - name: Run tests
        run: pytest tests/

  train-and-log:
    runs-on: ubuntu-latest
    needs: test
    steps:
      - name: Train and log model
        run: python fraud_detection_pipeline/model/train.py
```
