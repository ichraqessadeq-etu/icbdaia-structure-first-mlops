def detect_concept_drift(old_data, new_data) -> bool:
    """Detect concept drift between old and new datasets."""
    old_mean = old_data.mean()
    new_mean = new_data.mean()
    drift_detected = (abs(new_mean - old_mean) > 0.1 * old_mean).any()

import mlflow

def log_metrics_to_mlflow(metrics: dict) -> None:
    """Log performance metrics to MLflow."""
    for key, value in metrics.items():
        if isinstance(value, dict):
            for sub_key, sub_value in value.items():
                mlflow.log_metric(f"{key}_{sub_key}", sub_value)
        else:
            mlflow.log_metric(key, value)
