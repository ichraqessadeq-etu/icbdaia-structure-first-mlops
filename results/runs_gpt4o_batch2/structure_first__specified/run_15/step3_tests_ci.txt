```python
# tests/test_pipeline.py
import pytest
import pandas as pd
from sklearn.ensemble import RandomForestClassifier
from sklearn.datasets import make_classification
from src.data_loader import load_data
from src.model import train_model, detect_concept_drift
from src.evaluation import evaluate_model
from src.pipeline import run_pipeline

@pytest.fixture
def sample_data():
    """Create a sample dataset for testing."""
    X, y = make_classification(n_samples=100, n_features=20, n_classes=2, random_state=42)
    data = pd.DataFrame(X, columns=[f"feature_{i}" for i in range(X.shape[1])])
    data['Class'] = y
    return data

def test_load_data(sample_data, tmp_path):
    """Test the data loading function."""
    file_path = tmp_path / "sample_data.csv"
    sample_data.to_csv(file_path, index=False)
    X, y = load_data(file_path)
    assert not X.empty
    assert not y.empty
    assert 'Class' not in X.columns

def test_train_model(sample_data):
    """Test the model training function."""
    X = sample_data.drop('Class', axis=1)
    y = sample_data['Class']
    model = train_model(X, y)
    assert isinstance(model, RandomForestClassifier)
    assert hasattr(model, "predict")

def test_evaluate_model(sample_data):
    """Test the model evaluation function."""
    X = sample_data.drop('Class', axis=1)
    y = sample_data['Class']
    model = train_model(X, y)
    metrics = evaluate_model(model, X, y)
    assert isinstance(metrics, dict)
    assert 'accuracy' in metrics['accuracy']

def test_detect_concept_drift(sample_data):
    """Test the concept drift detection function."""
    X = sample_data.drop('Class', axis=1)
    drift = detect_concept_drift(X, X*1.1)
    assert drift is True

def test_run_pipeline(sample_data, tmp_path):
    """Test the entire pipeline."""
    file_path = tmp_path / "sample_data.csv"
    sample_data.to_csv(file_path, index=False)
    run_pipeline(file_path)
```

### (b) GitHub Actions Workflow

```yaml
# .github/workflows/python-package.yml
name: Python package

on:
  push:
    branches: [ main ]
  pull_request:
    branches: [ main ]

jobs:
  test:

    runs-on: ubuntu-latest

    steps:
    - uses: actions/checkout@v2

    - name: Set up Python
      uses: actions/setup-python@v2
      with:
        python-version: '3.8'

    - name: Install dependencies
      run: |
        python -m pip install --upgrade pip
        pip install pytest pandas scikit-learn mlflow

    - name: Test with pytest
      run: |
        pytest tests
```

### Explanation

- **Pytest Test File**: This file contains test functions for each module in the pipeline. It uses fixtures to create sample data and tests data loading, model training, evaluation, concept drift detection, and the entire pipeline.

- **GitHub Actions Workflow**: This workflow is triggered on push or pull requests to the `main` branch. It sets up a Python environment, installs dependencies, and runs the tests using pytest.
