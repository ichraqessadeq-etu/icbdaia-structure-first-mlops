### GitHub Actions Workflow (`.github/workflows/ci_cd_pipeline.yml`)
```yaml
name: CI/CD Pipeline

on:
  push:
    branches:
      - main
  pull_request:
    branches:
      - main

jobs:
  setup:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout code
        uses: actions/checkout@v2

      - name: Set up Python
        uses: actions/setup-python@v2
        with:
          python-version: '3.8'

      - name: Install dependencies
        run: |
          python -m pip install --upgrade pip
          pip install -r requirements.txt

  test:
    runs-on: ubuntu-latest
    needs: setup
    steps:
      - name: Run pytest
        run: |
          pytest tests/

  run_pipeline:
    runs-on: ubuntu-latest
    needs: test
    steps:
      - name: Execute pipeline
        run: |
          python src/pipeline.py --data_path data/creditcard.csv
```
