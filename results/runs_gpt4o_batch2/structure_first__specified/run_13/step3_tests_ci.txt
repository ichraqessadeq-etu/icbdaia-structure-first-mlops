# test_pipeline.py

import pytest
import pandas as pd
import numpy as np
from sklearn.ensemble import RandomForestClassifier
from pipeline import load_data, preprocess_data, train_model, evaluate_model, log_to_mlflow

def test_load_data():
    # Assuming there is a valid CSV file at 'data/test_dataset.csv'
    data = load_data('data/test_dataset.csv')
    assert isinstance(data, pd.DataFrame)
    assert not data.empty

def test_preprocess_data():
    # Create a simple DataFrame with missing values and an imbalanced target
    data = pd.DataFrame({
        'feature1': [1, 2, np.nan, 4, 5],
        'feature2': [5, 4, 3, np.nan, 1],
        'target': [0, 0, 0, 1, 1]
    })
    
    X_resampled, y_resampled = preprocess_data(data)
    assert len(X_resampled) == len(y_resampled)
    assert np.sum(y_resampled == 0) == np.sum(y_resampled == 1)  # Balanced classes

def test_train_model():
    # Generate dummy data
    X_train = np.array([[1, 2], [3, 4], [5, 6], [7, 8]])
    y_train = np.array([0, 1, 0, 1])
    
    model = train_model(X_train, y_train)
    assert isinstance(model, RandomForestClassifier)
    assert model.n_classes_ == 2

def test_evaluate_model():
    # Generate dummy data and a simple model
    model = RandomForestClassifier()
    X_train = np.array([[1, 2], [3, 4], [5, 6], [7, 8]])
    y_train = np.array([0, 1, 0, 1])
    model.fit(X_train, y_train)
    
    X_test = np.array([[2, 3], [4, 5]])
    y_test = np.array([0, 1])
    
    metrics = evaluate_model(model, X_test, y_test)
    
    assert 'accuracy' in metrics
    assert 'precision' in metrics
    assert 'recall' in metrics
    assert 'f1_score' in metrics

def test_log_to_mlflow():
    # Test if logging to MLflow runs without exceptions
    params = {'n_estimators': 100, 'random_state': 42}
    metrics = {'accuracy': 0.9, 'precision': 0.9, 'recall': 0.9, 'f1_score': 0.9}
    model = RandomForestClassifier()
    
    try:
        log_to_mlflow(params, metrics, model)
        success = True
    except Exception:
        success = False
    
    assert success

### (b) GitHub Actions Workflow

```yaml
# .github/workflows/test_pipeline.yml

name: Test Python Pipeline

on:
  push:
    branches:
      - main
  pull_request:
    branches:
      - main

jobs:
  test:
    runs-on: ubuntu-latest

    steps:
    - name: Checkout code
      uses: actions/checkout@v2

    - name: Set up Python
      uses: actions/setup-python@v2
      with:
        python-version: '3.8' # Specify the version of Python

    - name: Install dependencies
      run: |
        python -m pip install --upgrade pip
        pip install pandas scikit-learn imbalanced-learn mlflow pytest

    - name: Run tests
      run: |
        pytest test_pipeline.py --maxfail=1 --disable-warnings
```

### Explanation

- **Test File**: Each function in the pipeline module is tested for basic functionality, including checks on data types, data integrity, and expected outcomes (e.g., balanced classes after SMOTE).
  
- **GitHub Actions Workflow**: This workflow triggers on pushes and pull requests to the `main` branch. It sets up Python, installs dependencies, and runs the tests using `pytest`. If any test fails, the workflow stops at the first failure (`--maxfail=1`).
