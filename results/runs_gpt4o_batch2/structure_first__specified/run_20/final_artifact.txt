import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler
from imblearn.over_sampling import SMOTE

def load_data(file_path: str):
    """Load the dataset from the specified file path."""
    data = pd.read_csv(file_path)
    return data

def preprocess_data(data):
    """Preprocess the dataset to handle imbalances and prepare for training."""
    X = data.drop('label', axis=1)
    y = data['label']
    
    # Handle data imbalance using SMOTE
    smote = SMOTE()
    X_res, y_res = smote.fit_resample(X, y)
    
    # Standardize the features
    scaler = StandardScaler()
    X_res_scaled = scaler.fit_transform(X_res)
    
    X_train, X_test, y_train, y_test = train_test_split(X_res_scaled, y_res, test_size=0.2, random_state=42)
    return X_train, X_test, y_train, y_test

from scipy.spatial.distance import jensenshannon
import numpy as np

def detect_concept_drift(reference_data, new_data):
    """Detect concept drift between reference and new data."""
    drift_detected = False
    threshold = 0.1  # This threshold can be adjusted based on the sensitivity you want
    for col in reference_data.columns:
        ref_dist = np.histogram(reference_data[col], bins=10, density=True)[0]
        new_dist = np.histogram(new_data[col], bins=10, density=True)[0]
        distance = jensenshannon(ref_dist, new_dist)
        if distance > threshold:
            drift_detected = True
            print(f"Drift detected in column {col} with JS distance: {distance}")
    return drift_detected
