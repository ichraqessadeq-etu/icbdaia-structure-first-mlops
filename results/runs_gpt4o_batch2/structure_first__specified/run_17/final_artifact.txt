import mlflow
import pandas as pd
from sklearn.ensemble import RandomForestClassifier
from sklearn.model_selection import train_test_split
from sklearn.metrics import classification_report
from sklearn.preprocessing import StandardScaler
from sklearn.utils import resample
from typing import Tuple
import numpy as np

def load_data() -> Tuple[pd.DataFrame, pd.Series]:
    """Load the credit card fraud dataset."""
    # Assuming a CSV file named 'creditcard.csv' in the current directory
    data = pd.read_csv('creditcard.csv')
    X = data.drop('Class', axis=1)
    y = data['Class']
    return X, y

def preprocess_data(X: pd.DataFrame, y: pd.Series) -> Tuple[np.ndarray, np.ndarray, np.ndarray, np.ndarray]:
    """Preprocess the dataset, handling imbalanced classes."""
    # Normalize the features
    scaler = StandardScaler()
    X_scaled = scaler.fit_transform(X)

    # Balance the dataset using resampling
    X_resampled, y_resampled = resample(X_scaled[y == 1],
                                        y[y == 1],
                                        replace=True,
                                        n_samples=X_scaled[y == 0].shape[0],
                                        random_state=42)
    X_balanced = np.vstack((X_scaled[y == 0], X_resampled))
    y_balanced = np.hstack((y[y == 0], y_resampled))

    # Split the data
    X_train, X_test, y_train, y_test = train_test_split(X_balanced, y_balanced, test_size=0.2, random_state=42)
    return X_train, X_test, y_train, y_test

def train_model(X_train: np.ndarray, y_train: np.ndarray) -> RandomForestClassifier:
    """Train a RandomForestClassifier on the given data."""
    model = RandomForestClassifier(n_estimators=100, random_state=42)
    model.fit(X_train, y_train)
    return model

def evaluate_model(model: RandomForestClassifier, X_test: np.ndarray, y_test: np.ndarray) -> dict:
    """Evaluate the model and return performance metrics."""
    y_pred = model.predict(X_test)
    report = classification_report(y_test, y_pred, output_dict=True)
    return report

def log_to_mlflow(model: RandomForestClassifier, metrics: dict):
    """Log the model and metrics to MLflow."""
    mlflow.sklearn.log_model(model, "model")
    mlflow.log_metrics({
        "precision": metrics['1']['precision'],
        "recall": metrics['1']['recall'],
        "f1-score": metrics['1']['f1-score']
    })

def detect_concept_drift(data: pd.DataFrame) -> bool:

    return False

def main():
    """Execute the complete pipeline."""
    mlflow.start_run()

    # Load and preprocess data
    X, y = load_data()
    X_train, X_test, y_train, y_test = preprocess_data(X, y)

    # Train and evaluate model
    model = train_model(X_train, y_train)
    metrics = evaluate_model(model, X_test, y_test)

    # Log to MLflow
    log_to_mlflow(model, metrics)

    else:
