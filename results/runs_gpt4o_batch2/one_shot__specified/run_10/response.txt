### 1. Python Pipeline Script

```python
# fraud_detection_pipeline.py

import os
import sys
import pandas as pd
import numpy as np
from sklearn.model_selection import train_test_split
from sklearn.ensemble import RandomForestClassifier
from sklearn.metrics import classification_report, accuracy_score
from sklearn.utils import resample
import mlflow
import mlflow.sklearn
from mlflow.exceptions import MlflowException

DATA_PATH = 'creditcard.csv'
CONCEPT_DRIFT_THRESHOLD = 0.05

def load_data(data_path):
    try:
        data = pd.read_csv(data_path)
        return data
    except Exception as e:
        print(f"Error loading data: {e}")
        sys.exit(1)

def preprocess_data(data):
    # Handle class imbalance
    try:
        fraud = data[data.Class == 1]
        non_fraud = data[data.Class == 0]
        non_fraud_downsampled = resample(non_fraud, replace=False, n_samples=len(fraud), random_state=42)
        balanced_data = pd.concat([fraud, non_fraud_downsampled])
        X = balanced_data.drop('Class', axis=1)
        y = balanced_data['Class']
        return train_test_split(X, y, test_size=0.3, random_state=42)
    except Exception as e:
        print(f"Error in preprocessing: {e}")
        sys.exit(1)

def train_model(X_train, y_train):
    try:
        clf = RandomForestClassifier(random_state=42)
        clf.fit(X_train, y_train)
        return clf
    except Exception as e:
        print(f"Error training model: {e}")
        sys.exit(1)

def evaluate_model(clf, X_test, y_test):
    try:
        y_pred = clf.predict(X_test)
        accuracy = accuracy_score(y_test, y_pred)
        report = classification_report(y_test, y_pred)
        return accuracy, report
    except Exception as e:
        print(f"Error evaluating model: {e}")
        sys.exit(1)

def log_to_mlflow(clf, accuracy, report):
    try:
        mlflow.set_experiment("Fraud Detection")
        with mlflow.start_run():
            mlflow.sklearn.log_model(clf, "random_forest_model")
            mlflow.log_metric("accuracy", accuracy)
            mlflow.log_text(report, "classification_report.txt")
    except MlflowException as e:
        print(f"Error logging to MLflow: {e}")
        sys.exit(1)

def detect_concept_drift(new_accuracy, old_accuracy):
    return abs(new_accuracy - old_accuracy) > CONCEPT_DRIFT_THRESHOLD

def main():
    data = load_data(DATA_PATH)
    X_train, X_test, y_train, y_test = preprocess_data(data)
    
    clf = train_model(X_train, y_train)
    
    accuracy, report = evaluate_model(clf, X_test, y_test)
    print(f"Model Accuracy: {accuracy}")
    print(f"Classification Report: \n{report}")
    
    log_to_mlflow(clf, accuracy, report)
    
    # Placeholder for drift detection logic with previous accuracy
    # old_accuracy = fetch_previous_accuracy_from_storage()
    old_accuracy = 0.90  # Dummy value for illustration
    if detect_concept_drift(accuracy, old_accuracy):
        print("Concept drift detected. Retraining model.")
        # Retraining logic
        clf = train_model(X_train, y_train)
        accuracy, report = evaluate_model(clf, X_test, y_test)
        log_to_mlflow(clf, accuracy, report)

if __name__ == "__main__":
    main()
```

```yaml
# .github/workflows/ci-cd-pipeline.yml

name: CI/CD for Fraud Detection Model

on:
  push:
    branches:
      - main
  pull_request:
    branches:
      - main
  schedule:
    - cron: '0 0 * * 0'  # Weekly schedule

jobs:
  build:
    runs-on: ubuntu-latest

    steps:
    - name: Checkout code
      uses: actions/checkout@v2

    - name: Set up Python
      uses: actions/setup-python@v2
      with:
        python-version: '3.8'

    - name: Install dependencies
      run: |
        python -m pip install --upgrade pip
        pip install -r requirements.txt

    - name: Run tests
      run: |
        pytest test_fraud_detection_pipeline.py

    - name: Run pipeline
      run: |
        python fraud_detection_pipeline.py

    - name: Publish to MLflow
      env:
        MLFLOW_TRACKING_URI: ${{ secrets.MLFLOW_TRACKING_URI }}
      run: |
        python fraud_detection_pipeline.py
```
