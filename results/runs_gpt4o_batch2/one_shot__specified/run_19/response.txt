Below is a complete solution for a CI/CD pipeline for a fraud detection model using a `RandomForestClassifier` from scikit-learn, with MLflow for experiment tracking, pytest for testing, and GitHub Actions for continuous integration.

### Python Pipeline Script

```python
import os
import joblib
import mlflow
import mlflow.sklearn
import numpy as np
import pandas as pd
from sklearn.ensemble import RandomForestClassifier
from sklearn.metrics import classification_report, roc_auc_score
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler
from sklearn.utils import resample
from sklearn.exceptions import NotFittedError

def load_data():
    # Replace with actual data loading logic
    try:
        df = pd.read_csv('creditcard.csv')  # Ensure the dataset is in the root directory
        return df
    except Exception as e:
        raise RuntimeError(f"Error loading data: {e}")

def preprocess_data(df):
    try:
        # Separate input features and target
        X = df.drop('Class', axis=1)
        y = df['Class']
        
        # Normalize features
        scaler = StandardScaler()
        X_scaled = scaler.fit_transform(X)
        
        # Handle imbalance by down-sampling majority class
        df_majority = df[df['Class'] == 0]
        df_minority = df[df['Class'] == 1]
        
        df_majority_downsampled = resample(df_majority, 
                                           replace=False,    
                                           n_samples=len(df_minority),    
                                           random_state=42)
        
        df_downsampled = pd.concat([df_majority_downsampled, df_minority])
        X_downsampled = df_downsampled.drop('Class', axis=1)
        y_downsampled = df_downsampled['Class']
        
        return train_test_split(X_downsampled, y_downsampled, test_size=0.2, random_state=42)
    except Exception as e:
        raise RuntimeError(f"Error preprocessing data: {e}")

def train_model(X_train, y_train):
    try:
        model = RandomForestClassifier(random_state=42)
        model.fit(X_train, y_train)
        return model
    except Exception as e:
        raise RuntimeError(f"Error training model: {e}")

def evaluate_model(model, X_test, y_test):
    try:
        predictions = model.predict(X_test)
        auc_score = roc_auc_score(y_test, predictions)
        report = classification_report(y_test, predictions, output_dict=True)
        return auc_score, report
    except NotFittedError as e:
        raise RuntimeError(f"Model is not fitted: {e}")
    except Exception as e:
        raise RuntimeError(f"Error evaluating model: {e}")

def log_to_mlflow(model, auc_score, report):
    try:
        mlflow.start_run()
        mlflow.log_param("model_type", "RandomForestClassifier")
        mlflow.log_metric("auc_score", auc_score)
        mlflow.log_metrics(report['1'])
        mlflow.sklearn.log_model(model, "model")
        mlflow.end_run()
    except Exception as e:
        raise RuntimeError(f"Error logging to MLflow: {e}")

def main():
    df = load_data()
    X_train, X_test, y_train, y_test = preprocess_data(df)
    model = train_model(X_train, y_train)
    auc_score, report = evaluate_model(model, X_test, y_test)
    log_to_mlflow(model, auc_score, report)
    joblib.dump(model, "model.pkl")

if __name__ == "__main__":
    main()
```

### GitHub Actions YAML Workflow

```yaml
name: CI/CD Pipeline

on:
  push:
    branches:
      - main
  workflow_dispatch:

jobs:
  build:
    runs-on: ubuntu-latest

    steps:
    - name: Checkout code
      uses: actions/checkout@v2

    - name: Set up Python
      uses: actions/setup-python@v2
      with:
        python-version: '3.8'

    - name: Install dependencies
      run: |
        python -m pip install --upgrade pip
        pip install -r requirements.txt

    - name: Run tests
      run: |
        pytest tests/

    - name: Run pipeline
      run: |
        python pipeline.py

    - name: Upload Model to Artifacts
      uses: actions/upload-artifact@v2
      with:
        name: model
        path: model.pkl
```

### Notes

- Ensure that the `requirements.txt` file includes necessary dependencies such as `scikit-learn`, `pandas`, `numpy`, `mlflow`, `pytest`, and any other dependencies used in your pipeline.
- Replace `'creditcard.csv'` with the actual path to your dataset.
- The `GitHub Actions` workflow triggers on push to the `main` branch and can be manually triggered via `workflow_dispatch`.
- The `pipeline.py` script handles data loading, preprocessing, model training, evaluation, and logging to MLflow.
- The `pytest` test file includes checks for data loading, preprocessing, model training, and evaluation functions.
