class TestDataSplitting:
    """Test suite for data splitting."""
    
    def test_split_data_returns_four_sets(self, sample_data):
        """Test that split_data returns 4 datasets."""
        X, y = preprocess_data(sample_data)
        X_train, X_test, y_train, y_test = split_data(X, y)
        
        assert isinstance(X_train, pd.DataFrame)
        assert isinstance(X_test, pd.DataFrame)
        assert isinstance(y_train, pd.Series)
        assert isinstance(y_test, pd.Series)
    
    def test_split_data_correct_proportions(self, sample_data):
        """Test that data is split in correct proportions."""
        X, y = preprocess_data(sample_data)
        test_size = 0.2
        X_train, X_test, y_train, y_test = split_data(X, y, test_size=test_size)
        
        total_size = len(X)
        expected_test_size = int(total_size * test_size)
        
        # Allow small tolerance due to rounding
        assert abs(len(X_test) - expected_test_size) <= 1
        assert len(X_train) + len(X_test) == total_size
    
    def test_split_data_no_data_leakage(self, sample_data):
        """Test that train and test sets don't overlap."""
        X, y = preprocess_data(sample_data)
        X_train, X_test, y_train, y_test = split_data(X, y)
        
        train_indices = set(X_train.index)
        test_indices = set(X_test.index)
        
        assert len(train_indices.intersection(test_indices)) == 0
    
    def test_split_data_stratification(self, sample_data):
        """Test that stratification maintains class distribution."""
        X, y = preprocess_data(sample_data)
        X_train, X_test, y_train, y_test = split_data(X, y)
        
        # Calculate class proportions
        original_fraud_ratio = (y == 1).sum() / len(y)
        train_fraud_ratio = (y_train == 1).sum() / len(y_train)
        test_fraud_ratio = (y_test == 1).sum() / len(y_test)
        
        # Ratios should be similar (within 5% tolerance)
        assert abs(train_fraud_ratio - original_fraud_ratio) < 0.05
        assert abs(test_fraud_ratio - original_fraud_ratio) < 0.05
    
    def test_split_data_invalid_test_size(self, sample_data):
        """Test error handling for invalid test_size."""
        X, y = preprocess_data(sample_data)
        
        with pytest.raises(ValueError):
            split_data(X, y, test_size=1.5)
        
        with pytest.raises(ValueError):
            split_data(X, y, test_size=-0.1)
    
    def test_split_data_mismatched_lengths(self):
        """Test error when X and y have different lengths."""
        X = pd.DataFrame({'V1': [1, 2, 3]})
        y = pd.Series([0, 1])  # Different length
        
        with pytest.raises(ValueError, match="same length"):
            split_data(X, y)

class TestImbalanceHandling:
    """Test suite for class imbalance handling."""
    
    def test_handle_imbalance_smote(self, sample_data):
        """Test SMOTE oversampling."""
        X, y = preprocess_data(sample_data)
        X_train, X_test, y_train, y_test = split_data(X, y)
        
        X_balanced, y_balanced = handle_imbalance(X_train, y_train, strategy="smote")
        
        # After SMOTE, classes should be more balanced
        class_counts = y_balanced.value_counts()
        assert len(class_counts) == 2
        # Classes should be equal or nearly equal
        assert abs(class_counts[0] - class_counts[1]) < len(y_balanced) * 0.1
    
    def test_handle_imbalance_undersample(self, sample_data):
        """Test random undersampling."""
        X, y = preprocess_data(sample_data)
        X_train, X_test, y_train, y_test = split_data(X, y)
        
        X_balanced, y_balanced = handle_imbalance(X_train, y_train, strategy="undersample")
        
        # After undersampling, dataset should be smaller
        assert len(y_balanced) <= len(y_train)
        
        # Classes should be balanced
        class_counts = y_balanced.value_counts()
        assert abs(class_counts[0] - class_counts[1]) < len(y_balanced) * 0.1
    
    def test_handle_imbalance_smote_tomek(self, sample_data):
        """Test SMOTE + Tomek links."""
        X, y = preprocess_data(sample_data)
        X_train, X_test, y_train, y_test = split_data(X, y)
        
        X_balanced, y_balanced = handle_imbalance(X_train, y_train, strategy="smote_tomek")
        
        # Should return balanced data
        assert len(X_balanced) == len(y_balanced)
        assert isinstance(X_balanced, pd.DataFrame)
        assert isinstance(y_balanced, pd.Series)
    
    def test_handle_imbalance_none(self, sample_data):
        """Test no balancing strategy."""
        X, y = preprocess_data(sample_data)
        X_train, X_test, y_train, y_test = split_data(X, y)
        
        X_result, y_result = handle_imbalance(X_train, y_train, strategy="none")
        
        # Data should remain unchanged
        assert len(X_result) == len(X_train)
        assert len(y_result) == len(y_train)
    
    def test_handle_imbalance_invalid_strategy(self, sample_data):
        """Test error handling for invalid strategy."""
        X, y = preprocess_data(sample_data)
        X_train, X_test, y_train, y_test = split_data(X, y)
        
        with pytest.raises(ValueError, match="Unknown balancing strategy"):
            handle_imbalance(X_train, y_train, strategy="invalid_strategy")
    
    def test_handle_imbalance_preserves_features(self, sample_data):
        """Test that feature columns are preserved after balancing."""
        X, y = preprocess_data(sample_data)
        X_train, X_test, y_train, y_test = split_data(X, y)
        
        original_columns = X_train.columns.tolist()
        X_balanced, y_balanced = handle_imbalance(X_train, y_train, strategy="smote")
        
        assert X_balanced.columns.tolist() == original_columns

class TestModelEvaluation:
    """Test suite for model evaluation."""
    
    def test_evaluate_model_returns_metrics_dict(self, trained_model, sample_data):
        """Test that evaluate_model returns a dictionary of metrics."""
        X, y = preprocess_data(sample_data)
        X_train, X_test, y_train, y_test = split_data(X, y)
        
        metrics = evaluate_model(trained_model, X_test, y_test)
        
        assert isinstance(metrics, dict)
        assert len(metrics) > 0
    
    def test_evaluate_model_contains_required_metrics(self, trained_model, sample_data):
        """Test that all required metrics are present."""
        X, y = preprocess_data(sample_data)
        X_train, X_test, y_train, y_test = split_data(X, y)
        
        metrics = evaluate_model(trained_model, X_test, y_test)
        
        required_metrics = [
            'accuracy', 'precision_overall', 'recall_overall', 'f1_overall',
            'precision_class_0', 'recall_class_0', 'f1_class_0',
            'precision_class_1', 'recall_class_1', 'f1_class_1',
            'roc_auc', 'pr_auc', 'confusion_matrix'
        ]
        
        for metric in required_metrics:
            assert metric in metrics
    
    def test_evaluate_model_confusion_matrix_structure(self, trained_model, sample_data):
        """Test confusion matrix has correct structure."""
        X, y = preprocess_data(sample_data)
        X_train, X_test, y_train, y_test = split_data(X, y)
        
        metrics = evaluate_model(trained_model, X_test, y_test)
        cm = metrics['confusion_matrix']
        
        assert 'true_negative' in cm
        assert 'false_positive' in cm
        assert 'false_negative' in cm
        assert 'true_positive' in cm
        
        # All values should be non-negative integers
        for value in cm.values():
            assert isinstance(value, int)
            assert value >= 0
    
    def test_evaluate_model_metric_ranges(self, trained_model, sample_data):
        """Test that metrics are within valid ranges."""
        X, y = preprocess_data(sample_data)
        X_train, X_test, y_train, y_test = split_data(X, y)
        
        metrics = evaluate_model(trained_model, X_test, y_test)
        
        # Metrics should be between 0 and 1
        for key, value in metrics.items():
            if key != 'confusion_matrix' and not key.startswith('support'):
                assert 0 <= value <= 1, f"{key} = {value} is out of range [0, 1]"
    
    def test_evaluate_model_support_counts(self, trained_model, sample_data):
        """Test that support counts are correct."""
        X, y = preprocess_data(sample_data)
        X_train, X_test, y_train, y_test = split_data(X, y)
        
        metrics = evaluate_model(trained_model, X_test, y_test)
        
        assert 'support_class_0' in metrics
        assert 'support_class_1' in metrics
        assert metrics['support_class_0'] + metrics['support_class_1'] == len(y_test)
