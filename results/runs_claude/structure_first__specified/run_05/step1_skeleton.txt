## GitHub Actions Workflow (`..github/workflows/ml_pipeline.yml`)

```yaml
# Outline structure (not complete YAML yet)

name: ML Pipeline CI/CD

on:
  push:
    branches: [main, develop]
  pull_request:
    branches: [main]
  schedule:
    - cron: "0 2 * * *"  # Daily drift check
  workflow_dispatch:
    inputs:
      force_retrain:
        description: 'Force model retraining'
        required: false
        default: 'false'

env:
  PYTHON_VERSION: "3.9"
  MLFLOW_TRACKING_URI: ${{ secrets.MLFLOW_TRACKING_URI }}

jobs:
  # Job 1: Code Quality and Tests
  test:
    steps:
      - Checkout code
      - Setup Python
      - Cache dependencies
      - Install dependencies
      - Run linting (flake8, black)
      - Run type checking (mypy)
      - Run unit tests with pytest
      - Upload coverage reports

  # Job 2: Drift Detection
  drift-detection:
    needs: [test]
    steps:
      - Checkout code
      - Setup Python
      - Install dependencies
      - Download latest production data
      - Load reference statistics
      - Run drift detection
      - Set output variable (drift_detected)
      - Upload drift report

  # Job 3: Model Training (conditional on drift or manual trigger)
  train:
    needs: [drift-detection]
    if: needs.drift-detection.outputs.drift_detected == 'true' || github.event.inputs.force_retrain == 'true'
    steps:
      - Checkout code
      - Setup Python
      - Install dependencies
      - Download training data
      - Run training pipeline
      - Log to MLflow
      - Save model artifacts
      - Upload artifacts

  # Job 4: Model Evaluation and Validation
  evaluate:
    needs: [train]
    steps:
      - Checkout code
      - Setup Python
      - Download model artifacts
      - Download test data
      - Run model evaluation
      - Validate performance thresholds
      - Generate evaluation report
      - Upload report

  # Job 5: Model Registration (if evaluation passes)
  register:
    needs: [evaluate]
    steps:
      - Checkout code
      - Setup Python
      - Download model artifacts
      - Register model in MLflow Model Registry
      - Tag model version
      - Transition to staging

  # Job 6: Deploy to Staging
  deploy-staging:
    needs: [register]
    environment: staging
    steps:
      - Checkout code
      - Download registered model
      - Deploy to staging endpoint
      - Run smoke tests
      - Update deployment status

  # Job 7: Deploy to Production (manual approval)
  deploy-production:
    needs: [deploy-staging]
    environment: production
    if: github.ref == 'refs/heads/main'
    steps:
      - Checkout code
      - Download registered model
      - Deploy to production endpoint
      - Run smoke tests
      - Update production statistics
      - Send notification
```
