## GitHub Actions Workflow Structure

### `.github/workflows/ml_pipeline.yml`

```yaml
name: Fraud Detection ML Pipeline

on:
  # Trigger events
  push:
    branches: [main, develop]
  pull_request:
    branches: [main]
  schedule:
    - cron: '0 2 * * *'  # Daily at 2 AM
  workflow_dispatch:
    inputs:
      mode:
        description: 'Pipeline mode'
        required: true
        default: 'train'
        type: choice
        options:
          - train
          - evaluate
          - drift_check

env:
  # Environment variables

jobs:
  # Job 1: Code Quality & Linting
  code-quality:
    name: Code Quality Checks
    runs-on: ubuntu-latest
    steps:
      - name: Checkout code
      - name: Set up Python
      - name: Install dependencies
      - name: Run black formatter check
      - name: Run flake8 linter
      - name: Run mypy type checker
      - name: Run bandit security check

  # Job 2: Unit Tests
  unit-tests:
    name: Unit Tests
    runs-on: ubuntu-latest
    needs: code-quality
    steps:
      - name: Checkout code
      - name: Set up Python
      - name: Install dependencies
      - name: Run pytest with coverage
      - name: Upload coverage reports
      - name: Check coverage threshold

  # Job 3: Data Validation
  data-validation:
    name: Data Validation
    runs-on: ubuntu-latest
    needs: unit-tests
    steps:
      - name: Checkout code
      - name: Set up Python
      - name: Install dependencies
      - name: Download dataset
      - name: Validate data schema
      - name: Check data quality
      - name: Generate data profile report

  # Job 4: Drift Detection
  drift-detection:
    name: Concept Drift Detection
    runs-on: ubuntu-latest
    needs: data-validation
    if: github.event_name == 'schedule' || github.event.inputs.mode == 'drift_check'
    steps:
      - name: Checkout code
      - name: Set up Python
      - name: Install dependencies
      - name: Download reference data
      - name: Download current data
      - name: Load production model
      - name: Run drift detection
      - name: Parse drift results
      - name: Set retraining flag

  # Job 5: Model Training
  model-training:
    name: Model Training
    runs-on: ubuntu-latest
    needs: [data-validation, drift-detection]
    if: |
      always() &&
      (needs.data-validation.result == 'success') &&
      (needs.drift-detection.result == 'success' || needs.drift-detection.result == 'skipped')
    steps:
      - name: Checkout code
      - name: Set up Python
      - name: Install dependencies
      - name: Configure MLflow
      - name: Download training data
      - name: Run training pipeline
      - name: Save model artifacts
      - name: Upload model to artifacts

  # Job 6: Model Evaluation
  model-evaluation:
    name: Model Evaluation
    runs-on: ubuntu-latest
    needs: model-training
    steps:
      - name: Checkout code
      - name: Set up Python
      - name: Install dependencies
      - name: Download model artifacts
      - name: Download test data
      - name: Run evaluation pipeline
      - name: Generate evaluation report
      - name: Check performance thresholds
      - name: Upload evaluation results

  # Job 7: Model Registration
  model-registration:
    name: Model Registration
    runs-on: ubuntu-latest
    needs: model-evaluation
    if: github.ref == 'refs/heads/main'
    steps:
      - name: Checkout code
      - name: Set up Python
      - name: Install dependencies
      - name: Configure MLflow
      - name: Download model artifacts
      - name: Register model in MLflow
      - name: Transition model to staging
      - name: Tag model version

  # Job 8: Integration Tests
  integration-tests:
    name: Integration Tests
    runs-on: ubuntu-latest
    needs: model-registration
    steps:
      - name: Checkout code
      - name: Set up Python
      - name: Install dependencies
      - name: Load registered model
      - name: Run end-to-end tests
      - name: Test API endpoints
      - name: Validate predictions

  # Job 9: Deploy to Staging
  deploy-staging:
    name: Deploy to Staging
    runs-on: ubuntu-latest
    needs: integration-tests
    if: github.ref == 'refs/heads/main'
    steps:
      - name: Checkout code
      - name: Configure AWS credentials
      - name: Download model artifacts
      - name: Deploy to staging environment
      - name: Run smoke tests
      - name: Update deployment status

  # Job 10: Notification
  notification:
    name: Send Notifications
    runs-on: ubuntu-latest
    needs: [deploy-staging]
    if: always()
    steps:
      - name: Determine workflow status
      - name: Send Slack notification
      - name: Update monitoring dashboard
```
