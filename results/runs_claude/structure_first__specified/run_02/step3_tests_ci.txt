            test_size=0.2,
            random_state=42
        )
        
        # Check shapes
        assert len(X_train) + len(X_test) == len(sample_fraud_data)
        assert len(y_train) == len(X_train)
        assert len(y_test) == len(X_test)
        
        # Check train/test split ratio
        assert abs(len(X_test) / len(sample_fraud_data) - 0.2) < 0.01

        train_fraud_rate = y_train.sum() / len(y_train)

    def test_balance_dataset_smote(self, sample_fraud_data):
        """Test SMOTE balancing strategy."""
        X = sample_fraud_data.drop(columns=['Class']).values[:, :10]
        y = sample_fraud_data['Class'].values
        
        X_balanced, y_balanced = balance_dataset(X, y, strategy='smote')
        
        # Check classes are balanced
        class_counts = np.bincount(y_balanced)
        assert abs(class_counts[0] - class_counts[1]) < 10
        
        # Check dataset size increased
        assert len(y_balanced) > len(y)
    
    def test_balance_dataset_undersample(self, sample_fraud_data):
        """Test undersampling strategy."""
        X = sample_fraud_data.drop(columns=['Class']).values[:, :10]
        y = sample_fraud_data['Class'].values
        
        X_balanced, y_balanced = balance_dataset(X, y, strategy='undersample')
        
        # Check classes are balanced
        class_counts = np.bincount(y_balanced)
        assert abs(class_counts[0] - class_counts[1]) < 10
        
        # Check dataset size decreased
        assert len(y_balanced) < len(y)
    
    def test_balance_dataset_invalid_strategy(self, sample_fraud_data):

        y = sample_fraud_data['Class'].values

        model.train(X_train, y_train)

    def test_model_predict_before_training(self):
        """Test prediction fails before training."""
        model = FraudDetectionModel({'n_estimators': 10})
        X = np.random.rand(10, 5)
        
        with pytest.raises(ValueError, match="Model must be trained"):
            model.predict(X)
    
    def test_model_predict(self, trained_model):
        """Test model prediction."""
        model, X_test, y_test, _ = trained_model
        predictions = model.predict(X_test)
        
        assert len(predictions) == len(X_test)
        assert set(predictions).issubset({0, 1})
    
    def test_model_predict_proba(self, trained_model):
        """Test model probability prediction."""
        model, X_test, y_test, _ = trained_model
        probabilities = model.predict_proba(X_test)
        
        assert probabilities.shape == (len(X_test), 2)
        assert np.allclose(probabilities.sum(axis=1), 1.0)
        assert np.all(probabilities >= 0) and np.all(probabilities <= 1)
    
    def test_model_save_and_load(self, trained_model, temp_model_dir):
        """Test model serialization and deserialization."""
        model, X_test, y_test, _ = trained_model
        model_path = temp_model_dir / "test_model.pkl"
        
        # Save model
        model.save(str(model_path))
        assert model_path.exists()
        
        # Load model
        loaded_model = FraudDetectionModel.load(str(model_path))
        assert loaded_model.is_trained is True
        assert loaded_model.config == model.config
        
        # Test predictions are the same
        original_pred = model.predict(X_test)
        loaded_pred = loaded_model.predict(X_test)
        assert np.array_equal(original_pred, loaded_pred)

        metrics = evaluate_model(model, X_test, y_test, threshold=0.5)

        assert 'recall' in metrics
        assert 'f1_score' in metrics
        assert 'roc_auc' in metrics
        assert 'confusion_matrix' in metrics
        
        # Check metric values are valid
        assert 0 <= metrics['accuracy'] <= 1
        assert 0 <= metrics['precision'] <= 1
        assert 0 <= metrics['recall'] <= 1
        assert 0 <= metrics['f1_score'] <= 1
        assert 0 <= metrics['roc_auc'] <= 1

class TestDriftDetection:
    """Test drift detection functionality."""
    
    def test_drift_detector_initialization(self):
        """Test drift detector initialization."""
        reference_data = np.random.randn(100, 10)
        config = {
            'ks_test_threshold': 0.05,
            'psi_threshold': 0.2,
            'min_drift_features': 3
        }
        
        detector = DriftDetector(reference_data, config)
        assert detector.ks_threshold == 0.05
        assert detector.psi_threshold == 0.2
        assert detector.min_drift_features == 3
    
    def test_detect_feature_drift_no_drift(self):
        """Test feature drift detection when no drift exists."""
        reference_data = np.random.randn(1000, 10)
        current_data = np.random.randn(1000, 10)
        
        config = {'ks_test_threshold': 0.05, 'psi_threshold': 0.2, 'min_drift_features': 3}
        detector = DriftDetector(reference_data, config)
        
        drift_results = detector.detect_feature_drift(current_data)
        
        assert 'overall_drift_detected' in drift_results
        assert 'n_drifted_features' in drift_results
        assert drift_results['total_features'] == 10
        # With random data, should have minimal drift
        assert drift_results['n_drifted_features'] < 5
    
    def test_detect_feature_drift_with_drift(self):
        """Test feature drift detection when drift exists."""
        reference_data = np.random.randn(1000, 10)
        # Create drifted data by shifting mean
        current_data = np.random.randn(1000, 10) + 2.0
        
        config = {'ks_test_threshold': 0.05, 'psi_threshold': 0.2, 'min_drift_features': 3}
        detector = DriftDetector(reference_data, config)
        
        drift_results = detector.detect_feature_drift(current_data)
        
        assert drift_results['overall_drift_detected'] is True
        assert drift_results['n_drifted_features'] >= 3
    
    def test_detect_feature_drift_dimension_mismatch(self):
        """Test error handling for dimension mismatch."""
        reference_data = np.random.randn(100, 10)
        current_data = np.random.randn(100, 5)
        
        config = {'ks_test_threshold': 0.05, 'psi_threshold': 0.2, 'min_drift_features': 3}
        detector = DriftDetector(reference_data, config)
        
        with pytest.raises(ValueError, match="Feature dimension mismatch"):
            detector.detect_feature_drift(current_data)
    
    def test_detect_prediction_drift_no_drift(self):
        """Test prediction drift detection when no drift exists."""
        reference_data = np.random.randn(1000, 10)
        reference_predictions = np.random.rand(1000)
        current_predictions = np.random.rand(1000)
        
        config = {'ks_test_threshold': 0.05, 'psi_threshold': 0.2}
        detector = DriftDetector(reference_data, config)
        
        drift_results = detector.detect_prediction_drift(
            reference_predictions, current_predictions
        )
        
        assert 'drift_detected' in drift_results
        assert 'psi' in drift_results
        assert 'kl_divergence' in drift_results
    
    def test_detect_prediction_drift_with_drift(self):
        """Test prediction drift detection when drift exists."""
        reference_data = np.random.randn(1000, 10)
        reference_predictions = np.random.rand(1000) * 0.3  # Low predictions
        current_predictions = np.random.rand(1000) * 0.7 + 0.3  # High predictions
        
        config = {'ks_test_threshold': 0.05, 'psi_threshold': 0.2}
        detector = DriftDetector(reference_data, config)
        
        drift_results = detector.detect_prediction_drift(
            reference_predictions, current_predictions
        )
        
        assert drift_results['drift_detected'] is True
        assert drift_results['mean_shift'] > 0.2
    
    def test_detect_performance_drift_no_degradation(self):
        """Test performance drift when no degradation exists."""
        reference_data = np.random.randn(100, 10)
        reference_metrics = {
            'precision': 0.85,
            'recall': 0.80,
            'f1_score': 0.82,
            'roc_auc': 0.90
        }
        current_metrics = {
            'precision': 0.84,
            'recall': 0.81,
            'f1_score': 0.82,
            'roc_auc': 0.89
        }
        
        config = {'performance_degradation_threshold': 0.05}
        detector = DriftDetector(reference_data, config)
        
        drift_results = detector.detect_performance_drift(
            reference_metrics, current_metrics, threshold=0.05
        )
        
        assert drift_results['overall_degradation'] is False
    
    def test_detect_performance_drift_with_degradation(self):
        """Test performance drift when degradation exists."""
        reference_data = np.random.randn(100, 10)
        reference_metrics = {
            'precision': 0.85,
            'recall': 0.80,
            'f1_score': 0.82,
            'roc_auc': 0.90
        }
        current_metrics = {
            'precision': 0.70,  # Significant drop
            'recall': 0.65,     # Significant drop
            'f1_score': 0.67,
            'roc_auc': 0.75
        }
        
        config = {'performance_degradation_threshold': 0.05}
        detector = DriftDetector(reference_data, config)
        
        drift_results = detector.detect_performance_drift(
            reference_metrics, current_metrics, threshold=0.05
        )
        
        assert drift_results['overall_degradation'] is True
        assert drift_results['precision_degraded'] is True
    
    def test_should_retrain_no_drift(self):
        """Test retraining decision when no drift detected."""
        reference_data = np.random.randn(100, 10)
        config = {}
        detector = DriftDetector(reference_data, config)
        
        drift_results = {
            'feature_drift': {'overall_drift_detected': False},
            'prediction_drift': {'drift_detected': False},
            'performance_drift': {'overall_degradation': False}
        }
        
        should_retrain = detector.should_retrain(drift_results)
        assert should_retrain is False
    
    def test_should_retrain_with_drift(self):
        """Test retraining decision when drift detected."""
        reference_data = np.random.randn(100, 10)
        config = {}
        detector = DriftDetector(reference_data, config)
        
        drift_results = {
            'feature_drift': {'overall_drift_detected': True},
            'prediction_drift': {'drift_detected': False},
            'performance_drift': {'overall_degradation': False}
        }
        
        should_retrain = detector.should_retrain(drift_results)
        assert should_retrain is True
    
    def test_calculate_psi(self):
        """Test PSI calculation."""
        reference = np.random.randn(1000)
        current = np.random.randn(1000)
        
        psi = calculate_psi(reference, current)
        
        assert isinstance(psi, float)
        assert psi >= 0
        # PSI should be low for similar distributions
        assert psi < 0.5
    
    def test_calculate_psi_with_drift(self):
        """Test PSI calculation with drifted data."""
        reference = np.random.randn(1000)
        current = np.random.randn(1000) + 2.0  # Shifted distribution
        
        psi = calculate_psi(reference, current)
        
        # PSI should be higher for different distributions

    def test_calculate_kl_divergence(self):
        """Test KL divergence calculation."""

        current = np.random.randn(1000)
        
        kl = calculate_kl_divergence(reference, current)
