## GitHub Actions Workflow (`.github/workflows/ml_pipeline.yml`)

```yaml
# ML Pipeline CI/CD Workflow Structure

name: Fraud Detection ML Pipeline

on:
  # Triggers...

env:
  # Environment variables...

jobs:
  
  # Job 1: Code Quality and Testing
  test-and-lint:
    runs-on: ubuntu-latest
    steps:
      - Checkout code
      - Setup Python
      - Cache dependencies
      - Install dependencies
      - Run linting (flake8, black)
      - Run type checking (mypy)
      - Run unit tests with coverage
      - Upload coverage reports
  
  # Job 2: Drift Detection
  detect-drift:
    runs-on: ubuntu-latest
    needs: test-and-lint
    outputs:
      should_retrain: # output flag
    steps:
      - Checkout code
      - Setup Python
      - Install dependencies
      - Download reference data
      - Download current production data
      - Run drift detection script
      - Set output flag for retraining
      - Upload drift report as artifact
  
  # Job 3: Model Training (conditional)
  train-model:
    runs-on: ubuntu-latest
    needs: detect-drift
    if: needs.detect-drift.outputs.should_retrain == 'true' || github.event_name == 'workflow_dispatch'
    steps:
      - Checkout code
      - Setup Python
      - Install dependencies
      - Configure MLflow
      - Download training data
      - Run training pipeline
      - Log metrics to MLflow
      - Save model artifacts
      - Upload model artifacts
  
  # Job 4: Model Evaluation
  evaluate-model:
    runs-on: ubuntu-latest
    needs: train-model
    steps:
      - Checkout code
      - Setup Python
      - Install dependencies
      - Download model artifacts
      - Download test data
      - Run evaluation script
      - Generate evaluation report
      - Check performance thresholds
      - Upload evaluation report
  
  # Job 5: Model Registration (conditional)
  register-model:
    runs-on: ubuntu-latest
    needs: evaluate-model
    if: success()
    steps:
      - Checkout code
      - Setup Python
      - Install dependencies
      - Configure MLflow
      - Download model artifacts
      - Register model to MLflow Registry
      - Tag model version
      - Transition to Staging
  
  # Job 6: Deploy to Staging
  deploy-staging:
    runs-on: ubuntu-latest
    needs: register-model
    environment: staging
    steps:
      - Checkout code
      - Setup deployment tools
      - Pull model from MLflow
      - Build Docker image
      - Push to container registry
      - Deploy to staging environment
      - Run smoke tests
      - Update deployment status
  
  # Job 7: Deploy to Production (manual approval)
  deploy-production:
    runs-on: ubuntu-latest
    needs: deploy-staging
    environment: production
    if: github.ref == 'refs/heads/main'
    steps:
      - Checkout code
      - Setup deployment tools
      - Pull model from MLflow (Production stage)
      - Build Docker image
      - Push to container registry
      - Deploy to production with blue-green strategy
      - Run integration tests
      - Monitor initial metrics
      - Update deployment status
  
  # Job 8: Notify
  notify:
    runs-on: ubuntu-latest
    needs: [test-and-lint, detect-drift, train-model, evaluate-model, register-model, deploy-staging, deploy-production]
    if: always()
    steps:
      - Send Slack notification
      - Send email notification
      - Update dashboard
```
