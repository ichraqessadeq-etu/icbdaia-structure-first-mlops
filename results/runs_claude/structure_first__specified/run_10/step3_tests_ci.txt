class TestPreprocessData:
    """Test suite for data preprocessing."""
    
    def test_preprocess_returns_four_arrays(self, sample_fraud_data):
        """Test that preprocessing returns X_train, X_test, y_train, y_test."""
        result = preprocess_data(sample_fraud_data)
        assert len(result) == 4
    
    def test_preprocess_train_test_split(self, sample_fraud_data):
        """Test train/test split proportions."""
        X_train, X_test, y_train, y_test = preprocess_data(
            sample_fraud_data, test_size=0.2
        )
        
        total_samples = len(sample_fraud_data)
        assert len(X_train) == int(total_samples * 0.8)
        assert len(X_test) == int(total_samples * 0.2)
    
    def test_preprocess_array_types(self, sample_fraud_data):
        """Test that output arrays are numpy arrays."""
        X_train, X_test, y_train, y_test = preprocess_data(sample_fraud_data)
        
        assert isinstance(X_train, np.ndarray)
        assert isinstance(X_test, np.ndarray)
        assert isinstance(y_train, np.ndarray)
        assert isinstance(y_test, np.ndarray)
    
    def test_preprocess_feature_count(self, sample_fraud_data):
        """Test that features are correctly separated from target."""
        X_train, X_test, y_train, y_test = preprocess_data(sample_fraud_data)
        
        # Should have 30 features (all columns except Class)
        assert X_train.shape[1] == 30
        assert X_test.shape[1] == 30
    
    def test_preprocess_stratification(self, sample_fraud_data):
        """Test that stratification maintains class distribution."""
        X_train, X_test, y_train, y_test = preprocess_data(sample_fraud_data)
        
        original_fraud_rate = sample_fraud_data["Class"].mean()
        train_fraud_rate = y_train.mean()
        test_fraud_rate = y_test.mean()
        
        # Fraud rates should be similar (within 5% tolerance)
        assert abs(train_fraud_rate - original_fraud_rate) < 0.05
        assert abs(test_fraud_rate - original_fraud_rate) < 0.05
    
    def test_preprocess_custom_test_size(self, sample_fraud_data):
        """Test custom test size parameter."""
        X_train, X_test, y_train, y_test = preprocess_data(
            sample_fraud_data, test_size=0.3
        )
        
        total = len(X_train) + len(X_test)
        test_ratio = len(X_test) / total
        
        assert abs(test_ratio - 0.3) < 0.01
    
    def test_preprocess_reproducibility(self, sample_fraud_data):
        """Test that same random_state produces same split."""
        result1 = preprocess_data(sample_fraud_data, random_state=42)
        result2 = preprocess_data(sample_fraud_data, random_state=42)
        
        np.testing.assert_array_equal(result1[0], result2[0])  # X_train
        np.testing.assert_array_equal(result1[2], result2[2])  # y_train

class TestEvaluateModel:
    """Test suite for model evaluation."""
    
    def test_evaluate_returns_dict(self, trained_model, train_test_data):
        """Test that evaluation returns a dictionary."""
        X_train, X_test, y_train, y_test = train_test_data
        
        metrics = evaluate_model(trained_model, X_test, y_test)
        
        assert isinstance(metrics, dict)
    
    def test_evaluate_required_metrics(self, trained_model, train_test_data):
        """Test that all required metrics are present."""
        X_train, X_test, y_train, y_test = train_test_data
        
        metrics = evaluate_model(trained_model, X_test, y_test)
        
        required_metrics = [
            "accuracy", "precision", "recall", "f1_score", "roc_auc",
            "specificity", "true_positives", "true_negatives",
            "false_positives", "false_negatives"
        ]
        
        for metric in required_metrics:
            assert metric in metrics
    
    def test_evaluate_metric_ranges(self, trained_model, train_test_data):
        """Test that metrics are within valid ranges."""
        X_train, X_test, y_train, y_test = train_test_data
        
        metrics = evaluate_model(trained_model, X_test, y_test)
        
        # Percentage metrics should be between 0 and 1
        for metric in ["accuracy", "precision", "recall", "f1_score", "roc_auc", "specificity"]:
            assert 0 <= metrics[metric] <= 1
    
    def test_evaluate_confusion_matrix_sum(self, trained_model, train_test_data):
        """Test that confusion matrix values sum to total samples."""
        X_train, X_test, y_train, y_test = train_test_data
        
        metrics = evaluate_model(trained_model, X_test, y_test)
        
        total = (
            metrics["true_positives"] +
            metrics["true_negatives"] +
            metrics["false_positives"] +
            metrics["false_negatives"]
        )
        
        assert total == len(y_test)
    
    def test_evaluate_metric_types(self, trained_model, train_test_data):
        """Test that metrics have correct data types."""
        X_train, X_test, y_train, y_test = train_test_data
        
        metrics = evaluate_model(trained_model, X_test, y_test)
        
        float_metrics = ["accuracy", "precision", "recall", "f1_score", "roc_auc", "specificity"]
        int_metrics = ["true_positives", "true_negatives", "false_positives", "false_negatives"]
        
        for metric in float_metrics:
            assert isinstance(metrics[metric], float)
        
        for metric in int_metrics:
            assert isinstance(metrics[metric], int)
    
    def test_evaluate_perfect_predictions(self, train_test_data):
        """Test evaluation with perfect predictions."""
        X_train, X_test, y_train, y_test = train_test_data
        
        # Create a mock model that predicts perfectly

        assert metrics["accuracy"] == 1.0
        assert metrics["false_positives"] == 0
        assert metrics["false_negatives"] == 0

  # Job 6: Model Validation and Testing
  # ============================================================================
  validate-model:
    name: Validate Trained Model
    runs-on: ubuntu-latest
    needs: [train-model]
    
    steps:
      - name: Checkout code
        uses: actions/checkout@v3
      
      - name: Set up Python
        uses: actions/setup-python@v4
        with:
          python-version: ${{ env.PYTHON_VERSION }}
      
      - name: Install dependencies
        run: |
          python -m pip install --upgrade pip
          if [ -f requirements.txt ]; then pip install -r requirements.txt; fi
      
      - name: Download model artifacts
        uses: actions/download-artifact@v3
        with:
          name: trained-model
          path: models/
      
      - name: Load and test model
        run: |
          python -c "
          import pickle
          import numpy as np
          import json
          
          # Load model
          with open('models/fraud_model.pkl', 'rb') as f:
              model = pickle.load(f)
          
          # Load metadata
          with open('models/fraud_model_metadata.json', 'r') as f:
              metadata = json.load(f)
          
          # Test prediction
          n_features = metadata['n_features']
          test_input = np.random.randn(1, n_features)
          prediction = model.predict(test_input)
          proba = model.predict_proba(test_input)
          
          print(f'Test prediction: {prediction[0]}')
          print(f'Prediction probabilities: {proba[0]}')
          print('✓ Model loaded and inference successful')
          "
      
      - name: Run model performance tests
        run: |
          python -c "
          import json
          
          with open('models/fraud_model_metadata.json', 'r') as f:
              metadata = json.load(f)
          
          metrics = metadata['metrics']
          
          # Check for model degradation
          tests = [
              ('Recall >= 0.50', metrics['recall'] >= 0.50),
              ('Precision >= 0.50', metrics['precision'] >= 0.50),
              ('ROC-AUC >= 0.70', metrics['roc_auc'] >= 0.70),
              ('F1-Score >= 0.50', metrics['f1_score'] >= 0.50),
          ]
          
          all_passed = True
          for test_name, result in tests:
              status = '✓' if result else '✗'
              print(f'{status} {test_name}')
              if not result:
                  all_passed = False
          
          if not all_passed:
              print('✗ Some performance tests failed')
              exit(1)
          else:
              print('✓ All performance tests passed')
          "

          name: trained-model
          path: models/
      
      - name: Run inference benchmarks
        run: |
          python -c "
          import pickle
          import numpy as np
          import time
          
          # Load model
          with open('models/fraud_model.pkl', 'rb') as f:
              model = pickle.load(f)
          
          # Generate test data
          n_samples = 10000
          n_features = 30
          X_test = np.random.randn(n_samples, n_features)
          
          # Benchmark single prediction
          start = time.time()
          for _ in range(100):
              _ = model.predict(X_test[0:1])
          single_time = (time.time() - start) / 100
          
          # Benchmark batch prediction
          start = time.time()
          _ = model.predict(X_test)
          batch_time = time.time() - start
          
          print(f'Single prediction time: {single_time*1000:.2f} ms')
          print(f'Batch prediction time ({n_samples} samples): {batch_time:.2f} s')
          print(f'Throughput: {n_samples/batch_time:.0f} predictions/sec')
          
          # Performance assertions
          assert single_time < 0.1, 'Single prediction too slow'
          assert batch_time < 5.0, 'Batch prediction too slow'
          print('✓ Performance benchmarks passed')
          "
