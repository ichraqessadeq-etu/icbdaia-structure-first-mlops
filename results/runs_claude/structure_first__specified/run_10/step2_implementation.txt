```python
"""Main ML pipeline for fraud detection model training and evaluation."""

import logging
import os
import pickle
from typing import Tuple, Dict, Any
import pandas as pd
import numpy as np
from sklearn.ensemble import RandomForestClassifier
from sklearn.model_selection import train_test_split
from sklearn.metrics import (
    precision_recall_fscore_support,
    roc_auc_score,
    accuracy_score,
    confusion_matrix,
)
from sklearn.utils.class_weight import compute_class_weight
import mlflow
import mlflow.sklearn
import json
from datetime import datetime


def setup_logging(log_level: str = "INFO") -> logging.Logger:
    """Configure and return logger for pipeline execution."""
    logger = logging.getLogger("fraud_detection_pipeline")
    logger.setLevel(getattr(logging, log_level.upper()))
    
    # Remove existing handlers to avoid duplicates
    logger.handlers = []
    
    # Console handler
    console_handler = logging.StreamHandler()
    console_handler.setLevel(getattr(logging, log_level.upper()))
    
    # Formatter
    formatter = logging.Formatter(
        "%(asctime)s - %(name)s - %(levelname)s - %(message)s",
        datefmt="%Y-%m-%d %H:%M:%S",
    )
    console_handler.setFormatter(formatter)
    
    logger.addHandler(console_handler)
    
    return logger


def load_data(data_path: str) -> pd.DataFrame:
    """Load credit card fraud dataset from specified path."""
    logger = logging.getLogger("fraud_detection_pipeline")
    
    if not os.path.exists(data_path):
        logger.error(f"Data file not found: {data_path}")
        raise FileNotFoundError(f"Data file not found: {data_path}")
    
    try:
        logger.info(f"Loading data from {data_path}")
        df = pd.read_csv(data_path)
        logger.info(f"Successfully loaded {len(df)} records with {len(df.columns)} columns")
        return df
    except Exception as e:
        logger.error(f"Error loading data: {str(e)}")
        raise


def validate_data(df: pd.DataFrame) -> None:
    """Validate data schema, types, and basic quality checks."""
    logger = logging.getLogger("fraud_detection_pipeline")
    logger.info("Validating data schema and quality")
    
    # Check if dataframe is empty
    if df.empty:
        raise ValueError("Dataset is empty")
    
    # Check for required target column
    if "Class" not in df.columns:
        raise ValueError("Required column 'Class' not found in dataset")
    
    # Check that Class column contains binary values
    unique_classes = df["Class"].unique()
    if not set(unique_classes).issubset({0, 1}):
        raise ValueError(f"Class column must contain only 0 and 1, found: {unique_classes}")
    
    # Check for feature columns (should have V1-V28, Time, Amount)
    expected_features = ["Time", "Amount"] + [f"V{i}" for i in range(1, 29)]
    missing_features = [col for col in expected_features if col not in df.columns]
    
    if missing_features:
        raise ValueError(f"Missing expected feature columns: {missing_features}")
    
    # Check data types - all features should be numeric
    non_numeric_cols = df.select_dtypes(exclude=[np.number]).columns.tolist()
    if non_numeric_cols:
        raise ValueError(f"Non-numeric columns found: {non_numeric_cols}")
    
    # Check for excessive missing values
    missing_percentage = (df.isnull().sum() / len(df)) * 100
    high_missing_cols = missing_percentage[missing_percentage > 50].index.tolist()
    
    if high_missing_cols:
        logger.warning(f"Columns with >50% missing values: {high_missing_cols}")
    
    # Check for minimum dataset size
    if len(df) < 100:
        raise ValueError(f"Dataset too small: {len(df)} records (minimum 100 required)")
    
    # Check class distribution
    fraud_rate = df["Class"].mean()
    logger.info(f"Fraud rate: {fraud_rate:.4f} ({df['Class'].sum()} frauds out of {len(df)} transactions)")
    
    if fraud_rate == 0 or fraud_rate == 1:
        raise ValueError("Dataset contains only one class")
    
    logger.info("Data validation passed")


def preprocess_data(
    df: pd.DataFrame,
    target_column: str = "Class",
    test_size: float = 0.2,
    random_state: int = 42
) -> Tuple[np.ndarray, np.ndarray, np.ndarray, np.ndarray]:
    """Split and preprocess data into train/test sets."""
    logger = logging.getLogger("fraud_detection_pipeline")
    logger.info("Preprocessing data")
    
    # Separate features and target
    X = df.drop(columns=[target_column]).values
    y = df[target_column].values
    
    # Split into train and test sets
    X_train, X_test, y_train, y_test = train_test_split(
        X, y, test_size=test_size, random_state=random_state, stratify=y
    )
    
    logger.info(f"Train set: {X_train.shape[0]} samples")
    logger.info(f"Test set: {X_test.shape[0]} samples")
    logger.info(f"Train fraud rate: {y_train.mean():.4f}")
    logger.info(f"Test fraud rate: {y_test.mean():.4f}")
    
    return X_train, X_test, y_train, y_test


def handle_class_imbalance(
    X_train: np.ndarray,
    y_train: np.ndarray,
    strategy: str = "class_weight"
) -> Dict[str, Any]:
    """Calculate parameters to handle class imbalance."""
    logger = logging.getLogger("fraud_detection_pipeline")
    logger.info(f"Handling class imbalance using strategy: {strategy}")
    
    if strategy == "class_weight":
        # Compute class weights
        classes = np.unique(y_train)
        class_weights = compute_class_weight(
            class_weight="balanced",
            classes=classes,
            y=y_train
        )
        
        class_weight_dict = {int(classes[i]): class_weights[i] for i in range(len(classes))}
        
        logger.info(f"Computed class weights: {class_weight_dict}")
        
        return {
            "class_weight": class_weight_dict,
            "strategy": strategy
        }
    else:
        raise ValueError(f"Unknown imbalance handling strategy: {strategy}")


def train_model(
    X_train: np.ndarray,
    y_train: np.ndarray,
    hyperparameters: Dict[str, Any]
) -> RandomForestClassifier:
    """Train RandomForestClassifier with given hyperparameters."""
    logger = logging.getLogger("fraud_detection_pipeline")
    logger.info("Training RandomForestClassifier")
    logger.info(f"Hyperparameters: {hyperparameters}")
    
    try:
        model = RandomForestClassifier(**hyperparameters)
        model.fit(X_train, y_train)
        logger.info("Model training completed successfully")
        return model
    except Exception as e:
        logger.error(f"Error during model training: {str(e)}")
        raise


def evaluate_model(
    model: RandomForestClassifier,
    X_test: np.ndarray,
    y_test: np.ndarray
) -> Dict[str, float]:
    """Evaluate model and return comprehensive metrics."""
    logger = logging.getLogger("fraud_detection_pipeline")
    logger.info("Evaluating model performance")
    
    # Generate predictions
    y_pred = model.predict(X_test)
    y_pred_proba = model.predict_proba(X_test)[:, 1]
    
    # Calculate metrics
    precision, recall, f1, _ = precision_recall_fscore_support(
        y_test, y_pred, average="binary", zero_division=0
    )
    
    accuracy = accuracy_score(y_test, y_pred)
    roc_auc = roc_auc_score(y_test, y_pred_proba)
    
    # Confusion matrix
    tn, fp, fn, tp = confusion_matrix(y_test, y_pred).ravel()
    
    # Specificity
    specificity = tn / (tn + fp) if (tn + fp) > 0 else 0.0
    
    metrics = {
        "accuracy": float(accuracy),
        "precision": float(precision),
        "recall": float(recall),
        "f1_score": float(f1),
        "roc_auc": float(roc_auc),
        "specificity": float(specificity),
        "true_positives": int(tp),
        "true_negatives": int(tn),
        "false_positives": int(fp),
        "false_negatives": int(fn),
    }
    
    logger.info("Evaluation metrics:")
    for metric_name, metric_value in metrics.items():
        logger.info(f"  {metric_name}: {metric_value}")
    
    return metrics


def check_model_quality(metrics: Dict[str, float], thresholds: Dict[str, float]) -> bool:
    """Verify model meets minimum quality thresholds."""
    logger = logging.getLogger("fraud_detection_pipeline")
    logger.info("Checking model quality against thresholds")
    
    passed = True
    
    for metric_name, threshold_value in thresholds.items():
        if metric_name not in metrics:
            logger.warning(f"Threshold specified for unknown metric: {metric_name}")
            continue
        
        metric_value = metrics[metric_name]
        
        if metric_value < threshold_value:
            logger.error(
                f"Quality check FAILED: {metric_name} = {metric_value:.4f} < {threshold_value:.4f}"
            )
            passed = False
        else:
            logger.info(
                f"Quality check PASSED: {metric_name} = {metric_value:.4f} >= {threshold_value:.4f}"
            )
    
    return passed


def log_to_mlflow(
    model: RandomForestClassifier,
    metrics: Dict[str, float],
    hyperparameters: Dict[str, Any],
    artifacts: Dict[str, str],
    experiment_name: str
) -> str:
    """Log model, metrics, parameters, and artifacts to MLflow."""
    logger = logging.getLogger("fraud_detection_pipeline")
    logger.info(f"Logging to MLflow experiment: {experiment_name}")
    
    # Set experiment
    mlflow.set_experiment(experiment_name)
    
    # Start MLflow run
    with mlflow.start_run() as run:
        run_id = run.info.run_id
        logger.info(f"MLflow run ID: {run_id}")
        
        # Log hyperparameters
        for param_name, param_value in hyperparameters.items():
            # MLflow has limitations on parameter types
            if isinstance(param_value, (dict, list)):
                mlflow.log_param(param_name, str(param_value))
            else:
                mlflow.log_param(param_name, param_value)
        
        # Log metrics
        for metric_name, metric_value in metrics.items():
            if isinstance(metric_value, (int, float)):
                mlflow.log_metric(metric_name, metric_value)
        
        # Log model
        mlflow.sklearn.log_model(
            model,
            "model",
            registered_model_name=f"{experiment_name}_model"
        )
        
        # Log artifacts
        for artifact_name, artifact_path in artifacts.items():
            if os.path.exists(artifact_path):
                mlflow.log_artifact(artifact_path, artifact_name)
                logger.info(f"Logged artifact: {artifact_name}")
            else:
                logger.warning(f"Artifact not found: {artifact_path}")
        
        # Log additional tags
        mlflow.set_tag("pipeline_version", "1.0")
        mlflow.set_tag("timestamp", datetime.now().isoformat())
        
        logger.info(f"Successfully logged to MLflow with run ID: {run_id}")
        
        return run_id


def save_model_locally(
    model: RandomForestClassifier,
    model_path: str,
    metadata: Dict[str, Any]
) -> None:
    """Save trained model and metadata to local filesystem."""
    logger = logging.getLogger("fraud_detection_pipeline")
    logger.info(f"Saving model to {model_path}")
    
    # Create directory if it doesn't exist
    os.makedirs(os.path.dirname(model_path), exist_ok=True)
    
    # Save model
    with open(model_path, "wb") as f:
        pickle.dump(model, f)
    
    logger.info(f"Model saved to {model_path}")
    
    # Save metadata
    metadata_path = model_path.replace(".pkl", "_metadata.json")
    
    # Add timestamp to metadata
    metadata["saved_at"] = datetime.now().isoformat()
    
    with open(metadata_path, "w") as f:
        json.dump(metadata, f, indent=2)
    
    logger.info(f"Metadata saved to {metadata_path}")


def run_pipeline(
    data_path: str,
    experiment_name: str = "fraud_detection",
    model_output_path: str = "models/fraud_model.pkl",
    mlflow_tracking_uri: str = "http://localhost:5000"
) -> Dict[str, Any]:
    """Execute complete ML pipeline from data loading to model logging."""
    logger = setup_logging()
    logger.info("=" * 80)
    logger.info("Starting Fraud Detection ML Pipeline")
    logger.info("=" * 80)
    
    try:
        # Set MLflow tracking URI
        mlflow.set_tracking_uri(mlflow_tracking_uri)
        logger.info(f"MLflow tracking URI: {mlflow_tracking_uri}")
        
        # Load data
        df = load_data(data_path)
        
        # Validate data
        validate_data(df)
        
        # Preprocess data
        X_train, X_test, y_train, y_test = preprocess_data(df)
        
        # Handle class imbalance
        imbalance_params = handle_class_imbalance(X_train, y_train)
        
        # Define hyperparameters
        hyperparameters = {
            "n_estimators": 100,
            "max_depth": 10,
            "min_samples_split": 10,
            "min_samples_leaf": 4,
            "random_state": 42,
            "n_jobs": -1,
            "class_weight": imbalance_params["class_weight"],
        }
        
        # Train model
        model = train_model(X_train, y_train, hyperparameters)
        
        # Evaluate model
        metrics = evaluate_model(model, X_test, y_test)
        
        # Define quality thresholds
        quality_thresholds = {
            "recall": 0.75,
            "precision": 0.70,
            "roc_auc": 0.85,
        }
        
        # Check model quality
        quality_passed = check_model_quality(metrics, quality_thresholds)
        
        if not quality_passed:
            logger.warning("Model did not meet quality thresholds")
        
        # Prepare metadata
        metadata = {
            "experiment_name": experiment_name,
            "hyperparameters": {k: str(v) if isinstance(v, dict) else v 
                               for k, v in hyperparameters.items()},
            "metrics": metrics,
            "quality_passed": quality_passed,
            "train_samples": int(X_train.shape[0]),
            "test_samples": int(X_test.shape[0]),
            "n_features": int(X_train.shape[1]),
        }
        
        # Save model locally
        save_model_locally(model, model_output_path, metadata)
        
        # Log to MLflow
        artifacts = {
            "model_file": model_output_path,
            "metadata_file": model_output_path.replace(".pkl", "_metadata.json"),
        }
        
        run_id = log_to_mlflow(
            model=model,
            metrics=metrics,
            hyperparameters=hyperparameters,
            artifacts=artifacts,
            experiment_name=experiment_name,
        )
        
        # Prepare pipeline results
        results = {
            "status": "success",
            "quality_passed": quality_passed,
            "metrics": metrics,
            "model_path": model_output_path,
            "mlflow_run_id": run_id,
            "experiment_name": experiment_name,
        }
        
        logger.info("=" * 80)
        logger.info("Pipeline completed successfully")
        logger.info("=" * 80)
        
        return results
        
    except Exception as e:
        logger.error(f"Pipeline failed with error: {str(e)}", exc_info=True)
        
        return {
            "status": "failed",
            "error": str(e),
            "quality_passed": False,
        }


def main() -> None:
    """Entry point for pipeline execution."""
    import argparse
    
    parser = argparse.ArgumentParser(description="Fraud Detection ML Pipeline")
    parser.add_argument(
        "--data-path",
        type=str,
        required=True,
        help="Path to the fraud detection dataset CSV file",
    )
    parser.add_argument(
        "--experiment-name",
        type=str,
        default="fraud_detection",
        help="MLflow experiment name",
    )
    parser.add_argument(
        "--model-output-path",
        type=str,
        default="models/fraud_model.pkl",
        help="Path to save the trained model",
    )
    parser.add_argument(
        "--mlflow-tracking-uri",
        type=str,
        default="http://localhost:5000",
        help="MLflow tracking server URI",
    )
    parser.add_argument(
        "--log-level",
        type=str,
        default="INFO",
        choices=["DEBUG", "INFO", "WARNING", "ERROR"],
        help="Logging level",
    )
    
    args = parser.parse_args()
    
    # Setup logging with specified level
    setup_logging(args.log_level)
    
    # Run pipeline
    results = run_pipeline(
        data_path=args.data_path,
        experiment_name=args.experiment_name,
        model_output_path=args.model_output_path,
        mlflow_tracking_uri=args.mlflow_tracking_uri,
    )
    
    # Exit with appropriate code
    if results["status"] == "success" and results["quality_passed"]:
        exit(0)
    elif results["status"] == "success" and not results["quality_passed"]:
        exit(1)
    else:
        exit(2)


if __name__ == "__main__":
    main()
```
