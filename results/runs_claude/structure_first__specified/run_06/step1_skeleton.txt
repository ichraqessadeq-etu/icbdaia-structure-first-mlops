## GitHub Actions Workflow Structure

### `.github/workflows/ml_pipeline.yml`

```yaml
name: ML Pipeline CI/CD

on:
  push:
    branches: [main, develop]
  pull_request:
    branches: [main]
  schedule:
    - cron: '0 0 * * *'  # Daily drift check
  workflow_dispatch:
    inputs:
      force_retrain:
        description: 'Force model retraining'
        required: false
        default: 'false'

env:
  PYTHON_VERSION: '3.9'
  MLFLOW_TRACKING_URI: ${{ secrets.MLFLOW_TRACKING_URI }}

jobs:
  # Job 1: Code Quality and Linting
  code-quality:
    steps:
      - Checkout code
      - Setup Python
      - Install dependencies
      - Run black formatter check
      - Run flake8 linting
      - Run mypy type checking
      - Run bandit security checks

  # Job 2: Unit Tests
  unit-tests:
    steps:
      - Checkout code
      - Setup Python
      - Install dependencies
      - Run pytest with coverage
      - Upload coverage reports
      - Fail if coverage below threshold

  # Job 3: Data Validation
  data-validation:
    steps:
      - Checkout code
      - Setup Python
      - Install dependencies
      - Download latest dataset
      - Run data validation checks
      - Generate data quality report

  # Job 4: Drift Detection
  drift-detection:
    steps:
      - Checkout code
      - Setup Python
      - Install dependencies
      - Download reference dataset
      - Download current dataset
      - Run drift detection
      - Set output variable for retrain decision
      - Upload drift report

  # Job 5: Model Training
  model-training:
    needs: [code-quality, unit-tests, data-validation]
    if: needs.drift-detection.outputs.should_retrain == 'true' || github.event.inputs.force_retrain == 'true'
    steps:
      - Checkout code
      - Setup Python
      - Install dependencies
      - Configure MLflow
      - Download training data
      - Run training pipeline
      - Log model to MLflow
      - Save model artifacts

  # Job 6: Model Evaluation
  model-evaluation:
    needs: [model-training]
    steps:
      - Checkout code
      - Setup Python
      - Install dependencies
      - Download test data
      - Load trained model
      - Run evaluation pipeline
      - Generate evaluation reports
      - Check performance thresholds
      - Upload evaluation artifacts

  # Job 7: Model Registration
  model-registration:
    needs: [model-evaluation]
    if: success()
    steps:
      - Checkout code
      - Setup Python
      - Configure MLflow
      - Register model to MLflow Model Registry
      - Transition model to Staging
      - Tag model with metadata

  # Job 8: Integration Tests
  integration-tests:
    needs: [model-registration]
    steps:
      - Checkout code
      - Setup Python
      - Install dependencies
      - Load staged model
      - Run integration tests
      - Test API endpoints
      - Validate predictions

  # Job 9: Deploy to Staging
  deploy-staging:
    needs: [integration-tests]
    if: github.ref == 'refs/heads/develop'
    steps:
      - Checkout code
      - Configure cloud credentials
      - Deploy model to staging environment
      - Run smoke tests
      - Update deployment status

  # Job 10: Deploy to Production
  deploy-production:
    needs: [integration-tests]
    if: github.ref == 'refs/heads/main'
    environment: production
    steps:
      - Checkout code
      - Configure cloud credentials
      - Transition model to Production in MLflow
      - Deploy model to production environment
      - Run smoke tests
      - Setup monitoring
      - Send deployment notification

  # Job 11: Monitoring Setup
  setup-monitoring:
    needs: [deploy-production]
    if: success()
    steps:
      - Configure performance monitoring
      - Configure drift monitoring
      - Setup alerting rules
      - Create monitoring dashboard
```
