class TestDataPreprocessing:
    """Tests for data preprocessing functionality."""
    
    def test_preprocess_data_output_shapes(self, sample_fraud_data):
        """Test that preprocessing returns correct shapes."""
        X_train, X_test, y_train, y_test = preprocess_data(
            sample_fraud_data,
            test_size=0.2,
            random_state=42
        )
        
        assert X_train.shape[0] == 800
        assert X_test.shape[0] == 200
        assert y_train.shape[0] == 800
        assert y_test.shape[0] == 200
        assert X_train.shape[1] == 30  # 28 V columns + Time + Amount
    
    def test_preprocess_data_stratified_split(self, sample_fraud_data):
        """Test that split maintains class distribution."""
        X_train, X_test, y_train, y_test = preprocess_data(
            sample_fraud_data,
            test_size=0.2,
            random_state=42
        )
        
        train_fraud_ratio = y_train.sum() / len(y_train)
        test_fraud_ratio = y_test.sum() / len(y_test)
        
        # Ratios should be similar (within 2%)
        assert abs(train_fraud_ratio - test_fraud_ratio) < 0.02
    
    def test_preprocess_data_scaling(self, sample_fraud_data):
        """Test that Time and Amount are properly scaled."""
        X_train, X_test, y_train, y_test = preprocess_data(
            sample_fraud_data,
            test_size=0.2,
            random_state=42
        )
        
        # After scaling, mean should be close to 0 and std close to 1
        # Time is column 0, Amount is column 1
        assert abs(X_train[:, 0].mean()) < 0.1
        assert abs(X_train[:, 1].mean()) < 0.1
        assert abs(X_train[:, 0].std() - 1.0) < 0.1
        assert abs(X_train[:, 1].std() - 1.0) < 0.1
    
    def test_preprocess_data_no_data_leakage(self, sample_fraud_data):
        """Test that scaler is fit only on training data."""
        # This is verified by ensuring test set statistics differ from training
        X_train, X_test, y_train, y_test = preprocess_data(
            sample_fraud_data,
            test_size=0.2,
            random_state=42
        )
        
        # Test set mean might not be exactly 0 since scaler was fit on train
        # This is expected and correct behavior
        assert X_train.shape[0] > X_test.shape[0]
    
    def test_preprocess_data_custom_test_size(self, sample_fraud_data):
        """Test preprocessing with custom test size."""
        X_train, X_test, y_train, y_test = preprocess_data(
            sample_fraud_data,
            test_size=0.3,
            random_state=42
        )
        
        assert X_train.shape[0] == 700
        assert X_test.shape[0] == 300

class TestImbalanceHandling:
    """Tests for class imbalance handling."""
    
    def test_handle_imbalance_class_weight(self, sample_fraud_data):
        """Test class_weight strategy."""
        X = sample_fraud_data.drop('Class', axis=1).values
        y = sample_fraud_data['Class'].values
        
        X_res, y_res, info = handle_imbalance(X, y, strategy="class_weight")
        
        assert X_res.shape == X.shape
        assert y_res.shape == y.shape
        assert info["strategy"] == "class_weight"
        assert "class_weights" in info
        assert 0 in info["class_weights"]
        assert 1 in info["class_weights"]
    
    def test_handle_imbalance_smote(self, sample_fraud_data):
        """Test SMOTE oversampling strategy."""
        X = sample_fraud_data.drop('Class', axis=1).values
        y = sample_fraud_data['Class'].values
        
        X_res, y_res, info = handle_imbalance(X, y, strategy="smote")
        
        assert X_res.shape[0] > X.shape[0]  # More samples after SMOTE
        assert y_res.shape[0] > y.shape[0]
        assert info["strategy"] == "smote"
        assert "resampled_samples" in info
        assert info["resampled_fraud_ratio"] > info["original_fraud_ratio"]
    
    def test_handle_imbalance_invalid_strategy(self, sample_fraud_data):
        """Test error handling for invalid strategy."""
        X = sample_fraud_data.drop('Class', axis=1).values
        y = sample_fraud_data['Class'].values
        
        with pytest.raises(ValueError, match="Unknown imbalance strategy"):
            handle_imbalance(X, y, strategy="invalid_strategy")
    
    def test_handle_imbalance_smote_balances_classes(self, sample_fraud_data):
        """Test that SMOTE creates balanced classes."""
        X = sample_fraud_data.drop('Class', axis=1).values
        y = sample_fraud_data['Class'].values
        
        X_res, y_res, info = handle_imbalance(X, y, strategy="smote")
        
        # After SMOTE, classes should be balanced
        fraud_ratio = y_res.sum() / len(y_res)
        assert 0.4 < fraud_ratio < 0.6  # Should be close to 0.5

class TestModelEvaluation:
    """Tests for model evaluation functionality."""
    
    def test_evaluate_model_returns_all_metrics(self, trained_model, sample_fraud_data):
        """Test that evaluation returns all required metrics."""
        X = sample_fraud_data.drop('Class', axis=1).values
        y = sample_fraud_data['Class'].values
        
        metrics = evaluate_model(trained_model, X, y)
        
        required_metrics = [
            "precision", "recall", "f1_score", "roc_auc",
            "average_precision", "accuracy", "specificity",
            "true_positives", "true_negatives",
            "false_positives", "false_negatives"
        ]
        
        for metric in required_metrics:
            assert metric in metrics
    
    def test_evaluate_model_metric_ranges(self, trained_model, sample_fraud_data):
        """Test that metrics are in valid ranges."""
        X = sample_fraud_data.drop('Class', axis=1).values
        y = sample_fraud_data['Class'].values
        
        metrics = evaluate_model(trained_model, X, y)
        
        # Probability metrics should be between 0 and 1
        prob_metrics = ["precision", "recall", "f1_score", "roc_auc", 
                       "average_precision", "accuracy", "specificity"]
        
        for metric in prob_metrics:
            assert 0 <= metrics[metric] <= 1
    
    def test_evaluate_model_confusion_matrix_sum(self, trained_model, sample_fraud_data):
        """Test that confusion matrix values sum to total samples."""
        X = sample_fraud_data.drop('Class', axis=1).values
        y = sample_fraud_data['Class'].values
        
        metrics = evaluate_model(trained_model, X, y)
        
        total = (metrics["true_positives"] + metrics["true_negatives"] +
                metrics["false_positives"] + metrics["false_negatives"])
        
        assert total == len(y)
    
    def test_evaluate_model_handles_zero_division(self):
        """Test that evaluation handles edge cases with zero division."""
        # Create a model that predicts all negatives
        model = Mock(spec=RandomForestClassifier)
        model.predict.return_value = np.zeros(100)
        model.predict_proba.return_value = np.column_stack([
            np.ones(100), np.zeros(100)
        ])
        
        X_test = np.random.randn(100, 30)
        y_test = np.zeros(100)
        
        metrics = evaluate_model(model, X_test, y_test)
        
        # Should not raise error and return valid metrics
        assert "precision" in metrics
        assert "recall" in metrics
