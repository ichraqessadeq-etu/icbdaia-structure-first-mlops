## GitHub Actions Workflow (`github/workflows/ml_pipeline.yml`)

```yaml
name: ML Pipeline CI/CD

on:
  # Trigger events (outline)
  push:
    branches: [main, develop]
  pull_request:
    branches: [main]
  schedule:
    # Daily drift check
  workflow_dispatch:
    # Manual trigger

env:
  # Environment variables

jobs:
  # Job 1: Code Quality & Linting
  code-quality:
    steps:
      - Checkout code
      - Setup Python
      - Install linting dependencies
      - Run flake8
      - Run black check
      - Run isort check
      - Run mypy type checking

  # Job 2: Unit & Integration Tests
  test:
    steps:
      - Checkout code
      - Setup Python
      - Cache dependencies
      - Install dependencies
      - Run pytest with coverage
      - Upload coverage reports
      - Check coverage threshold

  # Job 3: Drift Detection
  drift-detection:
    steps:
      - Checkout code
      - Setup Python
      - Install dependencies
      - Download reference data
      - Download current production data
      - Run drift detection script
      - Set output flag for retraining
      - Upload drift report

  # Job 4: Model Training (conditional on drift or manual trigger)
  train-model:
    needs: [test, drift-detection]
    if: needs.drift-detection.outputs.requires_retraining == 'true' || github.event_name == 'workflow_dispatch'
    steps:
      - Checkout code
      - Setup Python
      - Install dependencies
      - Download training data
      - Setup MLflow tracking
      - Run training pipeline
      - Validate model performance
      - Register model in MLflow
      - Save model artifacts
      - Upload artifacts to storage

  # Job 5: Model Validation & Testing
  validate-model:
    needs: train-model
    steps:
      - Checkout code
      - Setup Python
      - Install dependencies
      - Download trained model
      - Run model validation tests
      - Performance benchmark tests
      - Bias and fairness checks
      - Generate validation report

  # Job 6: Deploy to Staging
  deploy-staging:
    needs: validate-model
    if: github.ref == 'refs/heads/develop'
    steps:
      - Checkout code
      - Setup cloud CLI
      - Authenticate
      - Download model artifacts
      - Deploy to staging endpoint
      - Run smoke tests
      - Update model registry

  # Job 7: Deploy to Production
  deploy-production:
    needs: validate-model
    if: github.ref == 'refs/heads/main'
    environment: production
    steps:
      - Checkout code
      - Setup cloud CLI
      - Authenticate
      - Download model artifacts
      - Deploy with blue-green strategy
      - Run integration tests
      - Monitor initial metrics
      - Update production registry
      - Send notification

  # Job 8: Monitoring Setup
  setup-monitoring:
    needs: deploy-production
    steps:
      - Setup performance monitoring
      - Setup drift monitoring
      - Configure alerts
      - Update dashboard
```
