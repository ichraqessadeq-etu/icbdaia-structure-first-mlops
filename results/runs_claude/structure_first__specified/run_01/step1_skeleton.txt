## GitHub Actions Workflow (`..github/workflows/ml_pipeline.yml`)

```yaml
# High-level structure outline

name: Fraud Detection ML Pipeline

on:
  # Triggers
  
jobs:
  
  # Job 1: code-quality
  code-quality:
    steps:
      - Checkout code
      - Setup Python
      - Install dependencies
      - Run linting (flake8/black)
      - Run type checking (mypy)
  
  # Job 2: unit-tests
  unit-tests:
    steps:
      - Checkout code
      - Setup Python
      - Install dependencies
      - Run pytest with coverage
      - Upload coverage reports
  
  # Job 3: drift-detection
  drift-detection:
    steps:
      - Checkout code
      - Setup Python
      - Install dependencies
      - Download production data
      - Run drift detection script
      - Check if retraining needed
      - Set output flag for retraining
  
  # Job 4: train-model (conditional on drift detection)
  train-model:
    needs: [unit-tests, drift-detection]
    if: needs.drift-detection.outputs.retrain == 'true'
    steps:
      - Checkout code
      - Setup Python
      - Install dependencies
      - Setup MLflow tracking
      - Download training data
      - Run training pipeline
      - Log metrics to MLflow
      - Save model artifacts
  
  # Job 5: model-evaluation
  model-evaluation:
    needs: train-model
    steps:
      - Checkout code
      - Setup Python
      - Load trained model
      - Run evaluation on test set
      - Check performance thresholds
      - Generate evaluation report
  
  # Job 6: model-registration
  model-registration:
    needs: model-evaluation
    steps:
      - Checkout code
      - Setup Python
      - Register model in MLflow Model Registry
      - Tag model version
      - Transition to staging
  
  # Job 7: deploy-staging
  deploy-staging:
    needs: model-registration
    steps:
      - Checkout code
      - Deploy model to staging environment
      - Run smoke tests
      - Update deployment metadata
  
  # Job 8: notify
  notify:
    needs: [train-model, model-evaluation, model-registration]
    if: always()
    steps:
      - Send notification (Slack/Email)
      - Update dashboard
```
