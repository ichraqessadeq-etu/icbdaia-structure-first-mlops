def test_split_data_stratification(sample_dataframe):
    """Test that stratification maintains class distribution."""
    X, y = preprocess_data(sample_dataframe)
    X_train, X_test, y_train, y_test = split_data(X, y)
    
    original_ratio = y.mean()
    train_ratio = y_train.mean()
    test_ratio = y_test.mean()
    
    # Ratios should be similar (within 50% relative difference)
    assert train_ratio == pytest.approx(original_ratio, rel=0.5)
    assert test_ratio == pytest.approx(original_ratio, rel=0.5)


def test_split_data_reproducibility(sample_dataframe):
    """Test that splitting is reproducible with same random state."""
    X, y = preprocess_data(sample_dataframe)
    
    X_train1, X_test1, y_train1, y_test1 = split_data(X, y, random_state=42)
    X_train2, X_test2, y_train2, y_test2 = split_data(X, y, random_state=42)
    
    pd.testing.assert_frame_equal(X_train1, X_train2)
    pd.testing.assert_frame_equal(X_test1, X_test2)

def test_handle_imbalance_smote(sample_dataframe):
    """Test SMOTE oversampling."""
    X, y = preprocess_data(sample_dataframe)
    X_train, X_test, y_train, y_test = split_data(X, y)
    
    X_resampled, y_resampled = handle_imbalance(X_train, y_train, strategy="smote")
    
    assert len(X_resampled) >= len(X_train)
    assert isinstance(X_resampled, pd.DataFrame)
    assert isinstance(y_resampled, pd.Series)
    
    # Check class balance improved
    original_ratio = y_train.value_counts().min() / y_train.value_counts().max()
    new_ratio = pd.Series(y_resampled).value_counts().min() / pd.Series(y_resampled).value_counts().max()
    assert new_ratio > original_ratio


def test_handle_imbalance_undersample(sample_dataframe):
    """Test random undersampling."""
    X, y = preprocess_data(sample_dataframe)
    X_train, X_test, y_train, y_test = split_data(X, y)
    
    X_resampled, y_resampled = handle_imbalance(X_train, y_train, strategy="undersample")
    
    assert len(X_resampled) <= len(X_train)
    assert isinstance(X_resampled, pd.DataFrame)
    assert isinstance(y_resampled, pd.Series)


def test_handle_imbalance_none(sample_dataframe):

def test_evaluate_model_returns_metrics(sample_model, sample_dataframe):
    """Test that evaluation returns all expected metrics."""
    X, y = preprocess_data(sample_dataframe)
    X_train, X_test, y_train, y_test = split_data(X, y)
    
    metrics = evaluate_model(sample_model, X_test, y_test)
    
    required_metrics = [
        'precision', 'recall', 'f1_score', 'auc_roc', 'average_precision',
        'true_negatives', 'false_positives', 'false_negatives', 'true_positives',
        'accuracy', 'specificity', 'classification_report'
    ]
    
    for metric in required_metrics:
        assert metric in metrics


def test_evaluate_model_metric_ranges(sample_model, balanced_dataframe):
    """Test that metrics are within valid ranges."""
    X, y = preprocess_data(balanced_dataframe)
    X_train, X_test, y_train, y_test = split_data(X, y)
    
    metrics = evaluate_model(sample_model, X_test, y_test)
    
    # Check ranges for probability-based metrics
    assert 0 <= metrics['precision'] <= 1
    assert 0 <= metrics['recall'] <= 1
    assert 0 <= metrics['f1_score'] <= 1
    assert 0 <= metrics['auc_roc'] <= 1
    assert 0 <= metrics['accuracy'] <= 1
    assert 0 <= metrics['specificity'] <= 1


def test_evaluate_model_confusion_matrix(sample_model, balanced_dataframe):
    """Test confusion matrix values are non-negative."""
    X, y = preprocess_data(balanced_dataframe)
    X_train, X_test, y_train, y_test = split_data(X, y)
    
    metrics = evaluate_model(sample_model, X_test, y_test)
    
    assert metrics['true_negatives'] >= 0
    assert metrics['false_positives'] >= 0
    assert metrics['false_negatives'] >= 0
    assert metrics['true_positives'] >= 0
    
    # Sum should equal test set size
    total = (metrics['true_negatives'] + metrics['false_positives'] + 
             metrics['false_negatives'] + metrics['true_positives'])
    assert total == len(y_test)
