    """Test that test size ratio is respected."""
    X, y = preprocess_data(balanced_fraud_data)
    test_size = 0.3
    
    X_train, X_test, y_train, y_test = split_data(X, y, test_size=test_size, random_state=42)

def test_split_data_stratification(balanced_fraud_data):
    """Test that stratification maintains class distribution."""
    X, y = preprocess_data(balanced_fraud_data)
    
    X_train, X_test, y_train, y_test = split_data(X, y, test_size=0.2, random_state=42)
    
    # Check that class distribution is similar in train and test
    train_ratio = y_train.value_counts(normalize=True)
    test_ratio = y_test.value_counts(normalize=True)
    
    assert abs(train_ratio[1] - test_ratio[1]) < 0.1  # Within 10%
