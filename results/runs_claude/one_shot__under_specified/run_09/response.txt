@dataclass
class ModelMetrics:
    """Container for model evaluation metrics"""
    accuracy: float
    precision: float
    recall: float
    f1_score: float
    roc_auc: float
    confusion_matrix: list
    timestamp: str
    
    def to_dict(self) -> Dict:
        return asdict(self)
    
    def passes_thresholds(self, thresholds: Dict[str, float]) -> bool:
        """Check if metrics meet minimum thresholds"""
        checks = [
            self.accuracy >= thresholds.get('accuracy', 0.85),
            self.precision >= thresholds.get('precision', 0.80),
            self.recall >= thresholds.get('recall', 0.75),
            self.f1_score >= thresholds.get('f1_score', 0.77),
            self.roc_auc >= thresholds.get('roc_auc', 0.85)
        ]
        return all(checks)

class PipelineConfig:

class DataValidator:
    """Validates input data quality and schema"""
    
    def __init__(self, expected_features: list):
        self.expected_features = expected_features
    
    def validate_schema(self, df: pd.DataFrame) -> Tuple[bool, str]:
        """Validate dataframe schema"""
        missing_features = set(self.expected_features) - set(df.columns)
        if missing_features:
            return False, f"Missing features: {missing_features}"
        
        return True, "Schema validation passed"
    
    def validate_data_quality(self, df: pd.DataFrame) -> Tuple[bool, str]:
        """Check data quality issues"""
        issues = []
        
        # Check for excessive missing values
        null_pct = df.isnull().sum() / len(df)
        high_null_cols = null_pct[null_pct > 0.3].index.tolist()
        if high_null_cols:
            issues.append(f"High null percentage in columns: {high_null_cols}")
        
        # Check for duplicate rows
        dup_count = df.duplicated().sum()
        if dup_count > len(df) * 0.1:
            issues.append(f"High duplicate count: {dup_count}")
        
        # Check for infinite values
        numeric_cols = df.select_dtypes(include=[np.number]).columns
        inf_cols = [col for col in numeric_cols if np.isinf(df[col]).any()]
        if inf_cols:
            issues.append(f"Infinite values in columns: {inf_cols}")
        
        if issues:
            return False, "; ".join(issues)
        
        return True, "Data quality validation passed"
    
    def validate(self, df: pd.DataFrame) -> Tuple[bool, str]:
        """Run all validations"""
        schema_valid, schema_msg = self.validate_schema(df)
        if not schema_valid:
            return False, schema_msg
        
        quality_valid, quality_msg = self.validate_data_quality(df)
        if not quality_valid:
            return False, quality_msg
        
        return True, "All validations passed"


class DataDriftDetector:
    """Detect data drift between reference and current data"""
    
    def __init__(self, reference_data: pd.DataFrame):
        self.reference_data = reference_data
    
    def detect_drift(self, current_data: pd.DataFrame) -> Tuple[bool, Dict]:
        """Detect data drift using Evidently"""
        try:
            report = Report(metrics=[DataDriftPreset()])
            report.run(
                reference_data=self.reference_data,
                current_data=current_data
            )
            
            result = report.as_dict()
            
            # Extract drift metrics
            drift_detected = result['metrics'][0]['result']['dataset_drift']
            drift_share = result['metrics'][0]['result']['share_of_drifted_columns']
            
            drift_info = {
                'drift_detected': drift_detected,
                'drift_share': drift_share,
                'timestamp': datetime.now().isoformat()
            }
            
            return drift_detected, drift_info
            
        except Exception as e:
            logger.error(f"Error detecting drift: {str(e)}")
            return False, {'error': str(e)}


class FraudDetectionPipeline:
    """End-to-end ML pipeline for fraud detection"""
    
    def __init__(self, config: PipelineConfig):
        self.config = config
        self.model = None
        self.scaler = StandardScaler()
        self.feature_names = None
        self.metrics = None
        
        # Setup MLflow
        mlflow.set_tracking_uri(config.mlflow_tracking_uri)
        mlflow.set_experiment(config.mlflow_experiment_name)
        
        # Create artifact directory
        Path(config.artifact_path).mkdir(parents=True, exist_ok=True)
    
    def _get_model(self):
        """Get model based on configuration"""
        if self.config.model_type == "random_forest":
            return RandomForestClassifier(**self.config.rf_params)
        elif self.config.model_type == "gradient_boosting":
            return GradientBoostingClassifier(**self.config.gb_params)
        else:
            raise ValueError(f"Unknown model type: {self.config.model_type}")
    
    def preprocess_data(
        self, 
        X: pd.DataFrame, 
        y: Optional[pd.Series] = None,
        fit_scaler: bool = False
    ) -> Tuple[np.ndarray, Optional[np.ndarray]]:
        """Preprocess features"""
        # Store feature names
        if self.feature_names is None:
            self.feature_names = X.columns.tolist()
        
        # Handle missing values
        X_processed = X.fillna(X.median())
        
        # Scale features
        if fit_scaler:
            X_scaled = self.scaler.fit_transform(X_processed)
        else:
            X_scaled = self.scaler.transform(X_processed)
        
        if y is not None:
            return X_scaled, y.values
        
        return X_scaled, None
    
    def train(
        self, 
        X_train: pd.DataFrame, 
        y_train: pd.Series,
        X_val: pd.DataFrame,
        y_val: pd.Series
    ) -> ModelMetrics:
        """Train the fraud detection model"""
        logger.info("Starting model training...")
        
        with mlflow.start_run() as run:
            # Preprocess data
            X_train_processed, y_train_processed = self.preprocess_data(
                X_train, y_train, fit_scaler=True
            )
            X_val_processed, y_val_processed = self.preprocess_data(
                X_val, y_val, fit_scaler=False
            )
            
            # Initialize and train model
            self.model = self._get_model()
            self.model.fit(X_train_processed, y_train_processed)
            
            # Cross-validation
            cv_scores = cross_val_score(
                self.model, 
                X_train_processed, 
                y_train_processed,
                cv=self.config.cv_folds,
                scoring='f1'
            )
            logger.info(f"CV F1 Scores: {cv_scores.mean():.4f} (+/- {cv_scores.std():.4f})")
            
            # Evaluate on validation set
            self.metrics = self.evaluate(X_val_processed, y_val_processed)
            
            # Log parameters and metrics to MLflow
            mlflow.log_params(self.config.__dict__)
            mlflow.log_metrics({
                'accuracy': self.metrics.accuracy,
                'precision': self.metrics.precision,
                'recall': self.metrics.recall,
                'f1_score': self.metrics.f1_score,
                'roc_auc': self.metrics.roc_auc,
                'cv_f1_mean': cv_scores.mean(),
                'cv_f1_std': cv_scores.std()
            })
            
            # Log model
            mlflow.sklearn.log_model(
                self.model,
                "model",
                registered_model_name=self.config.model_registry_name
            )
            
            # Log scaler
            mlflow.sklearn.log_model(self.scaler, "scaler")
            
            # Log feature importance
            if hasattr(self.model, 'feature_importances_'):
                feature_importance = pd.DataFrame({
                    'feature': self.feature_names,
                    'importance': self.model.feature_importances_
                }).sort_values('importance', ascending=False)
                
                feature_importance.to_csv(
                    f"{self.config.artifact_path}/feature_importance.csv",
                    index=False
                )
                mlflow.log_artifact(f"{self.config.artifact_path}/feature_importance.csv")
            
            logger.info(f"Training completed. Run ID: {run.info.run_id}")
        
        return self.metrics
    
    def evaluate(self, X: np.ndarray, y: np.ndarray) -> ModelMetrics:
        """Evaluate model performance"""
        y_pred = self.model.predict(X)
        y_pred_proba = self.model.predict_proba(X)[:, 1]
        
        metrics = ModelMetrics(
            accuracy=accuracy_score(y, y_pred),
            precision=precision_score(y, y_pred, zero_division=0),
            recall=recall_score(y, y_pred, zero_division=0),
            f1_score=f1_score(y, y_pred, zero_division=0),
            roc_auc=roc_auc_score(y, y_pred_proba),
            confusion_matrix=confusion_matrix(y, y_pred).tolist(),
            timestamp=datetime.now().isoformat()
        )
        
        logger.info(f"Model Metrics: {metrics}")
        return metrics
    
    def validate_model(self) -> Tuple[bool, str]:
        """Validate model meets performance thresholds"""
        if self.metrics is None:
            return False, "No metrics available for validation"
        
        if self.metrics.passes_thresholds(self.config.performance_thresholds):
            return True, "Model passes all performance thresholds"
        
        return False, f"Model fails to meet thresholds. Metrics: {self.metrics.to_dict()}"
    
    def save_model(self, path: str):
        """Save model and scaler to disk"""
        model_data = {
            'model': self.model,
            'scaler': self.scaler,
            'feature_names': self.feature_names,
            'config': asdict(self.config),
            'metrics': self.metrics.to_dict() if self.metrics else None
        }
        
        joblib.dump(model_data, path)
        logger.info(f"Model saved to {path}")
    
    def load_model(self, path: str):
        """Load model and scaler from disk"""
        model_data = joblib.load(path)
        
        self.model = model_data['model']
        self.scaler = model_data['scaler']
        self.feature_names = model_data['feature_names']
        
        if model_data.get('metrics'):
            self.metrics = ModelMetrics(**model_data['metrics'])
        
        logger.info(f"Model loaded from {path}")
    
    def predict(self, X: pd.DataFrame) -> Tuple[np.ndarray, np.ndarray]:
        """Make predictions on new data"""
        if self.model is None:
            raise ValueError("Model not trained or loaded")
        
        X_processed, _ = self.preprocess_data(X, fit_scaler=False)
        predictions = self.model.predict(X_processed)
        probabilities = self.model.predict_proba(X_processed)[:, 1]
        
        return predictions, probabilities
    
    def run_pipeline(
        self, 
        X: pd.DataFrame, 
        y: pd.Series,
        validator: Optional[DataValidator] = None
    ) -> Tuple[bool, ModelMetrics]:
        """Run the complete pipeline"""
        logger.info("Starting fraud detection pipeline...")
        
        # Validate data if validator provided
        if validator:
            is_valid, message = validator.validate(X)
            if not is_valid:
                raise ValueError(f"Data validation failed: {message}")
            logger.info(message)
        
        # Split data
        X_train, X_val, y_train, y_val = train_test_split(
            X, y,
            test_size=self.config.test_size,
            random_state=self.config.random_state,
            stratify=y
        )
        
        logger.info(f"Training set: {len(X_train)}, Validation set: {len(X_val)}")
        logger.info(f"Fraud rate - Train: {y_train.mean():.2%}, Val: {y_val.mean():.2%}")
        
        # Train model
        metrics = self.train(X_train, y_train, X_val, y_val)
        
        # Validate model performance
        is_valid, message = self.validate_model()
        logger.info(message)
        
        # Save model
        model_path = f"{self.config.artifact_path}/model_{datetime.now().strftime('%Y%m%d_%H%M%S')}.pkl"
        self.save_model(model_path)
        
        # Save metrics
        metrics_path = f"{self.config.artifact_path}/metrics.json"
        with open(metrics_path, 'w') as f:
            json.dump(metrics.to_dict(), f, indent=2)
        
        return is_valid, metrics


def main():
    """Example usage of the pipeline"""
    # Generate sample data (replace with actual data loading)
    np.random.seed(42)
    n_samples = 10000
    n_features = 20
    
    X = pd.DataFrame(
        np.random.randn(n_samples, n_features),
        columns=[f'feature_{i}' for i in range(n_features)]
    )
    # Imbalanced target (5% fraud rate)
    y = pd.Series(np.random.choice([0, 1], size=n_samples, p=[0.95, 0.05]))
    
    # Setup pipeline
    config = PipelineConfig(model_type="random_forest")
    pipeline = FraudDetectionPipeline(config)
    
    # Setup validator
    validator = DataValidator(expected_features=X.columns.tolist())
    
    # Run pipeline
    is_valid, metrics = pipeline.run_pipeline(X, y, validator)
    
    print(f"\nPipeline completed successfully: {is_valid}")
    print(f"Final metrics: {metrics.to_dict()}")


if __name__ == "__main__":
    main()

            --environment ${{ github.ref == 'refs/heads/main' && 'production' || 'staging' }}

      - name: Create CloudWatch alarms
        run: |
          aws cloudwatch put-metric-alarm \
            --alarm-name fraud-detection-high-error-rate \
            --alarm-description "Alert when error rate exceeds threshold" \
            --metric-name ErrorRate \
            --namespace FraudDetection \
            --statistic Average \
            --period 300 \
            --threshold 5 \
