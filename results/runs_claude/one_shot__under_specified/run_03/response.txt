```python
# pipeline/fraud_detection_pipeline.py
"""
Fraud Detection Model CI/CD Pipeline
Handles training, validation, deployment, and monitoring of fraud detection models.
"""

import os
import json
import logging
from typing import Dict, Tuple, Any, Optional
from datetime import datetime
from pathlib import Path
import pickle

import pandas as pd
import numpy as np
from sklearn.ensemble import RandomForestClassifier, GradientBoostingClassifier
from sklearn.model_selection import train_test_split, cross_val_score
from sklearn.metrics import (
    precision_score, recall_score, f1_score, roc_auc_score,
    confusion_matrix, classification_report, precision_recall_curve
)
from sklearn.preprocessing import StandardScaler
import joblib
import mlflow
import mlflow.sklearn
from evidently.report import Report
from evidently.metric_preset import DataDriftPreset, ClassificationPreset
import great_expectations as ge

logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)


class FraudDetectionPipeline:
    """End-to-end pipeline for fraud detection model."""
    
    def __init__(
        self,
        model_type: str = "random_forest",
        experiment_name: str = "fraud_detection",
        tracking_uri: str = "sqlite:///mlflow.db"
    ):
        """
        Initialize the pipeline.
        
        Args:
            model_type: Type of model ('random_forest' or 'gradient_boosting')
            experiment_name: MLflow experiment name
            tracking_uri: MLflow tracking URI
        """
        self.model_type = model_type
        self.experiment_name = experiment_name
        self.model = None
        self.scaler = StandardScaler()
        self.feature_names = None
        self.metrics = {}
        
        # MLflow setup
        mlflow.set_tracking_uri(tracking_uri)
        mlflow.set_experiment(experiment_name)
        
        # Model registry
        self.model_registry_path = Path("models")
        self.model_registry_path.mkdir(exist_ok=True)
        
    def validate_data(self, df: pd.DataFrame) -> Tuple[bool, Dict[str, Any]]:
        """
        Validate input data using Great Expectations.
        
        Args:
            df: Input dataframe
            
        Returns:
            Tuple of (is_valid, validation_results)
        """
        logger.info("Validating input data...")
        
        validation_results = {
            "timestamp": datetime.now().isoformat(),
            "checks": {}
        }
        
        try:
            # Convert to GE dataset
            ge_df = ge.from_pandas(df)
            
            # Define expectations
            expectations = [
                ("expect_table_row_count_to_be_between", {"min_value": 100}),
                ("expect_table_column_count_to_equal", {"value": len(df.columns)}),
            ]
            
            # Check for required columns
            required_columns = ['amount', 'is_fraud'] if 'is_fraud' in df.columns else ['amount']
            for col in required_columns:
                expectations.append(
                    ("expect_column_to_exist", {"column": col})
                )
                
            # Check for nulls in critical columns
            for col in df.columns:
                result = ge_df.expect_column_values_to_not_be_null(column=col)
                validation_results["checks"][f"{col}_not_null"] = result.success
                
            # Check numeric columns are within reasonable ranges
            numeric_cols = df.select_dtypes(include=[np.number]).columns
            for col in numeric_cols:
                if col != 'is_fraud':
                    result = ge_df.expect_column_values_to_be_between(
                        column=col,
                        min_value=df[col].quantile(0.001),
                        max_value=df[col].quantile(0.999),
                        mostly=0.95
                    )
                    validation_results["checks"][f"{col}_range"] = result.success
            
            is_valid = all(validation_results["checks"].values())
            validation_results["overall_status"] = "PASSED" if is_valid else "FAILED"
            
            logger.info(f"Data validation {'passed' if is_valid else 'failed'}")
            return is_valid, validation_results
            
        except Exception as e:
            logger.error(f"Data validation error: {str(e)}")
            validation_results["error"] = str(e)
            validation_results["overall_status"] = "ERROR"
            return False, validation_results
    
    def prepare_features(
        self,
        df: pd.DataFrame,
        target_col: str = 'is_fraud',
        test_size: float = 0.2,
        random_state: int = 42
    ) -> Tuple[np.ndarray, np.ndarray, np.ndarray, np.ndarray]:
        """
        Prepare features for training.
        
        Args:
            df: Input dataframe
            target_col: Target column name
            test_size: Test set size
            random_state: Random seed
            
        Returns:
            Tuple of (X_train, X_test, y_train, y_test)
        """
        logger.info("Preparing features...")
        
        # Separate features and target
        X = df.drop(columns=[target_col])
        y = df[target_col]
        
        # Store feature names
        self.feature_names = X.columns.tolist()
        
        # Split data
        X_train, X_test, y_train, y_test = train_test_split(
            X, y, test_size=test_size, random_state=random_state, stratify=y
        )
        
        # Scale features
        X_train_scaled = self.scaler.fit_transform(X_train)
        X_test_scaled = self.scaler.transform(X_test)
        
        logger.info(f"Training set: {X_train_scaled.shape}, Test set: {X_test_scaled.shape}")
        logger.info(f"Fraud rate - Train: {y_train.mean():.2%}, Test: {y_test.mean():.2%}")
        
        return X_train_scaled, X_test_scaled, y_train.values, y_test.values
    
    def train_model(
        self,
        X_train: np.ndarray,
        y_train: np.ndarray,
        hyperparameters: Optional[Dict] = None
    ) -> None:
        """
        Train the fraud detection model.
        
        Args:
            X_train: Training features
            y_train: Training labels
            hyperparameters: Model hyperparameters
        """
        logger.info(f"Training {self.model_type} model...")
        
        default_params = {
            "random_forest": {
                "n_estimators": 100,
                "max_depth": 10,
                "min_samples_split": 5,
                "min_samples_leaf": 2,
                "class_weight": "balanced",
                "random_state": 42,
                "n_jobs": -1
            },
            "gradient_boosting": {
                "n_estimators": 100,
                "learning_rate": 0.1,
                "max_depth": 5,
                "min_samples_split": 5,
                "min_samples_leaf": 2,
                "random_state": 42
            }
        }
        
        params = hyperparameters or default_params.get(self.model_type, {})
        
        # Initialize model
        if self.model_type == "random_forest":
            self.model = RandomForestClassifier(**params)
        elif self.model_type == "gradient_boosting":
            self.model = GradientBoostingClassifier(**params)
        else:
            raise ValueError(f"Unknown model type: {self.model_type}")
        
        # Train model
        with mlflow.start_run():
            self.model.fit(X_train, y_train)
            
            # Log parameters
            mlflow.log_params(params)
            mlflow.log_param("model_type", self.model_type)
            
            # Cross-validation
            cv_scores = cross_val_score(
                self.model, X_train, y_train, cv=5, scoring='roc_auc'
            )
            mlflow.log_metric("cv_roc_auc_mean", cv_scores.mean())
            mlflow.log_metric("cv_roc_auc_std", cv_scores.std())
            
            logger.info(f"Cross-validation ROC-AUC: {cv_scores.mean():.4f} (+/- {cv_scores.std():.4f})")
    
    def evaluate_model(
        self,
        X_test: np.ndarray,
        y_test: np.ndarray,
        threshold: float = 0.5
    ) -> Dict[str, float]:
        """
        Evaluate model performance.
        
        Args:
            X_test: Test features
            y_test: Test labels
            threshold: Classification threshold
            
        Returns:
            Dictionary of metrics
        """
        logger.info("Evaluating model...")
        
        # Predictions
        y_pred_proba = self.model.predict_proba(X_test)[:, 1]
        y_pred = (y_pred_proba >= threshold).astype(int)
        
        # Calculate metrics
        self.metrics = {
            "accuracy": float(np.mean(y_pred == y_test)),
            "precision": float(precision_score(y_test, y_pred, zero_division=0)),
            "recall": float(recall_score(y_test, y_pred, zero_division=0)),
            "f1_score": float(f1_score(y_test, y_pred, zero_division=0)),
            "roc_auc": float(roc_auc_score(y_test, y_pred_proba)),
            "threshold": threshold
        }
        
        # Confusion matrix
        tn, fp, fn, tp = confusion_matrix(y_test, y_pred).ravel()
        self.metrics.update({
            "true_negatives": int(tn),
            "false_positives": int(fp),
            "false_negatives": int(fn),
            "true_positives": int(tp),
            "false_positive_rate": float(fp / (fp + tn)) if (fp + tn) > 0 else 0,
            "false_negative_rate": float(fn / (fn + tp)) if (fn + tp) > 0 else 0
        })
        
        # Business metrics for fraud detection
        self.metrics["fraud_detection_rate"] = self.metrics["recall"]
        self.metrics["false_alarm_rate"] = self.metrics["false_positive_rate"]
        
        # Log to MLflow
        with mlflow.start_run():
            for metric_name, metric_value in self.metrics.items():
                if isinstance(metric_value, (int, float)):
                    mlflow.log_metric(metric_name, metric_value)
        
        logger.info(f"Model Performance:")
        logger.info(f"  ROC-AUC: {self.metrics['roc_auc']:.4f}")
        logger.info(f"  Precision: {self.metrics['precision']:.4f}")
        logger.info(f"  Recall: {self.metrics['recall']:.4f}")
        logger.info(f"  F1-Score: {self.metrics['f1_score']:.4f}")
        
        return self.metrics
    
    def check_model_quality_gates(
        self,
        min_roc_auc: float = 0.85,
        min_recall: float = 0.75,
        max_false_positive_rate: float = 0.05
    ) -> Tuple[bool, Dict[str, Any]]:
        """
        Check if model meets quality gates.
        
        Args:
            min_roc_auc: Minimum required ROC-AUC
            min_recall: Minimum required recall (fraud detection rate)
            max_false_positive_rate: Maximum allowed false positive rate
            
        Returns:
            Tuple of (passed, gate_results)
        """
        logger.info("Checking model quality gates...")
        
        gate_results = {
            "timestamp": datetime.now().isoformat(),
            "gates": {}
        }
        
        # Check ROC-AUC
        gate_results["gates"]["roc_auc"] = {
            "threshold": min_roc_auc,
            "value": self.metrics["roc_auc"],
            "passed": self.metrics["roc_auc"] >= min_roc_auc
        }
        
        # Check recall (critical for fraud detection)
        gate_results["gates"]["recall"] = {
            "threshold": min_recall,
            "value": self.metrics["recall"],
            "passed": self.metrics["recall"] >= min_recall
        }
        
        # Check false positive rate
        gate_results["gates"]["false_positive_rate"] = {
            "threshold": max_false_positive_rate,
            "value": self.metrics["false_positive_rate"],
            "passed": self.metrics["false_positive_rate"] <= max_false_positive_rate
        }
        
        # Overall result
        all_passed = all(gate["passed"] for gate in gate_results["gates"].values())
        gate_results["overall_status"] = "PASSED" if all_passed else "FAILED"
        
        logger.info(f"Quality gates: {gate_results['overall_status']}")
        for gate_name, gate_info in gate_results["gates"].items():
            status = "✓" if gate_info["passed"] else "✗"
            logger.info(f"  {status} {gate_name}: {gate_info['value']:.4f} (threshold: {gate_info['threshold']})")
        
        return all_passed, gate_results
    
    def detect_data_drift(
        self,
        reference_data: pd.DataFrame,
        current_data: pd.DataFrame,
        target_col: str = 'is_fraud'
    ) -> Tuple[bool, Dict[str, Any]]:
        """
        Detect data drift using Evidently.
        
        Args:
            reference_data: Reference (training) dataset
            current_data: Current (production) dataset
            target_col: Target column name
            
        Returns:
            Tuple of (has_drift, drift_report)
        """
        logger.info("Checking for data drift...")
        
        try:
            # Create drift report
            drift_report = Report(metrics=[
                DataDriftPreset(),
            ])
            
            drift_report.run(
                reference_data=reference_data,
                current_data=current_data
            )
            
            # Extract results
            report_dict = drift_report.as_dict()
            
            drift_results = {
                "timestamp": datetime.now().isoformat(),
                "has_drift": False,
                "drifted_features": [],
                "drift_scores": {}
            }
            
            # Parse drift metrics
            for metric in report_dict.get("metrics", []):
                if "DatasetDriftMetric" in str(metric.get("metric")):
                    drift_results["has_drift"] = metric.get("result", {}).get("dataset_drift", False)
                    drift_results["drift_share"] = metric.get("result", {}).get("drift_share", 0)
            
            logger.info(f"Data drift detected: {drift_results['has_drift']}")
            
            return drift_results["has_drift"], drift_results
            
        except Exception as e:
            logger.error(f"Error detecting data drift: {str(e)}")
            return False, {"error": str(e)}
    
    def save_model(
        self,
        version: str,
        metadata: Optional[Dict] = None
    ) -> Path:
        """
        Save model to disk and MLflow.
        
        Args:
            version: Model version
            metadata: Additional metadata
            
        Returns:
            Path to saved model
        """
        logger.info(f"Saving model version {version}...")
        
        # Create version directory
        model_dir = self.model_registry_path / version
        model_dir.mkdir(exist_ok=True)
        
        # Save model and scaler
        model_path = model_dir / "model.pkl"
        scaler_path = model_dir / "scaler.pkl"
        
        joblib.dump(self.model, model_path)
        joblib.dump(self.scaler, scaler_path)
        
        # Save metadata
        model_metadata = {
            "version": version,
            "model_type": self.model_type,
            "feature_names": self.feature_names,
            "metrics": self.metrics,
            "timestamp": datetime.now().isoformat(),
            **(metadata or {})
        }
        
        metadata_path = model_dir / "metadata.json"
        with open(metadata_path, 'w') as f:
            json.dump(model_metadata, f, indent=2)
        
        # Log to MLflow
        with mlflow.start_run():
            mlflow.sklearn.log_model(self.model, "model")
            mlflow.log_artifact(str(scaler_path))
            mlflow.log_artifact(str(metadata_path))
            mlflow.set_tag("version", version)
        
        logger.info(f"Model saved to {model_dir}")
        return model_dir
    
    def load_model(self, version: str) -> None:
        """
        Load model from disk.
        
        Args:
            version: Model version to load
        """
        logger.info(f"Loading model version {version}...")
        
        model_dir = self.model_registry_path / version
        
        if not model_dir.exists():
            raise FileNotFoundError(f"Model version {version} not found")
        
        # Load model and scaler
        model_path = model_dir / "model.pkl"
        scaler_path = model_dir / "scaler.pkl"
        
        self.model = joblib.load(model_path)
        self.scaler = joblib.load(scaler_path)
        
        # Load metadata
        metadata_path = model_dir / "metadata.json"
        with open(metadata_path, 'r') as f:
            metadata = json.load(f)
        
        self.feature_names = metadata.get("feature_names")
        self.metrics = metadata.get("metrics", {})
        
        logger.info(f"Model version {version} loaded successfully")
    
    def predict(
        self,
        X: np.ndarray,
        threshold: float = 0.5
    ) -> Tuple[np.ndarray, np.ndarray]:
        """
        Make predictions.
        
        Args:
            X: Input features
            threshold: Classification threshold
            
        Returns:
            Tuple of (predictions, probabilities)
        """
        if self.model is None:
            raise ValueError("Model not trained or loaded")
        
        # Scale features
        X_scaled = self.scaler.transform(X)
        
        # Predict
        probabilities = self.model.predict_proba(X_scaled)[:, 1]
        predictions = (probabilities >= threshold).astype(int)
        
        return predictions, probabilities
    
    def run_full_pipeline(
        self,
        train_data: pd.DataFrame,
        version: str,
        target_col: str = 'is_fraud',
        hyperparameters: Optional[Dict] = None,
        quality_gates: Optional[Dict] = None
    ) -> Dict[str, Any]:
        """
        Run the complete pipeline.
        
        Args:
            train_data: Training dataset
            version: Model version
            target_col: Target column name
            hyperparameters: Model hyperparameters
            quality_gates: Quality gate thresholds
            
        Returns:
            Pipeline results dictionary
        """
        results = {
            "version": version,
            "timestamp": datetime.now().isoformat(),
            "status": "FAILED"
        }
        
        try:
            # 1. Validate data
            is_valid, validation_results = self.validate_data(train_data)
            results["validation"] = validation_results
            
            if not is_valid:
                results["error"] = "Data validation failed"
                return results
            
            # 2. Prepare features
            X_train, X_test, y_train, y_test = self.prepare_features(
                train_data, target_col=target_col
            )
            
            # 3. Train model
            self.train_model(X_train, y_train, hyperparameters)
            
            # 4. Evaluate model
            metrics = self.evaluate_model(X_test, y_test)
            results["metrics"] = metrics
            
            # 5. Check quality gates
            quality_gates = quality_gates or {}
            passed, gate_results = self.check_model_quality_gates(**quality_gates)
            results["quality_gates"] = gate_results
            
            if not passed:
                results["error"] = "Quality gates not met"
                return results
            
            # 6. Save model
            model_path = self.save_model(version)
            results["model_path"] = str(model_path)
            results["status"] = "SUCCESS"
            
            logger.info("Pipeline completed successfully")
            
        except Exception as e:
            logger.error(f"Pipeline failed: {str(e)}")
            results["error"] = str(e)
            results["status"] = "FAILED"
        
        return results


def main():
    """Example usage of the pipeline."""
    # Generate sample data
    np.random.seed(42)
    n_samples = 10000
    
    data = pd.DataFrame({
        'amount': np.random.exponential(100, n_samples),
        'time': np.random.uniform(0, 24, n_samples),
        'merchant_risk_score': np.random.uniform(0, 1, n_samples),
        'customer_age': np.random.randint(18, 80, n_samples),
        'transaction_count_24h': np.random.poisson(5, n_samples),
    })
    
    # Generate fraud labels (imbalanced)
    fraud_probability = (
        0.01 + 
        0.1 * (data['amount'] > 500).astype(int) +
        0.15 * (data['merchant_risk_score'] > 0.7).astype(int)
    )
    data['is_fraud'] = np.random.binomial(1, fraud_probability)
    
    # Initialize and run pipeline
    pipeline = FraudDetectionPipeline(model_type="random_forest")
    
    results = pipeline.run_full_pipeline(
        train_data=data,
        version="v1.0.0",
        quality_gates={
            "min_roc_auc": 0.80,
            "min_recall": 0.70,
            "max_false_positive_rate": 0.10
        }
    )
    
    print(json.dumps(results, indent=2))


if __name__ == "__main__":
    main()
```

```python
class TestModelEvaluation:
    """Tests for model evaluation."""
    
    def test_evaluate_model_metrics(self, trained_pipeline, sample_data):
        """Test model evaluation returns correct metrics."""
        X_train, X_test, y_train, y_test = trained_pipeline.prepare_features(sample_data)
        
        metrics = trained_pipeline.evaluate_model(X_test, y_test)
        
        required_metrics = [
            'accuracy', 'precision', 'recall', 'f1_score', 'roc_auc',
            'true_positives', 'true_negatives', 'false_positives', 'false_negatives',
            'false_positive_rate', 'false_negative_rate'
        ]
        
        for metric in required_metrics:
            assert metric in metrics
            assert isinstance(metrics[metric], (int, float))
    
    def test_evaluate_model_metric_ranges(self, trained_pipeline, sample_data):
        """Test metrics are within valid ranges."""
        X_train, X_test, y_train, y_test = trained_pipeline.prepare_features(sample_data)
        
        metrics = trained_pipeline.evaluate_model(X_test, y_test)
        
        # Metrics should be between 0 and 1
        for metric in ['accuracy', 'precision', 'recall', 'f1_score', 'roc_auc']:
            assert 0 <= metrics[metric] <= 1
        
        # Rates should be between 0 and 1
        assert 0 <= metrics['false_positive_rate'] <= 1
        assert 0 <= metrics['false_negative_rate'] <= 1
    
    def test_evaluate_model_confusion_matrix(self, trained_pipeline, sample_data):
        """Test confusion matrix values."""
        X_train, X_test, y_train, y_test = trained_pipeline.prepare_features(sample_data)
        
        metrics = trained_pipeline.evaluate_model(X_test, y_test)
        
        total = (
            metrics['true_positives'] + metrics['true_negatives'] +
            metrics['false_positives'] + metrics['false_negatives']
        )
        
        assert total == len(y_test)
    
    def test_evaluate_with_different_threshold(self, trained_pipeline, sample_data):
        """Test evaluation with different classification threshold."""
        X_train, X_test, y_train, y_test = trained_pipeline.prepare_features(sample_data)
        
        metrics_05 = trained_pipeline.evaluate_model(X_test, y_test, threshold=0.5)
        metrics_03 = trained_pipeline.evaluate_model(X_test, y_test, threshold=0.3)
        
        # Lower threshold should increase recall
        assert metrics_03['recall'] >= metrics_05['recall']
```
