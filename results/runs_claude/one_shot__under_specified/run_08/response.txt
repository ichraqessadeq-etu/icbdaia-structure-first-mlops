```python
"""

class ModelConfig:
    """Configuration for fraud detection model."""
    
    def __init__(self, config_path: Optional[str] = None):
        self.config = self._load_config(config_path)
        
    def _load_config(self, config_path: Optional[str]) -> Dict[str, Any]:
        """Load configuration from file or use defaults."""
        default_config = {
            "model_type": "random_forest",
            "model_params": {
                "n_estimators": 100,
                "max_depth": 10,
                "min_samples_split": 5,
                "random_state": 42
            },
            "training": {
                "test_size": 0.2,
                "validation_size": 0.1,
                "cv_folds": 5
            },
            "thresholds": {
                "min_accuracy": 0.85,
                "min_precision": 0.80,
                "min_recall": 0.75,
                "min_f1": 0.77,
                "min_auc": 0.85,
                "max_data_drift": 0.3
            },
            "features": [
                "transaction_amount",
                "transaction_hour",
                "transaction_day",
                "merchant_category",
                "card_present",
                "international",
                "num_transactions_24h",
                "avg_transaction_amount_30d"
            ],
            "target": "is_fraud",
            "mlflow": {
                "tracking_uri": "http://localhost:5000",
                "experiment_name": "fraud_detection"
            },
            "deployment": {
                "model_registry_name": "fraud_detection_model",
                "s3_bucket": "fraud-detection-models",
                "endpoint_name": "fraud-detection-endpoint"
            }
        }
        
        if config_path and os.path.exists(config_path):
            with open(config_path, 'r') as f:
                loaded_config = json.load(f)
                default_config.update(loaded_config)
        
        return default_config
    
    def get(self, key: str, default: Any = None) -> Any:
        """Get configuration value."""
        keys = key.split('.')
        value = self.config
        for k in keys:
            value = value.get(k, default)
            if value is None:
                return default
        return value


class DataValidator:
    """Validate data quality and schema."""
    
    def __init__(self, config: ModelConfig):
        self.config = config
        
    def validate_schema(self, df: pd.DataFrame) -> Tuple[bool, str]:
        """Validate that dataframe has required columns and types."""
        required_features = self.config.get('features', [])
        target = self.config.get('target')
        
        missing_cols = set(required_features + [target]) - set(df.columns)
        if missing_cols:
            return False, f"Missing required columns: {missing_cols}"
        
        # Check for null values
        null_counts = df[required_features + [target]].isnull().sum()
        if null_counts.any():
            return False, f"Null values found: {null_counts[null_counts > 0].to_dict()}"
        
        # Check target is binary
        unique_targets = df[target].unique()
        if not set(unique_targets).issubset({0, 1}):
            return False, f"Target must be binary (0/1), found: {unique_targets}"
        
        return True, "Schema validation passed"
    
    def validate_data_quality(self, df: pd.DataFrame) -> Tuple[bool, str]:
        """Validate data quality metrics."""
        features = self.config.get('features', [])
        
        # Check for duplicates
        dup_count = df.duplicated().sum()
        if dup_count > len(df) * 0.01:  # More than 1% duplicates
            logger.warning(f"High duplicate rate: {dup_count} ({dup_count/len(df)*100:.2f}%)")
        
        # Check for outliers (simple IQR method)
        numeric_features = df[features].select_dtypes(include=[np.number]).columns
        for col in numeric_features:
            Q1 = df[col].quantile(0.25)
            Q3 = df[col].quantile(0.75)
            IQR = Q3 - Q1
            outliers = ((df[col] < (Q1 - 3 * IQR)) | (df[col] > (Q3 + 3 * IQR))).sum()
            if outliers > len(df) * 0.05:  # More than 5% outliers
                logger.warning(f"High outlier rate in {col}: {outliers} ({outliers/len(df)*100:.2f}%)")
        
        # Check class imbalance
        target = self.config.get('target')
        fraud_rate = df[target].mean()
        if fraud_rate < 0.001 or fraud_rate > 0.5:
            logger.warning(f"Unusual fraud rate: {fraud_rate:.4f}")
        
        return True, "Data quality validation passed"


class ModelTrainer:
    """Train and validate fraud detection model."""
    
    def __init__(self, config: ModelConfig):
        self.config = config
        self.model = None
        self.scaler = StandardScaler()
        self.metrics = {}
        
    def prepare_data(
        self, df: pd.DataFrame
    ) -> Tuple[np.ndarray, np.ndarray, np.ndarray, np.ndarray, np.ndarray, np.ndarray]:
        """Prepare data for training."""
        features = self.config.get('features', [])
        target = self.config.get('target')
        test_size = self.config.get('training.test_size', 0.2)
        val_size = self.config.get('training.validation_size', 0.1)
        
        X = df[features].values
        y = df[target].values
        
        # Split into train+val and test
        X_temp, X_test, y_temp, y_test = train_test_split(
            X, y, test_size=test_size, random_state=42, stratify=y
        )
        
        # Split train+val into train and validation
        val_size_adjusted = val_size / (1 - test_size)
        X_train, X_val, y_train, y_val = train_test_split(
            X_temp, y_temp, test_size=val_size_adjusted, random_state=42, stratify=y_temp
        )
        
        # Scale features
        X_train = self.scaler.fit_transform(X_train)
        X_val = self.scaler.transform(X_val)
        X_test = self.scaler.transform(X_test)
        
        logger.info(f"Data split - Train: {len(X_train)}, Val: {len(X_val)}, Test: {len(X_test)}")
        logger.info(f"Fraud rate - Train: {y_train.mean():.4f}, Val: {y_val.mean():.4f}, Test: {y_test.mean():.4f}")
        
        return X_train, X_val, X_test, y_train, y_val, y_test
    
    def train(
        self, X_train: np.ndarray, y_train: np.ndarray
    ) -> None:
        """Train the model."""
        model_type = self.config.get('model_type', 'random_forest')
        model_params = self.config.get('model_params', {})
        
        logger.info(f"Training {model_type} model with params: {model_params}")
        
        if model_type == 'random_forest':
            self.model = RandomForestClassifier(**model_params)
        elif model_type == 'gradient_boosting':
            self.model = GradientBoostingClassifier(**model_params)
        else:
            raise ValueError(f"Unknown model type: {model_type}")
        
        self.model.fit(X_train, y_train)
        
        # Cross-validation
        cv_folds = self.config.get('training.cv_folds', 5)
        cv_scores = cross_val_score(
            self.model, X_train, y_train, cv=cv_folds, scoring='f1'
        )
        logger.info(f"Cross-validation F1 scores: {cv_scores}")
        logger.info(f"Mean CV F1: {cv_scores.mean():.4f} (+/- {cv_scores.std() * 2:.4f})")
        
        self.metrics['cv_f1_mean'] = float(cv_scores.mean())
        self.metrics['cv_f1_std'] = float(cv_scores.std())
    
    def evaluate(
        self, X: np.ndarray, y: np.ndarray, dataset_name: str = 'test'
    ) -> Dict[str, float]:
        """Evaluate model performance."""
        if self.model is None:
            raise ValueError("Model not trained yet")
        
        y_pred = self.model.predict(X)
        y_pred_proba = self.model.predict_proba(X)[:, 1]
        
        metrics = {
            f'{dataset_name}_accuracy': float(accuracy_score(y, y_pred)),
            f'{dataset_name}_precision': float(precision_score(y, y_pred, zero_division=0)),
            f'{dataset_name}_recall': float(recall_score(y, y_pred, zero_division=0)),
            f'{dataset_name}_f1': float(f1_score(y, y_pred, zero_division=0)),
            f'{dataset_name}_auc': float(roc_auc_score(y, y_pred_proba))
        }
        
        cm = confusion_matrix(y, y_pred)
        logger.info(f"\n{dataset_name.upper()} Set Metrics:")
        logger.info(f"Accuracy: {metrics[f'{dataset_name}_accuracy']:.4f}")
        logger.info(f"Precision: {metrics[f'{dataset_name}_precision']:.4f}")
        logger.info(f"Recall: {metrics[f'{dataset_name}_recall']:.4f}")
        logger.info(f"F1 Score: {metrics[f'{dataset_name}_f1']:.4f}")
        logger.info(f"AUC-ROC: {metrics[f'{dataset_name}_auc']:.4f}")
        logger.info(f"\nConfusion Matrix:\n{cm}")
        
        self.metrics.update(metrics)
        return metrics
    
    def validate_metrics(self) -> Tuple[bool, str]:
        """Validate that metrics meet minimum thresholds."""
        thresholds = self.config.get('thresholds', {})
        
        checks = [
            ('test_accuracy', 'min_accuracy'),
            ('test_precision', 'min_precision'),
            ('test_recall', 'min_recall'),
            ('test_f1', 'min_f1'),
            ('test_auc', 'min_auc')
        ]
        
        for metric_name, threshold_name in checks:
            metric_value = self.metrics.get(metric_name, 0)
            threshold_value = thresholds.get(threshold_name, 0)
            
            if metric_value < threshold_value:
                return False, f"{metric_name} ({metric_value:.4f}) below threshold ({threshold_value:.4f})"
        
        return True, "All metrics meet thresholds"


class DataDriftDetector:
    """Detect data drift between reference and production data."""
    
    def __init__(self, config: ModelConfig):
        self.config = config
        
    def detect_drift(
        self, reference_data: pd.DataFrame, current_data: pd.DataFrame
    ) -> Tuple[bool, Dict[str, Any]]:
        """Detect data drift using Evidently."""
        features = self.config.get('features', [])
        target = self.config.get('target')
        
        # Create drift report
        report = Report(metrics=[
            DataDriftPreset(),
        ])
        
        report.run(
            reference_data=reference_data[features + [target]],
            current_data=current_data[features + [target]]
        )
        
        # Extract drift results
        report_dict = report.as_dict()
        
        # Check for drift
        drift_score = report_dict['metrics'][0]['result']['dataset_drift']
        max_drift = self.config.get('thresholds.max_data_drift', 0.3)
        
        drift_detected = drift_score > max_drift
        
        drift_info = {
            'drift_score': drift_score,
            'drift_detected': drift_detected,
            'threshold': max_drift,
            'report': report_dict
        }
        
        logger.info(f"Data drift score: {drift_score:.4f} (threshold: {max_drift})")
        
        return drift_detected, drift_info


class ModelRegistry:
    """Manage model versioning and deployment."""
    
    def __init__(self, config: ModelConfig):
        self.config = config
        self._setup_mlflow()
        
    def _setup_mlflow(self):
        """Setup MLflow tracking."""
        tracking_uri = self.config.get('mlflow.tracking_uri')
        experiment_name = self.config.get('mlflow.experiment_name')
        
        mlflow.set_tracking_uri(tracking_uri)
        mlflow.set_experiment(experiment_name)
        
    def log_model(
        self,
        model: Any,
        scaler: Any,
        metrics: Dict[str, float],
        artifacts: Dict[str, str] = None
    ) -> str:
        """Log model to MLflow."""
        with mlflow.start_run() as run:
            # Log parameters
            model_params = self.config.get('model_params', {})
            mlflow.log_params(model_params)
            
            # Log metrics
            mlflow.log_metrics(metrics)
            
            # Log model
            mlflow.sklearn.log_model(
                model,
                "model",
                registered_model_name=self.config.get('deployment.model_registry_name')
            )
            
            # Log scaler
            with open("scaler.pkl", "wb") as f:
                pickle.dump(scaler, f)
            mlflow.log_artifact("scaler.pkl")
            
            # Log additional artifacts
            if artifacts:
                for name, path in artifacts.items():
                    mlflow.log_artifact(path, name)
            
            # Log config
            config_path = "model_config.json"
            with open(config_path, "w") as f:
                json.dump(self.config.config, f, indent=2)
            mlflow.log_artifact(config_path)
            
            run_id = run.info.run_id
            logger.info(f"Model logged with run_id: {run_id}")
            
            return run_id
    
    def promote_model(self, run_id: str, stage: str = "Production") -> None:
        """Promote model to specific stage."""
        client = mlflow.tracking.MlflowClient()
        model_name = self.config.get('deployment.model_registry_name')
        
        # Get model version
        model_versions = client.search_model_versions(f"run_id='{run_id}'")
        
        if not model_versions:
            raise ValueError(f"No model version found for run_id: {run_id}")
        
        version = model_versions[0].version
        
        # Transition to stage
        client.transition_model_version_stage(
            name=model_name,
            version=version,
            stage=stage
        )
        
        logger.info(f"Model version {version} promoted to {stage}")


class ModelDeployer:
    """Deploy model to production environment."""
    
    def __init__(self, config: ModelConfig):
        self.config = config
        self.s3_client = boto3.client('s3')
        
    def upload_to_s3(self, local_path: str, s3_key: str) -> str:
        """Upload model artifacts to S3."""
        bucket = self.config.get('deployment.s3_bucket')
        
        try:
            self.s3_client.upload_file(local_path, bucket, s3_key)
            s3_uri = f"s3://{bucket}/{s3_key}"
            logger.info(f"Uploaded {local_path} to {s3_uri}")
            return s3_uri
        except ClientError as e:
            logger.error(f"Failed to upload to S3: {e}")
            raise
    
    def create_model_package(
        self, model_path: str, scaler_path: str, output_dir: str = "model_package"
    ) -> str:
        """Create deployment package."""
        Path(output_dir).mkdir(exist_ok=True)
        
        # Copy model and scaler
        import shutil
        shutil.copy(model_path, os.path.join(output_dir, "model.pkl"))
        shutil.copy(scaler_path, os.path.join(output_dir, "scaler.pkl"))
        
        # Save config
        config_path = os.path.join(output_dir, "config.json")
        with open(config_path, "w") as f:
            json.dump(self.config.config, f, indent=2)
        
        # Create metadata
        metadata = {
            "created_at": datetime.now().isoformat(),
            "model_type": self.config.get('model_type'),
            "features": self.config.get('features'),
            "target": self.config.get('target')
        }
        
        metadata_path = os.path.join(output_dir, "metadata.json")
        with open(metadata_path, "w") as f:
            json.dump(metadata, f, indent=2)
        
        logger.info(f"Model package created at {output_dir}")
        return output_dir


class FraudDetectionPipeline:
    """Main pipeline orchestrator."""
    
    def __init__(self, config_path: Optional[str] = None):
        self.config = ModelConfig(config_path)
        self.validator = DataValidator(self.config)
        self.trainer = ModelTrainer(self.config)
        self.drift_detector = DataDriftDetector(self.config)
        self.registry = ModelRegistry(self.config)
        self.deployer = ModelDeployer(self.config)
        
    def run_training_pipeline(
        self, 
        train_data_path: str,
        reference_data_path: Optional[str] = None
    ) -> Dict[str, Any]:
        """Run complete training pipeline."""
        logger.info("=" * 80)
        logger.info("Starting Fraud Detection Training Pipeline")
        logger.info("=" * 80)
        
        results = {
            'success': False,
            'metrics': {},
            'run_id': None,
            'errors': []
        }
        
        try:
            # Load data
            logger.info(f"Loading training data from {train_data_path}")
            df = pd.read_csv(train_data_path)
            logger.info(f"Loaded {len(df)} records")
            
            # Validate schema
            valid, message = self.validator.validate_schema(df)
            if not valid:
                results['errors'].append(f"Schema validation failed: {message}")
                return results
            logger.info(f"✓ {message}")
            
            # Validate data quality
            valid, message = self.validator.validate_data_quality(df)
            if not valid:
                results['errors'].append(f"Data quality validation failed: {message}")
                return results
            logger.info(f"✓ {message}")
            
            # Check for data drift if reference data provided
            if reference_data_path and os.path.exists(reference_data_path):
                logger.info("Checking for data drift...")
                reference_df = pd.read_csv(reference_data_path)
                drift_detected, drift_info = self.drift_detector.detect_drift(
                    reference_df, df
                )
                
                if drift_detected:
                    logger.warning(f"⚠ Data drift detected! Score: {drift_info['drift_score']:.4f}")
                else:
                    logger.info(f"✓ No significant data drift detected")
                
                results['drift_info'] = drift_info
            
            # Prepare data
            logger.info("Preparing data for training...")
            X_train, X_val, X_test, y_train, y_val, y_test = self.trainer.prepare_data(df)
            
            # Train model
            logger.info("Training model...")
            self.trainer.train(X_train, y_train)
            logger.info("✓ Model training completed")
            
            # Evaluate on validation set
            logger.info("Evaluating on validation set...")
            val_metrics = self.trainer.evaluate(X_val, y_val, 'validation')
            
            # Evaluate on test set
            logger.info("Evaluating on test set...")
            test_metrics = self.trainer.evaluate(X_test, y_test, 'test')
            
            # Validate metrics
            valid, message = self.trainer.validate_metrics()
            if not valid:
                results['errors'].append(f"Metrics validation failed: {message}")
                logger.error(f"✗ {message}")
                return results
            logger.info(f"✓ {message}")
            
            # Log to MLflow
            logger.info("Logging model to MLflow...")
            run_id = self.registry.log_model(
                self.trainer.model,
                self.trainer.scaler,
                self.trainer.metrics
            )
            
            results['success'] = True
            results['metrics'] = self.trainer.metrics
            results['run_id'] = run_id
            
            logger.info("=" * 80)
            logger.info("✓ Training Pipeline Completed Successfully")
            logger.info(f"Run ID: {run_id}")
            logger.info("=" * 80)
            
        except Exception as e:
            logger.error(f"Pipeline failed with error: {str(e)}", exc_info=True)
            results['errors'].append(str(e))
        
        return results
    
    def run_deployment_pipeline(
        self, run_id: str, stage: str = "Production"
    ) -> Dict[str, Any]:
        """Run deployment pipeline."""
        logger.info("=" * 80)
        logger.info("Starting Fraud Detection Deployment Pipeline")
        logger.info("=" * 80)
        
        results = {
            'success': False,
            'model_uri': None,
            'errors': []
        }
        
        try:
            # Promote model
            logger.info(f"Promoting model {run_id} to {stage}...")
            self.registry.promote_model(run_id, stage)
            logger.info("✓ Model promoted")
            
            # Create deployment package
            logger.info("Creating deployment package...")
            # In real scenario, download from MLflow
            package_dir = self.deployer.create_model_package(
                "model.pkl",  # These would be downloaded from MLflow
                "scaler.pkl"
            )
            logger.info("✓ Deployment package created")
            
            # Upload to S3
            timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
            s3_key = f"models/fraud_detection/{timestamp}/model_package.tar.gz"
            
            # In real scenario, tar the package and upload
            logger.info(f"Uploading to S3: {s3_key}")
            # model_uri = self.deployer.upload_to_s3(package_path, s3_key)
            model_uri = f"s3://{self.config.get('deployment.s3_bucket')}/{s3_key}"
            
            results['success'] = True
            results['model_uri'] = model_uri
            
            logger.info("=" * 80)
            logger.info("✓ Deployment Pipeline Completed Successfully")
            logger.info(f"Model URI: {model_uri}")
            logger.info("=" * 80)
            
        except Exception as e:
            logger.error(f"Deployment failed with error: {str(e)}", exc_info=True)
            results['errors'].append(str(e))
        
        return results


def main():
    """Main entry point for CLI usage."""
    import argparse
    
    parser = argparse.ArgumentParser(description="Fraud Detection ML Pipeline")
    parser.add_argument(
        '--mode',
        choices=['train', 'deploy'],
        required=True,
        help='Pipeline mode'
    )
    parser.add_argument(
        '--config',
        type=str,
        help='Path to configuration file'
    )
    parser.add_argument(
        '--train-data',
        type=str,
        help='Path to training data (for train mode)'
    )
    parser.add_argument(
        '--reference-data',
        type=str,
        help='Path to reference data for drift detection'
    )
    parser.add_argument(
        '--run-id',
        type=str,
        help='MLflow run ID (for deploy mode)'
    )
    parser.add_argument(
        '--stage',
        type=str,
        default='Production',
        help='Deployment stage'
    )
    
    args = parser.parse_args()
    
    pipeline = FraudDetectionPipeline(args.config)
    
    if args.mode == 'train':
        if not args.train_data:
            parser.error("--train-data is required for train mode")
        
        results = pipeline.run_training_pipeline(
            args.train_data,
            args.reference_data
        )
        
        if results['success']:
            print(f"\n✓ Training completed successfully!")
            print(f"Run ID: {results['run_id']}")
            print(f"\nMetrics:")
            for key, value in results['metrics'].items():
                print(f"  {key}: {value:.4f}")
        else:
            print(f"\n✗ Training failed!")
            print(f"Errors: {results['errors']}")
            exit(1)
    
    elif args.mode == 'deploy':
        if not args.run_id:
            parser.error("--run-id is required for deploy mode")
        
        results = pipeline.run_deployment_pipeline(args.run_id, args.stage)
        
        if results['success']:
            print(f"\n✓ Deployment completed successfully!")
            print(f"Model URI: {results['model_uri']}")
        else:
            print(f"\n✗ Deployment failed!")
            print(f"Errors: {results['errors']}")
            exit(1)


if __name__ == "__main__":
    main()
```
