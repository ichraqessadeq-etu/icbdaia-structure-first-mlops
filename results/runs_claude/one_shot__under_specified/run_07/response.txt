## 1. Pipeline Module (`pipeline/fraud_detection_pipeline.py`)

```python
"""
Fraud Detection ML Pipeline
Handles training, validation, and deployment of fraud detection models
"""

import os
import json
import pickle
import logging
from pathlib import Path
from typing import Dict, Tuple, Any, Optional
from datetime import datetime

import numpy as np
import pandas as pd
from sklearn.model_selection import train_test_split, cross_val_score
from sklearn.ensemble import RandomForestClassifier, GradientBoostingClassifier
from sklearn.preprocessing import StandardScaler
from sklearn.metrics import (
    accuracy_score, precision_score, recall_score, 
    f1_score, roc_auc_score, confusion_matrix,
    precision_recall_curve
)
import joblib
import mlflow
import mlflow.sklearn
from evidently.report import Report
from evidently.metric_preset import DataDriftPreset, ClassificationPreset

logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)


class FraudDetectionPipeline:
    """End-to-end pipeline for fraud detection model"""
    
    def __init__(
        self,
        model_type: str = "random_forest",
        experiment_name: str = "fraud_detection",
        tracking_uri: Optional[str] = None
    ):
        """
        Initialize the pipeline
        
        Args:
            model_type: Type of model ('random_forest' or 'gradient_boosting')
            experiment_name: MLflow experiment name
            tracking_uri: MLflow tracking URI
        """
        self.model_type = model_type
        self.model = None
        self.scaler = StandardScaler()
        self.feature_names = None
        self.metrics = {}
        
        # MLflow setup
        if tracking_uri:
            mlflow.set_tracking_uri(tracking_uri)
        mlflow.set_experiment(experiment_name)
        
        # Model registry
        self.models = {
            "random_forest": RandomForestClassifier(
                n_estimators=100,
                max_depth=10,
                min_samples_split=5,
                min_samples_leaf=2,
                random_state=42,
                n_jobs=-1
            ),
            "gradient_boosting": GradientBoostingClassifier(
                n_estimators=100,
                learning_rate=0.1,
                max_depth=5,
                random_state=42
            )
        }
    
    def load_data(
        self, 
        data_path: str,
        target_column: str = "is_fraud"
    ) -> Tuple[pd.DataFrame, pd.Series]:
        """
        Load and validate data
        
        Args:
            data_path: Path to the data file
            target_column: Name of the target column
            
        Returns:
            Features and target
        """
        logger.info(f"Loading data from {data_path}")
        
        if not os.path.exists(data_path):
            raise FileNotFoundError(f"Data file not found: {data_path}")
        
        df = pd.read_csv(data_path)
        
        if target_column not in df.columns:
            raise ValueError(f"Target column '{target_column}' not found in data")
        
        # Basic validation
        if df.isnull().any().any():
            logger.warning("Missing values detected. Dropping rows with missing values.")
            df = df.dropna()
        
        X = df.drop(columns=[target_column])
        y = df[target_column]
        
        self.feature_names = X.columns.tolist()
        
        logger.info(f"Loaded {len(df)} samples with {len(X.columns)} features")
        logger.info(f"Fraud rate: {y.mean():.2%}")
        
        return X, y
    
    def preprocess_data(
        self,
        X: pd.DataFrame,
        fit: bool = True
    ) -> np.ndarray:
        """
        Preprocess features
        
        Args:
            X: Feature dataframe
            fit: Whether to fit the scaler
            
        Returns:
            Scaled features
        """
        logger.info("Preprocessing data")
        
        if fit:
            X_scaled = self.scaler.fit_transform(X)
        else:
            X_scaled = self.scaler.transform(X)
        
        return X_scaled
    
    def validate_data_quality(
        self,
        X: pd.DataFrame,
        y: pd.Series,
        min_samples: int = 1000,
        min_fraud_rate: float = 0.001,
        max_fraud_rate: float = 0.5
    ) -> bool:
        """
        Validate data quality
        
        Args:
            X: Features
            y: Target
            min_samples: Minimum number of samples required
            min_fraud_rate: Minimum fraud rate
            max_fraud_rate: Maximum fraud rate
            
        Returns:
            True if data passes validation
        """
        logger.info("Validating data quality")
        
        # Check sample size
        if len(X) < min_samples:
            raise ValueError(f"Insufficient samples: {len(X)} < {min_samples}")
        
        # Check fraud rate
        fraud_rate = y.mean()
        if fraud_rate < min_fraud_rate or fraud_rate > max_fraud_rate:
            raise ValueError(
                f"Fraud rate {fraud_rate:.2%} outside acceptable range "
                f"[{min_fraud_rate:.2%}, {max_fraud_rate:.2%}]"
            )
        
        # Check for duplicate features
        duplicate_cols = X.columns[X.columns.duplicated()].tolist()
        if duplicate_cols:
            raise ValueError(f"Duplicate columns found: {duplicate_cols}")
        
        # Check for zero variance features
        zero_var_cols = X.columns[X.var() == 0].tolist()
        if zero_var_cols:
            logger.warning(f"Zero variance columns found: {zero_var_cols}")
        
        logger.info("Data quality validation passed")
        return True
    
    def train(
        self,
        X_train: np.ndarray,
        y_train: pd.Series,
        X_val: np.ndarray,
        y_val: pd.Series,
        **model_params
    ) -> Dict[str, float]:
        """
        Train the model
        
        Args:
            X_train: Training features
            y_train: Training target
            X_val: Validation features
            y_val: Validation target
            **model_params: Additional model parameters
            
        Returns:
            Training metrics
        """
        logger.info(f"Training {self.model_type} model")
        
        with mlflow.start_run():
            # Log parameters
            mlflow.log_param("model_type", self.model_type)
            mlflow.log_param("n_samples_train", len(X_train))
            mlflow.log_param("n_samples_val", len(X_val))
            mlflow.log_param("fraud_rate_train", y_train.mean())
            
            # Initialize model
            self.model = self.models[self.model_type]
            if model_params:
                self.model.set_params(**model_params)
                mlflow.log_params(model_params)
            
            # Train model
            self.model.fit(X_train, y_train)
            
            # Evaluate on training set
            train_metrics = self._evaluate(X_train, y_train, prefix="train")
            
            # Evaluate on validation set
            val_metrics = self._evaluate(X_val, y_val, prefix="val")
            
            # Combined metrics
            self.metrics = {**train_metrics, **val_metrics}
            
            # Log metrics to MLflow
            mlflow.log_metrics(self.metrics)
            
            # Log model
            mlflow.sklearn.log_model(
                self.model,
                "model",
                registered_model_name=f"fraud_detection_{self.model_type}"
            )
            
            # Log feature importance
            if hasattr(self.model, 'feature_importances_'):
                importance_df = pd.DataFrame({
                    'feature': self.feature_names,
                    'importance': self.model.feature_importances_
                }).sort_values('importance', ascending=False)
                
                importance_dict = importance_df.set_index('feature')['importance'].to_dict()
                mlflow.log_dict(importance_dict, "feature_importance.json")
            
            logger.info(f"Training completed. Val AUC: {self.metrics['val_roc_auc']:.4f}")
        
        return self.metrics
    
    def _evaluate(
        self,
        X: np.ndarray,
        y: pd.Series,
        prefix: str = ""
    ) -> Dict[str, float]:
        """
        Evaluate model performance
        
        Args:
            X: Features
            y: Target
            prefix: Metric name prefix
            
        Returns:
            Dictionary of metrics
        """
        y_pred = self.model.predict(X)
        y_pred_proba = self.model.predict_proba(X)[:, 1]
        
        metrics = {
            f"{prefix}_accuracy": accuracy_score(y, y_pred),
            f"{prefix}_precision": precision_score(y, y_pred, zero_division=0),
            f"{prefix}_recall": recall_score(y, y_pred, zero_division=0),
            f"{prefix}_f1": f1_score(y, y_pred, zero_division=0),
            f"{prefix}_roc_auc": roc_auc_score(y, y_pred_proba),
        }
        
        return metrics
    
    def validate_model_performance(
        self,
        min_auc: float = 0.75,
        min_precision: float = 0.70,
        min_recall: float = 0.60,
        max_train_val_auc_diff: float = 0.1
    ) -> bool:
        """
        Validate model performance meets requirements
        
        Args:
            min_auc: Minimum required AUC
            min_precision: Minimum required precision
            min_recall: Minimum required recall
            max_train_val_auc_diff: Maximum allowed AUC difference (overfitting check)
            
        Returns:
            True if model passes validation
        """
        logger.info("Validating model performance")
        
        val_auc = self.metrics.get('val_roc_auc', 0)
        val_precision = self.metrics.get('val_precision', 0)
        val_recall = self.metrics.get('val_recall', 0)
        train_auc = self.metrics.get('train_roc_auc', 0)
        
        # Check minimum performance
        if val_auc < min_auc:
            raise ValueError(f"Validation AUC {val_auc:.4f} < {min_auc}")
        
        if val_precision < min_precision:
            raise ValueError(f"Validation Precision {val_precision:.4f} < {min_precision}")
        
        if val_recall < min_recall:
            raise ValueError(f"Validation Recall {val_recall:.4f} < {min_recall}")
        
        # Check for overfitting
        auc_diff = train_auc - val_auc
        if auc_diff > max_train_val_auc_diff:
            logger.warning(
                f"Potential overfitting detected. "
                f"Train-Val AUC difference: {auc_diff:.4f}"
            )
        
        logger.info("Model performance validation passed")
        return True
    
    def detect_data_drift(
        self,
        reference_data: pd.DataFrame,
        current_data: pd.DataFrame,
        reference_target: pd.Series,
        current_target: pd.Series
    ) -> Dict[str, Any]:
        """
        Detect data drift using Evidently
        
        Args:
            reference_data: Reference dataset features
            current_data: Current dataset features
            reference_target: Reference target
            current_target: Current target
            
        Returns:
            Drift report metrics
        """
        logger.info("Detecting data drift")
        
        # Prepare data
        reference_df = reference_data.copy()
        reference_df['target'] = reference_target
        reference_df['prediction'] = self.model.predict(
            self.scaler.transform(reference_data)
        )
        
        current_df = current_data.copy()
        current_df['target'] = current_target
        current_df['prediction'] = self.model.predict(
            self.scaler.transform(current_data)
        )
        
        # Generate drift report
        report = Report(metrics=[
            DataDriftPreset(),
            ClassificationPreset(),
        ])
        
        report.run(
            reference_data=reference_df,
            current_data=current_df
        )
        
        # Extract metrics
        report_dict = report.as_dict()
        
        drift_detected = report_dict['metrics'][0]['result']['dataset_drift']
        
        logger.info(f"Data drift detected: {drift_detected}")
        
        return {
            'drift_detected': drift_detected,
            'report': report_dict
        }
    
    def save_model(
        self,
        output_dir: str,
        version: Optional[str] = None
    ) -> str:
        """
        Save model artifacts
        
        Args:
            output_dir: Output directory
            version: Model version (defaults to timestamp)
            
        Returns:
            Path to saved model
        """
        if version is None:
            version = datetime.now().strftime("%Y%m%d_%H%M%S")
        
        output_path = Path(output_dir) / version
        output_path.mkdir(parents=True, exist_ok=True)
        
        # Save model
        model_path = output_path / "model.pkl"
        joblib.dump(self.model, model_path)
        
        # Save scaler
        scaler_path = output_path / "scaler.pkl"
        joblib.dump(self.scaler, scaler_path)
        
        # Save metadata
        metadata = {
            'model_type': self.model_type,
            'version': version,
            'timestamp': datetime.now().isoformat(),
            'feature_names': self.feature_names,
            'metrics': self.metrics,
            'model_params': self.model.get_params()
        }
        
        metadata_path = output_path / "metadata.json"
        with open(metadata_path, 'w') as f:
            json.dump(metadata, f, indent=2)
        
        logger.info(f"Model saved to {output_path}")
        
        return str(output_path)
    
    def load_model(self, model_dir: str) -> None:
        """
        Load model artifacts
        
        Args:
            model_dir: Directory containing model artifacts
        """
        model_path = Path(model_dir)
        
        # Load model
        self.model = joblib.load(model_path / "model.pkl")
        
        # Load scaler
        self.scaler = joblib.load(model_path / "scaler.pkl")
        
        # Load metadata
        with open(model_path / "metadata.json", 'r') as f:
            metadata = json.load(f)
        
        self.model_type = metadata['model_type']
        self.feature_names = metadata['feature_names']
        self.metrics = metadata['metrics']
        
        logger.info(f"Model loaded from {model_dir}")
    
    def predict(
        self,
        X: pd.DataFrame,
        return_proba: bool = True
    ) -> np.ndarray:
        """
        Make predictions
        
        Args:
            X: Features
            return_proba: Whether to return probabilities
            
        Returns:
            Predictions or probabilities
        """
        if self.model is None:
            raise ValueError("Model not trained or loaded")
        
        X_scaled = self.preprocess_data(X, fit=False)
        
        if return_proba:
            return self.model.predict_proba(X_scaled)[:, 1]
        else:
            return self.model.predict(X_scaled)
    
    def run_pipeline(
        self,
        data_path: str,
        output_dir: str,
        test_size: float = 0.2,
        val_size: float = 0.2,
        random_state: int = 42,
        **model_params
    ) -> Dict[str, Any]:
        """
        Run the complete pipeline
        
        Args:
            data_path: Path to training data
            output_dir: Output directory for model artifacts
            test_size: Test set size
            val_size: Validation set size (from training set)
            random_state: Random state for reproducibility
            **model_params: Additional model parameters
            
        Returns:
            Pipeline results
        """
        logger.info("Starting fraud detection pipeline")
        
        # Load data
        X, y = self.load_data(data_path)
        
        # Validate data quality
        self.validate_data_quality(X, y)
        
        # Split data
        X_temp, X_test, y_temp, y_test = train_test_split(
            X, y, test_size=test_size, random_state=random_state, stratify=y
        )
        
        X_train, X_val, y_train, y_val = train_test_split(
            X_temp, y_temp, test_size=val_size, random_state=random_state, stratify=y_temp
        )
        
        # Preprocess
        X_train_scaled = self.preprocess_data(X_train, fit=True)
        X_val_scaled = self.preprocess_data(X_val, fit=False)
        X_test_scaled = self.preprocess_data(X_test, fit=False)
        
        # Train
        metrics = self.train(
            X_train_scaled, y_train,
            X_val_scaled, y_val,
            **model_params
        )
        
        # Validate performance
        self.validate_model_performance()
        
        # Test set evaluation
        test_metrics = self._evaluate(X_test_scaled, y_test, prefix="test")
        metrics.update(test_metrics)
        
        # Save model
        model_path = self.save_model(output_dir)
        
        logger.info("Pipeline completed successfully")
        
        return {
            'metrics': metrics,
            'model_path': model_path,
            'feature_names': self.feature_names
        }


def main():
    """Main execution function"""
    import argparse
    
    parser = argparse.ArgumentParser(description="Fraud Detection Pipeline")
    parser.add_argument("--data-path", required=True, help="Path to training data")
    parser.add_argument("--output-dir", default="models", help="Output directory")
    parser.add_argument("--model-type", default="random_forest", 
                       choices=["random_forest", "gradient_boosting"])
    parser.add_argument("--test-size", type=float, default=0.2)
    parser.add_argument("--mlflow-uri", default=None, help="MLflow tracking URI")
    
    args = parser.parse_args()
    
    # Initialize pipeline
    pipeline = FraudDetectionPipeline(
        model_type=args.model_type,
        tracking_uri=args.mlflow_uri
    )
    
    # Run pipeline
    results = pipeline.run_pipeline(
        data_path=args.data_path,
        output_dir=args.output_dir,
        test_size=args.test_size
    )
    
    print("\n" + "="*50)
    print("Pipeline Results")
    print("="*50)
    print(f"Model saved to: {results['model_path']}")
    print("\nMetrics:")
    for metric, value in results['metrics'].items():
        print(f"  {metric}: {value:.4f}")


if __name__ == "__main__":
    main()
```
