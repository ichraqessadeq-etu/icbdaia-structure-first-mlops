# ml_pipeline/fraud_detection_pipeline.py
"""
Fraud Detection Model CI/CD Pipeline
Handles training, validation, deployment, and monitoring
"""

import os
import json
import logging
from pathlib import Path
from typing import Dict, Tuple, Any, Optional
from datetime import datetime
import joblib
import numpy as np
import pandas as pd
from sklearn.ensemble import RandomForestClassifier, GradientBoostingClassifier
from sklearn.model_selection import train_test_split, cross_val_score
from sklearn.metrics import (
    precision_score, recall_score, f1_score, roc_auc_score,
    confusion_matrix, classification_report, average_precision_score
)
from sklearn.preprocessing import StandardScaler
import mlflow
import mlflow.sklearn
from evidently.report import Report
from evidently.metric_preset import DataDriftPreset, ClassificationPreset
import great_expectations as ge

logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)


class FraudDetectionPipeline:
    """Complete MLOps pipeline for fraud detection model"""
    
    def __init__(
        self,
        model_name: str = "fraud_detector",
        experiment_name: str = "fraud_detection",
        model_registry_path: str = "./models",
        data_path: str = "./data",
        mlflow_tracking_uri: Optional[str] = None
    ):
        self.model_name = model_name
        self.experiment_name = experiment_name
        self.model_registry_path = Path(model_registry_path)
        self.data_path = Path(data_path)
        self.model_registry_path.mkdir(parents=True, exist_ok=True)
        self.data_path.mkdir(parents=True, exist_ok=True)
        
        # MLflow setup
        if mlflow_tracking_uri:
            mlflow.set_tracking_uri(mlflow_tracking_uri)
        mlflow.set_experiment(experiment_name)
        
        self.model = None
        self.scaler = StandardScaler()
        self.metadata = {}
        
        # Performance thresholds
        self.thresholds = {
            "min_precision": 0.85,
            "min_recall": 0.80,
            "min_f1": 0.82,
            "min_roc_auc": 0.90,
            "max_false_positive_rate": 0.05
        }
    
    def validate_data(self, df: pd.DataFrame, is_training: bool = True) -> Tuple[bool, Dict]:
        """Validate data quality using Great Expectations"""
        logger.info("Validating data quality...")
        
        validation_results = {
            "passed": True,
            "checks": []
        }
        
        try:
            ge_df = ge.from_pandas(df)
            
            # Schema validation
            expected_columns = ['amount', 'merchant_id', 'customer_id', 
                              'transaction_time', 'is_fraud']
            if is_training:
                assert all(col in df.columns for col in expected_columns), \
                    f"Missing required columns. Expected: {expected_columns}"
            
            # Data quality checks
            checks = [
                ("no_null_values", ge_df.expect_column_values_to_not_be_null("amount")),
                ("amount_positive", ge_df.expect_column_values_to_be_between("amount", min_value=0)),
                ("valid_fraud_labels", ge_df.expect_column_values_to_be_in_set("is_fraud", [0, 1]))
            ]
            
            for check_name, result in checks:
                validation_results["checks"].append({
                    "name": check_name,
                    "passed": result.success
                })
                if not result.success:
                    validation_results["passed"] = False
                    logger.error(f"Data validation failed: {check_name}")
            
            # Check class imbalance
            if is_training:
                fraud_ratio = df['is_fraud'].mean()
                if fraud_ratio < 0.001 or fraud_ratio > 0.5:
                    logger.warning(f"Unusual fraud ratio: {fraud_ratio:.4f}")
            
            return validation_results["passed"], validation_results
            
        except Exception as e:
            logger.error(f"Data validation error: {str(e)}")
            validation_results["passed"] = False
            validation_results["error"] = str(e)
            return False, validation_results
    
    def preprocess_data(
        self, 
        df: pd.DataFrame, 
        fit_scaler: bool = True
    ) -> Tuple[np.ndarray, np.ndarray]:
        """Preprocess data for model training/inference"""
        logger.info("Preprocessing data...")
        
        # Feature engineering
        df = df.copy()
        df['hour'] = pd.to_datetime(df['transaction_time']).dt.hour
        df['day_of_week'] = pd.to_datetime(df['transaction_time']).dt.dayofweek
        df['is_weekend'] = df['day_of_week'].isin([5, 6]).astype(int)
        
        # Select features
        feature_columns = ['amount', 'merchant_id', 'customer_id', 
                          'hour', 'day_of_week', 'is_weekend']
        
        X = df[feature_columns].values
        y = df['is_fraud'].values if 'is_fraud' in df.columns else None
        
        # Scale features
        if fit_scaler:
            X = self.scaler.fit_transform(X)
        else:
            X = self.scaler.transform(X)
        
        return X, y
    
    def train_model(
        self, 
        X_train: np.ndarray, 
        y_train: np.ndarray,
        model_type: str = "random_forest"
    ) -> Dict[str, Any]:
        """Train fraud detection model"""
        logger.info(f"Training {model_type} model...")
        
        with mlflow.start_run(run_name=f"{self.model_name}_{datetime.now().strftime('%Y%m%d_%H%M%S')}"):
            # Model selection
            if model_type == "random_forest":
                self.model = RandomForestClassifier(
                    n_estimators=200,
                    max_depth=15,
                    min_samples_split=10,
                    min_samples_leaf=5,
                    class_weight='balanced',
                    random_state=42,
                    n_jobs=-1
                )
            elif model_type == "gradient_boosting":
                self.model = GradientBoostingClassifier(
                    n_estimators=150,
                    learning_rate=0.1,
                    max_depth=7,
                    min_samples_split=10,
                    random_state=42
                )
            else:
                raise ValueError(f"Unknown model type: {model_type}")
            
            # Log parameters
            mlflow.log_params(self.model.get_params())
            
            # Train model
            self.model.fit(X_train, y_train)
            
            # Cross-validation
            cv_scores = cross_val_score(
                self.model, X_train, y_train, 
                cv=5, scoring='f1', n_jobs=-1
            )
            
            training_metrics = {
                "cv_f1_mean": cv_scores.mean(),
                "cv_f1_std": cv_scores.std(),
                "model_type": model_type,
                "training_samples": len(X_train)
            }
            
            # Log metrics
            mlflow.log_metrics({
                "cv_f1_mean": cv_scores.mean(),
                "cv_f1_std": cv_scores.std()
            })
            
            # Log model
            mlflow.sklearn.log_model(self.model, "model")
            
            logger.info(f"Training complete. CV F1: {cv_scores.mean():.4f} (+/- {cv_scores.std():.4f})")
            
            return training_metrics
    
    def evaluate_model(
        self, 
        X_test: np.ndarray, 
        y_test: np.ndarray
    ) -> Tuple[bool, Dict[str, float]]:
        """Evaluate model performance against thresholds"""
        logger.info("Evaluating model performance...")
        
        if self.model is None:
            raise ValueError("Model not trained. Call train_model first.")
        
        # Predictions
        y_pred = self.model.predict(X_test)
        y_pred_proba = self.model.predict_proba(X_test)[:, 1]
        
        # Calculate metrics
        metrics = {
            "precision": precision_score(y_test, y_pred),
            "recall": recall_score(y_test, y_pred),
            "f1_score": f1_score(y_test, y_pred),
            "roc_auc": roc_auc_score(y_test, y_pred_proba),
            "average_precision": average_precision_score(y_test, y_pred_proba)
        }
        
        # Confusion matrix
        tn, fp, fn, tp = confusion_matrix(y_test, y_pred).ravel()
        metrics["true_negatives"] = int(tn)
        metrics["false_positives"] = int(fp)
        metrics["false_negatives"] = int(fn)
        metrics["true_positives"] = int(tp)
        metrics["false_positive_rate"] = fp / (fp + tn)
        
        # Log metrics to MLflow
        mlflow.log_metrics(metrics)
        
        # Check thresholds
        passed = (
            metrics["precision"] >= self.thresholds["min_precision"] and
            metrics["recall"] >= self.thresholds["min_recall"] and
            metrics["f1_score"] >= self.thresholds["min_f1"] and
            metrics["roc_auc"] >= self.thresholds["min_roc_auc"] and
            metrics["false_positive_rate"] <= self.thresholds["max_false_positive_rate"]
        )
        
        logger.info(f"Model evaluation {'PASSED' if passed else 'FAILED'}")
        logger.info(f"Metrics: {json.dumps(metrics, indent=2)}")
        
        return passed, metrics
    
    def detect_data_drift(
        self, 
        reference_data: pd.DataFrame, 
        current_data: pd.DataFrame
    ) -> Tuple[bool, Dict]:
        """Detect data drift using Evidently"""
        logger.info("Checking for data drift...")
        
        try:
            # Create drift report
            drift_report = Report(metrics=[
                DataDriftPreset(),
            ])
            
            drift_report.run(
                reference_data=reference_data,
                current_data=current_data
            )
            
            # Extract results
            report_dict = drift_report.as_dict()
            
            # Check if drift detected
            drift_detected = report_dict['metrics'][0]['result']['dataset_drift']
            
            drift_results = {
                "drift_detected": drift_detected,
                "drift_score": report_dict['metrics'][0]['result'].get('drift_share', 0),
                "timestamp": datetime.now().isoformat()
            }
            
            if drift_detected:
                logger.warning("Data drift detected!")
            else:
                logger.info("No significant data drift detected")
            
            return drift_detected, drift_results
            
        except Exception as e:
            logger.error(f"Drift detection error: {str(e)}")
            return False, {"error": str(e)}
    
    def save_model(self, version: str = "latest") -> Path:
        """Save model and artifacts to disk"""
        logger.info(f"Saving model version: {version}")
        
        if self.model is None:
            raise ValueError("No model to save")
        
        model_dir = self.model_registry_path / version
        model_dir.mkdir(parents=True, exist_ok=True)
        
        # Save model
        model_path = model_dir / "model.joblib"
        joblib.dump(self.model, model_path)
        
        # Save scaler
        scaler_path = model_dir / "scaler.joblib"
        joblib.dump(self.scaler, scaler_path)
        
        # Save metadata
        self.metadata.update({
            "version": version,
            "saved_at": datetime.now().isoformat(),
            "model_type": type(self.model).__name__
        })
        
        metadata_path = model_dir / "metadata.json"
        with open(metadata_path, 'w') as f:
            json.dump(self.metadata, f, indent=2)
        
        logger.info(f"Model saved to {model_dir}")
        return model_dir
    
    def load_model(self, version: str = "latest") -> bool:
        """Load model and artifacts from disk"""
        logger.info(f"Loading model version: {version}")
        
        model_dir = self.model_registry_path / version
        
        if not model_dir.exists():
            logger.error(f"Model version {version} not found")
            return False
        
        try:
            # Load model
            model_path = model_dir / "model.joblib"
            self.model = joblib.load(model_path)
            
            # Load scaler
            scaler_path = model_dir / "scaler.joblib"
            self.scaler = joblib.load(scaler_path)
            
            # Load metadata
            metadata_path = model_dir / "metadata.json"
            with open(metadata_path, 'r') as f:
                self.metadata = json.load(f)
            
            logger.info(f"Model loaded successfully from {model_dir}")
            return True
            
        except Exception as e:
            logger.error(f"Error loading model: {str(e)}")
            return False
    
    def run_full_pipeline(
        self, 
        train_data: pd.DataFrame,
        test_data: pd.DataFrame,
        model_version: str = "latest"
    ) -> Dict[str, Any]:
        """Execute complete CI/CD pipeline"""
        logger.info("Starting full ML pipeline...")
        
        pipeline_results = {
            "success": False,
            "stages": {}
        }
        
        try:
            # Stage 1: Data Validation
            logger.info("Stage 1: Data Validation")
            train_valid, train_validation = self.validate_data(train_data, is_training=True)
            test_valid, test_validation = self.validate_data(test_data, is_training=False)
            
            pipeline_results["stages"]["data_validation"] = {
                "train": train_validation,
                "test": test_validation,
                "passed": train_valid and test_valid
            }
            
            if not (train_valid and test_valid):
                logger.error("Data validation failed")
                return pipeline_results
            
            # Stage 2: Data Preprocessing
            logger.info("Stage 2: Data Preprocessing")
            X_train, y_train = self.preprocess_data(train_data, fit_scaler=True)
            X_test, y_test = self.preprocess_data(test_data, fit_scaler=False)
            
            pipeline_results["stages"]["preprocessing"] = {
                "train_shape": X_train.shape,
                "test_shape": X_test.shape,
                "passed": True
            }
            
            # Stage 3: Model Training
            logger.info("Stage 3: Model Training")
            training_metrics = self.train_model(X_train, y_train)
            pipeline_results["stages"]["training"] = {
                "metrics": training_metrics,
                "passed": True
            }
            
            # Stage 4: Model Evaluation
            logger.info("Stage 4: Model Evaluation")
            eval_passed, eval_metrics = self.evaluate_model(X_test, y_test)
            pipeline_results["stages"]["evaluation"] = {
                "metrics": eval_metrics,
                "thresholds": self.thresholds,
                "passed": eval_passed
            }
            
            if not eval_passed:
                logger.error("Model evaluation failed to meet thresholds")
                return pipeline_results
            
            # Stage 5: Drift Detection
            logger.info("Stage 5: Drift Detection")
            drift_detected, drift_results = self.detect_data_drift(
                train_data.head(1000), 
                test_data.head(1000)
            )
            pipeline_results["stages"]["drift_detection"] = {
                "results": drift_results,
                "passed": not drift_detected
            }
            
            # Stage 6: Model Saving
            logger.info("Stage 6: Model Saving")
            model_path = self.save_model(version=model_version)
            pipeline_results["stages"]["model_saving"] = {
                "path": str(model_path),
                "passed": True
            }
            
            pipeline_results["success"] = True
            logger.info("Pipeline completed successfully!")
            
        except Exception as e:
            logger.error(f"Pipeline failed: {str(e)}")
            pipeline_results["error"] = str(e)
        
        return pipeline_results


def generate_sample_data(n_samples: int = 10000, fraud_ratio: float = 0.02) -> pd.DataFrame:
    """Generate synthetic fraud detection data for testing"""
    np.random.seed(42)
    
    n_fraud = int(n_samples * fraud_ratio)
    n_legit = n_samples - n_fraud
    
    # Legitimate transactions
    legit_data = {
        'amount': np.random.lognormal(3, 1, n_legit),
        'merchant_id': np.random.randint(1, 1000, n_legit),
        'customer_id': np.random.randint(1, 5000, n_legit),
        'transaction_time': pd.date_range('2024-01-01', periods=n_legit, freq='1min'),
        'is_fraud': np.zeros(n_legit, dtype=int)
    }
    
    # Fraudulent transactions (higher amounts, unusual patterns)
    fraud_data = {
        'amount': np.random.lognormal(5, 1.5, n_fraud),
        'merchant_id': np.random.randint(1, 1000, n_fraud),
        'customer_id': np.random.randint(1, 5000, n_fraud),
        'transaction_time': pd.date_range('2024-01-01', periods=n_fraud, freq='1min'),
        'is_fraud': np.ones(n_fraud, dtype=int)
    }
    
    df_legit = pd.DataFrame(legit_data)
    df_fraud = pd.DataFrame(fraud_data)
    
    df = pd.concat([df_legit, df_fraud], ignore_index=True)
    df = df.sample(frac=1, random_state=42).reset_index(drop=True)
    
    return df


if __name__ == "__main__":
    # Example usage
    print("Generating sample data...")
    data = generate_sample_data(n_samples=10000)
    train_data, test_data = train_test_split(data, test_size=0.2, random_state=42)
    
    print("Initializing pipeline...")
    pipeline = FraudDetectionPipeline()
    
    print("Running full pipeline...")
    results = pipeline.run_full_pipeline(train_data, test_data, model_version="v1.0.0")
    
    print("\nPipeline Results:")
    print(json.dumps(results, indent=2, default=str))
