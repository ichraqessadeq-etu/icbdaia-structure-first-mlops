## 1. Pipeline Module

```python
# pipeline.py
"""
MLOps Pipeline for Fraud Detection Model
Handles training, validation, deployment, and monitoring
"""

import os
import json
import pickle
import logging
from typing import Dict, Tuple, Any, Optional
from datetime import datetime
from pathlib import Path

import pandas as pd
import numpy as np
from sklearn.ensemble import RandomForestClassifier, GradientBoostingClassifier
from sklearn.model_selection import train_test_split, cross_val_score
from sklearn.metrics import (
    precision_score, recall_score, f1_score, roc_auc_score,
    confusion_matrix, classification_report, average_precision_score
)
from sklearn.preprocessing import StandardScaler
import mlflow
import mlflow.sklearn
from evidently.report import Report
from evidently.metric_preset import DataDriftPreset, ClassificationPreset
import joblib

logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)


class FraudDetectionPipeline:
    """Complete MLOps pipeline for fraud detection model"""
    
    def __init__(
        self,
        model_name: str = "fraud_detection_model",
        experiment_name: str = "fraud_detection",
        artifact_path: str = "./artifacts",
        min_precision: float = 0.85,
        min_recall: float = 0.75,
        min_f1: float = 0.80,
        max_data_drift: float = 0.3
    ):
        self.model_name = model_name
        self.experiment_name = experiment_name
        self.artifact_path = Path(artifact_path)
        self.artifact_path.mkdir(parents=True, exist_ok=True)
        
        # Performance thresholds
        self.min_precision = min_precision
        self.min_recall = min_recall
        self.min_f1 = min_f1
        self.max_data_drift = max_data_drift
        
        # Model components
        self.model = None
        self.scaler = None
        self.feature_names = None
        self.metadata = {}
        
        # MLflow setup
        mlflow.set_experiment(self.experiment_name)
    
    def load_data(self, data_path: str) -> pd.DataFrame:
        """Load and validate input data"""
        logger.info(f"Loading data from {data_path}")
        
        if data_path.endswith('.csv'):
            df = pd.read_csv(data_path)
        elif data_path.endswith('.parquet'):
            df = pd.read_parquet(data_path)
        else:
            raise ValueError(f"Unsupported file format: {data_path}")
        
        logger.info(f"Loaded {len(df)} records with {len(df.columns)} columns")
        return df
    
    def preprocess_data(
        self,
        df: pd.DataFrame,
        target_col: str = 'is_fraud',
        test_size: float = 0.2,
        random_state: int = 42
    ) -> Tuple[np.ndarray, np.ndarray, np.ndarray, np.ndarray]:
        """Preprocess data and split into train/test sets"""
        logger.info("Preprocessing data")
        
        # Separate features and target
        X = df.drop(columns=[target_col])
        y = df[target_col]
        
        # Store feature names
        self.feature_names = X.columns.tolist()
        
        # Handle missing values
        X = X.fillna(X.median(numeric_only=True))
        
        # Split data
        X_train, X_test, y_train, y_test = train_test_split(
            X, y, test_size=test_size, random_state=random_state, stratify=y
        )
        
        # Scale features
        self.scaler = StandardScaler()
        X_train_scaled = self.scaler.fit_transform(X_train)
        X_test_scaled = self.scaler.transform(X_test)
        
        logger.info(f"Train set: {len(X_train)} samples, Test set: {len(X_test)} samples")
        logger.info(f"Fraud rate - Train: {y_train.mean():.2%}, Test: {y_test.mean():.2%}")
        
        return X_train_scaled, X_test_scaled, y_train.values, y_test.values
    
    def train_model(
        self,
        X_train: np.ndarray,
        y_train: np.ndarray,
        model_type: str = "random_forest",
        **model_params
    ) -> None:
        """Train the fraud detection model"""
        logger.info(f"Training {model_type} model")
        
        with mlflow.start_run(run_name=f"{self.model_name}_{datetime.now().strftime('%Y%m%d_%H%M%S')}"):
            # Select model
            if model_type == "random_forest":
                default_params = {
                    'n_estimators': 100,
                    'max_depth': 10,
                    'min_samples_split': 5,
                    'min_samples_leaf': 2,
                    'class_weight': 'balanced',
                    'random_state': 42,
                    'n_jobs': -1
                }
                default_params.update(model_params)
                self.model = RandomForestClassifier(**default_params)
            
            elif model_type == "gradient_boosting":
                default_params = {
                    'n_estimators': 100,
                    'learning_rate': 0.1,
                    'max_depth': 5,
                    'min_samples_split': 5,
                    'min_samples_leaf': 2,
                    'random_state': 42
                }
                default_params.update(model_params)
                self.model = GradientBoostingClassifier(**default_params)
            
            else:
                raise ValueError(f"Unsupported model type: {model_type}")
            
            # Train model
            self.model.fit(X_train, y_train)
            
            # Cross-validation
            cv_scores = cross_val_score(
                self.model, X_train, y_train, cv=5, scoring='f1'
            )
            
            # Log parameters
            mlflow.log_params(default_params)
            mlflow.log_param("model_type", model_type)
            mlflow.log_metric("cv_f1_mean", cv_scores.mean())
            mlflow.log_metric("cv_f1_std", cv_scores.std())
            
            logger.info(f"Cross-validation F1: {cv_scores.mean():.4f} (+/- {cv_scores.std():.4f})")
    
    def evaluate_model(
        self,
        X_test: np.ndarray,
        y_test: np.ndarray
    ) -> Dict[str, float]:
        """Evaluate model performance"""
        logger.info("Evaluating model")
        
        # Predictions
        y_pred = self.model.predict(X_test)
        y_pred_proba = self.model.predict_proba(X_test)[:, 1]
        
        # Calculate metrics
        metrics = {
            'precision': precision_score(y_test, y_pred),
            'recall': recall_score(y_test, y_pred),
            'f1_score': f1_score(y_test, y_pred),
            'roc_auc': roc_auc_score(y_test, y_pred_proba),
            'avg_precision': average_precision_score(y_test, y_pred_proba)
        }
        
        # Confusion matrix
        cm = confusion_matrix(y_test, y_pred)
        tn, fp, fn, tp = cm.ravel()
        
        metrics.update({
            'true_negatives': int(tn),
            'false_positives': int(fp),
            'false_negatives': int(fn),
            'true_positives': int(tp),
            'false_positive_rate': fp / (fp + tn),
            'false_negative_rate': fn / (fn + tp)
        })
        
        # Log metrics to MLflow
        with mlflow.start_run(nested=True):
            for metric_name, metric_value in metrics.items():
                if isinstance(metric_value, (int, float)):
                    mlflow.log_metric(metric_name, metric_value)
        
        # Log detailed report
        logger.info(f"\n{classification_report(y_test, y_pred)}")
        logger.info(f"Metrics: {json.dumps(metrics, indent=2)}")
        
        return metrics
    
    def validate_model(self, metrics: Dict[str, float]) -> Tuple[bool, list]:
        """Validate model meets performance thresholds"""
        logger.info("Validating model performance")
        
        validation_results = []
        is_valid = True
        
        checks = [
            ('precision', metrics['precision'], self.min_precision, '>='),
            ('recall', metrics['recall'], self.min_recall, '>='),
            ('f1_score', metrics['f1_score'], self.min_f1, '>='),
        ]
        
        for metric_name, value, threshold, operator in checks:
            passed = value >= threshold if operator == '>=' else value <= threshold
            
            validation_results.append({
                'metric': metric_name,
                'value': value,
                'threshold': threshold,
                'passed': passed
            })
            
            if not passed:
                is_valid = False
                logger.warning(
                    f"Validation failed: {metric_name}={value:.4f} "
                    f"does not meet threshold {operator} {threshold}"
                )
            else:
                logger.info(
                    f"Validation passed: {metric_name}={value:.4f} "
                    f"meets threshold {operator} {threshold}"
                )
        
        return is_valid, validation_results
    
    def check_data_drift(
        self,
        reference_data: pd.DataFrame,
        current_data: pd.DataFrame,
        target_col: str = 'is_fraud'
    ) -> Tuple[bool, Dict]:
        """Check for data drift using Evidently"""
        logger.info("Checking for data drift")
        
        # Remove target column for drift detection
        reference_features = reference_data.drop(columns=[target_col])
        current_features = current_data.drop(columns=[target_col])
        
        # Create drift report
        drift_report = Report(metrics=[DataDriftPreset()])
        drift_report.run(
            reference_data=reference_features,
            current_data=current_features
        )
        
        # Extract drift results
        drift_results = drift_report.as_dict()
        
        # Calculate overall drift score
        drift_metrics = drift_results['metrics'][0]['result']
        n_drifted_features = drift_metrics.get('number_of_drifted_columns', 0)
        n_total_features = drift_metrics.get('number_of_columns', len(reference_features.columns))
        drift_share = n_drifted_features / n_total_features if n_total_features > 0 else 0
        
        no_drift = drift_share <= self.max_data_drift
        
        drift_summary = {
            'drift_share': drift_share,
            'n_drifted_features': n_drifted_features,
            'n_total_features': n_total_features,
            'passed': no_drift
        }
        
        if not no_drift:
            logger.warning(
                f"Data drift detected: {drift_share:.2%} of features drifted "
                f"(threshold: {self.max_data_drift:.2%})"
            )
        else:
            logger.info(f"No significant data drift detected: {drift_share:.2%}")
        
        return no_drift, drift_summary
    
    def save_model(self, version: Optional[str] = None) -> str:
        """Save model and artifacts"""
        if version is None:
            version = datetime.now().strftime('%Y%m%d_%H%M%S')
        
        model_dir = self.artifact_path / f"model_v{version}"
        model_dir.mkdir(parents=True, exist_ok=True)
        
        logger.info(f"Saving model to {model_dir}")
        
        # Save model
        model_path = model_dir / "model.pkl"
        joblib.dump(self.model, model_path)
        
        # Save scaler
        scaler_path = model_dir / "scaler.pkl"
        joblib.dump(self.scaler, scaler_path)
        
        # Save metadata
        metadata = {
            'model_name': self.model_name,
            'version': version,
            'timestamp': datetime.now().isoformat(),
            'feature_names': self.feature_names,
            'model_type': type(self.model).__name__,
            'model_params': self.model.get_params(),
        }
        
        metadata_path = model_dir / "metadata.json"
        with open(metadata_path, 'w') as f:
            json.dump(metadata, f, indent=2)
        
        # Log to MLflow
        with mlflow.start_run(nested=True):
            mlflow.sklearn.log_model(self.model, "model")
            mlflow.log_artifact(str(scaler_path))
            mlflow.log_artifact(str(metadata_path))
        
        logger.info(f"Model saved successfully to {model_dir}")
        return str(model_dir)
    
    def load_model(self, model_dir: str) -> None:
        """Load model and artifacts"""
        logger.info(f"Loading model from {model_dir}")
        
        model_path = Path(model_dir)
        
        # Load model
        self.model = joblib.load(model_path / "model.pkl")
        
        # Load scaler
        self.scaler = joblib.load(model_path / "scaler.pkl")
        
        # Load metadata
        with open(model_path / "metadata.json", 'r') as f:
            self.metadata = json.load(f)
        
        self.feature_names = self.metadata.get('feature_names', [])
        
        logger.info(f"Model loaded successfully: {self.metadata.get('model_type')}")
    
    def predict(self, X: np.ndarray) -> Tuple[np.ndarray, np.ndarray]:
        """Make predictions"""
        if self.model is None:
            raise ValueError("Model not trained or loaded")
        
        X_scaled = self.scaler.transform(X)
        predictions = self.model.predict(X_scaled)
        probabilities = self.model.predict_proba(X_scaled)[:, 1]
        
        return predictions, probabilities
    
    def run_pipeline(
        self,
        train_data_path: str,
        target_col: str = 'is_fraud',
        model_type: str = "random_forest",
        validate: bool = True,
        save: bool = True,
        **model_params
    ) -> Dict[str, Any]:
        """Run the complete ML pipeline"""
        logger.info("Starting fraud detection pipeline")
        
        try:
            # Load data
            df = self.load_data(train_data_path)
            
            # Preprocess
            X_train, X_test, y_train, y_test = self.preprocess_data(df, target_col)
            
            # Train
            self.train_model(X_train, y_train, model_type, **model_params)
            
            # Evaluate
            metrics = self.evaluate_model(X_test, y_test)
            
            # Validate
            if validate:
                is_valid, validation_results = self.validate_model(metrics)
                if not is_valid:
                    raise ValueError("Model validation failed")
            else:
                validation_results = []
            
            # Save
            model_path = None
            if save:
                model_path = self.save_model()
            
            results = {
                'status': 'success',
                'metrics': metrics,
                'validation_results': validation_results,
                'model_path': model_path,
                'timestamp': datetime.now().isoformat()
            }
            
            logger.info("Pipeline completed successfully")
            return results
        
        except Exception as e:
            logger.error(f"Pipeline failed: {str(e)}")
            raise


def main():
    """Main execution function"""
    pipeline = FraudDetectionPipeline(
        min_precision=0.85,
        min_recall=0.75,
        min_f1=0.80
    )
    
    # Run pipeline
    results = pipeline.run_pipeline(
        train_data_path="data/train_data.csv",
        model_type="random_forest",
        n_estimators=100,
        max_depth=10
    )
    
    print(f"Pipeline Results: {json.dumps(results, indent=2)}")


if __name__ == "__main__":
    main()
```
