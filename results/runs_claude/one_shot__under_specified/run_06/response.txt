```python
# pipeline.py
"""
MLOps Pipeline for Fraud Detection Model
Handles training, validation, deployment, and monitoring
"""

import os
import json
import logging
from typing import Dict, Tuple, Any, Optional
from datetime import datetime
from pathlib import Path
import pickle

import numpy as np
import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.ensemble import RandomForestClassifier
from sklearn.metrics import (
    accuracy_score, precision_score, recall_score, 
    f1_score, roc_auc_score, confusion_matrix
)
import mlflow
import mlflow.sklearn
from evidently.report import Report
from evidently.metric_preset import DataDriftPreset, ClassificationPreset
import boto3
from botocore.exceptions import ClientError

logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)


class FraudDetectionPipeline:
    """End-to-end pipeline for fraud detection model"""
    
    def __init__(
        self,
        model_name: str = "fraud_detection_model",
        experiment_name: str = "fraud_detection",
        artifact_path: str = "./artifacts",
        mlflow_tracking_uri: Optional[str] = None,
        s3_bucket: Optional[str] = None
    ):
        self.model_name = model_name
        self.experiment_name = experiment_name
        self.artifact_path = Path(artifact_path)
        self.artifact_path.mkdir(parents=True, exist_ok=True)
        
        self.model = None
        self.metrics = {}
        self.s3_bucket = s3_bucket
        
        # Setup MLflow
        if mlflow_tracking_uri:
            mlflow.set_tracking_uri(mlflow_tracking_uri)
        mlflow.set_experiment(experiment_name)
        
        # Performance thresholds
        self.thresholds = {
            "min_precision": 0.85,
            "min_recall": 0.80,
            "min_f1": 0.82,
            "min_auc": 0.90,
            "max_drift_score": 0.3
        }
    
    def load_data(self, data_path: str) -> pd.DataFrame:
        """Load and validate data"""
        logger.info(f"Loading data from {data_path}")
        
        if data_path.endswith('.csv'):
            df = pd.read_csv(data_path)
        elif data_path.endswith('.parquet'):
            df = pd.read_parquet(data_path)
        else:
            raise ValueError(f"Unsupported file format: {data_path}")
        
        # Validate required columns
        required_cols = ['is_fraud']  # Target column
        missing_cols = set(required_cols) - set(df.columns)
        if missing_cols:
            raise ValueError(f"Missing required columns: {missing_cols}")
        
        logger.info(f"Loaded {len(df)} records with {len(df.columns)} features")
        return df
    
    def preprocess_data(
        self, 
        df: pd.DataFrame, 
        test_size: float = 0.2,
        random_state: int = 42
    ) -> Tuple[np.ndarray, np.ndarray, np.ndarray, np.ndarray]:
        """Preprocess and split data"""
        logger.info("Preprocessing data")
        
        # Separate features and target
        X = df.drop('is_fraud', axis=1)
        y = df['is_fraud']
        
        # Handle categorical variables
        X = pd.get_dummies(X, drop_first=True)
        
        # Handle missing values
        X = X.fillna(X.median())
        
        # Split data
        X_train, X_test, y_train, y_test = train_test_split(
            X, y, test_size=test_size, random_state=random_state, stratify=y
        )
        
        logger.info(f"Train set: {len(X_train)}, Test set: {len(X_test)}")
        logger.info(f"Fraud rate - Train: {y_train.mean():.3f}, Test: {y_test.mean():.3f}")
        
        return X_train, X_test, y_train, y_test
    
    def train_model(
        self, 
        X_train: np.ndarray, 
        y_train: np.ndarray,
        hyperparameters: Optional[Dict] = None
    ) -> RandomForestClassifier:
        """Train fraud detection model"""
        logger.info("Training model")
        
        if hyperparameters is None:
            hyperparameters = {
                'n_estimators': 100,
                'max_depth': 10,
                'min_samples_split': 5,
                'min_samples_leaf': 2,
                'class_weight': 'balanced',
                'random_state': 42,
                'n_jobs': -1
            }
        
        with mlflow.start_run(run_name=f"training_{datetime.now().strftime('%Y%m%d_%H%M%S')}"):
            # Log parameters
            mlflow.log_params(hyperparameters)
            
            # Train model
            self.model = RandomForestClassifier(**hyperparameters)
            self.model.fit(X_train, y_train)
            
            # Log model
            mlflow.sklearn.log_model(self.model, "model")
            
            logger.info("Model training completed")
        
        return self.model
    
    def evaluate_model(
        self, 
        X_test: np.ndarray, 
        y_test: np.ndarray
    ) -> Dict[str, float]:
        """Evaluate model performance"""
        logger.info("Evaluating model")
        
        if self.model is None:
            raise ValueError("Model not trained. Call train_model first.")
        
        # Predictions
        y_pred = self.model.predict(X_test)
        y_pred_proba = self.model.predict_proba(X_test)[:, 1]
        
        # Calculate metrics
        self.metrics = {
            'accuracy': accuracy_score(y_test, y_pred),
            'precision': precision_score(y_test, y_pred),
            'recall': recall_score(y_test, y_pred),
            'f1_score': f1_score(y_test, y_pred),
            'roc_auc': roc_auc_score(y_test, y_pred_proba)
        }
        
        # Confusion matrix
        cm = confusion_matrix(y_test, y_pred)
        self.metrics['true_negatives'] = int(cm[0, 0])
        self.metrics['false_positives'] = int(cm[0, 1])
        self.metrics['false_negatives'] = int(cm[1, 0])
        self.metrics['true_positives'] = int(cm[1, 1])
        
        # Log metrics to MLflow
        with mlflow.start_run(run_name=f"evaluation_{datetime.now().strftime('%Y%m%d_%H%M%S')}"):
            for metric_name, metric_value in self.metrics.items():
                mlflow.log_metric(metric_name, metric_value)
        
        logger.info(f"Metrics: {json.dumps(self.metrics, indent=2)}")
        
        return self.metrics
    
    def validate_model(self) -> Tuple[bool, Dict[str, Any]]:
        """Validate model against thresholds"""
        logger.info("Validating model against thresholds")
        
        validation_results = {
            'passed': True,
            'checks': {}
        }
        
        checks = [
            ('precision', self.metrics.get('precision', 0), self.thresholds['min_precision'], '>='),
            ('recall', self.metrics.get('recall', 0), self.thresholds['min_recall'], '>='),
            ('f1_score', self.metrics.get('f1_score', 0), self.thresholds['min_f1'], '>='),
            ('roc_auc', self.metrics.get('roc_auc', 0), self.thresholds['min_auc'], '>=')
        ]
        
        for metric_name, value, threshold, operator in checks:
            if operator == '>=':
                passed = value >= threshold
            else:
                passed = value <= threshold
            
            validation_results['checks'][metric_name] = {
                'value': value,
                'threshold': threshold,
                'passed': passed
            }
            
            if not passed:
                validation_results['passed'] = False
                logger.warning(
                    f"Validation failed for {metric_name}: "
                    f"{value:.4f} {operator} {threshold}"
                )
        
        logger.info(f"Validation result: {'PASSED' if validation_results['passed'] else 'FAILED'}")
        
        return validation_results['passed'], validation_results
    
    def check_data_drift(
        self, 
        reference_data: pd.DataFrame, 
        current_data: pd.DataFrame
    ) -> Tuple[bool, Dict]:
        """Check for data drift using Evidently"""
        logger.info("Checking for data drift")
        
        # Create drift report
        report = Report(metrics=[DataDriftPreset()])
        report.run(reference_data=reference_data, current_data=current_data)
        
        # Save report
        report_path = self.artifact_path / f"drift_report_{datetime.now().strftime('%Y%m%d_%H%M%S')}.html"
        report.save_html(str(report_path))
        
        # Extract drift metrics
        report_dict = report.as_dict()
        drift_share = report_dict['metrics'][0]['result']['drift_share']
        
        drift_results = {
            'drift_share': drift_share,
            'drift_detected': drift_share > self.thresholds['max_drift_score'],
            'report_path': str(report_path)
        }
        
        logger.info(f"Drift share: {drift_share:.4f}")
        
        return not drift_results['drift_detected'], drift_results
    
    def save_model(self, version: Optional[str] = None) -> str:
        """Save model to local artifacts"""
        if self.model is None:
            raise ValueError("No model to save")
        
        if version is None:
            version = datetime.now().strftime('%Y%m%d_%H%M%S')
        
        model_path = self.artifact_path / f"{self.model_name}_v{version}.pkl"
        
        with open(model_path, 'wb') as f:
            pickle.dump(self.model, f)
        
        # Save metrics
        metrics_path = self.artifact_path / f"{self.model_name}_v{version}_metrics.json"
        with open(metrics_path, 'w') as f:
            json.dump(self.metrics, f, indent=2)
        
        logger.info(f"Model saved to {model_path}")
        
        return str(model_path)
    
    def load_model(self, model_path: str):
        """Load model from file"""
        with open(model_path, 'rb') as f:
            self.model = pickle.load(f)
        
        logger.info(f"Model loaded from {model_path}")
        return self.model
    
    def deploy_to_s3(self, model_path: str, version: str) -> bool:
        """Deploy model to S3"""
        if not self.s3_bucket:
            logger.warning("No S3 bucket configured, skipping deployment")
            return False
        
        try:
            s3_client = boto3.client('s3')
            s3_key = f"models/{self.model_name}/v{version}/{Path(model_path).name}"
            
            s3_client.upload_file(model_path, self.s3_bucket, s3_key)
            
            # Upload metrics
            metrics_path = model_path.replace('.pkl', '_metrics.json')
            if os.path.exists(metrics_path):
                metrics_key = f"models/{self.model_name}/v{version}/metrics.json"
                s3_client.upload_file(metrics_path, self.s3_bucket, metrics_key)
            
            logger.info(f"Model deployed to s3://{self.s3_bucket}/{s3_key}")
            return True
            
        except ClientError as e:
            logger.error(f"Failed to deploy to S3: {e}")
            return False
    
    def register_model_mlflow(self, model_uri: str, stage: str = "Staging") -> str:
        """Register model in MLflow Model Registry"""
        try:
            model_version = mlflow.register_model(model_uri, self.model_name)
            
            # Transition to stage
            client = mlflow.tracking.MlflowClient()
            client.transition_model_version_stage(
                name=self.model_name,
                version=model_version.version,
                stage=stage
            )
            
            logger.info(f"Model registered: {self.model_name} v{model_version.version} -> {stage}")
            return model_version.version
            
        except Exception as e:
            logger.error(f"Failed to register model: {e}")
            raise
    
    def run_full_pipeline(
        self, 
        data_path: str,
        deploy: bool = False,
        register: bool = False
    ) -> Dict[str, Any]:
        """Execute full ML pipeline"""
        logger.info("=" * 50)
        logger.info("Starting Full ML Pipeline")
        logger.info("=" * 50)
        
        pipeline_results = {
            'status': 'success',
            'timestamp': datetime.now().isoformat(),
            'stages': {}
        }
        
        try:
            # 1. Load data
            df = self.load_data(data_path)
            pipeline_results['stages']['data_loading'] = {'status': 'success'}
            
            # 2. Preprocess
            X_train, X_test, y_train, y_test = self.preprocess_data(df)
            pipeline_results['stages']['preprocessing'] = {'status': 'success'}
            
            # 3. Train
            self.train_model(X_train, y_train)
            pipeline_results['stages']['training'] = {'status': 'success'}
            
            # 4. Evaluate
            metrics = self.evaluate_model(X_test, y_test)
            pipeline_results['stages']['evaluation'] = {
                'status': 'success',
                'metrics': metrics
            }
            
            # 5. Validate
            validation_passed, validation_results = self.validate_model()
            pipeline_results['stages']['validation'] = {
                'status': 'success',
                'passed': validation_passed,
                'results': validation_results
            }
            
            if not validation_passed:
                pipeline_results['status'] = 'validation_failed'
                logger.error("Pipeline stopped: Model validation failed")
                return pipeline_results
            
            # 6. Save model
            version = datetime.now().strftime('%Y%m%d_%H%M%S')
            model_path = self.save_model(version)
            pipeline_results['stages']['save_model'] = {
                'status': 'success',
                'model_path': model_path,
                'version': version
            }
            
            # 7. Deploy (optional)
            if deploy and self.s3_bucket:
                deploy_success = self.deploy_to_s3(model_path, version)
                pipeline_results['stages']['deployment'] = {
                    'status': 'success' if deploy_success else 'failed',
                    'deployed': deploy_success
                }
            
            # 8. Register (optional)
            if register:
                model_uri = f"runs:/{mlflow.active_run().info.run_id}/model"
                model_version = self.register_model_mlflow(model_uri, stage="Staging")
                pipeline_results['stages']['registration'] = {
                    'status': 'success',
                    'model_version': model_version
                }
            
            logger.info("=" * 50)
            logger.info("Pipeline Completed Successfully")
            logger.info("=" * 50)
            
        except Exception as e:
            logger.error(f"Pipeline failed: {str(e)}", exc_info=True)
            pipeline_results['status'] = 'failed'
            pipeline_results['error'] = str(e)
        
        return pipeline_results


if __name__ == "__main__":
    # Example usage
    pipeline = FraudDetectionPipeline(

    )
```
