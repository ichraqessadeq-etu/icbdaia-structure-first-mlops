@dataclass
class ModelMetrics:
    """Container for model performance metrics"""
    accuracy: float
    precision: float
    recall: float
    f1_score: float
    roc_auc: float
    confusion_matrix: List[List[int]]
    threshold: float = 0.5


@dataclass
class PipelineConfig:
    """Configuration for the ML pipeline"""
    model_name: str = "fraud_detection_model"
    model_version: str = "1.0.0"
    test_size: float = 0.2
    random_state: int = 42
    min_accuracy: float = 0.85
    min_precision: float = 0.80
    min_recall: float = 0.75

class DataValidator:
    """Validates input data quality and schema"""
    
    def __init__(self, expected_features: List[str]):
        self.expected_features = expected_features
    
    def validate_schema(self, df: pd.DataFrame) -> Tuple[bool, List[str]]:
        """Validate dataframe schema"""
        errors = []
        
        # Check for missing columns
        missing_cols = set(self.expected_features) - set(df.columns)
        if missing_cols:
            errors.append(f"Missing columns: {missing_cols}")
        
        # Check for extra columns (excluding target)
        extra_cols = set(df.columns) - set(self.expected_features) - {'is_fraud'}
        if extra_cols:
            errors.append(f"Unexpected columns: {extra_cols}")
        
        return len(errors) == 0, errors
    
    def validate_data_quality(self, df: pd.DataFrame) -> Tuple[bool, List[str]]:
        """Validate data quality"""
        errors = []
        
        # Check for null values
        null_counts = df.isnull().sum()
        if null_counts.any():
            errors.append(f"Null values found: {null_counts[null_counts > 0].to_dict()}")
        
        # Check for infinite values
        numeric_cols = df.select_dtypes(include=[np.number]).columns
        for col in numeric_cols:
            if np.isinf(df[col]).any():
                errors.append(f"Infinite values found in column: {col}")
        
        # Check for duplicate rows
        duplicates = df.duplicated().sum()
        if duplicates > 0:
            logger.warning(f"Found {duplicates} duplicate rows")
        
        return len(errors) == 0, errors
    
    def validate_target_distribution(self, y: pd.Series, min_fraud_rate: float = 0.001) -> Tuple[bool, List[str]]:
        """Validate target variable distribution"""
        errors = []
        
        fraud_rate = y.mean()
        if fraud_rate < min_fraud_rate:
            errors.append(f"Fraud rate too low: {fraud_rate:.4f}")
        
        if fraud_rate > 0.5:
            errors.append(f"Fraud rate suspiciously high: {fraud_rate:.4f}")
        
        logger.info(f"Fraud rate: {fraud_rate:.4f}")
        
        return len(errors) == 0, errors


class FeatureEngineer:
    """Handles feature engineering and preprocessing"""
    
    def __init__(self):
        self.scaler = StandardScaler()
        self.feature_names = None
    
    def fit_transform(self, X: pd.DataFrame) -> np.ndarray:
        """Fit and transform features"""
        self.feature_names = X.columns.tolist()
        X_scaled = self.scaler.fit_transform(X)
        logger.info(f"Fitted scaler on {len(self.feature_names)} features")
        return X_scaled
    
    def transform(self, X: pd.DataFrame) -> np.ndarray:
        """Transform features using fitted scaler"""
        if self.feature_names is None:
            raise ValueError("FeatureEngineer must be fitted before transform")
        
        # Ensure same feature order
        X = X[self.feature_names]
        return self.scaler.transform(X)
    
    def detect_feature_drift(self, X_reference: pd.DataFrame, X_current: pd.DataFrame) -> Dict[str, float]:
        """Detect feature drift using statistical tests"""
        drift_scores = {}
        
        for col in X_reference.columns:
            # Calculate mean absolute difference in distributions
            ref_mean = X_reference[col].mean()
            curr_mean = X_current[col].mean()
            ref_std = X_reference[col].std()
            
            if ref_std > 0:
                drift_score = abs(curr_mean - ref_mean) / ref_std
            else:
                drift_score = 0.0
            
            drift_scores[col] = drift_score
        
        return drift_scores


class ModelTrainer:
    """Handles model training and hyperparameter tuning"""
    
    def __init__(self, config: PipelineConfig):
        self.config = config
        self.model = None
    
    def train(self, X_train: np.ndarray, y_train: np.ndarray, 
              class_weight: str = 'balanced') -> RandomForestClassifier:
        """Train the fraud detection model"""
        logger.info("Training Random Forest model...")
        
        self.model = RandomForestClassifier(
            n_estimators=100,
            max_depth=10,
            min_samples_split=10,
            min_samples_leaf=5,
            class_weight=class_weight,
            random_state=self.config.random_state,
            n_jobs=-1
        )
        
        self.model.fit(X_train, y_train)
        logger.info("Model training completed")
        
        return self.model
    
    def get_feature_importance(self, feature_names: List[str]) -> pd.DataFrame:
        """Get feature importance from trained model"""
        if self.model is None:
            raise ValueError("Model must be trained first")
        
        importance_df = pd.DataFrame({
            'feature': feature_names,
            'importance': self.model.feature_importances_
        }).sort_values('importance', ascending=False)
        
        return importance_df


class ModelEvaluator:
    """Evaluates model performance"""
    
    def __init__(self, config: PipelineConfig):
        self.config = config
    
    def evaluate(self, model: RandomForestClassifier, 
                 X_test: np.ndarray, y_test: np.ndarray,
                 threshold: float = 0.5) -> ModelMetrics:
        """Evaluate model performance"""
        logger.info("Evaluating model...")
        
        # Get predictions
        y_pred_proba = model.predict_proba(X_test)[:, 1]
        y_pred = (y_pred_proba >= threshold).astype(int)
        
        # Calculate metrics
        metrics = ModelMetrics(
            accuracy=accuracy_score(y_test, y_pred),
            precision=precision_score(y_test, y_pred, zero_division=0),
            recall=recall_score(y_test, y_pred, zero_division=0),
            f1_score=f1_score(y_test, y_pred, zero_division=0),
            roc_auc=roc_auc_score(y_test, y_pred_proba),
            confusion_matrix=confusion_matrix(y_test, y_pred).tolist(),
            threshold=threshold
        )
        
        logger.info(f"Metrics - Accuracy: {metrics.accuracy:.4f}, "
                   f"Precision: {metrics.precision:.4f}, "
                   f"Recall: {metrics.recall:.4f}, "
                   f"F1: {metrics.f1_score:.4f}, "
                   f"ROC-AUC: {metrics.roc_auc:.4f}")
        
        return metrics
    
    def validate_metrics(self, metrics: ModelMetrics) -> Tuple[bool, List[str]]:
        """Validate metrics against thresholds"""
        errors = []
        
        if metrics.accuracy < self.config.min_accuracy:
            errors.append(f"Accuracy {metrics.accuracy:.4f} below threshold {self.config.min_accuracy}")
        
        if metrics.precision < self.config.min_precision:
            errors.append(f"Precision {metrics.precision:.4f} below threshold {self.config.min_precision}")
        
        if metrics.recall < self.config.min_recall:
            errors.append(f"Recall {metrics.recall:.4f} below threshold {self.config.min_recall}")
        
        if metrics.f1_score < self.config.min_f1:
            errors.append(f"F1 {metrics.f1_score:.4f} below threshold {self.config.min_f1}")
        
        if metrics.roc_auc < self.config.min_roc_auc:
            errors.append(f"ROC-AUC {metrics.roc_auc:.4f} below threshold {self.config.min_roc_auc}")
        
        return len(errors) == 0, errors
    
    def find_optimal_threshold(self, model: RandomForestClassifier,
                              X_test: np.ndarray, y_test: np.ndarray) -> float:
        """Find optimal classification threshold"""
        y_pred_proba = model.predict_proba(X_test)[:, 1]
        precision, recall, thresholds = precision_recall_curve(y_test, y_pred_proba)
        
        # Find threshold that maximizes F1 score
        f1_scores = 2 * (precision * recall) / (precision + recall + 1e-10)
        optimal_idx = np.argmax(f1_scores)
        optimal_threshold = thresholds[optimal_idx] if optimal_idx < len(thresholds) else 0.5
        
        logger.info(f"Optimal threshold: {optimal_threshold:.4f}")
        return optimal_threshold


class ModelRegistry:
    """Handles model versioning and storage"""
    
    def __init__(self, config: PipelineConfig):
        self.config = config
        self.models_dir = Path(config.models_dir)
        self.models_dir.mkdir(parents=True, exist_ok=True)
    
    def save_model(self, model: RandomForestClassifier, 
                   feature_engineer: FeatureEngineer,
                   metrics: ModelMetrics,
                   metadata: Optional[Dict[str, Any]] = None) -> str:
        """Save model with metadata"""
        timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
        model_path = self.models_dir / f"{self.config.model_name}_{timestamp}"
        model_path.mkdir(parents=True, exist_ok=True)
        
        # Save model
        model_file = model_path / "model.pkl"
        joblib.dump(model, model_file)
        
        # Save feature engineer
        fe_file = model_path / "feature_engineer.pkl"
        joblib.dump(feature_engineer, fe_file)
        
        # Save metadata
        metadata_dict = {
            "model_name": self.config.model_name,
            "model_version": self.config.model_version,
            "timestamp": timestamp,
            "metrics": {
                "accuracy": metrics.accuracy,
                "precision": metrics.precision,
                "recall": metrics.recall,
                "f1_score": metrics.f1_score,
                "roc_auc": metrics.roc_auc,
                "threshold": metrics.threshold
            },
            "config": {
                "test_size": self.config.test_size,
                "random_state": self.config.random_state
            }
        }
        
        if metadata:
            metadata_dict.update(metadata)
        
        metadata_file = model_path / "metadata.json"
        with open(metadata_file, 'w') as f:
            json.dump(metadata_dict, f, indent=2)
        
        logger.info(f"Model saved to {model_path}")
        return str(model_path)
    
    def load_model(self, model_path: str) -> Tuple[RandomForestClassifier, FeatureEngineer, Dict]:
        """Load model with metadata"""
        model_path = Path(model_path)
        
        # Load model
        model = joblib.load(model_path / "model.pkl")
        
        # Load feature engineer
        feature_engineer = joblib.load(model_path / "feature_engineer.pkl")
        
        # Load metadata
        with open(model_path / "metadata.json", 'r') as f:
            metadata = json.load(f)
        
        logger.info(f"Model loaded from {model_path}")
        return model, feature_engineer, metadata


class MLFlowTracker:
    """Handles MLflow experiment tracking"""
    
    def __init__(self, experiment_name: str = "fraud_detection"):
        self.experiment_name = experiment_name
        mlflow.set_experiment(experiment_name)
    
    def log_run(self, model: RandomForestClassifier,
                metrics: ModelMetrics,
                params: Dict[str, Any],
                artifacts: Optional[Dict[str, str]] = None):
        """Log a training run to MLflow"""
        with mlflow.start_run():
            # Log parameters
            mlflow.log_params(params)
            
            # Log metrics
            mlflow.log_metric("accuracy", metrics.accuracy)
            mlflow.log_metric("precision", metrics.precision)
            mlflow.log_metric("recall", metrics.recall)
            mlflow.log_metric("f1_score", metrics.f1_score)
            mlflow.log_metric("roc_auc", metrics.roc_auc)
            mlflow.log_metric("threshold", metrics.threshold)
            
            # Log model
            mlflow.sklearn.log_model(model, "model")
            
            # Log artifacts
            if artifacts:
                for name, path in artifacts.items():
                    mlflow.log_artifact(path, name)
            
            logger.info("Run logged to MLflow")


class FraudDetectionPipeline:
    """Main pipeline orchestrator"""
    
    def __init__(self, config: Optional[PipelineConfig] = None):
        self.config = config or PipelineConfig()
        self.data_validator = None
        self.feature_engineer = FeatureEngineer()
        self.model_trainer = ModelTrainer(self.config)
        self.model_evaluator = ModelEvaluator(self.config)
        self.model_registry = ModelRegistry(self.config)
        self.mlflow_tracker = MLFlowTracker()
        
        # Create directories
        Path(self.config.artifacts_dir).mkdir(parents=True, exist_ok=True)
        Path(self.config.models_dir).mkdir(parents=True, exist_ok=True)
        Path(self.config.data_dir).mkdir(parents=True, exist_ok=True)
    
    def run_training_pipeline(self, data_path: str, 
                             expected_features: List[str]) -> Dict[str, Any]:
        """Execute the complete training pipeline"""
        logger.info("=" * 80)
        logger.info("Starting Fraud Detection Training Pipeline")
        logger.info("=" * 80)
        
        # Load data
        logger.info(f"Loading data from {data_path}")
        df = pd.read_csv(data_path)
        logger.info(f"Loaded {len(df)} records")
        
        # Initialize data validator
        self.data_validator = DataValidator(expected_features)
        
        # Validate schema
        logger.info("Validating data schema...")
        schema_valid, schema_errors = self.data_validator.validate_schema(df)
        if not schema_valid:
            raise ValueError(f"Schema validation failed: {schema_errors}")
        
        # Validate data quality
        logger.info("Validating data quality...")
        quality_valid, quality_errors = self.data_validator.validate_data_quality(df)
        if not quality_valid:
            raise ValueError(f"Data quality validation failed: {quality_errors}")
        
        # Prepare features and target
        X = df[expected_features]
        y = df['is_fraud']
        
        # Validate target distribution
        logger.info("Validating target distribution...")
        target_valid, target_errors = self.data_validator.validate_target_distribution(y)
        if not target_valid:
            raise ValueError(f"Target validation failed: {target_errors}")
        
        # Split data
        logger.info("Splitting data...")
        X_train, X_test, y_train, y_test = train_test_split(
            X, y, 
            test_size=self.config.test_size,
            random_state=self.config.random_state,
            stratify=y
        )
        logger.info(f"Train set: {len(X_train)}, Test set: {len(X_test)}")
        
        # Feature engineering
        logger.info("Engineering features...")
        X_train_scaled = self.feature_engineer.fit_transform(X_train)
        X_test_scaled = self.feature_engineer.transform(X_test)
        
        # Train model
        model = self.model_trainer.train(X_train_scaled, y_train)
        
        # Find optimal threshold
        optimal_threshold = self.model_evaluator.find_optimal_threshold(
            model, X_test_scaled, y_test
        )
        
        # Evaluate model
        metrics = self.model_evaluator.evaluate(
            model, X_test_scaled, y_test, threshold=optimal_threshold
        )
        
        # Validate metrics
        logger.info("Validating metrics against thresholds...")
        metrics_valid, metrics_errors = self.model_evaluator.validate_metrics(metrics)
        if not metrics_valid:
            raise ValueError(f"Metrics validation failed: {metrics_errors}")
        
        # Get feature importance
        feature_importance = self.model_trainer.get_feature_importance(expected_features)
        logger.info(f"Top 5 features:\n{feature_importance.head()}")
        
        # Save model
        model_path = self.model_registry.save_model(
            model, 
            self.feature_engineer,
            metrics,
            metadata={
                "feature_importance": feature_importance.to_dict('records'),
                "train_size": len(X_train),
                "test_size": len(X_test)
            }
        )
        
        # Log to MLflow
        self.mlflow_tracker.log_run(
            model,
            metrics,
            params={
                "test_size": self.config.test_size,
                "random_state": self.config.random_state,
                "n_estimators": model.n_estimators,
                "max_depth": model.max_depth,
                "threshold": optimal_threshold
            }
        )
        
        logger.info("=" * 80)
        logger.info("Training Pipeline Completed Successfully")
        logger.info("=" * 80)
        
        return {
            "model_path": model_path,
            "metrics": metrics,
            "feature_importance": feature_importance,
            "status": "success"
        }
    
    def run_inference_pipeline(self, model_path: str, 
                               data_path: str) -> pd.DataFrame:
        """Execute inference pipeline"""
        logger.info("Starting inference pipeline...")
        
        # Load model
        model, feature_engineer, metadata = self.model_registry.load_model(model_path)
        
        # Load data
        df = pd.read_csv(data_path)
        X = df[feature_engineer.feature_names]
        
        # Transform features
        X_scaled = feature_engineer.transform(X)
        
        # Predict
        threshold = metadata['metrics']['threshold']
        y_pred_proba = model.predict_proba(X_scaled)[:, 1]
        y_pred = (y_pred_proba >= threshold).astype(int)
        
        # Create results dataframe
        results = df.copy()
        results['fraud_probability'] = y_pred_proba
        results['fraud_prediction'] = y_pred
        
        logger.info(f"Inference completed on {len(results)} records")
        return results
    
    def run_monitoring_pipeline(self, model_path: str,
                               reference_data_path: str,
                               current_data_path: str) -> Dict[str, Any]:
        """Execute monitoring pipeline for drift detection"""
        logger.info("Starting monitoring pipeline...")
        
        # Load model
        _, feature_engineer, _ = self.model_registry.load_model(model_path)
        
        # Load data
        df_reference = pd.read_csv(reference_data_path)
        df_current = pd.read_csv(current_data_path)
        
        X_reference = df_reference[feature_engineer.feature_names]
        X_current = df_current[feature_engineer.feature_names]
        
        # Detect drift
        drift_scores = feature_engineer.detect_feature_drift(X_reference, X_current)
        
        # Check for significant drift
        max_drift = max(drift_scores.values())
        drift_detected = max_drift > self.config.max_feature_drift
        
        monitoring_report = {
            "drift_detected": drift_detected,
            "max_drift_score": max_drift,
            "drift_threshold": self.config.max_feature_drift,
            "feature_drift_scores": drift_scores,
            "timestamp": datetime.now().isoformat()
        }
        
        if drift_detected:
            logger.warning(f"Drift detected! Max drift score: {max_drift:.4f}")
        else:
            logger.info(f"No significant drift detected. Max drift score: {max_drift:.4f}")
        
        return monitoring_report


def main():
    """Main entry point for the pipeline"""
    # Example usage
    config = PipelineConfig()
    pipeline = FraudDetectionPipeline(config)
    
    # Define expected features
    expected_features = [
        'transaction_amount',
        'transaction_hour',
        'days_since_last_transaction',
        'avg_transaction_amount',
        'transaction_count_24h',
        'merchant_risk_score'
    ]
    
    # Run training pipeline
    results = pipeline.run_training_pipeline(
        data_path="data/training_data.csv",
        expected_features=expected_features
    )
    
    print(f"Model saved to: {results['model_path']}")
    print(f"Metrics: {results['metrics']}")


if __name__ == "__main__":
    main()
