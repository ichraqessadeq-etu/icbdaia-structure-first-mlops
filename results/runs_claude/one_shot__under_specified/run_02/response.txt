```python
# pipeline.py
"""
MLOps Pipeline for Fraud Detection Model
Handles training, validation, deployment, and monitoring
"""

import os
import json
import logging
from typing import Dict, Any, Tuple, Optional
from dataclasses import dataclass, asdict
from datetime import datetime
from pathlib import Path

import pandas as pd
import numpy as np
from sklearn.model_selection import train_test_split
from sklearn.ensemble import RandomForestClassifier
from sklearn.metrics import (
    precision_score, recall_score, f1_score, 
    roc_auc_score, confusion_matrix, classification_report
)
from sklearn.preprocessing import StandardScaler
import joblib
import mlflow
import mlflow.sklearn
from evidently.report import Report
from evidently.metric_preset import DataDriftPreset, ClassificationPreset
import boto3
from botocore.exceptions import ClientError

logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)


@dataclass
class ModelMetrics:
    """Model performance metrics"""
    precision: float
    recall: float
    f1_score: float
    roc_auc: float
    accuracy: float
    confusion_matrix: list
    timestamp: str
    model_version: str


@dataclass
class PipelineConfig:
    """Pipeline configuration"""
    model_name: str = "fraud_detection_model"
    model_version: str = "1.0.0"
    test_size: float = 0.2
    random_state: int = 42
    min_precision: float = 0.85
    min_recall: float = 0.80
    min_f1_score: float = 0.82
    max_drift_score: float = 0.15
    artifact_path: str = "./artifacts"
    mlflow_tracking_uri: str = "http://localhost:5000"
    s3_bucket: Optional[str] = None
    environment: str = "development"


class FraudDetectionPipeline:
    """End-to-end ML pipeline for fraud detection"""
    
    def __init__(self, config: PipelineConfig):
        self.config = config
        self.model = None
        self.scaler = StandardScaler()
        self.feature_columns = None
        self.metrics = None
        
        # Setup paths
        self.artifact_path = Path(config.artifact_path)
        self.artifact_path.mkdir(parents=True, exist_ok=True)
        
        # Setup MLflow
        mlflow.set_tracking_uri(config.mlflow_tracking_uri)
        mlflow.set_experiment(f"{config.model_name}_{config.environment}")
        
        logger.info(f"Initialized pipeline for {config.model_name} v{config.model_version}")
    
    def load_data(self, data_path: str) -> pd.DataFrame:
        """Load and validate data"""
        logger.info(f"Loading data from {data_path}")
        
        try:
            if data_path.endswith('.csv'):
                df = pd.read_csv(data_path)
            elif data_path.endswith('.parquet'):
                df = pd.read_parquet(data_path)
            else:
                raise ValueError(f"Unsupported file format: {data_path}")
            
            logger.info(f"Loaded {len(df)} records with {len(df.columns)} columns")
            return df
        
        except Exception as e:
            logger.error(f"Error loading data: {str(e)}")
            raise
    
    def preprocess_data(
        self, 
        df: pd.DataFrame, 
        target_column: str = 'is_fraud'
    ) -> Tuple[np.ndarray, np.ndarray]:
        """Preprocess data for training"""
        logger.info("Preprocessing data")
        
        # Validate target column
        if target_column not in df.columns:
            raise ValueError(f"Target column '{target_column}' not found in dataframe")
        
        # Separate features and target
        X = df.drop(columns=[target_column])
        y = df[target_column]
        
        # Store feature columns for later use
        self.feature_columns = X.columns.tolist()
        
        # Handle missing values
        X = X.fillna(X.median())
        
        # Scale features
        X_scaled = self.scaler.fit_transform(X)
        
        logger.info(f"Preprocessed {X_scaled.shape[0]} samples with {X_scaled.shape[1]} features")
        logger.info(f"Class distribution: {y.value_counts().to_dict()}")
        
        return X_scaled, y.values
    
    def train_model(
        self, 
        X_train: np.ndarray, 
        y_train: np.ndarray,
        hyperparameters: Optional[Dict[str, Any]] = None
    ) -> RandomForestClassifier:
        """Train the fraud detection model"""
        logger.info("Training model")
        
        # Default hyperparameters
        if hyperparameters is None:
            hyperparameters = {
                'n_estimators': 100,
                'max_depth': 10,
                'min_samples_split': 5,
                'min_samples_leaf': 2,
                'class_weight': 'balanced',
                'random_state': self.config.random_state,
                'n_jobs': -1
            }
        
        with mlflow.start_run(run_name=f"training_{datetime.now().strftime('%Y%m%d_%H%M%S')}"):
            # Log parameters
            mlflow.log_params(hyperparameters)
            mlflow.log_param("model_version", self.config.model_version)
            
            # Train model
            self.model = RandomForestClassifier(**hyperparameters)
            self.model.fit(X_train, y_train)
            
            # Log model
            mlflow.sklearn.log_model(
                self.model, 
                "model",
                registered_model_name=self.config.model_name
            )
            
            logger.info("Model training completed")
            
        return self.model
    
    def evaluate_model(
        self, 
        X_test: np.ndarray, 
        y_test: np.ndarray
    ) -> ModelMetrics:
        """Evaluate model performance"""
        logger.info("Evaluating model")
        
        if self.model is None:
            raise ValueError("Model not trained. Call train_model() first.")
        
        # Make predictions
        y_pred = self.model.predict(X_test)
        y_pred_proba = self.model.predict_proba(X_test)[:, 1]
        
        # Calculate metrics
        metrics = ModelMetrics(
            precision=precision_score(y_test, y_pred),
            recall=recall_score(y_test, y_pred),
            f1_score=f1_score(y_test, y_pred),
            roc_auc=roc_auc_score(y_test, y_pred_proba),
            accuracy=self.model.score(X_test, y_test),
            confusion_matrix=confusion_matrix(y_test, y_pred).tolist(),
            timestamp=datetime.now().isoformat(),
            model_version=self.config.model_version
        )
        
        self.metrics = metrics
        
        # Log metrics to MLflow
        with mlflow.start_run(run_name=f"evaluation_{datetime.now().strftime('%Y%m%d_%H%M%S')}"):
            mlflow.log_metrics({
                'precision': metrics.precision,
                'recall': metrics.recall,
                'f1_score': metrics.f1_score,
                'roc_auc': metrics.roc_auc,
                'accuracy': metrics.accuracy
            })
        
        logger.info(f"Model Metrics: Precision={metrics.precision:.4f}, "
                   f"Recall={metrics.recall:.4f}, F1={metrics.f1_score:.4f}, "
                   f"ROC-AUC={metrics.roc_auc:.4f}")
        
        return metrics
    
    def validate_model(self) -> bool:
        """Validate model meets minimum performance thresholds"""
        logger.info("Validating model performance")
        
        if self.metrics is None:
            raise ValueError("No metrics available. Run evaluate_model() first.")
        
        validations = {
            'precision': self.metrics.precision >= self.config.min_precision,
            'recall': self.metrics.recall >= self.config.min_recall,
            'f1_score': self.metrics.f1_score >= self.config.min_f1_score
        }
        
        all_passed = all(validations.values())
        
        for metric_name, passed in validations.items():
            status = "✓" if passed else "✗"
            logger.info(f"{status} {metric_name}: {getattr(self.metrics, metric_name):.4f}")
        
        if all_passed:
            logger.info("✓ Model validation PASSED")
        else:
            logger.warning("✗ Model validation FAILED")
        
        return all_passed
    
    def check_data_drift(
        self, 
        reference_data: pd.DataFrame, 
        current_data: pd.DataFrame
    ) -> Dict[str, Any]:
        """Check for data drift between reference and current data"""
        logger.info("Checking for data drift")
        
        # Create Evidently report
        report = Report(metrics=[
            DataDriftPreset(),
        ])
        
        report.run(
            reference_data=reference_data,
            current_data=current_data
        )
        
        # Extract drift results
        drift_results = report.as_dict()
        
        # Calculate drift score (simplified)
        drift_score = drift_results['metrics'][0]['result'].get('dataset_drift', False)
        
        logger.info(f"Data drift detected: {drift_score}")
        
        return {
            'drift_detected': drift_score,
            'report': drift_results,
            'timestamp': datetime.now().isoformat()
        }
    
    def save_artifacts(self) -> Dict[str, str]:
        """Save model artifacts locally"""
        logger.info("Saving model artifacts")
        
        if self.model is None:
            raise ValueError("No model to save. Train model first.")
        
        timestamp = datetime.now().strftime('%Y%m%d_%H%M%S')
        version_path = self.artifact_path / f"v{self.config.model_version}_{timestamp}"
        version_path.mkdir(parents=True, exist_ok=True)
        
        artifacts = {}
        
        # Save model
        model_path = version_path / "model.joblib"
        joblib.dump(self.model, model_path)
        artifacts['model'] = str(model_path)
        logger.info(f"Saved model to {model_path}")
        
        # Save scaler
        scaler_path = version_path / "scaler.joblib"
        joblib.dump(self.scaler, scaler_path)
        artifacts['scaler'] = str(scaler_path)
        logger.info(f"Saved scaler to {scaler_path}")
        
        # Save feature columns
        features_path = version_path / "features.json"
        with open(features_path, 'w') as f:
            json.dump(self.feature_columns, f, indent=2)
        artifacts['features'] = str(features_path)
        
        # Save metrics
        if self.metrics:
            metrics_path = version_path / "metrics.json"
            with open(metrics_path, 'w') as f:
                json.dump(asdict(self.metrics), f, indent=2)
            artifacts['metrics'] = str(metrics_path)
        
        # Save config
        config_path = version_path / "config.json"
        with open(config_path, 'w') as f:
            json.dump(asdict(self.config), f, indent=2)
        artifacts['config'] = str(config_path)
        
        logger.info(f"All artifacts saved to {version_path}")
        return artifacts
    
    def upload_to_s3(self, artifacts: Dict[str, str]) -> bool:
        """Upload artifacts to S3"""
        if not self.config.s3_bucket:
            logger.warning("No S3 bucket configured. Skipping upload.")
            return False
        
        logger.info(f"Uploading artifacts to S3 bucket: {self.config.s3_bucket}")
        
        try:
            s3_client = boto3.client('s3')
            
            for artifact_type, local_path in artifacts.items():
                s3_key = f"{self.config.model_name}/{self.config.model_version}/{Path(local_path).name}"
                
                s3_client.upload_file(
                    local_path,
                    self.config.s3_bucket,
                    s3_key
                )
                
                logger.info(f"Uploaded {artifact_type} to s3://{self.config.s3_bucket}/{s3_key}")
            
            return True
        
        except ClientError as e:
            logger.error(f"Error uploading to S3: {str(e)}")
            return False
    
    def load_model(self, artifact_path: str) -> bool:
        """Load a saved model"""
        logger.info(f"Loading model from {artifact_path}")
        
        try:
            model_path = Path(artifact_path) / "model.joblib"
            scaler_path = Path(artifact_path) / "scaler.joblib"
            features_path = Path(artifact_path) / "features.json"
            
            self.model = joblib.load(model_path)
            self.scaler = joblib.load(scaler_path)
            
            with open(features_path, 'r') as f:
                self.feature_columns = json.load(f)
            
            logger.info("Model loaded successfully")
            return True
        
        except Exception as e:
            logger.error(f"Error loading model: {str(e)}")
            return False
    
    def predict(self, X: pd.DataFrame) -> Tuple[np.ndarray, np.ndarray]:
        """Make predictions on new data"""
        if self.model is None:
            raise ValueError("No model loaded. Load or train a model first.")
        
        # Ensure features match training data
        if self.feature_columns:
            X = X[self.feature_columns]
        
        # Preprocess
        X_scaled = self.scaler.transform(X)
        
        # Predict
        predictions = self.model.predict(X_scaled)
        probabilities = self.model.predict_proba(X_scaled)[:, 1]
        
        return predictions, probabilities
    
    def run_pipeline(
        self, 
        data_path: str,
        target_column: str = 'is_fraud',
        hyperparameters: Optional[Dict[str, Any]] = None
    ) -> Dict[str, Any]:
        """Run the complete ML pipeline"""
        logger.info("=" * 80)
        logger.info("STARTING FRAUD DETECTION ML PIPELINE")
        logger.info("=" * 80)
        
        try:
            # Load data
            df = self.load_data(data_path)
            
            # Preprocess
            X, y = self.preprocess_data(df, target_column)
            
            # Split data
            X_train, X_test, y_train, y_test = train_test_split(
                X, y, 
                test_size=self.config.test_size,
                random_state=self.config.random_state,
                stratify=y
            )
            
            # Train
            self.train_model(X_train, y_train, hyperparameters)
            
            # Evaluate
            metrics = self.evaluate_model(X_test, y_test)
            
            # Validate
            validation_passed = self.validate_model()
            
            # Save artifacts
            artifacts = self.save_artifacts()
            
            # Upload to S3 (if configured)
            s3_uploaded = self.upload_to_s3(artifacts)
            
            result = {
                'success': validation_passed,
                'metrics': asdict(metrics),
                'artifacts': artifacts,
                's3_uploaded': s3_uploaded,
                'timestamp': datetime.now().isoformat()
            }
            
            logger.info("=" * 80)
            logger.info("PIPELINE COMPLETED SUCCESSFULLY")
            logger.info("=" * 80)
            
            return result
        
        except Exception as e:
            logger.error(f"Pipeline failed: {str(e)}")
            raise


def main():
    """Main entry point for the pipeline"""
    import argparse
    
    parser = argparse.ArgumentParser(description='Fraud Detection ML Pipeline')
    parser.add_argument('--data-path', required=True, help='Path to training data')
    parser.add_argument('--target-column', default='is_fraud', help='Target column name')
    parser.add_argument('--model-version', default='1.0.0', help='Model version')
    parser.add_argument('--environment', default='development', help='Environment (development/staging/production)')
    parser.add_argument('--s3-bucket', help='S3 bucket for artifact storage')
    parser.add_argument('--min-precision', type=float, default=0.85, help='Minimum precision threshold')
    parser.add_argument('--min-recall', type=float, default=0.80, help='Minimum recall threshold')
    parser.add_argument('--min-f1', type=float, default=0.82, help='Minimum F1 score threshold')
    
    args = parser.parse_args()
    
    # Create config
    config = PipelineConfig(
        model_version=args.model_version,
        environment=args.environment,
        s3_bucket=args.s3_bucket,
        min_precision=args.min_precision,
        min_recall=args.min_recall,
        min_f1_score=args.min_f1
    )
    
    # Run pipeline
    pipeline = FraudDetectionPipeline(config)
    result = pipeline.run_pipeline(args.data_path, args.target_column)
    
    # Exit with appropriate code
    exit(0 if result['success'] else 1)


if __name__ == '__main__':
    main()
```

```yaml
          aws-access-key-id: ${{ secrets.AWS_ACCESS_KEY_ID }}
          aws-secret-access-key: ${{ secrets.AWS_SECRET_ACCESS_KEY }}
          aws-region: ${{ env.AWS_REGION }}

      - name: Delete old model versions
        run: |
          python scripts/cleanup_old_models.py \
            --model-name ${{ env.MODEL_NAME }} \
            --keep-last 5

      - name: Archive old artifacts
        run: |
          python scripts/archive_artifacts.py \
            --bucket ${{ secrets.ARCHIVE_BUCKET }} \
            --retention-days 90
```
