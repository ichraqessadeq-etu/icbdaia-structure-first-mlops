@dataclass
class ModelMetrics:
    """Container for model performance metrics."""
    accuracy: float
    precision: float
    recall: float
    f1_score: float
    roc_auc: float
    timestamp: str
    
    def to_dict(self) -> Dict[str, Any]:
        return {
            "accuracy": self.accuracy,
            "precision": self.precision,
            "recall": self.recall,
            "f1_score": self.f1_score,
            "roc_auc": self.roc_auc,
            "timestamp": self.timestamp,
        }
    
    def meets_thresholds(self, thresholds: Dict[str, float]) -> bool:
        """Check if metrics meet minimum thresholds."""
        checks = [
            self.accuracy >= thresholds.get("accuracy", 0.0),
            self.precision >= thresholds.get("precision", 0.0),
            self.recall >= thresholds.get("recall", 0.0),
            self.f1_score >= thresholds.get("f1_score", 0.0),
            self.roc_auc >= thresholds.get("roc_auc", 0.0),
        ]
        return all(checks)


class DataValidator:
    """Validates input data quality and schema."""
    
    def __init__(self, expected_features: List[str]):
        self.expected_features = expected_features
    
    def validate_schema(self, df: pd.DataFrame) -> Tuple[bool, List[str]]:
        """Validate dataframe schema."""
        errors = []
        
        # Check required columns
        missing_cols = set(self.expected_features) - set(df.columns)
        if missing_cols:
            errors.append(f"Missing columns: {missing_cols}")
        
        # Check for null values
        null_counts = df[self.expected_features].isnull().sum()
        if null_counts.any():
            errors.append(f"Null values found: {null_counts[null_counts > 0].to_dict()}")
        
        # Check data types
        for col in self.expected_features:
            if col in df.columns and not pd.api.types.is_numeric_dtype(df[col]):
                errors.append(f"Column {col} is not numeric")
        
        return len(errors) == 0, errors
    
    def validate_data_quality(self, df: pd.DataFrame) -> Tuple[bool, List[str]]:
        """Validate data quality metrics."""
        errors = []
        
        # Check for duplicates
        duplicates = df.duplicated().sum()
        if duplicates > len(df) * 0.01:  # More than 1% duplicates
            errors.append(f"High duplicate rate: {duplicates} rows")
        
        # Check value ranges
        for col in self.expected_features:
            if col in df.columns:
                if (df[col] == np.inf).any() or (df[col] == -np.inf).any():
                    errors.append(f"Infinite values in column: {col}")
        
        return len(errors) == 0, errors


class ModelTrainer:
    """Handles model training and hyperparameter tuning."""
    
    def __init__(self, model_config: Optional[Dict[str, Any]] = None):
        self.model_config = model_config or {
            "n_estimators": 100,
            "max_depth": 10,
            "min_samples_split": 5,
            "random_state": 42,
            "class_weight": "balanced",
        }
        self.model = None
        self.scaler = StandardScaler()
    
    def train(
        self,
        X_train: pd.DataFrame,
        y_train: pd.Series,
        X_val: Optional[pd.DataFrame] = None,
        y_val: Optional[pd.Series] = None,
    ) -> "ModelTrainer":
        """Train the fraud detection model."""
        logger.info("Starting model training...")
        
        # Scale features
        X_train_scaled = self.scaler.fit_transform(X_train)
        
        # Train model
        self.model = RandomForestClassifier(**self.model_config)
        self.model.fit(X_train_scaled, y_train)
        
        logger.info("Model training completed")
        
        # Validate on validation set if provided
        if X_val is not None and y_val is not None:
            X_val_scaled = self.scaler.transform(X_val)
            val_score = self.model.score(X_val_scaled, y_val)
            logger.info(f"Validation accuracy: {val_score:.4f}")
        
        return self
    
    def predict(self, X: pd.DataFrame) -> np.ndarray:
        """Make predictions."""
        if self.model is None:
            raise ValueError("Model not trained yet")
        X_scaled = self.scaler.transform(X)
        return self.model.predict(X_scaled)
    
    def predict_proba(self, X: pd.DataFrame) -> np.ndarray:
        """Get prediction probabilities."""
        if self.model is None:
            raise ValueError("Model not trained yet")
        X_scaled = self.scaler.transform(X)
        return self.model.predict_proba(X_scaled)


class ModelEvaluator:
    """Evaluates model performance."""
    
    @staticmethod
    def evaluate(
        y_true: np.ndarray,
        y_pred: np.ndarray,
        y_pred_proba: Optional[np.ndarray] = None,
    ) -> ModelMetrics:
        """Calculate comprehensive model metrics."""
        metrics = ModelMetrics(
            accuracy=accuracy_score(y_true, y_pred),
            precision=precision_score(y_true, y_pred, zero_division=0),
            recall=recall_score(y_true, y_pred, zero_division=0),
            f1_score=f1_score(y_true, y_pred, zero_division=0),
            roc_auc=roc_auc_score(y_true, y_pred_proba[:, 1]) if y_pred_proba is not None else 0.0,
            timestamp=datetime.now().isoformat(),
        )
        
        logger.info(f"Model Metrics: {metrics.to_dict()}")
        return metrics
    
    @staticmethod
    def compare_models(
        current_metrics: ModelMetrics,
        baseline_metrics: ModelMetrics,
        comparison_metric: str = "f1_score",
    ) -> bool:
        """Compare current model with baseline."""
        current_value = getattr(current_metrics, comparison_metric)
        baseline_value = getattr(baseline_metrics, comparison_metric)
        
        improvement = current_value - baseline_value
        logger.info(
            f"Model comparison on {comparison_metric}: "
            f"Current={current_value:.4f}, Baseline={baseline_value:.4f}, "
            f"Improvement={improvement:.4f}"
        )
        
        return current_value >= baseline_value


class ModelRegistry:
    """Manages model versioning and storage."""
    
    def __init__(self, registry_path: str = "models"):
        self.registry_path = Path(registry_path)
        self.registry_path.mkdir(parents=True, exist_ok=True)
        self.metadata_file = self.registry_path / "metadata.json"
        self._load_metadata()
    
    def _load_metadata(self):
        """Load model registry metadata."""
        if self.metadata_file.exists():
            with open(self.metadata_file, "r") as f:
                self.metadata = json.load(f)
        else:
            self.metadata = {"models": [], "production_model": None}
    
    def _save_metadata(self):
        """Save model registry metadata."""
        with open(self.metadata_file, "w") as f:
            json.dump(self.metadata, f, indent=2)
    
    def save_model(
        self,
        trainer: ModelTrainer,
        metrics: ModelMetrics,
        version: str,
        tags: Optional[Dict[str, str]] = None,
    ) -> str:
        """Save model to registry."""
        model_dir = self.registry_path / version
        model_dir.mkdir(parents=True, exist_ok=True)
        
        # Save model and scaler
        model_path = model_dir / "model.pkl"
        scaler_path = model_dir / "scaler.pkl"
        
        joblib.dump(trainer.model, model_path)
        joblib.dump(trainer.scaler, scaler_path)
        
        # Save metrics
        metrics_path = model_dir / "metrics.json"
        with open(metrics_path, "w") as f:
            json.dump(metrics.to_dict(), f, indent=2)
        
        # Update metadata
        model_info = {
            "version": version,
            "path": str(model_dir),
            "metrics": metrics.to_dict(),
            "tags": tags or {},
            "created_at": datetime.now().isoformat(),
        }
        self.metadata["models"].append(model_info)
        self._save_metadata()
        
        logger.info(f"Model saved: {version}")
        return str(model_path)
    
    def load_model(self, version: str) -> Tuple[ModelTrainer, ModelMetrics]:
        """Load model from registry."""
        model_dir = self.registry_path / version
        
        if not model_dir.exists():
            raise ValueError(f"Model version {version} not found")
        
        # Load model and scaler
        trainer = ModelTrainer()
        trainer.model = joblib.load(model_dir / "model.pkl")
        trainer.scaler = joblib.load(model_dir / "scaler.pkl")
        
        # Load metrics
        with open(model_dir / "metrics.json", "r") as f:
            metrics_dict = json.load(f)
        
        metrics = ModelMetrics(**metrics_dict)
        
        logger.info(f"Model loaded: {version}")
        return trainer, metrics
    
    def promote_to_production(self, version: str):
        """Promote a model version to production."""
        model_dir = self.registry_path / version
        if not model_dir.exists():
            raise ValueError(f"Model version {version} not found")
        
        self.metadata["production_model"] = version
        self._save_metadata()
        
        logger.info(f"Model promoted to production: {version}")
    
    def get_production_model(self) -> Optional[Tuple[ModelTrainer, ModelMetrics]]:
        """Get the current production model."""
        prod_version = self.metadata.get("production_model")
        if prod_version:
            return self.load_model(prod_version)
        return None


class FraudDetectionPipeline:
    """Main pipeline orchestrator."""
    
    def __init__(
        self,
        expected_features: List[str],
        model_config: Optional[Dict[str, Any]] = None,
        registry_path: str = "models",
    ):
        self.expected_features = expected_features
        self.validator = DataValidator(expected_features)
        self.trainer = ModelTrainer(model_config)
        self.evaluator = ModelEvaluator()
        self.registry = ModelRegistry(registry_path)
    
    def prepare_data(
        self,
        data_path: str,
        target_column: str = "is_fraud",
        test_size: float = 0.2,
    ) -> Tuple[pd.DataFrame, pd.DataFrame, pd.Series, pd.Series]:
        """Load and prepare data for training."""
        logger.info(f"Loading data from {data_path}")
        
        # Load data
        df = pd.read_csv(data_path)
        
        # Validate schema
        schema_valid, schema_errors = self.validator.validate_schema(df)
        if not schema_valid:
            raise ValueError(f"Schema validation failed: {schema_errors}")
        
        # Validate data quality
        quality_valid, quality_errors = self.validator.validate_data_quality(df)
        if not quality_valid:
            logger.warning(f"Data quality issues: {quality_errors}")
        
        # Split features and target
        X = df[self.expected_features]
        y = df[target_column]
        
        # Train-test split
        X_train, X_test, y_train, y_test = train_test_split(
            X, y, test_size=test_size, random_state=42, stratify=y
        )
        
        logger.info(
            f"Data prepared: Train={len(X_train)}, Test={len(X_test)}, "
            f"Fraud rate={y.mean():.4f}"
        )
        
        return X_train, X_test, y_train, y_test
    
    def train_and_evaluate(
        self,
        X_train: pd.DataFrame,
        X_test: pd.DataFrame,
        y_train: pd.Series,
        y_test: pd.Series,
        version: str,
        metric_thresholds: Optional[Dict[str, float]] = None,
    ) -> Tuple[ModelTrainer, ModelMetrics, bool]:
        """Train and evaluate model."""
        # Train model
        self.trainer.train(X_train, y_train)
        
        # Make predictions
        y_pred = self.trainer.predict(X_test)
        y_pred_proba = self.trainer.predict_proba(X_test)
        
        # Evaluate
        metrics = self.evaluator.evaluate(y_test, y_pred, y_pred_proba)
        
        # Check thresholds
        thresholds_met = True
        if metric_thresholds:
            thresholds_met = metrics.meets_thresholds(metric_thresholds)
            if not thresholds_met:
                logger.warning("Model does not meet metric thresholds")
        
        # Save model
        self.registry.save_model(
            self.trainer,
            metrics,
            version,
            tags={"status": "trained"},
        )
        
        return self.trainer, metrics, thresholds_met
    
    def run_pipeline(
        self,
        data_path: str,
        version: str,
        target_column: str = "is_fraud",
        metric_thresholds: Optional[Dict[str, float]] = None,
        promote_to_prod: bool = False,
        compare_with_baseline: bool = True,
    ) -> Dict[str, Any]:
        """Run the complete ML pipeline."""
        logger.info("=" * 50)
        logger.info(f"Starting Fraud Detection Pipeline - Version {version}")
        logger.info("=" * 50)
        
        # Prepare data
        X_train, X_test, y_train, y_test = self.prepare_data(
            data_path, target_column
        )
        
        # Train and evaluate
        trainer, metrics, thresholds_met = self.train_and_evaluate(
            X_train, X_test, y_train, y_test, version, metric_thresholds
        )
        
        # Compare with baseline if requested
        baseline_comparison_passed = True
        if compare_with_baseline:
            baseline_model = self.registry.get_production_model()
            if baseline_model:
                baseline_trainer, baseline_metrics = baseline_model
                baseline_comparison_passed = self.evaluator.compare_models(
                    metrics, baseline_metrics
                )
            else:
                logger.info("No baseline model found for comparison")
        
        # Promote to production if all checks pass
        should_promote = (
            promote_to_prod
            and thresholds_met
            and baseline_comparison_passed
        )
        
        if should_promote:
            self.registry.promote_to_production(version)
            logger.info(f"Model {version} promoted to production")
        
        result = {
            "version": version,
            "metrics": metrics.to_dict(),
            "thresholds_met": thresholds_met,
            "baseline_comparison_passed": baseline_comparison_passed,
            "promoted_to_production": should_promote,
        }
        
        logger.info("=" * 50)
        logger.info("Pipeline completed successfully")
        logger.info("=" * 50)
        
        return result


def main():
    """Main entry point for the pipeline."""
    # Configuration
    EXPECTED_FEATURES = [
        "transaction_amount",
        "transaction_hour",
        "transaction_day",
        "merchant_category",
        "customer_age",
        "account_age_days",
        "previous_transactions",
        "avg_transaction_amount",
    ]
    
    METRIC_THRESHOLDS = {
        "accuracy": 0.85,
        "precision": 0.80,
        "recall": 0.75,
        "f1_score": 0.77,
        "roc_auc": 0.85,
    }
    
    # Initialize pipeline
    pipeline = FraudDetectionPipeline(
        expected_features=EXPECTED_FEATURES,
        registry_path=os.getenv("MODEL_REGISTRY_PATH", "models"),
    )
    
    # Run pipeline
    version = os.getenv("MODEL_VERSION", f"v_{datetime.now().strftime('%Y%m%d_%H%M%S')}")
    data_path = os.getenv("DATA_PATH", "data/fraud_data.csv")
    
    result = pipeline.run_pipeline(
        data_path=data_path,
        version=version,
        metric_thresholds=METRIC_THRESHOLDS,
        promote_to_prod=os.getenv("PROMOTE_TO_PROD", "false").lower() == "true",
        compare_with_baseline=True,
    )
    
    # Save results
    results_path = Path("results")
    results_path.mkdir(exist_ok=True)
    
    with open(results_path / f"pipeline_result_{version}.json", "w") as f:
        json.dump(result, f, indent=2)
    
    # Exit with appropriate code
    if not result["thresholds_met"]:
        logger.error("Pipeline failed: Metric thresholds not met")
        exit(1)
    
    if not result["baseline_comparison_passed"]:
        logger.error("Pipeline failed: Model performance worse than baseline")
        exit(1)
    
    logger.info("Pipeline succeeded!")
    exit(0)


if __name__ == "__main__":
    main()
