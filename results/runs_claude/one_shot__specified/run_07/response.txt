```python
# fraud_detection_pipeline.py
"""
Production-ready fraud detection pipeline with MLflow integration.
Handles imbalanced dataset, drift detection, and comprehensive logging.
"""

import os
import sys
import json
import logging
from typing import Tuple, Dict, Any
from datetime import datetime
import numpy as np
import pandas as pd
from sklearn.ensemble import RandomForestClassifier
from sklearn.model_selection import train_test_split, StratifiedKFold
from sklearn.preprocessing import StandardScaler
from sklearn.metrics import (
    precision_score, recall_score, f1_score, roc_auc_score,
    average_precision_score, confusion_matrix, classification_report
)
from imblearn.over_sampling import SMOTE
from imblearn.pipeline import Pipeline as ImbPipeline
import mlflow
import mlflow.sklearn
from scipy import stats
import joblib

# Configure logging
logging.basicConfig(
    level=logging.INFO,
    format='%(asctime)s - %(name)s - %(levelname)s - %(message)s',
    handlers=[
        logging.FileHandler('pipeline.log'),
        logging.StreamHandler(sys.stdout)
    ]
)
logger = logging.getLogger(__name__)


class FraudDetectionPipeline:
    """End-to-end fraud detection pipeline with drift detection."""
    
    def __init__(
        self,
        data_path: str,
        mlflow_tracking_uri: str = "http://localhost:5000",
        experiment_name: str = "fraud_detection",
        drift_threshold: float = 0.05,
        random_state: int = 42
    ):
        """
        Initialize the fraud detection pipeline.
        
        Args:
            data_path: Path to the fraud dataset CSV
            mlflow_tracking_uri: MLflow tracking server URI
            experiment_name: Name of the MLflow experiment
            drift_threshold: P-value threshold for drift detection
            random_state: Random seed for reproducibility
        """
        self.data_path = data_path
        self.mlflow_tracking_uri = mlflow_tracking_uri
        self.experiment_name = experiment_name
        self.drift_threshold = drift_threshold
        self.random_state = random_state
        self.model = None
        self.scaler = None
        self.feature_names = None
        
        # Setup MLflow
        mlflow.set_tracking_uri(self.mlflow_tracking_uri)
        mlflow.set_experiment(self.experiment_name)
        
    def load_data(self) -> pd.DataFrame:
        """
        Load and validate fraud detection dataset.
        
        Returns:
            DataFrame containing the fraud data
            
        Raises:
            FileNotFoundError: If data file doesn't exist
            ValueError: If data validation fails
        """
        try:
            logger.info(f"Loading data from {self.data_path}")
            
            if not os.path.exists(self.data_path):
                raise FileNotFoundError(f"Data file not found: {self.data_path}")
            
            df = pd.read_csv(self.data_path)
            
            # Validate data
            if df.empty:
                raise ValueError("Dataset is empty")
            
            if 'Class' not in df.columns:
                raise ValueError("Target column 'Class' not found in dataset")
            
            # Check for missing values
            if df.isnull().any().any():
                logger.warning("Missing values detected. Filling with median.")
                df = df.fillna(df.median())
            
            # Log data statistics
            fraud_rate = df['Class'].mean()
            logger.info(f"Dataset loaded: {len(df)} samples, {fraud_rate*100:.3f}% fraud rate")
            
            return df
            
        except Exception as e:
            logger.error(f"Error loading data: {str(e)}")
            raise
    
    def detect_drift(
        self,
        reference_data: pd.DataFrame,
        current_data: pd.DataFrame,
        feature_cols: list
    ) -> Tuple[bool, Dict[str, float]]:
        """
        Detect concept drift using Kolmogorov-Smirnov test.
        
        Args:
            reference_data: Historical reference dataset
            current_data: Current dataset to compare
            feature_cols: List of feature column names
            
        Returns:
            Tuple of (drift_detected, drift_scores)
        """
        try:
            logger.info("Starting drift detection")
            drift_scores = {}
            drift_detected = False
            
            for col in feature_cols:
                # Perform KS test
                statistic, p_value = stats.ks_2samp(
                    reference_data[col],
                    current_data[col]
                )
                drift_scores[col] = {
                    'statistic': statistic,
                    'p_value': p_value
                }
                
                if p_value < self.drift_threshold:
                    drift_detected = True
                    logger.warning(
                        f"Drift detected in feature '{col}': "
                        f"p-value={p_value:.4f} < {self.drift_threshold}"
                    )
            
            drift_percentage = sum(
                1 for score in drift_scores.values()
                if score['p_value'] < self.drift_threshold
            ) / len(feature_cols) * 100
            
            logger.info(
                f"Drift detection complete: {drift_percentage:.1f}% "
                f"of features show drift"
            )
            
            return drift_detected, drift_scores
            
        except Exception as e:
            logger.error(f"Error in drift detection: {str(e)}")
            raise
    
    def prepare_data(
        self,
        df: pd.DataFrame,
        test_size: float = 0.2
    ) -> Tuple[np.ndarray, np.ndarray, np.ndarray, np.ndarray, list]:
        """
        Prepare data for training: split and scale.
        
        Args:
            df: Input DataFrame
            test_size: Proportion of data for testing
            
        Returns:
            Tuple of (X_train, X_test, y_train, y_test, feature_names)
        """
        try:
            logger.info("Preparing data for training")
            
            # Separate features and target
            X = df.drop('Class', axis=1)
            y = df['Class']
            
            self.feature_names = X.columns.tolist()
            
            # Split data with stratification
            X_train, X_test, y_train, y_test = train_test_split(
                X, y,
                test_size=test_size,
                random_state=self.random_state,
                stratify=y
            )
            
            logger.info(
                f"Data split: Train={len(X_train)}, Test={len(X_test)}, "
                f"Train fraud rate={y_train.mean()*100:.3f}%"
            )
            
            # Scale features
            self.scaler = StandardScaler()
            X_train_scaled = self.scaler.fit_transform(X_train)
            X_test_scaled = self.scaler.transform(X_test)
            
            return X_train_scaled, X_test_scaled, y_train.values, y_test.values, self.feature_names
            
        except Exception as e:
            logger.error(f"Error preparing data: {str(e)}")
            raise
    
    def train_model(
        self,
        X_train: np.ndarray,
        y_train: np.ndarray,
        use_smote: bool = True,
        n_estimators: int = 100,
        max_depth: int = 20,
        min_samples_split: int = 10,
        min_samples_leaf: int = 5,
        class_weight: str = 'balanced'
    ) -> RandomForestClassifier:
        """
        Train Random Forest model with SMOTE for imbalanced data.
        
        Args:
            X_train: Training features
            y_train: Training labels
            use_smote: Whether to apply SMOTE oversampling
            n_estimators: Number of trees
            max_depth: Maximum tree depth
            min_samples_split: Minimum samples to split
            min_samples_leaf: Minimum samples in leaf
            class_weight: Class weighting strategy
            
        Returns:
            Trained RandomForestClassifier
        """
        try:
            logger.info("Starting model training")
            
            # Initialize model
            rf_model = RandomForestClassifier(
                n_estimators=n_estimators,
                max_depth=max_depth,
                min_samples_split=min_samples_split,
                min_samples_leaf=min_samples_leaf,
                class_weight=class_weight,
                random_state=self.random_state,
                n_jobs=-1,
                verbose=0
            )
            
            if use_smote:
                logger.info("Applying SMOTE for handling class imbalance")
                # Create pipeline with SMOTE
                smote = SMOTE(random_state=self.random_state, k_neighbors=5)
                pipeline = ImbPipeline([
                    ('smote', smote),
                    ('classifier', rf_model)
                ])
                
                # Train with SMOTE
                pipeline.fit(X_train, y_train)
                self.model = pipeline.named_steps['classifier']
                
                # For prediction, we need the full pipeline
                self._pipeline = pipeline
            else:
                # Train without SMOTE
                rf_model.fit(X_train, y_train)
                self.model = rf_model
                self._pipeline = rf_model
            
            logger.info("Model training completed")
            return self.model
            
        except Exception as e:
            logger.error(f"Error training model: {str(e)}")
            raise
    
    def evaluate_model(
        self,
        X_test: np.ndarray,
        y_test: np.ndarray
    ) -> Dict[str, Any]:
        """
        Evaluate model performance with comprehensive metrics.
        
        Args:
            X_test: Test features
            y_test: Test labels
            
        Returns:
            Dictionary containing evaluation metrics
        """
        try:
            logger.info("Evaluating model performance")
            
            # Predictions
            y_pred = self._pipeline.predict(X_test)
            y_pred_proba = self._pipeline.predict_proba(X_test)[:, 1]
            
            # Calculate metrics
            metrics = {
                'precision': precision_score(y_test, y_pred, zero_division=0),
                'recall': recall_score(y_test, y_pred, zero_division=0),
                'f1_score': f1_score(y_test, y_pred, zero_division=0),
                'roc_auc': roc_auc_score(y_test, y_pred_proba),
                'average_precision': average_precision_score(y_test, y_pred_proba),
            }
            
            # Confusion matrix
            tn, fp, fn, tp = confusion_matrix(y_test, y_pred).ravel()
            metrics.update({
                'true_negatives': int(tn),
                'false_positives': int(fp),
                'false_negatives': int(fn),
                'true_positives': int(tp),
                'specificity': tn / (tn + fp) if (tn + fp) > 0 else 0,
            })
            
            # Feature importance
            if hasattr(self.model, 'feature_importances_'):
                feature_importance = dict(zip(
                    self.feature_names,
                    self.model.feature_importances_
                ))
                top_features = sorted(
                    feature_importance.items(),
                    key=lambda x: x[1],
                    reverse=True
                )[:10]
                metrics['top_10_features'] = dict(top_features)
            
            # Log detailed metrics
            logger.info("Model Evaluation Results:")
            logger.info(f"  Precision: {metrics['precision']:.4f}")
            logger.info(f"  Recall: {metrics['recall']:.4f}")
            logger.info(f"  F1-Score: {metrics['f1_score']:.4f}")
            logger.info(f"  ROC-AUC: {metrics['roc_auc']:.4f}")
            logger.info(f"  Average Precision: {metrics['average_precision']:.4f}")
            logger.info(f"  Confusion Matrix: TP={tp}, FP={fp}, TN={tn}, FN={fn}")
            
            return metrics
            
        except Exception as e:
            logger.error(f"Error evaluating model: {str(e)}")
            raise
    
    def log_to_mlflow(
        self,
        metrics: Dict[str, Any],
        params: Dict[str, Any],
        artifacts: Dict[str, str] = None
    ) -> str:
        """
        Log model, metrics, and artifacts to MLflow.
        
        Args:
            metrics: Dictionary of metrics to log
            params: Dictionary of parameters to log
            artifacts: Dictionary of artifact paths to log
            
        Returns:
            MLflow run ID
        """
        try:
            logger.info("Logging to MLflow")
            
            with mlflow.start_run() as run:
                # Log parameters
                mlflow.log_params(params)
                
                # Log metrics (filter out non-numeric values)
                numeric_metrics = {
                    k: v for k, v in metrics.items()
                    if isinstance(v, (int, float, np.number))
                }
                mlflow.log_metrics(numeric_metrics)
                
                # Log complex metrics as JSON
                if 'top_10_features' in metrics:
                    mlflow.log_dict(
                        metrics['top_10_features'],
                        'top_10_features.json'
                    )
                
                # Log model
                mlflow.sklearn.log_model(
                    self._pipeline,
                    "model",
                    registered_model_name="fraud_detection_rf"
                )
                
                # Log scaler
                mlflow.sklearn.log_model(
                    self.scaler,
                    "scaler"
                )
                
                # Log artifacts
                if artifacts:
                    for name, path in artifacts.items():
                        if os.path.exists(path):
                            mlflow.log_artifact(path, name)
                
                # Add tags
                mlflow.set_tags({
                    'model_type': 'RandomForestClassifier',
                    'problem_type': 'fraud_detection',
                    'framework': 'scikit-learn',
                    'timestamp': datetime.now().isoformat()
                })
                
                run_id = run.info.run_id
                logger.info(f"MLflow run completed: {run_id}")
                
                return run_id
                
        except Exception as e:
            logger.error(f"Error logging to MLflow: {str(e)}")
            raise
    
    def save_model(self, model_path: str = "models/fraud_detection_model.pkl"):
        """
        Save trained model to disk.
        
        Args:
            model_path: Path to save the model
        """
        try:
            os.makedirs(os.path.dirname(model_path), exist_ok=True)
            
            model_artifacts = {
                'pipeline': self._pipeline,
                'scaler': self.scaler,
                'feature_names': self.feature_names,
                'metadata': {
                    'timestamp': datetime.now().isoformat(),
                    'random_state': self.random_state
                }
            }
            
            joblib.dump(model_artifacts, model_path)
            logger.info(f"Model saved to {model_path}")
            
        except Exception as e:
            logger.error(f"Error saving model: {str(e)}")
            raise
    
    def run_pipeline(
        self,
        reference_data_path: str = None,
        check_drift: bool = True,
        save_model_path: str = "models/fraud_detection_model.pkl"
    ) -> Dict[str, Any]:
        """
        Execute the complete pipeline.
        
        Args:
            reference_data_path: Path to reference data for drift detection
            check_drift: Whether to perform drift detection
            save_model_path: Path to save the trained model
            
        Returns:
            Dictionary containing pipeline results
        """
        try:
            logger.info("=" * 80)
            logger.info("Starting Fraud Detection Pipeline")
            logger.info("=" * 80)
            
            # Load data
            df = self.load_data()
            
            # Drift detection
            drift_detected = False
            drift_scores = None
            
            if check_drift and reference_data_path and os.path.exists(reference_data_path):
                reference_df = pd.read_csv(reference_data_path)
                feature_cols = [col for col in df.columns if col != 'Class']
                drift_detected, drift_scores = self.detect_drift(
                    reference_df,
                    df,
                    feature_cols
                )
                
                if not drift_detected:
                    logger.info("No significant drift detected. Skipping retraining.")
                    return {
                        'status': 'skipped',
                        'reason': 'no_drift_detected',
                        'drift_scores': drift_scores
                    }
            
            # Prepare data
            X_train, X_test, y_train, y_test, feature_names = self.prepare_data(df)
            
            # Model parameters
            params = {
                'n_estimators': 100,
                'max_depth': 20,
                'min_samples_split': 10,
                'min_samples_leaf': 5,
                'class_weight': 'balanced',
                'use_smote': True,
                'test_size': 0.2,
                'random_state': self.random_state
            }
            
            # Train model
            self.train_model(
                X_train,
                y_train,
                use_smote=params['use_smote'],
                n_estimators=params['n_estimators'],
                max_depth=params['max_depth'],
                min_samples_split=params['min_samples_split'],
                min_samples_leaf=params['min_samples_leaf'],
                class_weight=params['class_weight']
            )
            
            # Evaluate model
            metrics = self.evaluate_model(X_test, y_test)
            
            # Add dataset info to metrics
            metrics['train_samples'] = len(X_train)
            metrics['test_samples'] = len(X_test)
            metrics['fraud_rate'] = float(y_train.mean())
            
            # Log to MLflow
            run_id = self.log_to_mlflow(metrics, params)
            
            # Save model
            self.save_model(save_model_path)
            
            # Prepare results
            results = {
                'status': 'success',
                'mlflow_run_id': run_id,
                'metrics': metrics,
                'params': params,
                'drift_detected': drift_detected,
                'drift_scores': drift_scores,
                'model_path': save_model_path
            }
            
            logger.info("=" * 80)
            logger.info("Pipeline execution completed successfully")
            logger.info("=" * 80)
            
            return results
            
        except Exception as e:
            logger.error(f"Pipeline execution failed: {str(e)}")
            raise


def main():
    """Main entry point for the pipeline."""
    import argparse
    
    parser = argparse.ArgumentParser(description='Fraud Detection Pipeline')
    parser.add_argument(
        '--data-path',
        type=str,
        required=True,
        help='Path to fraud detection dataset'
    )
    parser.add_argument(
        '--reference-data-path',
        type=str,
        default=None,
        help='Path to reference dataset for drift detection'
    )
    parser.add_argument(
        '--mlflow-uri',
        type=str,
        default='http://localhost:5000',
        help='MLflow tracking URI'
    )
    parser.add_argument(
        '--experiment-name',
        type=str,
        default='fraud_detection',
        help='MLflow experiment name'
    )
    parser.add_argument(
        '--no-drift-check',
        action='store_true',
        help='Skip drift detection'
    )
    parser.add_argument(
        '--model-output',
        type=str,
        default='models/fraud_detection_model.pkl',
        help='Path to save trained model'
    )
    
    args = parser.parse_args()
    
    # Initialize and run pipeline
    pipeline = FraudDetectionPipeline(
        data_path=args.data_path,
        mlflow_tracking_uri=args.mlflow_uri,
        experiment_name=args.experiment_name
    )
    
    results = pipeline.run_pipeline(
        reference_data_path=args.reference_data_path,
        check_drift=not args.no_drift_check,
        save_model_path=args.model_output
    )
    
    # Save results to JSON
    results_path = 'pipeline_results.json'
    with open(results_path, 'w') as f:
        # Convert numpy types to native Python types for JSON serialization
        json_results = json.loads(
            json.dumps(results, default=lambda x: float(x) if isinstance(x, np.number) else str(x))
        )
        json.dump(json_results, f, indent=2)
    
    logger.info(f"Results saved to {results_path}")
    
    # Exit with appropriate code
    if results['status'] == 'success':
        sys.exit(0)
    else:
        sys.exit(1)


if __name__ == '__main__':
    main()
```

```python
        # Create drifted data by adding noise
        drifted_data = sample_fraud_data.copy()
        for col in feature_cols[:5]:  # Drift first 5 features

            sample_fraud_data,
            drifted_data,
            feature_cols
        )
        
        assert drift_detected is True
        assert len(drift_scores) == len(feature_cols)
        
        # Check that drifted features have low p-values
        drifted_features = feature_cols[:5]
        for feature in drifted_features:
            assert drift_scores[feature]['p_value'] < 0.05
```

```yaml
        df = pd.DataFrame(X, columns=[f'V{i}' for i in range(1, 31)])
        df['Class'] = y
        df.to_csv('training_data.csv', index=False)

          --model-output models/fraud_detection_model.pkl

      run: |
        python -c "
        import json
        import sys
        
        with open('pipeline_results.json', 'r') as f:
            results = json.load(f)
        
        metrics = results['metrics']
        
        # Define minimum thresholds
        min_recall = 0.70  # Must catch at least 70% of fraud
        min_precision = 0.50  # At least 50% precision
        min_roc_auc = 0.80  # ROC-AUC should be at least 0.80

        print(f\"  Precision: {metrics['precision']:.4f} (min: {min_precision})\")
        print(f\"  ROC-AUC: {metrics['roc_auc']:.4f} (min: {min_roc_auc})\")
        print(f\"  F1-Score: {metrics['f1_score']:.4f}\")
        
        # Check thresholds
        if metrics['recall'] < min_recall:
            print(f\"ERROR: Recall {metrics['recall']:.4f} below threshold {min_recall}\")
            sys.exit(1)
        
        if metrics['precision'] < min_precision:
            print(f\"ERROR: Precision {metrics['precision']:.4f} below threshold {min_precision}\")
            sys.exit(1)
        
        if metrics['roc_auc'] < min_roc_auc:
            print(f\"ERROR: ROC-AUC {metrics['roc_auc']:.4f} below threshold {min_roc_auc}\")
            sys.exit(1)
        
        print('✅ Model meets all performance thresholds')
        "

      run: |
        python -c "
        import json
        import os

            results = json.load(f)

        ## Confusion Matrix
        
        |  | Predicted Negative | Predicted Positive |
        |---|---|---|
        | **Actual Negative** | {metrics['true_negatives']} (TN) | {metrics['false_positives']} (FP) |
        | **Actual Positive** | {metrics['false_negatives']} (FN) | {metrics['true_positives']} (TP) |
        
        ## Top 10 Important Features
        '''
        
        if 'top_10_features' in metrics:
            report += '\n| Feature | Importance |\n|---------|------------|\n'
            for feature, importance in metrics['top_10_features'].items():
                report += f'| {feature} | {importance:.4f} |\n'
        
        with open('model_report.md', 'w') as f:
            f.write(report)
```
