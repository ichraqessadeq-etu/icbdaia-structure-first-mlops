class ConceptDriftDetector:
    """Detect concept drift using statistical tests."""
    
    def __init__(self, reference_data: pd.DataFrame, threshold: float = 0.05):
        """
        Initialize drift detector.
        
        Args:
            reference_data: Reference dataset for comparison
            threshold: P-value threshold for drift detection
        """
        self.reference_data = reference_data
        self.threshold = threshold
        self.feature_stats = self._compute_feature_stats(reference_data)
    
    def _compute_feature_stats(self, data: pd.DataFrame) -> Dict[str, Dict[str, float]]:
        """Compute statistical features for each column."""
        stats_dict = {}
        for col in data.columns:
            if col != 'Class':  # Exclude target variable
                stats_dict[col] = {
                    'mean': data[col].mean(),
                    'std': data[col].std(),
                    'median': data[col].median()
                }
        return stats_dict
    
    def detect_drift(self, new_data: pd.DataFrame) -> Tuple[bool, Dict[str, Any]]:
        """
        Detect concept drift using Kolmogorov-Smirnov test.
        
        Args:
            new_data: New dataset to compare
            
        Returns:
            Tuple of (drift_detected, drift_report)
        """
        drift_report = {
            'timestamp': datetime.now().isoformat(),
            'drifted_features': [],
            'p_values': {}
        }
        
        drift_detected = False
        
        for col in self.reference_data.columns:
            if col != 'Class':
                try:
                    # Kolmogorov-Smirnov test
                    ks_statistic, p_value = stats.ks_2samp(
                        self.reference_data[col].dropna(),
                        new_data[col].dropna()
                    )
                    
                    drift_report['p_values'][col] = p_value
                    
                    if p_value < self.threshold:
                        drift_report['drifted_features'].append(col)
                        drift_detected = True
                        logger.warning(f"Drift detected in feature '{col}' (p-value: {p_value:.4f})")
                        
                except Exception as e:
                    logger.error(f"Error detecting drift for feature '{col}': {str(e)}")
        
        drift_report['drift_detected'] = drift_detected
        drift_report['num_drifted_features'] = len(drift_report['drifted_features'])
        
        return drift_detected, drift_report


class FraudDetectionPipeline:
    """Complete fraud detection pipeline with MLflow tracking."""
    
    def __init__(
        self,
        mlflow_tracking_uri: str = "http://localhost:5000",
        experiment_name: str = "fraud_detection",
        random_state: int = 42
    ):
        """
        Initialize pipeline.
        
        Args:
            mlflow_tracking_uri: MLflow tracking server URI
            experiment_name: Name of MLflow experiment
            random_state: Random seed for reproducibility
        """
        self.random_state = random_state
        self.model = None
        self.scaler = None
        self.drift_detector = None
        
        # Setup MLflow
        mlflow.set_tracking_uri(mlflow_tracking_uri)
        mlflow.set_experiment(experiment_name)
        
        logger.info(f"Initialized FraudDetectionPipeline with experiment: {experiment_name}")
    
    def load_data(self, data_path: str) -> pd.DataFrame:
        """
        Load and validate dataset.
        
        Args:
            data_path: Path to CSV file
            
        Returns:
            Loaded DataFrame
        """
        try:
            logger.info(f"Loading data from {data_path}")
            df = pd.read_csv(data_path)
            
            # Validate required columns
            if 'Class' not in df.columns:
                raise ValueError("Dataset must contain 'Class' column")
            
            # Log data statistics
            fraud_rate = df['Class'].mean()
            logger.info(f"Dataset loaded: {len(df)} samples, fraud rate: {fraud_rate:.4%}")
            
            if fraud_rate > 0.05:
                logger.warning(f"Unusually high fraud rate detected: {fraud_rate:.4%}")
            
            # Check for missing values
            missing_values = df.isnull().sum().sum()
            if missing_values > 0:
                logger.warning(f"Found {missing_values} missing values. Filling with median.")
                df = df.fillna(df.median())
            
            return df
            
        except FileNotFoundError:
            logger.error(f"Data file not found: {data_path}")
            raise
        except Exception as e:
            logger.error(f"Error loading data: {str(e)}")
            raise
    
    def preprocess_data(
        self, 
        df: pd.DataFrame,
        test_size: float = 0.2,
        validation_size: float = 0.1
    ) -> Tuple[np.ndarray, np.ndarray, np.ndarray, np.ndarray, np.ndarray, np.ndarray]:
        """
        Preprocess data with train/validation/test split.
        
        Args:
            df: Input DataFrame
            test_size: Proportion of data for testing
            validation_size: Proportion of training data for validation
            
        Returns:
            Tuple of (X_train, X_val, X_test, y_train, y_val, y_test)
        """
        try:
            logger.info("Preprocessing data...")
            
            # Separate features and target
            X = df.drop('Class', axis=1).values
            y = df['Class'].values
            
            # First split: train+val vs test
            X_temp, X_test, y_temp, y_test = train_test_split(
                X, y, 
                test_size=test_size,
                stratify=y,
                random_state=self.random_state
            )
            
            # Second split: train vs validation
            val_size_adjusted = validation_size / (1 - test_size)
            X_train, X_val, y_train, y_val = train_test_split(
                X_temp, y_temp,
                test_size=val_size_adjusted,
                stratify=y_temp,
                random_state=self.random_state
            )
            
            # Scale features
            self.scaler = StandardScaler()
            X_train = self.scaler.fit_transform(X_train)
            X_val = self.scaler.transform(X_val)
            X_test = self.scaler.transform(X_test)
            
            logger.info(f"Data split - Train: {len(X_train)}, Val: {len(X_val)}, Test: {len(X_test)}")
            logger.info(f"Class distribution - Train: {y_train.mean():.4%}, Val: {y_val.mean():.4%}, Test: {y_test.mean():.4%}")
            
            return X_train, X_val, X_test, y_train, y_val, y_test
            
        except Exception as e:
            logger.error(f"Error in preprocessing: {str(e)}")
            raise
    
    def handle_imbalance(
        self,
        X_train: np.ndarray,
        y_train: np.ndarray,
        sampling_strategy: str = 'combined'
    ) -> Tuple[np.ndarray, np.ndarray]:
        """
        Handle class imbalance using SMOTE and undersampling.
        
        Args:
            X_train: Training features
            y_train: Training labels
            sampling_strategy: 'smote', 'undersample', or 'combined'
            
        Returns:
            Resampled (X_train, y_train)
        """
        try:
            logger.info(f"Handling class imbalance using {sampling_strategy} strategy...")
            original_fraud_rate = y_train.mean()
            
            if sampling_strategy == 'smote':
                sampler = SMOTE(random_state=self.random_state)
                X_resampled, y_resampled = sampler.fit_resample(X_train, y_train)
                
            elif sampling_strategy == 'undersample':
                sampler = RandomUnderSampler(random_state=self.random_state)
                X_resampled, y_resampled = sampler.fit_resample(X_train, y_train)
                
            elif sampling_strategy == 'combined':
                # First oversample minority class to 0.3, then undersample majority
                smote = SMOTE(sampling_strategy=0.3, random_state=self.random_state)
                undersampler = RandomUnderSampler(sampling_strategy=0.5, random_state=self.random_state)
                
                X_resampled, y_resampled = smote.fit_resample(X_train, y_train)
                X_resampled, y_resampled = undersampler.fit_resample(X_resampled, y_resampled)
            else:
                raise ValueError(f"Unknown sampling strategy: {sampling_strategy}")
            
            new_fraud_rate = y_resampled.mean()
            logger.info(f"Resampling complete - Original fraud rate: {original_fraud_rate:.4%}, New: {new_fraud_rate:.4%}")
            logger.info(f"Dataset size - Original: {len(X_train)}, Resampled: {len(X_resampled)}")
            
            return X_resampled, y_resampled
            
        except Exception as e:
            logger.error(f"Error in handling imbalance: {str(e)}")
            raise
    
    def train_model(
        self,
        X_train: np.ndarray,
        y_train: np.ndarray,
        X_val: np.ndarray,
        y_val: np.ndarray,
        hyperparameters: Dict[str, Any] = None
    ) -> RandomForestClassifier:
        """
        Train Random Forest model with validation.
        
        Args:
            X_train: Training features
            y_train: Training labels
            X_val: Validation features
            y_val: Validation labels
            hyperparameters: Model hyperparameters
            
        Returns:
            Trained model
        """
        try:
            logger.info("Training Random Forest model...")
            
            # Default hyperparameters
            if hyperparameters is None:
                hyperparameters = {
                    'n_estimators': 100,
                    'max_depth': 20,
                    'min_samples_split': 10,
                    'min_samples_leaf': 4,
                    'max_features': 'sqrt',
                    'class_weight': 'balanced',
                    'random_state': self.random_state,
                    'n_jobs': -1
                }
            
            self.model = RandomForestClassifier(**hyperparameters)
            
            # Train model
            self.model.fit(X_train, y_train)
            
            # Validation predictions
            val_predictions = self.model.predict(X_val)
            val_proba = self.model.predict_proba(X_val)[:, 1]
            
            # Validation metrics
            val_f1 = f1_score(y_val, val_predictions)
            val_recall = recall_score(y_val, val_predictions)
            val_precision = precision_score(y_val, val_predictions)
            val_auc = roc_auc_score(y_val, val_proba)
            
            logger.info(f"Validation Metrics - F1: {val_f1:.4f}, Recall: {val_recall:.4f}, Precision: {val_precision:.4f}, AUC: {val_auc:.4f}")
            
            return self.model
            
        except Exception as e:
            logger.error(f"Error in training: {str(e)}")
            raise
    
    def evaluate_model(
        self,
        X_test: np.ndarray,
        y_test: np.ndarray
    ) -> Dict[str, Any]:
        """
        Comprehensive model evaluation.
        
        Args:
            X_test: Test features
            y_test: Test labels
            
        Returns:
            Dictionary of evaluation metrics
        """
        try:
            logger.info("Evaluating model on test set...")
            
            if self.model is None:
                raise ValueError("Model not trained yet")
            
            # Predictions
            y_pred = self.model.predict(X_test)
            y_proba = self.model.predict_proba(X_test)[:, 1]
            
            # Compute metrics
            metrics = {
                'accuracy': (y_pred == y_test).mean(),
                'precision': precision_score(y_test, y_pred, zero_division=0),
                'recall': recall_score(y_test, y_pred, zero_division=0),
                'f1_score': f1_score(y_test, y_pred, zero_division=0),
                'roc_auc': roc_auc_score(y_test, y_proba),
                'average_precision': average_precision_score(y_test, y_proba)
            }
            
            # Confusion matrix
            tn, fp, fn, tp = confusion_matrix(y_test, y_pred).ravel()
            metrics['true_negatives'] = int(tn)
            metrics['false_positives'] = int(fp)
            metrics['false_negatives'] = int(fn)
            metrics['true_positives'] = int(tp)
            
            # Specificity
            metrics['specificity'] = tn / (tn + fp) if (tn + fp) > 0 else 0
            
            # Log metrics
            logger.info("Test Set Metrics:")
            for metric_name, metric_value in metrics.items():
                if isinstance(metric_value, float):
                    logger.info(f"  {metric_name}: {metric_value:.4f}")
                else:
                    logger.info(f"  {metric_name}: {metric_value}")
            
            # Classification report
            report = classification_report(y_test, y_pred, output_dict=True)
            metrics['classification_report'] = report
            
            return metrics
            
        except Exception as e:
            logger.error(f"Error in evaluation: {str(e)}")
            raise
    
    def log_to_mlflow(
        self,
        metrics: Dict[str, Any],
        hyperparameters: Dict[str, Any],
        model_name: str = "fraud_detection_rf"
    ) -> str:
        """
        Log model, metrics, and artifacts to MLflow.
        
        Args:
            metrics: Evaluation metrics
            hyperparameters: Model hyperparameters
            model_name: Name for registered model
            
        Returns:
            Run ID
        """
        try:
            logger.info("Logging to MLflow...")
            
            with mlflow.start_run() as run:
                # Log parameters
                mlflow.log_params(hyperparameters)
                
                # Log metrics (only numeric values)
                numeric_metrics = {
                    k: v for k, v in metrics.items() 
                    if isinstance(v, (int, float)) and k != 'classification_report'
                }
                mlflow.log_metrics(numeric_metrics)
                
                # Log model
                mlflow.sklearn.log_model(
                    self.model,
                    "model",
                    registered_model_name=model_name
                )
                
                # Log scaler
                mlflow.sklearn.log_model(
                    self.scaler,
                    "scaler"
                )
                
                # Log classification report as artifact
                report_path = "classification_report.json"
                with open(report_path, 'w') as f:
                    json.dump(metrics['classification_report'], f, indent=2)
                mlflow.log_artifact(report_path)
                os.remove(report_path)
                
                # Log feature importance
                if hasattr(self.model, 'feature_importances_'):
                    feature_importance_path = "feature_importance.json"
                    importance_dict = {
                        f'feature_{i}': float(imp) 
                        for i, imp in enumerate(self.model.feature_importances_)
                    }
                    with open(feature_importance_path, 'w') as f:
                        json.dump(importance_dict, f, indent=2)
                    mlflow.log_artifact(feature_importance_path)
                    os.remove(feature_importance_path)
                
                # Tag run
                mlflow.set_tag("model_type", "RandomForestClassifier")
                mlflow.set_tag("task", "fraud_detection")
                mlflow.set_tag("timestamp", datetime.now().isoformat())
                
                run_id = run.info.run_id
                logger.info(f"MLflow run completed. Run ID: {run_id}")
                
                return run_id
                
        except Exception as e:
            logger.error(f"Error logging to MLflow: {str(e)}")
            raise
    
    def check_drift_and_retrain(
        self,
        reference_data_path: str,
        new_data_path: str,
        drift_threshold: float = 0.05
    ) -> Tuple[bool, Dict[str, Any]]:
        """
        Check for concept drift and trigger retraining if detected.
        
        Args:
            reference_data_path: Path to reference dataset
            new_data_path: Path to new dataset
            drift_threshold: P-value threshold for drift detection
            
        Returns:
            Tuple of (drift_detected, drift_report)
        """
        try:
            logger.info("Checking for concept drift...")
            
            # Load datasets
            reference_df = self.load_data(reference_data_path)
            new_df = self.load_data(new_data_path)
            
            # Initialize drift detector
            self.drift_detector = ConceptDriftDetector(
                reference_df.drop('Class', axis=1),
                threshold=drift_threshold
            )
            
            # Detect drift
            drift_detected, drift_report = self.drift_detector.detect_drift(
                new_df.drop('Class', axis=1)
            )
            
            # Save drift report
            drift_report_path = f"drift_report_{datetime.now().strftime('%Y%m%d_%H%M%S')}.json"
            with open(drift_report_path, 'w') as f:
                json.dump(drift_report, f, indent=2)
            
            logger.info(f"Drift report saved to {drift_report_path}")
            
            if drift_detected:
                logger.warning(f"CONCEPT DRIFT DETECTED! {drift_report['num_drifted_features']} features drifted.")
                logger.info("Triggering model retraining...")
            else:
                logger.info("No concept drift detected.")
            
            return drift_detected, drift_report
            
        except Exception as e:
            logger.error(f"Error in drift detection: {str(e)}")
            raise
    
    def run_pipeline(
        self,
        data_path: str,
        hyperparameters: Dict[str, Any] = None,
        sampling_strategy: str = 'combined',
        test_size: float = 0.2,
        validation_size: float = 0.1
    ) -> Dict[str, Any]:
        """
        Execute complete pipeline.
        
        Args:
            data_path: Path to dataset
            hyperparameters: Model hyperparameters
            sampling_strategy: Strategy for handling imbalance
            test_size: Test set proportion
            validation_size: Validation set proportion
            
        Returns:
            Pipeline results dictionary
        """
        try:
            logger.info("=" * 80)
            logger.info("STARTING FRAUD DETECTION PIPELINE")
            logger.info("=" * 80)
            
            start_time = datetime.now()
            
            # Load data
            df = self.load_data(data_path)
            
            # Preprocess
            X_train, X_val, X_test, y_train, y_val, y_test = self.preprocess_data(
                df, test_size, validation_size
            )
            
            # Handle imbalance
            X_train_resampled, y_train_resampled = self.handle_imbalance(
                X_train, y_train, sampling_strategy
            )
            
            # Train model
            self.train_model(
                X_train_resampled, y_train_resampled,
                X_val, y_val,
                hyperparameters
            )
            
            # Evaluate model
            metrics = self.evaluate_model(X_test, y_test)
            
            # Log to MLflow
            if hyperparameters is None:
                hyperparameters = {
                    'n_estimators': 100,
                    'max_depth': 20,
                    'min_samples_split': 10,
                    'min_samples_leaf': 4,
                    'max_features': 'sqrt',
                    'class_weight': 'balanced',
                    'random_state': self.random_state
                }
            
            run_id = self.log_to_mlflow(metrics, hyperparameters)
            
            end_time = datetime.now()
            duration = (end_time - start_time).total_seconds()
            
            results = {
                'run_id': run_id,
                'metrics': metrics,
                'duration_seconds': duration,
                'timestamp': end_time.isoformat()
            }
            
            logger.info("=" * 80)
            logger.info(f"PIPELINE COMPLETED SUCCESSFULLY in {duration:.2f} seconds")
            logger.info("=" * 80)
            
            return results
            
        except Exception as e:
            logger.error(f"Pipeline failed: {str(e)}")
            raise


def main():
    """Main execution function."""
    try:
        # Configuration
        DATA_PATH = os.getenv('DATA_PATH', 'data/creditcard.csv')
        MLFLOW_URI = os.getenv('MLFLOW_TRACKING_URI', 'http://localhost:5000')
        
        # Initialize pipeline
        pipeline = FraudDetectionPipeline(
            mlflow_tracking_uri=MLFLOW_URI,
            experiment_name="fraud_detection_production"
        )
        
        # Check for drift if reference data exists
        reference_data_path = os.getenv('REFERENCE_DATA_PATH')
        if reference_data_path and os.path.exists(reference_data_path):
            drift_detected, drift_report = pipeline.check_drift_and_retrain(
                reference_data_path=reference_data_path,
                new_data_path=DATA_PATH,
                drift_threshold=0.05
            )
            
            # Exit if no drift detected (skip retraining)
            if not drift_detected and os.getenv('SKIP_TRAINING_IF_NO_DRIFT', 'false').lower() == 'true':
                logger.info("Skipping training - no drift detected")
                return
        
        # Run pipeline
        results = pipeline.run_pipeline(
            data_path=DATA_PATH,
            sampling_strategy='combined',
            test_size=0.2,
            validation_size=0.1
        )
        
        # Check if model meets minimum performance threshold
        min_recall = float(os.getenv('MIN_RECALL_THRESHOLD', '0.75'))
        min_precision = float(os.getenv('MIN_PRECISION_THRESHOLD', '0.10'))
        
        if results['metrics']['recall'] < min_recall:
            logger.error(f"Model recall {results['metrics']['recall']:.4f} below threshold {min_recall}")
            sys.exit(1)
        
        if results['metrics']['precision'] < min_precision:
            logger.error(f"Model precision {results['metrics']['precision']:.4f} below threshold {min_precision}")
            sys.exit(1)
        
        logger.info("Model meets all performance thresholds")
        
    except Exception as e:
        logger.error(f"Main execution failed: {str(e)}")
        sys.exit(1)


if __name__ == "__main__":
    main()

          aws-secret-access-key: ${{ secrets.AWS_SECRET_ACCESS_KEY }}
          aws-region: ${{ env.AWS_REGION }}
      
      - name: Download training data from S3
        run: |
          mkdir -p data
          aws s3 cp s3://${{ secrets.S3_BUCKET }}/training_data/creditcard.csv ./data/creditcard.csv
      
      - name: Start MLflow tracking server (if self-hosted)
        if: env.MLFLOW_TRACKING_URI == ''
        run: |
          mlflow server --host 0.0.0.0 --port 5000 &
          sleep 5
          echo "MLFLOW_TRACKING_URI=http://localhost:5000" >> $GITHUB_ENV
      
      - name: Train model
        id: train
        run: |
          python fraud_detection_pipeline.py
        env:
          DATA_PATH: ./data/creditcard.csv
          MLFLOW_TRACKING_URI: ${{ env.MLFLOW_TRACKING_URI }}
          MIN_RECALL_THRESHOLD: ${{ env.MIN_RECALL_THRESHOLD }}
          MIN_PRECISION_THRESHOLD: ${{ env.MIN_PRECISION_THRESHOLD }}
      
      - name: Extract model metrics
        id: metrics
        run: |
          # Parse logs to extract metrics (in production, use MLflow API)
          recall=$(grep -oP 'recall: \K[0-9.]+' fraud_detection_pipeline.log | tail -1)
          precision=$(grep -oP 'precision: \K[0-9.]+' fraud_detection_pipeline.log | tail -1)
          f1=$(grep -oP 'f1_score: \K[0-9.]+' fraud_detection_pipeline.log | tail -1)
          auc=$(grep -oP 'roc_auc: \K[0-9.]+' fraud_detection_pipeline.log | tail -1)
          
          echo "::set-output name=recall::${recall}"
          echo "::set-output name=precision::${precision}"
          echo "::set-output name=f1::${f1}"
          echo "::set-output name=auc::${auc}"
      
      - name: Upload training logs
        uses: actions/upload-artifact@v3
        with:
          name: training-logs
          path: |
            fraud_detection_pipeline.log
            drift_report_*.json
      
      - name: Comment PR with metrics
        if: github.event_name == 'pull_request'
        uses: actions/github-script@v6
        with:
          script: |
            github.rest.issues.createComment({
              issue_number: context.issue.number,
              owner: context.repo.owner,
              repo: context.repo.repo,
              body: `## 🤖 Model Training Results
              
              | Metric | Value |
              |--------|-------|
              | Recall | ${{ steps.metrics.outputs.recall }} |
              | Precision | ${{ steps.metrics.outputs.precision }} |
              | F1 Score | ${{ steps.metrics.outputs.f1 }} |
              | ROC AUC | ${{ steps.metrics.outputs.auc }} |
              
              ✅ Model meets performance thresholds
              `
            })

  # Job 4: Model Validation
  validate:
    name: Validate Model Performance
    runs-on: ubuntu-latest
    needs: train
    
    steps:
      - name: Checkout code
        uses: actions/checkout@v3
      
      - name: Set up Python
        uses: actions/setup-python@v4
        with:
          python-version: ${{ env.PYTHON_VERSION }}
      
      - name: Install dependencies
        run: |
          pip install -r requirements.txt
      
      - name: Download validation data
        uses: aws-actions/configure-aws-credentials@v2
        with:
          aws-access-key-id: ${{ secrets.AWS_ACCESS_KEY_ID }}
          aws-secret-access-key: ${{ secrets.AWS_SECRET_ACCESS_KEY }}
          aws-region: ${{ env.AWS_REGION }}
      
      - name: Run validation tests
        run: |
          aws s3 cp s3://${{ secrets.S3_BUCKET }}/validation_data/creditcard_validation.csv ./data/validation.csv
          
          python -c "
          import mlflow
          import pandas as pd
          import numpy as np
          from sklearn.metrics import recall_score, precision_score
          
          # Load latest model from MLflow
          mlflow.set_tracking_uri('${{ env.MLFLOW_TRACKING_URI }}')
          model = mlflow.sklearn.load_model('models:/${{ env.MODEL_NAME }}/latest')
          
          # Load validation data
          df = pd.read_csv('./data/validation.csv')
          X = df.drop('Class', axis=1).values
          y = df['Class'].values
          
          # Predict
          predictions = model.predict(X)
          
          # Calculate metrics
          recall = recall_score(y, predictions)
          precision = precision_score(y, predictions)
          
          print(f'Validation Recall: {recall:.4f}')
          print(f'Validation Precision: {precision:.4f}')
          
          # Check thresholds
          assert recall >= ${{ env.MIN_RECALL_THRESHOLD }}, f'Recall {recall} below threshold'
          assert precision >= ${{ env.MIN_PRECISION_THRESHOLD }}, f'Precision {precision} below threshold'
          
          print('✅ Validation passed!')
          "
      
      - name: Performance regression check
        run: |
          python -c "
          import mlflow
          
          mlflow.set_tracking_uri('${{ env.MLFLOW_TRACKING_URI }}')
          client = mlflow.tracking.MlflowClient()
          
          # Get latest two model versions
          versions = client.search_model_versions('name=\"${{ env.MODEL_NAME }}\"')
          
          if len(versions) >= 2:
              latest_metrics = versions[0].run_data.metrics
              previous_metrics = versions[1].run_data.metrics
              
              # Check for significant degradation
              recall_diff = latest_metrics.get('recall', 0) - previous_metrics.get('recall', 0)
              
              if recall_diff < -0.05:
                  print(f'⚠️ Warning: Recall decreased by {abs(recall_diff):.2%}')
                  exit(1)
              else:
                  print('✅ No performance regression detected')
          else:
              print('ℹ️ Not enough versions for comparison')
          "
        continue-on-error: true

  # Job 5: Model Deployment
  deploy:
    name: Deploy Model to Production
    runs-on: ubuntu-latest
    needs: validate
    if: github.ref == 'refs/heads/main' && github.event_name == 'push'
    environment:
      name: production
      url: https://fraud-detection.example.com
    
    steps:
      - name: Checkout code
        uses: actions/checkout@v3
      
      - name: Set up Python
        uses: actions/setup-python@v4
        with:
          python-version: ${{ env.PYTHON_VERSION }}
      
      - name: Install dependencies
        run: |
          pip install -r requirements.txt
          pip install boto3
      
      - name: Configure AWS credentials
        uses: aws-actions/configure-aws-credentials@v2
        with:
          aws-access-key-id: ${{ secrets.AWS_ACCESS_KEY_ID }}
          aws-secret-access-key: ${{ secrets.AWS_SECRET_ACCESS_KEY }}
          aws-region: ${{ env.AWS_REGION }}
      
      - name: Promote model to production
        run: |
          python -c "
          import mlflow
          from mlflow.tracking import MlflowClient
          
          mlflow.set_tracking_uri('${{ env.MLFLOW_TRACKING_URI }}')
          client = MlflowClient()
          
          # Get latest model version
          latest_version = client.get_latest_versions('${{ env.MODEL_NAME }}', stages=['None'])[0]
          
          # Transition to production
          client.transition_model_version_stage(
              name='${{ env.MODEL_NAME }}',
              version=latest_version.version,
              stage='Production',
              archive_existing_versions=True
          )
          
          print(f'✅ Model version {latest_version.version} promoted to Production')
          "
      
      - name: Deploy to SageMaker (optional)
        if: env.DEPLOY_TO_SAGEMAKER == 'true'
        run: |
          python -c "
          import boto3
          import mlflow
          
          # Download model from MLflow
          mlflow.set_tracking_uri('${{ env.MLFLOW_TRACKING_URI }}')
          model_uri = 'models:/${{ env.MODEL_NAME }}/Production'
          
          # Deploy to SageMaker
          # (Implementation depends on your SageMaker setup)
          print('Deploying to SageMaker...')
          "
      
      - name: Update model registry
        run: |
          python -c "
          import json
          from datetime import datetime
          
          registry = {
              'model_name': '${{ env.MODEL_NAME }}',
              'version': 'latest',
              'deployed_at': datetime.now().isoformat(),
              'deployed_by': '${{ github.actor }}',
              'commit_sha': '${{ github.sha }}',
              'environment': 'production'
          }
          
          with open('model_registry.json', 'w') as f:
              json.dump(registry, f, indent=2)
          "
      
      - name: Upload registry to S3
        run: |
          aws s3 cp model_registry.json s3://${{ secrets.S3_BUCKET }}/model_registry/latest.json
      
      - name: Create GitHub release
        uses: actions/create-release@v1
        env:
          GITHUB_TOKEN: ${{ secrets.GITHUB_TOKEN }}
        with:
          tag_name: model-v${{ github.run_number }}
          release_name: Model Release v${{ github.run_number }}
          body: |
            ## Fraud Detection Model Release
            
            **Deployed:** ${{ github.event.head_commit.timestamp }}
            **Commit:** ${{ github.sha }}
            
            ### Performance Metrics
            - Recall: ${{ needs.train.outputs.recall }}
            - Precision: ${{ needs.train.outputs.precision }}
            
            See MLflow for detailed metrics and artifacts.
          draft: false
          prerelease: false
      
      - name: Send deployment notification
        uses: slackapi/slack-github-action@v1.24.0
        with:
          webhook: ${{ secrets.SLACK_WEBHOOK }}
          payload: |
            {
              "text": "✅ Fraud Detection Model Deployed to Production",
              "blocks": [
                {
                  "type": "section",
                  "text": {
                    "type": "mrkdwn",
                    "text": "*Model Deployment Successful*\n\nVersion: v${{ github.run_number }}\nEnvironment: Production\nDeployed by: ${{ github.actor }}"
                  }
                }
              ]
            }
