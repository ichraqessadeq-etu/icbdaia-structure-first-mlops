Handles imbalanced data, concept drift detection, and comprehensive logging.

class ConceptDriftDetector:
    """Detects concept drift using statistical tests."""
    
    def __init__(self, significance_level: float = 0.05):
        self.significance_level = significance_level
        self.reference_distribution = None
        
    def fit(self, X: pd.DataFrame):
        """Store reference distribution."""
        self.reference_distribution = X.copy()
        logger.info("Reference distribution stored for drift detection")
        
    def detect_drift(self, X: pd.DataFrame) -> Tuple[bool, Dict]:
        """
        Detect drift using Kolmogorov-Smirnov test.
        
        Returns:
            Tuple of (drift_detected, drift_metrics)
        """
        if self.reference_distribution is None:
            raise ValueError("Reference distribution not set. Call fit() first.")
        
        drift_detected = False
        drift_metrics = {}
        
        for column in X.columns:
            if column in self.reference_distribution.columns:
                # Perform KS test
                statistic, p_value = stats.ks_2samp(
                    self.reference_distribution[column].dropna(),
                    X[column].dropna()
                )
                
                drift_metrics[column] = {
                    'ks_statistic': float(statistic),
                    'p_value': float(p_value),
                    'drift': p_value < self.significance_level
                }
                
                if p_value < self.significance_level:
                    drift_detected = True
                    logger.warning(f"Drift detected in feature '{column}': p-value={p_value:.4f}")
        
        drift_metrics['overall_drift'] = drift_detected
        return drift_detected, drift_metrics


class FraudDetectionPipeline:
    """Complete fraud detection pipeline with MLflow integration."""
    
    def __init__(
        self,
        experiment_name: str = "fraud_detection",
        mlflow_tracking_uri: str = "sqlite:///mlflow.db",
        random_state: int = 42
    ):
        self.experiment_name = experiment_name
        self.mlflow_tracking_uri = mlflow_tracking_uri
        self.random_state = random_state
        self.model = None
        self.scaler = None
        self.drift_detector = ConceptDriftDetector()
        
        # Set MLflow tracking
        mlflow.set_tracking_uri(self.mlflow_tracking_uri)
        mlflow.set_experiment(self.experiment_name)
        
    def load_data(self, data_path: str) -> pd.DataFrame:
        """
        Load fraud detection data.
        
        Args:
            data_path: Path to CSV file
            
        Returns:
            DataFrame with loaded data
        """
        try:
            logger.info(f"Loading data from {data_path}")
            df = pd.read_csv(data_path)
            
            # Validate data
            if df.empty:
                raise ValueError("Loaded dataset is empty")
            
            if 'Class' not in df.columns:
                raise ValueError("Target column 'Class' not found in dataset")
            
            # Log data statistics
            fraud_rate = df['Class'].mean()
            logger.info(f"Data loaded successfully: {len(df)} samples")
            logger.info(f"Fraud rate: {fraud_rate:.4%}")
            logger.info(f"Features: {df.shape[1] - 1}")
            
            return df
            
        except FileNotFoundError:
            logger.error(f"Data file not found: {data_path}")
            raise
        except Exception as e:
            logger.error(f"Error loading data: {str(e)}")
            raise
    
    def preprocess_data(
        self,
        df: pd.DataFrame,
        test_size: float = 0.2
    ) -> Tuple[pd.DataFrame, pd.DataFrame, pd.Series, pd.Series]:
        """
        Preprocess and split data.
        
        Args:
            df: Input DataFrame
            test_size: Proportion of test set
            
        Returns:
            X_train, X_test, y_train, y_test
        """
        try:
            logger.info("Preprocessing data")
            
            # Separate features and target
            X = df.drop('Class', axis=1)
            y = df['Class']
            
            # Handle missing values
            if X.isnull().any().any():
                logger.warning("Missing values detected, filling with median")
                X = X.fillna(X.median())
            
            # Split data with stratification
            X_train, X_test, y_train, y_test = train_test_split(
                X, y,
                test_size=test_size,
                random_state=self.random_state,
                stratify=y
            )
            
            logger.info(f"Train set: {len(X_train)} samples")
            logger.info(f"Test set: {len(X_test)} samples")
            logger.info(f"Train fraud rate: {y_train.mean():.4%}")
            logger.info(f"Test fraud rate: {y_test.mean():.4%}")
            
            return X_train, X_test, y_train, y_test
            
        except Exception as e:
            logger.error(f"Error in preprocessing: {str(e)}")
            raise
    
    def train_model(
        self,
        X_train: pd.DataFrame,
        y_train: pd.Series,
        n_estimators: int = 100,
        max_depth: int = 10,
        min_samples_split: int = 10,
        class_weight: str = 'balanced',
        use_smote: bool = True
    ) -> ImbPipeline:
        """
        Train Random Forest model with SMOTE for handling imbalanced data.
        
        Args:
            X_train: Training features
            y_train: Training labels
            n_estimators: Number of trees
            max_depth: Maximum tree depth
            min_samples_split: Minimum samples to split
            class_weight: Class weight strategy
            use_smote: Whether to use SMOTE oversampling
            
        Returns:
            Trained pipeline
        """
        try:
            logger.info("Training model")
            
            # Create pipeline steps
            steps = [
                ('scaler', StandardScaler())
            ]
            
            if use_smote:
                # Calculate sampling strategy for SMOTE
                fraud_count = y_train.sum()
                non_fraud_count = len(y_train) - fraud_count
                sampling_ratio = min(0.5, fraud_count / non_fraud_count * 10)
                
                logger.info(f"Applying SMOTE with sampling ratio: {sampling_ratio:.4f}")
                steps.append(('smote', SMOTE(
                    sampling_strategy=sampling_ratio,
                    random_state=self.random_state
                )))
            
            # Add classifier
            steps.append(('classifier', RandomForestClassifier(
                n_estimators=n_estimators,
                max_depth=max_depth,
                min_samples_split=min_samples_split,
                class_weight=class_weight,
                random_state=self.random_state,
                n_jobs=-1,
                verbose=0
            )))
            
            # Create and train pipeline
            pipeline = ImbPipeline(steps)
            pipeline.fit(X_train, y_train)
            
            self.model = pipeline
            logger.info("Model training completed")
            
            return pipeline
            
        except Exception as e:
            logger.error(f"Error in model training: {str(e)}")
            raise
    
    def evaluate_model(
        self,
        X_test: pd.DataFrame,
        y_test: pd.Series,
        threshold: float = 0.5
    ) -> Dict:
        """
        Evaluate model performance with fraud-specific metrics.
        
        Args:
            X_test: Test features
            y_test: Test labels
            threshold: Classification threshold
            
        Returns:
            Dictionary of metrics
        """
        try:
            logger.info("Evaluating model")
            
            if self.model is None:
                raise ValueError("Model not trained. Call train_model() first.")
            
            # Get predictions
            y_pred_proba = self.model.predict_proba(X_test)[:, 1]
            y_pred = (y_pred_proba >= threshold).astype(int)
            
            # Calculate metrics
            metrics = {
                'accuracy': float(np.mean(y_pred == y_test)),
                'precision': float(precision_score(y_test, y_pred, zero_division=0)),
                'recall': float(recall_score(y_test, y_pred, zero_division=0)),
                'f1_score': float(f1_score(y_test, y_pred, zero_division=0)),
                'roc_auc': float(roc_auc_score(y_test, y_pred_proba)),
                'average_precision': float(average_precision_score(y_test, y_pred_proba)),
                'threshold': threshold
            }
            
            # Confusion matrix
            cm = confusion_matrix(y_test, y_pred)
            metrics['confusion_matrix'] = {
                'tn': int(cm[0, 0]),
                'fp': int(cm[0, 1]),
                'fn': int(cm[1, 0]),
                'tp': int(cm[1, 1])
            }
            
            # Calculate cost-based metrics (assuming fraud costs more)
            fp_cost = 1  # Cost of false positive
            fn_cost = 10  # Cost of missing fraud (false negative)
            total_cost = cm[0, 1] * fp_cost + cm[1, 0] * fn_cost
            metrics['total_cost'] = int(total_cost)
            
            # Log metrics
            logger.info(f"ROC-AUC: {metrics['roc_auc']:.4f}")
            logger.info(f"Average Precision: {metrics['average_precision']:.4f}")
            logger.info(f"Precision: {metrics['precision']:.4f}")
            logger.info(f"Recall: {metrics['recall']:.4f}")
            logger.info(f"F1-Score: {metrics['f1_score']:.4f}")
            logger.info(f"Total Cost: {metrics['total_cost']}")
            
            return metrics
            
        except Exception as e:
            logger.error(f"Error in model evaluation: {str(e)}")
            raise
    
    def cross_validate(
        self,
        X: pd.DataFrame,
        y: pd.Series,
        n_splits: int = 5
    ) -> Dict:
        """
        Perform stratified k-fold cross-validation.
        
        Args:
            X: Features
            y: Labels
            n_splits: Number of CV folds
            
        Returns:
            Dictionary of CV metrics
        """
        try:
            logger.info(f"Performing {n_splits}-fold cross-validation")
            
            skf = StratifiedKFold(n_splits=n_splits, shuffle=True, random_state=self.random_state)
            
            cv_scores = {
                'roc_auc': [],
                'precision': [],
                'recall': [],
                'f1': []
            }
            
            for fold, (train_idx, val_idx) in enumerate(skf.split(X, y), 1):
                X_train_cv, X_val_cv = X.iloc[train_idx], X.iloc[val_idx]
                y_train_cv, y_val_cv = y.iloc[train_idx], y.iloc[val_idx]
                
                # Train model
                pipeline = self.train_model(X_train_cv, y_train_cv)
                
                # Evaluate
                y_pred_proba = pipeline.predict_proba(X_val_cv)[:, 1]
                y_pred = pipeline.predict(X_val_cv)
                
                cv_scores['roc_auc'].append(roc_auc_score(y_val_cv, y_pred_proba))
                cv_scores['precision'].append(precision_score(y_val_cv, y_pred, zero_division=0))
                cv_scores['recall'].append(recall_score(y_val_cv, y_pred, zero_division=0))
                cv_scores['f1'].append(f1_score(y_val_cv, y_pred, zero_division=0))
                
                logger.info(f"Fold {fold} - ROC-AUC: {cv_scores['roc_auc'][-1]:.4f}")
            
            # Calculate mean and std
            cv_results = {}
            for metric, scores in cv_scores.items():
                cv_results[f'{metric}_mean'] = float(np.mean(scores))
                cv_results[f'{metric}_std'] = float(np.std(scores))
            
            logger.info(f"CV ROC-AUC: {cv_results['roc_auc_mean']:.4f} ± {cv_results['roc_auc_std']:.4f}")
            
            return cv_results
            
        except Exception as e:
            logger.error(f"Error in cross-validation: {str(e)}")
            raise
    
    def log_to_mlflow(
        self,
        params: Dict,
        metrics: Dict,
        model: ImbPipeline,
        tags: Optional[Dict] = None
    ) -> str:
        """
        Log experiment to MLflow.
        
        Args:
            params: Model parameters
            metrics: Evaluation metrics
            model: Trained model
            tags: Optional tags
            
        Returns:
            Run ID
        """
        try:
            logger.info("Logging to MLflow")
            
            with mlflow.start_run() as run:
                # Log parameters
                mlflow.log_params(params)
                
                # Log metrics
                mlflow.log_metrics(metrics)
                
                # Log tags
                if tags:
                    mlflow.set_tags(tags)
                
                # Log model
                mlflow.sklearn.log_model(
                    model,
                    "model",
                    registered_model_name="fraud_detection_rf"
                )
                
                # Log feature importance
                if hasattr(model.named_steps['classifier'], 'feature_importances_'):
                    importance = model.named_steps['classifier'].feature_importances_
                    importance_dict = {f'feature_{i}': float(imp) for i, imp in enumerate(importance)}
                    mlflow.log_dict(importance_dict, "feature_importance.json")
                
                run_id = run.info.run_id
                logger.info(f"MLflow run ID: {run_id}")
                
                return run_id
                
        except Exception as e:
            logger.error(f"Error logging to MLflow: {str(e)}")
            raise
    
    def check_drift_and_retrain(
        self,
        reference_data_path: str,
        new_data_path: str,
        auto_retrain: bool = True
    ) -> Tuple[bool, Optional[str]]:
        """
        Check for concept drift and trigger retraining if detected.
        
        Args:
            reference_data_path: Path to reference dataset
            new_data_path: Path to new dataset
            auto_retrain: Whether to automatically retrain on drift
            
        Returns:
            Tuple of (drift_detected, run_id)
        """
        try:
            logger.info("Checking for concept drift")
            
            # Load datasets
            ref_df = self.load_data(reference_data_path)
            new_df = self.load_data(new_data_path)
            
            # Prepare features
            X_ref = ref_df.drop('Class', axis=1)
            X_new = new_df.drop('Class', axis=1)
            
            # Fit drift detector on reference data
            self.drift_detector.fit(X_ref)
            
            # Detect drift
            drift_detected, drift_metrics = self.drift_detector.detect_drift(X_new)
            
            # Log drift metrics
            logger.info(f"Drift detected: {drift_detected}")
            with open('drift_report.json', 'w') as f:
                json.dump(drift_metrics, f, indent=2)
            
            run_id = None
            
            if drift_detected and auto_retrain:
                logger.info("Triggering automatic retraining due to drift")
                run_id = self.run_pipeline(new_data_path, retrain_reason="concept_drift")
            
            return drift_detected, run_id
            
        except Exception as e:
            logger.error(f"Error in drift detection: {str(e)}")
            raise
    
    def run_pipeline(
        self,
        data_path: str,
        retrain_reason: str = "scheduled",
        perform_cv: bool = False
    ) -> str:
        """
        Run complete pipeline from data loading to MLflow logging.
        
        Args:
            data_path: Path to data file
            retrain_reason: Reason for training (scheduled, concept_drift, etc.)
            perform_cv: Whether to perform cross-validation
            
        Returns:
            MLflow run ID
        """
        try:
            logger.info("="*80)
            logger.info("Starting Fraud Detection Pipeline")
            logger.info("="*80)
            
            start_time = datetime.now()
            
            # Load data
            df = self.load_data(data_path)
            
            # Preprocess
            X_train, X_test, y_train, y_test = self.preprocess_data(df)
            
            # Model parameters
            params = {
                'n_estimators': 100,
                'max_depth': 10,
                'min_samples_split': 10,
                'class_weight': 'balanced',
                'use_smote': True,
                'random_state': self.random_state,
                'retrain_reason': retrain_reason
            }
            
            # Cross-validation (optional)
            cv_results = {}
            if perform_cv:
                cv_results = self.cross_validate(X_train, y_train)
            
            # Train model
            model = self.train_model(X_train, y_train, **{k: v for k, v in params.items() 
                                                          if k not in ['retrain_reason']})
            
            # Evaluate model
            metrics = self.evaluate_model(X_test, y_test)
            
            # Combine metrics
            all_metrics = {**metrics, **cv_results}
            
            # Add timing
            end_time = datetime.now()
            duration = (end_time - start_time).total_seconds()
            all_metrics['training_duration_seconds'] = duration
            
            # Log to MLflow
            tags = {
                'retrain_reason': retrain_reason,
                'timestamp': start_time.isoformat(),
                'data_path': data_path
            }
            
            run_id = self.log_to_mlflow(params, all_metrics, model, tags)
            
            logger.info("="*80)
            logger.info(f"Pipeline completed successfully in {duration:.2f} seconds")
            logger.info(f"MLflow Run ID: {run_id}")
            logger.info("="*80)
            
            return run_id
            
        except Exception as e:
            logger.error(f"Pipeline failed: {str(e)}")
            raise


def main():
    """Main execution function."""
    import argparse
    
    parser = argparse.ArgumentParser(description='Fraud Detection Pipeline')
    parser.add_argument('--data-path', type=str, required=True, help='Path to data file')
    parser.add_argument('--check-drift', action='store_true', help='Check for concept drift')
    parser.add_argument('--reference-data', type=str, help='Path to reference data for drift detection')
    parser.add_argument('--cv', action='store_true', help='Perform cross-validation')
    parser.add_argument('--experiment-name', type=str, default='fraud_detection', help='MLflow experiment name')
    
    args = parser.parse_args()
    
    # Initialize pipeline
    pipeline = FraudDetectionPipeline(experiment_name=args.experiment_name)
    
    try:
        if args.check_drift:
            if not args.reference_data:
                raise ValueError("--reference-data required when using --check-drift")
            
            drift_detected, run_id = pipeline.check_drift_and_retrain(
                args.reference_data,
                args.data_path,
                auto_retrain=True
            )
            
            if drift_detected:
                logger.info("Drift detected and model retrained")
                sys.exit(1)  # Exit with code 1 to signal drift
            else:
                logger.info("No drift detected")
                sys.exit(0)
        else:
            # Normal training
            run_id = pipeline.run_pipeline(args.data_path, perform_cv=args.cv)
            logger.info(f"Training completed. Run ID: {run_id}")
            sys.exit(0)
            
    except Exception as e:
        logger.error(f"Pipeline execution failed: {str(e)}")
        sys.exit(1)


if __name__ == "__main__":
    main()

class TestConceptDrift:
    """Test suite for concept drift detection."""
    
    def test_drift_detector_initialization(self):
        """Test drift detector initialization."""
        detector = ConceptDriftDetector(significance_level=0.05)
        assert detector.significance_level == 0.05
        assert detector.reference_distribution is None
    
    def test_fit_drift_detector(self, sample_fraud_data):
        """Test fitting drift detector."""
        detector = ConceptDriftDetector()
        X = sample_fraud_data.drop('Class', axis=1)
        
        detector.fit(X)
        
        assert detector.reference_distribution is not None
        assert len(detector.reference_distribution) == len(X)
    
    def test_no_drift_same_distribution(self, sample_fraud_data):
        """Test no drift detected for same distribution."""
        detector = ConceptDriftDetector()
        X = sample_fraud_data.drop('Class', axis=1)
        
        detector.fit(X)
        drift_detected, drift_metrics = detector.detect_drift(X)
        
        assert not drift_detected
        assert 'overall_drift' in drift_metrics
    
    def test_drift_detected_different_distribution(self, sample_fraud_data):
        """Test drift detected for different distribution."""
        detector = ConceptDriftDetector()
        X_ref = sample_fraud_data.drop('Class', axis=1)
        
        # Create shifted distribution
        X_new = X_ref.copy()
        X_new.iloc[:, 0] = X_new.iloc[:, 0] + 5  # Shift first feature
        
        detector.fit(X_ref)
        drift_detected, drift_metrics = detector.detect_drift(X_new)
        
        # Should detect drift in at least the shifted feature
        assert any(metrics['drift'] for col, metrics in drift_metrics.items() 
                  if col != 'overall_drift')
    
    def test_drift_without_fit(self, sample_fraud_data):
        """Test drift detection fails without fitting."""
        detector = ConceptDriftDetector()
        X = sample_fraud_data.drop('Class', axis=1)
        
        with pytest.raises(ValueError, match="Reference distribution not set"):
            detector.detect_drift(X)
    
    def test_drift_metrics_structure(self, sample_fraud_data):
        """Test drift metrics have correct structure."""
        detector = ConceptDriftDetector()
        X = sample_fraud_data.drop('Class', axis=1)
        
        detector.fit(X)
        drift_detected, drift_metrics = detector.detect_drift(X)
        
        # Check structure
        assert 'overall_drift' in drift_metrics
        for col in X.columns:
            assert col in drift_metrics
            assert 'ks_statistic' in drift_metrics[col]
            assert 'p_value' in drift_metrics[col]
            assert 'drift' in drift_metrics[col]

            --cv \
            --experiment-name fraud_detection_prod
      
      - name: Upload MLflow artifacts
        uses: actions/upload-artifact@v3
        with:
          name: mlflow-artifacts
          path: |
            mlflow.db
            mlruns/
      
      - name: Upload model artifacts
        uses: actions/upload-artifact@v3
        with:
          name: model-artifacts
          path: |
            fraud_detection_pipeline.log

  # Job 6: Model Evaluation and Validation
  model-evaluation:
    name: Model Evaluation
    runs-on: ubuntu-latest
    needs: model-training
    
    outputs:
      model_approved: ${{ steps.validate_metrics.outputs.approved }}
    
    steps:
      - name: Checkout code
        uses: actions/checkout@v3
      
      - name: Set up Python
        uses: actions/setup-python@v4
        with:
          python-version: ${{ env.PYTHON_VERSION }}
      
      - name: Install dependencies
        run: |
          python -m pip install --upgrade pip
          pip install -r requirements.txt
      
      - name: Download MLflow artifacts
        uses: actions/download-artifact@v3
        with:
          name: mlflow-artifacts
      
      - name: Validate model metrics
        id: validate_metrics
        run: |
          python -c "
          import mlflow
          import sys
          
          mlflow.set_tracking_uri('${{ env.MLFLOW_TRACKING_URI }}')
          
          # Get latest run
          experiment = mlflow.get_experiment_by_name('fraud_detection_prod')
          runs = mlflow.search_runs(experiment_ids=[experiment.experiment_id], order_by=['start_time DESC'], max_results=1)
          
          if runs.empty:
              print('No runs found')
              sys.exit(1)
          
          latest_run = runs.iloc[0]
          
          # Define thresholds
          min_roc_auc = 0.85
          min_recall = 0.70
          min_precision = 0.05
          
          roc_auc = latest_run['metrics.roc_auc']
          recall = latest_run['metrics.recall']
          precision = latest_run['metrics.precision']
          
          print(f'ROC-AUC: {roc_auc:.4f} (min: {min_roc_auc})')
          print(f'Recall: {recall:.4f} (min: {min_recall})')
          print(f'Precision: {precision:.4f} (min: {min_precision})')
          
          approved = (roc_auc >= min_roc_auc and recall >= min_recall and precision >= min_precision)
          
          if approved:
              print('✅ Model approved for deployment')
              print('approved=true' >> '$GITHUB_OUTPUT')
          else:
              print('❌ Model does not meet quality thresholds')
              print('approved=false' >> '$GITHUB_OUTPUT')
              sys.exit(1)
          "
      
      - name: Generate model report
        if: always()
        run: |
          python -c "
          import mlflow
          import json
          
          mlflow.set_tracking_uri('${{ env.MLFLOW_TRACKING_URI }}')
          
          experiment = mlflow.get_experiment_by_name('fraud_detection_prod')
          runs = mlflow.search_runs(experiment_ids=[experiment.experiment_id], order_by=['start_time DESC'], max_results=1)
          
          if not runs.empty:
              latest_run = runs.iloc[0]
              
              report = {
                  'run_id': latest_run['run_id'],
                  'metrics': {
                      'roc_auc': float(latest_run['metrics.roc_auc']),
                      'recall': float(latest_run['metrics.recall']),
                      'precision': float(latest_run['metrics.precision']),
                      'f1_score': float(latest_run['metrics.f1_score'])
                  },
                  'timestamp': str(latest_run['start_time'])
              }
              
              with open('model_report.json', 'w') as f:
                  json.dump(report, f, indent=2)
          "
      
      - name: Upload model report
        uses: actions/upload-artifact@v3
        if: always()
        with:
          name: model-report
          path: model_report.json

  deploy-staging:

              # Transition model to staging
