# fraud_detection_pipeline.py
"""
Production-ready fraud detection pipeline with MLflow integration.
Handles imbalanced datasets and includes comprehensive error handling.
"""

import os
import sys
import logging
import pickle
from typing import Tuple, Dict, Any
from datetime import datetime
import warnings

import numpy as np
import pandas as pd
from sklearn.ensemble import RandomForestClassifier
from sklearn.model_selection import train_test_split, StratifiedKFold
from sklearn.preprocessing import StandardScaler
from sklearn.metrics import (
    precision_score, recall_score, f1_score, roc_auc_score,
    average_precision_score, confusion_matrix, classification_report
)
from imblearn.over_sampling import SMOTE
from imblearn.pipeline import Pipeline as ImbPipeline
import mlflow
import mlflow.sklearn
from scipy import stats

warnings.filterwarnings('ignore')

# Configure logging
logging.basicConfig(
    level=logging.INFO,
    format='%(asctime)s - %(name)s - %(levelname)s - %(message)s',
    handlers=[
        logging.FileHandler('fraud_detection_pipeline.log'),
        logging.StreamHandler(sys.stdout)
    ]
)
logger = logging.getLogger(__name__)


class FraudDetectionPipeline:
    """End-to-end pipeline for fraud detection model training and evaluation."""
    
    def __init__(
        self,
        mlflow_tracking_uri: str = "http://localhost:5000",
        experiment_name: str = "fraud_detection",
        random_state: int = 42
    ):
        """
        Initialize the fraud detection pipeline.
        
        Args:
            mlflow_tracking_uri: MLflow tracking server URI
            experiment_name: Name of the MLflow experiment
            random_state: Random seed for reproducibility
        """
        self.mlflow_tracking_uri = mlflow_tracking_uri
        self.experiment_name = experiment_name
        self.random_state = random_state
        self.model = None
        self.scaler = None
        self.pipeline = None
        
        # Set up MLflow
        mlflow.set_tracking_uri(self.mlflow_tracking_uri)
        mlflow.set_experiment(self.experiment_name)
        
        logger.info(f"Initialized FraudDetectionPipeline with experiment: {experiment_name}")
    
    def load_data(self, data_path: str) -> pd.DataFrame:
        """
        Load fraud detection dataset with error handling.
        
        Args:
            data_path: Path to the CSV file
            
        Returns:
            Loaded DataFrame
            
        Raises:
            FileNotFoundError: If data file doesn't exist
            ValueError: If data is invalid
        """
        try:
            if not os.path.exists(data_path):
                raise FileNotFoundError(f"Data file not found: {data_path}")
            
            logger.info(f"Loading data from {data_path}")
            df = pd.read_csv(data_path)
            
            # Validate data
            if df.empty:
                raise ValueError("Loaded DataFrame is empty")
            
            if 'Class' not in df.columns:
                raise ValueError("Target column 'Class' not found in dataset")
            
            logger.info(f"Data loaded successfully. Shape: {df.shape}")
            logger.info(f"Fraud rate: {df['Class'].mean():.4f}")
            
            return df
            
        except Exception as e:
            logger.error(f"Error loading data: {str(e)}")
            raise
    
    def preprocess_data(
        self,
        df: pd.DataFrame,
        test_size: float = 0.2
    ) -> Tuple[np.ndarray, np.ndarray, np.ndarray, np.ndarray]:
        """
        Preprocess data: split and validate.
        
        Args:
            df: Input DataFrame
            test_size: Proportion of test set
            
        Returns:
            Tuple of (X_train, X_test, y_train, y_test)
        """
        try:
            logger.info("Preprocessing data")
            
            # Separate features and target
            X = df.drop('Class', axis=1).values
            y = df['Class'].values
            
            # Check for missing values
            if np.isnan(X).any():
                logger.warning("Missing values detected, filling with median")
                X = pd.DataFrame(X).fillna(pd.DataFrame(X).median()).values
            
            # Stratified split to maintain class distribution
            X_train, X_test, y_train, y_test = train_test_split(
                X, y,
                test_size=test_size,
                random_state=self.random_state,
                stratify=y
            )
            
            logger.info(f"Train set size: {X_train.shape[0]}, Test set size: {X_test.shape[0]}")
            logger.info(f"Train fraud rate: {y_train.mean():.4f}, Test fraud rate: {y_test.mean():.4f}")
            
            return X_train, X_test, y_train, y_test
            
        except Exception as e:
            logger.error(f"Error in preprocessing: {str(e)}")
            raise
    
    def detect_concept_drift(
        self,
        reference_data: np.ndarray,
        current_data: np.ndarray,
        threshold: float = 0.05
    ) -> Tuple[bool, Dict[str, float]]:
        """
        Detect concept drift using Kolmogorov-Smirnov test.
        
        Args:
            reference_data: Historical reference data
            current_data: Current production data
            threshold: P-value threshold for drift detection
            
        Returns:
            Tuple of (drift_detected, drift_metrics)
        """
        try:
            logger.info("Detecting concept drift")
            drift_metrics = {}
            drift_detected = False
            
            # Perform KS test for each feature
            n_features = reference_data.shape[1]
            p_values = []
            
            for i in range(n_features):
                statistic, p_value = stats.ks_2samp(
                    reference_data[:, i],
                    current_data[:, i]
                )
                p_values.append(p_value)
                
                if p_value < threshold:
                    drift_detected = True
                    logger.warning(f"Drift detected in feature {i}: p-value={p_value:.4f}")
            
            drift_metrics['min_p_value'] = float(np.min(p_values))
            drift_metrics['mean_p_value'] = float(np.mean(p_values))
            drift_metrics['drift_detected'] = drift_detected
            drift_metrics['n_drifted_features'] = int(np.sum(np.array(p_values) < threshold))
            
            logger.info(f"Drift detection complete. Drift detected: {drift_detected}")
            
            return drift_detected, drift_metrics
            
        except Exception as e:
            logger.error(f"Error in drift detection: {str(e)}")
            raise
    
    def build_model(self, hyperparameters: Dict[str, Any] = None) -> ImbPipeline:
        """
        Build fraud detection model with SMOTE and RandomForest.
        
        Args:
            hyperparameters: Model hyperparameters
            
        Returns:
            Scikit-learn pipeline
        """
        try:
            logger.info("Building model pipeline")
            
            if hyperparameters is None:
                hyperparameters = {
                    'n_estimators': 100,
                    'max_depth': 10,
                    'min_samples_split': 10,
                    'min_samples_leaf': 4,
                    'class_weight': 'balanced',
                    'random_state': self.random_state,
                    'n_jobs': -1
                }
            
            # Build pipeline with SMOTE and RandomForest
            pipeline = ImbPipeline([
                ('scaler', StandardScaler()),
                ('smote', SMOTE(random_state=self.random_state, k_neighbors=5)),
                ('classifier', RandomForestClassifier(**hyperparameters))
            ])
            
            logger.info("Model pipeline built successfully")
            return pipeline
            
        except Exception as e:
            logger.error(f"Error building model: {str(e)}")
            raise
    
    def train_model(
        self,
        X_train: np.ndarray,
        y_train: np.ndarray,
        hyperparameters: Dict[str, Any] = None
    ) -> ImbPipeline:
        """
        Train the fraud detection model.
        
        Args:
            X_train: Training features
            y_train: Training labels
            hyperparameters: Model hyperparameters
            
        Returns:
            Trained pipeline
        """
        try:
            logger.info("Starting model training")
            
            self.pipeline = self.build_model(hyperparameters)
            self.pipeline.fit(X_train, y_train)
            
            logger.info("Model training completed")
            return self.pipeline
            
        except Exception as e:
            logger.error(f"Error training model: {str(e)}")
            raise
    
    def evaluate_model(
        self,
        X_test: np.ndarray,
        y_test: np.ndarray
    ) -> Dict[str, float]:
        """
        Evaluate model performance with comprehensive metrics.
        
        Args:
            X_test: Test features
            y_test: Test labels
            
        Returns:
            Dictionary of evaluation metrics
        """
        try:
            logger.info("Evaluating model")
            
            if self.pipeline is None:
                raise ValueError("Model not trained. Call train_model first.")
            
            # Predictions
            y_pred = self.pipeline.predict(X_test)
            y_pred_proba = self.pipeline.predict_proba(X_test)[:, 1]
            
            # Calculate metrics
            metrics = {
                'precision': float(precision_score(y_test, y_pred, zero_division=0)),
                'recall': float(recall_score(y_test, y_pred, zero_division=0)),
                'f1_score': float(f1_score(y_test, y_pred, zero_division=0)),
                'roc_auc': float(roc_auc_score(y_test, y_pred_proba)),
                'pr_auc': float(average_precision_score(y_test, y_pred_proba)),
            }
            
            # Confusion matrix
            cm = confusion_matrix(y_test, y_pred)
            metrics['true_negatives'] = int(cm[0, 0])
            metrics['false_positives'] = int(cm[0, 1])
            metrics['false_negatives'] = int(cm[1, 0])
            metrics['true_positives'] = int(cm[1, 1])
            
            # Log metrics
            logger.info("Model Evaluation Metrics:")
            for metric, value in metrics.items():
                logger.info(f"  {metric}: {value}")
            
            # Classification report
            logger.info("\nClassification Report:")
            logger.info("\n" + classification_report(y_test, y_pred))
            
            return metrics
            
        except Exception as e:
            logger.error(f"Error evaluating model: {str(e)}")
            raise
    
    def log_to_mlflow(
        self,
        metrics: Dict[str, float],
        hyperparameters: Dict[str, Any],
        drift_metrics: Dict[str, float] = None,
        model_path: str = "model"
    ) -> str:
        """
        Log experiment to MLflow.
        
        Args:
            metrics: Evaluation metrics
            hyperparameters: Model hyperparameters
            drift_metrics: Drift detection metrics
            model_path: Path to save model artifacts
            
        Returns:
            MLflow run ID
        """
        try:
            logger.info("Logging to MLflow")
            
            with mlflow.start_run() as run:
                # Log parameters
                mlflow.log_params(hyperparameters)
                
                # Log metrics
                mlflow.log_metrics(metrics)
                
                # Log drift metrics if available
                if drift_metrics:
                    mlflow.log_metrics({f"drift_{k}": v for k, v in drift_metrics.items() 
                                       if isinstance(v, (int, float))})
                
                # Log model
                mlflow.sklearn.log_model(
                    self.pipeline,
                    model_path,
                    registered_model_name="fraud_detection_model"
                )
                
                # Log additional info
                mlflow.set_tag("model_type", "RandomForestClassifier")
                mlflow.set_tag("data_handling", "SMOTE")
                mlflow.set_tag("timestamp", datetime.now().isoformat())
                
                run_id = run.info.run_id
                logger.info(f"MLflow run ID: {run_id}")
                
                return run_id
                
        except Exception as e:
            logger.error(f"Error logging to MLflow: {str(e)}")
            raise
    
    def save_model(self, filepath: str = "fraud_detection_model.pkl"):
        """
        Save trained model to disk.
        
        Args:
            filepath: Path to save the model
        """
        try:
            if self.pipeline is None:
                raise ValueError("No trained model to save")
            
            logger.info(f"Saving model to {filepath}")
            
            with open(filepath, 'wb') as f:
                pickle.dump(self.pipeline, f)
            
            logger.info("Model saved successfully")
            
        except Exception as e:
            logger.error(f"Error saving model: {str(e)}")
            raise
    
    def load_model(self, filepath: str = "fraud_detection_model.pkl"):
        """
        Load trained model from disk.
        
        Args:
            filepath: Path to the saved model
        """
        try:
            if not os.path.exists(filepath):
                raise FileNotFoundError(f"Model file not found: {filepath}")
            
            logger.info(f"Loading model from {filepath}")
            
            with open(filepath, 'rb') as f:
                self.pipeline = pickle.load(f)
            
            logger.info("Model loaded successfully")
            
        except Exception as e:
            logger.error(f"Error loading model: {str(e)}")
            raise
    
    def run_pipeline(
        self,
        data_path: str,
        hyperparameters: Dict[str, Any] = None,
        reference_data_path: str = None,
        drift_threshold: float = 0.05,
        force_retrain: bool = False
    ) -> Dict[str, Any]:
        """
        Execute the complete pipeline.
        
        Args:
            data_path: Path to training data
            hyperparameters: Model hyperparameters
            reference_data_path: Path to reference data for drift detection
            drift_threshold: Threshold for drift detection
            force_retrain: Force model retraining
            
        Returns:
            Dictionary with pipeline results
        """
        try:
            logger.info("=" * 80)
            logger.info("Starting Fraud Detection Pipeline")
            logger.info("=" * 80)
            
            results = {
                'status': 'success',
                'timestamp': datetime.now().isoformat(),
                'retrained': False,
                'drift_detected': False
            }
            
            # Load data
            df = self.load_data(data_path)
            X_train, X_test, y_train, y_test = self.preprocess_data(df)
            
            # Drift detection
            drift_metrics = None
            if reference_data_path and os.path.exists(reference_data_path):
                logger.info("Checking for concept drift")
                reference_df = self.load_data(reference_data_path)
                reference_X = reference_df.drop('Class', axis=1).values
                
                drift_detected, drift_metrics = self.detect_concept_drift(
                    reference_X, X_train, drift_threshold
                )
                results['drift_detected'] = drift_detected
                results['drift_metrics'] = drift_metrics
                
                if drift_detected:
                    logger.warning("Concept drift detected! Retraining model...")
                    force_retrain = True
            
            # Train model if needed
            if force_retrain or not os.path.exists("fraud_detection_model.pkl"):
                logger.info("Training new model")
                self.train_model(X_train, y_train, hyperparameters)
                self.save_model()
                results['retrained'] = True
            else:
                logger.info("Loading existing model")
                self.load_model()
            
            # Evaluate model
            metrics = self.evaluate_model(X_test, y_test)
            results['metrics'] = metrics
            
            # Log to MLflow
            if hyperparameters is None:
                hyperparameters = {
                    'n_estimators': 100,
                    'max_depth': 10,
                    'min_samples_split': 10,
                    'min_samples_leaf': 4,
                }
            
            run_id = self.log_to_mlflow(metrics, hyperparameters, drift_metrics)
            results['mlflow_run_id'] = run_id
            
            logger.info("=" * 80)
            logger.info("Pipeline completed successfully")
            logger.info("=" * 80)
            
            return results
            
        except Exception as e:
            logger.error(f"Pipeline failed: {str(e)}")
            results['status'] = 'failed'
            results['error'] = str(e)
            raise


def main():
    """Main execution function."""
    import argparse
    
    parser = argparse.ArgumentParser(description='Fraud Detection Pipeline')
    parser.add_argument('--data-path', type=str, required=True, help='Path to training data')
    parser.add_argument('--reference-data', type=str, help='Path to reference data for drift detection')
    parser.add_argument('--mlflow-uri', type=str, default='http://localhost:5000', help='MLflow tracking URI')
    parser.add_argument('--force-retrain', action='store_true', help='Force model retraining')
    parser.add_argument('--drift-threshold', type=float, default=0.05, help='Drift detection threshold')
    
    args = parser.parse_args()
    
    # Initialize pipeline
    pipeline = FraudDetectionPipeline(
        mlflow_tracking_uri=args.mlflow_uri,
        experiment_name="fraud_detection"
    )
    
    # Run pipeline
    results = pipeline.run_pipeline(
        data_path=args.data_path,
        reference_data_path=args.reference_data,
        drift_threshold=args.drift_threshold,
        force_retrain=args.force_retrain
    )
    
    print("\n" + "=" * 80)
    print("PIPELINE RESULTS")
    print("=" * 80)
    print(f"Status: {results['status']}")
    print(f"Model Retrained: {results['retrained']}")
    print(f"Drift Detected: {results['drift_detected']}")
    print(f"\nMetrics:")
    for metric, value in results['metrics'].items():
        print(f"  {metric}: {value}")
    print("=" * 80)


if __name__ == "__main__":
    main()
