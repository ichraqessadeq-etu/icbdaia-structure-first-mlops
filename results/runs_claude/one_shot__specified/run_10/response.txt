```python
# ml_pipeline/fraud_detection_pipeline.py
"""
Production-ready fraud detection ML pipeline with MLflow tracking.
Handles imbalanced data, concept drift detection, and comprehensive logging.
"""

import logging
import sys
from pathlib import Path
from typing import Tuple, Dict, Any
import warnings

import numpy as np
import pandas as pd
from sklearn.ensemble import RandomForestClassifier
from sklearn.model_selection import train_test_split, StratifiedKFold
from sklearn.preprocessing import StandardScaler
from sklearn.metrics import (
    precision_recall_curve,
    roc_auc_score,
    average_precision_score,
    confusion_matrix,
    classification_report,
    f1_score,
    precision_score,
    recall_score
)
from imblearn.over_sampling import SMOTE
from imblearn.pipeline import Pipeline as ImbPipeline
import mlflow
import mlflow.sklearn
from scipy import stats
import joblib

warnings.filterwarnings('ignore')

# Configure logging
logging.basicConfig(
    level=logging.INFO,
    format='%(asctime)s - %(name)s - %(levelname)s - %(message)s',
    handlers=[
        logging.FileHandler('pipeline.log'),
        logging.StreamHandler(sys.stdout)
    ]
)
logger = logging.getLogger(__name__)


class FraudDetectionPipeline:
    """End-to-end fraud detection pipeline with MLflow tracking."""
    
    def __init__(
        self,
        experiment_name: str = "fraud_detection",
        model_name: str = "fraud_detector_rf",
        drift_threshold: float = 0.05,
        random_state: int = 42
    ):
        """
        Initialize the fraud detection pipeline.
        
        Args:
            experiment_name: MLflow experiment name
            model_name: Registered model name in MLflow
            drift_threshold: P-value threshold for drift detection
            random_state: Random seed for reproducibility
        """
        self.experiment_name = experiment_name
        self.model_name = model_name
        self.drift_threshold = drift_threshold
        self.random_state = random_state
        self.model = None
        self.scaler = None
        self.feature_names = None
        
        # Set MLflow experiment
        mlflow.set_experiment(experiment_name)
        logger.info(f"Initialized pipeline with experiment: {experiment_name}")
    
    def load_data(self, data_path: str) -> pd.DataFrame:
        """
        Load and validate fraud detection data.
        
        Args:
            data_path: Path to CSV data file
            
        Returns:
            DataFrame with loaded data
            
        Raises:
            FileNotFoundError: If data file doesn't exist
            ValueError: If data validation fails
        """
        try:
            logger.info(f"Loading data from {data_path}")
            
            if not Path(data_path).exists():
                raise FileNotFoundError(f"Data file not found: {data_path}")
            
            df = pd.read_csv(data_path)
            
            # Validate data
            if df.empty:
                raise ValueError("Loaded dataframe is empty")
            
            if 'Class' not in df.columns:
                raise ValueError("Target column 'Class' not found in data")
            
            # Check for missing values
            if df.isnull().sum().sum() > 0:
                logger.warning("Missing values detected, dropping rows")
                df = df.dropna()
            
            fraud_rate = df['Class'].mean()
            logger.info(f"Data loaded: {len(df)} records, fraud rate: {fraud_rate:.4f}")
            
            return df
            
        except Exception as e:
            logger.error(f"Error loading data: {str(e)}")
            raise
    
    def prepare_data(
        self,
        df: pd.DataFrame,
        test_size: float = 0.2
    ) -> Tuple[np.ndarray, np.ndarray, np.ndarray, np.ndarray]:
        """
        Prepare data for training: split and validate.
        
        Args:
            df: Input dataframe
            test_size: Proportion of data for testing
            
        Returns:
            Tuple of (X_train, X_test, y_train, y_test)
        """
        try:
            logger.info("Preparing data for training")
            
            # Separate features and target
            X = df.drop('Class', axis=1)
            y = df['Class']
            
            self.feature_names = X.columns.tolist()
            
            # Stratified split to maintain class distribution
            X_train, X_test, y_train, y_test = train_test_split(
                X, y,
                test_size=test_size,
                random_state=self.random_state,
                stratify=y
            )
            
            logger.info(f"Train set: {len(X_train)} samples, "
                       f"fraud rate: {y_train.mean():.4f}")
            logger.info(f"Test set: {len(X_test)} samples, "
                       f"fraud rate: {y_test.mean():.4f}")
            
            return X_train, X_test, y_train, y_test
            
        except Exception as e:
            logger.error(f"Error preparing data: {str(e)}")
            raise
    
    def build_model(
        self,
        n_estimators: int = 100,
        max_depth: int = 10,
        min_samples_split: int = 10,
        min_samples_leaf: int = 5,
        class_weight: str = 'balanced'
    ) -> ImbPipeline:
        """
        Build sklearn pipeline with SMOTE and RandomForest.
        
        Args:
            n_estimators: Number of trees in the forest
            max_depth: Maximum depth of trees
            min_samples_split: Minimum samples required to split
            min_samples_leaf: Minimum samples required at leaf node
            class_weight: Class weight strategy
            
        Returns:
            Pipeline with scaler, SMOTE, and classifier
        """
        try:
            logger.info("Building model pipeline")
            
            pipeline = ImbPipeline([
                ('scaler', StandardScaler()),
                ('smote', SMOTE(random_state=self.random_state, k_neighbors=5)),
                ('classifier', RandomForestClassifier(
                    n_estimators=n_estimators,
                    max_depth=max_depth,
                    min_samples_split=min_samples_split,
                    min_samples_leaf=min_samples_leaf,
                    class_weight=class_weight,
                    random_state=self.random_state,
                    n_jobs=-1,
                    verbose=0
                ))
            ])
            
            logger.info("Model pipeline built successfully")
            return pipeline
            
        except Exception as e:
            logger.error(f"Error building model: {str(e)}")
            raise
    
    def train_model(
        self,
        X_train: np.ndarray,
        y_train: np.ndarray,
        hyperparameters: Dict[str, Any] = None
    ) -> ImbPipeline:
        """
        Train the fraud detection model.
        
        Args:
            X_train: Training features
            y_train: Training labels
            hyperparameters: Model hyperparameters
            
        Returns:
            Trained pipeline
        """
        try:
            logger.info("Starting model training")
            
            if hyperparameters is None:
                hyperparameters = {}
            
            # Build model with hyperparameters
            self.model = self.build_model(**hyperparameters)
            
            # Train model
            self.model.fit(X_train, y_train)
            
            logger.info("Model training completed")
            return self.model
            
        except Exception as e:
            logger.error(f"Error training model: {str(e)}")
            raise
    
    def evaluate_model(
        self,
        X_test: np.ndarray,
        y_test: np.ndarray,
        threshold: float = 0.5
    ) -> Dict[str, float]:
        """
        Evaluate model performance on test data.
        
        Args:
            X_test: Test features
            y_test: Test labels
            threshold: Classification threshold
            
        Returns:
            Dictionary of evaluation metrics
        """
        try:
            logger.info("Evaluating model performance")
            
            if self.model is None:
                raise ValueError("Model not trained yet")
            
            # Predictions
            y_pred_proba = self.model.predict_proba(X_test)[:, 1]
            y_pred = (y_pred_proba >= threshold).astype(int)
            
            # Calculate metrics
            metrics = {
                'roc_auc': roc_auc_score(y_test, y_pred_proba),
                'average_precision': average_precision_score(y_test, y_pred_proba),
                'precision': precision_score(y_test, y_pred, zero_division=0),
                'recall': recall_score(y_test, y_pred, zero_division=0),
                'f1_score': f1_score(y_test, y_pred, zero_division=0),
                'threshold': threshold
            }
            
            # Confusion matrix
            tn, fp, fn, tp = confusion_matrix(y_test, y_pred).ravel()
            metrics.update({
                'true_negatives': int(tn),
                'false_positives': int(fp),
                'false_negatives': int(fn),
                'true_positives': int(tp)
            })
            
            # Log metrics
            logger.info(f"ROC-AUC: {metrics['roc_auc']:.4f}")
            logger.info(f"Average Precision: {metrics['average_precision']:.4f}")
            logger.info(f"Precision: {metrics['precision']:.4f}")
            logger.info(f"Recall: {metrics['recall']:.4f}")
            logger.info(f"F1-Score: {metrics['f1_score']:.4f}")
            
            return metrics
            
        except Exception as e:
            logger.error(f"Error evaluating model: {str(e)}")
            raise
    
    def detect_concept_drift(
        self,
        reference_data: pd.DataFrame,
        current_data: pd.DataFrame
    ) -> Tuple[bool, Dict[str, float]]:
        """
        Detect concept drift using Kolmogorov-Smirnov test.
        
        Args:
            reference_data: Historical reference data
            current_data: Current production data
            
        Returns:
            Tuple of (drift_detected, drift_metrics)
        """
        try:
            logger.info("Detecting concept drift")
            
            drift_metrics = {}
            drift_detected = False
            
            # Get common features
            common_features = set(reference_data.columns) & set(current_data.columns)
            common_features.discard('Class')
            
            for feature in common_features:
                # Perform KS test
                statistic, p_value = stats.ks_2samp(
                    reference_data[feature],
                    current_data[feature]
                )
                
                drift_metrics[f'{feature}_ks_statistic'] = statistic
                drift_metrics[f'{feature}_p_value'] = p_value
                
                # Check if drift detected
                if p_value < self.drift_threshold:
                    drift_detected = True
                    logger.warning(
                        f"Drift detected in feature '{feature}': "
                        f"p-value={p_value:.6f}"
                    )
            
            # Check target distribution drift
            if 'Class' in reference_data.columns and 'Class' in current_data.columns:
                ref_fraud_rate = reference_data['Class'].mean()
                curr_fraud_rate = current_data['Class'].mean()
                drift_metrics['fraud_rate_change'] = abs(curr_fraud_rate - ref_fraud_rate)
                
                if abs(curr_fraud_rate - ref_fraud_rate) > 0.001:  # 0.1% change
                    logger.warning(
                        f"Fraud rate changed from {ref_fraud_rate:.4f} "
                        f"to {curr_fraud_rate:.4f}"
                    )
            
            logger.info(f"Drift detection completed: drift_detected={drift_detected}")
            return drift_detected, drift_metrics
            
        except Exception as e:
            logger.error(f"Error detecting drift: {str(e)}")
            raise
    
    def log_to_mlflow(
        self,
        metrics: Dict[str, float],
        hyperparameters: Dict[str, Any],
        artifacts_dir: str = "artifacts"
    ) -> str:
        """
        Log model, metrics, and artifacts to MLflow.
        
        Args:
            metrics: Evaluation metrics
            hyperparameters: Model hyperparameters
            artifacts_dir: Directory to save artifacts
            
        Returns:
            MLflow run ID
        """
        try:
            logger.info("Logging to MLflow")
            
            with mlflow.start_run() as run:
                # Log hyperparameters
                mlflow.log_params(hyperparameters)
                
                # Log metrics
                mlflow.log_metrics(metrics)
                
                # Log model
                mlflow.sklearn.log_model(
                    self.model,
                    "model",
                    registered_model_name=self.model_name
                )
                
                # Save and log artifacts
                Path(artifacts_dir).mkdir(exist_ok=True)
                
                # Save feature importance
                if hasattr(self.model.named_steps['classifier'], 'feature_importances_'):
                    importance_df = pd.DataFrame({
                        'feature': self.feature_names,
                        'importance': self.model.named_steps['classifier'].feature_importances_
                    }).sort_values('importance', ascending=False)
                    
                    importance_path = Path(artifacts_dir) / 'feature_importance.csv'
                    importance_df.to_csv(importance_path, index=False)
                    mlflow.log_artifact(importance_path)
                
                # Log confusion matrix as text
                cm_text = (
                    f"Confusion Matrix:\n"
                    f"TN: {metrics['true_negatives']}, FP: {metrics['false_positives']}\n"
                    f"FN: {metrics['false_negatives']}, TP: {metrics['true_positives']}"
                )
                cm_path = Path(artifacts_dir) / 'confusion_matrix.txt'
                cm_path.write_text(cm_text)
                mlflow.log_artifact(cm_path)
                
                logger.info(f"MLflow run completed: {run.info.run_id}")
                return run.info.run_id
                
        except Exception as e:
            logger.error(f"Error logging to MLflow: {str(e)}")
            raise
    
    def save_model(self, model_path: str = "model.joblib"):
        """
        Save trained model to disk.
        
        Args:
            model_path: Path to save the model
        """
        try:
            if self.model is None:
                raise ValueError("No model to save")
            
            logger.info(f"Saving model to {model_path}")
            joblib.dump(self.model, model_path)
            logger.info("Model saved successfully")
            
        except Exception as e:
            logger.error(f"Error saving model: {str(e)}")
            raise
    
    def load_model(self, model_path: str = "model.joblib"):
        """
        Load trained model from disk.
        
        Args:
            model_path: Path to the saved model
        """
        try:
            logger.info(f"Loading model from {model_path}")
            
            if not Path(model_path).exists():
                raise FileNotFoundError(f"Model file not found: {model_path}")
            
            self.model = joblib.load(model_path)
            logger.info("Model loaded successfully")
            
        except Exception as e:
            logger.error(f"Error loading model: {str(e)}")
            raise
    
    def run_pipeline(
        self,
        data_path: str,
        hyperparameters: Dict[str, Any] = None,
        reference_data_path: str = None
    ) -> Dict[str, Any]:
        """
        Execute the complete ML pipeline.
        
        Args:
            data_path: Path to training data
            hyperparameters: Model hyperparameters
            reference_data_path: Path to reference data for drift detection
            
        Returns:
            Dictionary with run results
        """
        try:
            logger.info("=" * 80)
            logger.info("Starting fraud detection pipeline")
            logger.info("=" * 80)
            
            # Load data
            df = self.load_data(data_path)
            
            # Check for concept drift if reference data provided
            drift_detected = False
            if reference_data_path:
                reference_df = self.load_data(reference_data_path)
                drift_detected, drift_metrics = self.detect_concept_drift(
                    reference_df, df
                )
                
                if not drift_detected:
                    logger.info("No concept drift detected, skipping retraining")
                    return {
                        'status': 'skipped',
                        'reason': 'no_drift_detected',
                        'drift_metrics': drift_metrics
                    }
            
            # Prepare data
            X_train, X_test, y_train, y_test = self.prepare_data(df)
            
            # Set default hyperparameters
            if hyperparameters is None:
                hyperparameters = {
                    'n_estimators': 100,
                    'max_depth': 10,
                    'min_samples_split': 10,
                    'min_samples_leaf': 5,
                    'class_weight': 'balanced'
                }
            
            # Train model
            self.train_model(X_train, y_train, hyperparameters)
            
            # Evaluate model
            metrics = self.evaluate_model(X_test, y_test)
            
            # Log to MLflow
            run_id = self.log_to_mlflow(metrics, hyperparameters)
            
            # Save model
            self.save_model()
            
            logger.info("=" * 80)
            logger.info("Pipeline completed successfully")
            logger.info("=" * 80)
            
            return {
                'status': 'success',
                'run_id': run_id,
                'metrics': metrics,
                'drift_detected': drift_detected
            }
            
        except Exception as e:
            logger.error(f"Pipeline failed: {str(e)}")
            raise


def main():
    """Main entry point for the pipeline."""
    import argparse
    
    parser = argparse.ArgumentParser(description='Fraud Detection ML Pipeline')
    parser.add_argument(
        '--data-path',
        type=str,
        required=True,
        help='Path to training data CSV'
    )
    parser.add_argument(
        '--reference-data-path',
        type=str,
        default=None,
        help='Path to reference data for drift detection'
    )
    parser.add_argument(
        '--experiment-name',
        type=str,
        default='fraud_detection',
        help='MLflow experiment name'
    )
    parser.add_argument(
        '--model-name',
        type=str,
        default='fraud_detector_rf',
        help='Model name for MLflow registry'
    )
    
    args = parser.parse_args()
    
    # Initialize and run pipeline
    pipeline = FraudDetectionPipeline(
        experiment_name=args.experiment_name,
        model_name=args.model_name
    )
    
    results = pipeline.run_pipeline(
        data_path=args.data_path,
        reference_data_path=args.reference_data_path
    )
    
    print(f"\nPipeline Results: {results}")


if __name__ == "__main__":
    main()
```
