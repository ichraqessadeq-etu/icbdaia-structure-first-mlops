```python
# fraud_detection_pipeline.py
"""
Production-ready fraud detection pipeline with MLflow tracking.
Handles imbalanced data, concept drift detection, and comprehensive logging.
"""

import os
import sys
import logging
import json
from datetime import datetime
from typing import Tuple, Dict, Any, Optional
import warnings

import numpy as np
import pandas as pd
from sklearn.ensemble import RandomForestClassifier
from sklearn.model_selection import train_test_split, StratifiedKFold
from sklearn.metrics import (
    precision_score, recall_score, f1_score, roc_auc_score,
    average_precision_score, confusion_matrix, classification_report
)
from sklearn.preprocessing import StandardScaler
from imblearn.over_sampling import SMOTE
from imblearn.pipeline import Pipeline as ImbPipeline
import mlflow
import mlflow.sklearn
from scipy import stats
import joblib

warnings.filterwarnings('ignore')

# Configure logging
logging.basicConfig(
    level=logging.INFO,
    format='%(asctime)s - %(name)s - %(levelname)s - %(message)s',
    handlers=[
        logging.FileHandler('fraud_detection_pipeline.log'),
        logging.StreamHandler(sys.stdout)
    ]
)
logger = logging.getLogger(__name__)


class FraudDetectionPipeline:
    """Complete pipeline for fraud detection model training and evaluation."""
    
    def __init__(
        self,
        data_path: str,
        mlflow_tracking_uri: str = "http://localhost:5000",
        experiment_name: str = "fraud_detection",
        random_state: int = 42
    ):
        """
        Initialize the fraud detection pipeline.
        
        Args:
            data_path: Path to the credit card fraud dataset
            mlflow_tracking_uri: MLflow tracking server URI
            experiment_name: Name of the MLflow experiment
            random_state: Random seed for reproducibility
        """
        self.data_path = data_path
        self.mlflow_tracking_uri = mlflow_tracking_uri
        self.experiment_name = experiment_name
        self.random_state = random_state
        self.model = None
        self.scaler = None
        self.metrics = {}
        
        # Setup MLflow
        mlflow.set_tracking_uri(self.mlflow_tracking_uri)
        mlflow.set_experiment(self.experiment_name)
        
        logger.info(f"Initialized FraudDetectionPipeline with experiment: {experiment_name}")
    
    def load_data(self) -> pd.DataFrame:
        """
        Load and validate the fraud detection dataset.
        
        Returns:
            DataFrame containing the loaded data
            
        Raises:
            FileNotFoundError: If data file doesn't exist
            ValueError: If data validation fails
        """
        try:
            logger.info(f"Loading data from {self.data_path}")
            
            if not os.path.exists(self.data_path):
                raise FileNotFoundError(f"Data file not found: {self.data_path}")
            
            # Load data
            if self.data_path.endswith('.csv'):
                df = pd.read_csv(self.data_path)
            elif self.data_path.endswith('.parquet'):
                df = pd.read_parquet(self.data_path)
            else:
                raise ValueError("Unsupported file format. Use CSV or Parquet.")
            
            # Validate data
            self._validate_data(df)
            
            logger.info(f"Successfully loaded {len(df)} records")
            logger.info(f"Fraud rate: {df['Class'].mean() * 100:.4f}%")
            
            return df
            
        except Exception as e:
            logger.error(f"Error loading data: {str(e)}")
            raise
    
    def _validate_data(self, df: pd.DataFrame) -> None:
        """
        Validate the loaded dataset.
        
        Args:
            df: DataFrame to validate
            
        Raises:
            ValueError: If validation fails
        """
        # Check for required columns
        if 'Class' not in df.columns:
            raise ValueError("Target column 'Class' not found in dataset")
        
        # Check for missing values
        missing_pct = (df.isnull().sum() / len(df) * 100).max()
        if missing_pct > 5:
            raise ValueError(f"Too many missing values: {missing_pct:.2f}%")
        
        # Check class distribution
        fraud_rate = df['Class'].mean()
        if fraud_rate == 0 or fraud_rate == 1:
            raise ValueError("Invalid class distribution: only one class present")
        
        # Check for sufficient samples
        if len(df) < 1000:
            raise ValueError(f"Insufficient data: only {len(df)} samples")
        
        logger.info("Data validation passed")
    
    def preprocess_data(
        self, 
        df: pd.DataFrame,
        test_size: float = 0.2
    ) -> Tuple[np.ndarray, np.ndarray, np.ndarray, np.ndarray]:
        """
        Preprocess data with train-test split and feature scaling.
        
        Args:
            df: Input DataFrame
            test_size: Proportion of data for testing
            
        Returns:
            Tuple of (X_train, X_test, y_train, y_test)
        """
        try:
            logger.info("Preprocessing data...")
            
            # Separate features and target
            X = df.drop('Class', axis=1).values
            y = df['Class'].values
            
            # Handle any remaining missing values
            X = np.nan_to_num(X, nan=0.0)
            
            # Stratified split to maintain class distribution
            X_train, X_test, y_train, y_test = train_test_split(
                X, y,
                test_size=test_size,
                random_state=self.random_state,
                stratify=y
            )
            
            # Scale features
            self.scaler = StandardScaler()
            X_train = self.scaler.fit_transform(X_train)
            X_test = self.scaler.transform(X_test)
            
            logger.info(f"Train set: {len(X_train)} samples, Fraud rate: {y_train.mean()*100:.4f}%")
            logger.info(f"Test set: {len(X_test)} samples, Fraud rate: {y_test.mean()*100:.4f}%")
            
            return X_train, X_test, y_train, y_test
            
        except Exception as e:
            logger.error(f"Error in preprocessing: {str(e)}")
            raise
    
    def train_model(
        self,
        X_train: np.ndarray,
        y_train: np.ndarray,
        use_smote: bool = True,
        n_estimators: int = 100,
        max_depth: int = 20,
        min_samples_split: int = 10,
        class_weight: str = 'balanced'
    ) -> None:
        """
        Train Random Forest model with SMOTE for handling imbalanced data.
        
        Args:
            X_train: Training features
            y_train: Training labels
            use_smote: Whether to apply SMOTE oversampling
            n_estimators: Number of trees in the forest
            max_depth: Maximum depth of trees
            min_samples_split: Minimum samples required to split
            class_weight: Class weight strategy
        """
        try:
            logger.info("Training model...")
            
            # Create Random Forest classifier
            rf_classifier = RandomForestClassifier(
                n_estimators=n_estimators,
                max_depth=max_depth,
                min_samples_split=min_samples_split,
                class_weight=class_weight,
                random_state=self.random_state,
                n_jobs=-1,
                verbose=0
            )
            
            if use_smote:
                # Create pipeline with SMOTE
                logger.info("Applying SMOTE for handling class imbalance...")
                smote = SMOTE(
                    sampling_strategy=0.5,  # Make minority class 50% of majority
                    random_state=self.random_state
                )
                self.model = ImbPipeline([
                    ('smote', smote),
                    ('classifier', rf_classifier)
                ])
            else:
                self.model = rf_classifier
            
            # Train model
            self.model.fit(X_train, y_train)
            
            logger.info("Model training completed successfully")
            
        except Exception as e:
            logger.error(f"Error in model training: {str(e)}")
            raise
    
    def evaluate_model(
        self,
        X_test: np.ndarray,
        y_test: np.ndarray,
        threshold: float = 0.5
    ) -> Dict[str, Any]:
        """
        Evaluate model performance with comprehensive metrics.
        
        Args:
            X_test: Test features
            y_test: Test labels
            threshold: Classification threshold
            
        Returns:
            Dictionary containing evaluation metrics
        """
        try:
            logger.info("Evaluating model...")
            
            # Get predictions
            y_pred_proba = self.model.predict_proba(X_test)[:, 1]
            y_pred = (y_pred_proba >= threshold).astype(int)
            
            # Calculate metrics
            metrics = {
                'accuracy': float(np.mean(y_pred == y_test)),
                'precision': float(precision_score(y_test, y_pred, zero_division=0)),
                'recall': float(recall_score(y_test, y_pred, zero_division=0)),
                'f1_score': float(f1_score(y_test, y_pred, zero_division=0)),
                'roc_auc': float(roc_auc_score(y_test, y_pred_proba)),
                'average_precision': float(average_precision_score(y_test, y_pred_proba)),
                'threshold': threshold
            }
            
            # Confusion matrix
            tn, fp, fn, tp = confusion_matrix(y_test, y_pred).ravel()
            metrics.update({
                'true_negatives': int(tn),
                'false_positives': int(fp),
                'false_negatives': int(fn),
                'true_positives': int(tp),
                'false_positive_rate': float(fp / (fp + tn)) if (fp + tn) > 0 else 0.0,
                'false_negative_rate': float(fn / (fn + tp)) if (fn + tp) > 0 else 0.0
            })
            
            self.metrics = metrics
            
            # Log metrics
            logger.info("Evaluation Metrics:")
            logger.info(f"  ROC-AUC: {metrics['roc_auc']:.4f}")
            logger.info(f"  Average Precision: {metrics['average_precision']:.4f}")
            logger.info(f"  Precision: {metrics['precision']:.4f}")
            logger.info(f"  Recall: {metrics['recall']:.4f}")
            logger.info(f"  F1-Score: {metrics['f1_score']:.4f}")
            logger.info(f"  True Positives: {metrics['true_positives']}")
            logger.info(f"  False Positives: {metrics['false_positives']}")
            logger.info(f"  False Negatives: {metrics['false_negatives']}")
            
            return metrics
            
        except Exception as e:
            logger.error(f"Error in model evaluation: {str(e)}")
            raise
    
    def detect_concept_drift(
        self,
        X_current: np.ndarray,
        X_reference: np.ndarray,
        threshold: float = 0.05
    ) -> Dict[str, Any]:
        """
        Detect concept drift using Kolmogorov-Smirnov test.
        
        Args:
            X_current: Current data features
            X_reference: Reference (training) data features
            threshold: P-value threshold for drift detection
            
        Returns:
            Dictionary containing drift detection results
        """
        try:
            logger.info("Detecting concept drift...")
            
            drift_detected = False
            drift_features = []
            p_values = []
            
            # Test each feature for drift
            for i in range(X_current.shape[1]):
                statistic, p_value = stats.ks_2samp(
                    X_reference[:, i],
                    X_current[:, i]
                )
                p_values.append(p_value)
                
                if p_value < threshold:
                    drift_detected = True
                    drift_features.append(i)
            
            drift_results = {
                'drift_detected': drift_detected,
                'num_drifted_features': len(drift_features),
                'drifted_features': drift_features,
                'min_p_value': float(min(p_values)) if p_values else 1.0,
                'mean_p_value': float(np.mean(p_values)) if p_values else 1.0,
                'threshold': threshold
            }
            
            if drift_detected:
                logger.warning(f"Concept drift detected in {len(drift_features)} features!")
                logger.warning(f"Minimum p-value: {drift_results['min_p_value']:.6f}")
            else:
                logger.info("No significant concept drift detected")
            
            return drift_results
            
        except Exception as e:
            logger.error(f"Error in drift detection: {str(e)}")
            raise
    
    def log_to_mlflow(
        self,
        params: Dict[str, Any],
        metrics: Dict[str, Any],
        model_name: str = "fraud_detection_rf",
        tags: Optional[Dict[str, str]] = None
    ) -> str:
        """
        Log model, parameters, and metrics to MLflow.
        
        Args:
            params: Model parameters
            metrics: Evaluation metrics
            model_name: Name for the registered model
            tags: Additional tags for the run
            
        Returns:
            MLflow run ID
        """
        try:
            logger.info("Logging to MLflow...")
            
            with mlflow.start_run() as run:
                # Log parameters
                mlflow.log_params(params)
                
                # Log metrics
                mlflow.log_metrics(metrics)
                
                # Log tags
                if tags:
                    mlflow.set_tags(tags)
                
                # Log additional info
                mlflow.set_tag("model_type", "RandomForestClassifier")
                mlflow.set_tag("training_date", datetime.now().isoformat())
                
                # Log model
                mlflow.sklearn.log_model(
                    self.model,
                    "model",
                    registered_model_name=model_name
                )
                
                # Log scaler separately
                mlflow.sklearn.log_model(
                    self.scaler,
                    "scaler"
                )
                
                # Log classification report as artifact
                y_pred = self.model.predict(self.X_test_cached)
                report = classification_report(
                    self.y_test_cached,
                    y_pred,
                    target_names=['Legitimate', 'Fraud']
                )
                
                with open("classification_report.txt", "w") as f:
                    f.write(report)
                mlflow.log_artifact("classification_report.txt")
                os.remove("classification_report.txt")
                
                run_id = run.info.run_id
                logger.info(f"Successfully logged to MLflow. Run ID: {run_id}")
                
                return run_id
                
        except Exception as e:
            logger.error(f"Error logging to MLflow: {str(e)}")
            raise
    
    def save_model(self, output_path: str = "models") -> str:
        """
        Save model and scaler to disk.
        
        Args:
            output_path: Directory to save the model
            
        Returns:
            Path to saved model
        """
        try:
            os.makedirs(output_path, exist_ok=True)
            
            timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
            model_path = os.path.join(output_path, f"fraud_model_{timestamp}.pkl")
            scaler_path = os.path.join(output_path, f"scaler_{timestamp}.pkl")
            
            joblib.dump(self.model, model_path)
            joblib.dump(self.scaler, scaler_path)
            
            # Save metadata
            metadata = {
                'timestamp': timestamp,
                'metrics': self.metrics,
                'model_path': model_path,
                'scaler_path': scaler_path
            }
            
            metadata_path = os.path.join(output_path, f"metadata_{timestamp}.json")
            with open(metadata_path, 'w') as f:
                json.dump(metadata, f, indent=2)
            
            logger.info(f"Model saved to {model_path}")
            return model_path
            
        except Exception as e:
            logger.error(f"Error saving model: {str(e)}")
            raise
    
    def run_pipeline(
        self,
        test_size: float = 0.2,
        use_smote: bool = True,
        n_estimators: int = 100,
        max_depth: int = 20,
        detect_drift: bool = False,
        save_model: bool = True
    ) -> Dict[str, Any]:
        """
        Run the complete pipeline from data loading to model evaluation.
        
        Args:
            test_size: Proportion of data for testing
            use_smote: Whether to apply SMOTE
            n_estimators: Number of trees
            max_depth: Maximum tree depth
            detect_drift: Whether to perform drift detection
            save_model: Whether to save the model to disk
            
        Returns:
            Dictionary containing pipeline results
        """
        try:
            logger.info("="*50)
            logger.info("Starting Fraud Detection Pipeline")
            logger.info("="*50)
            
            # Load data
            df = self.load_data()
            
            # Preprocess
            X_train, X_test, y_train, y_test = self.preprocess_data(df, test_size)
            
            # Cache test data for MLflow logging
            self.X_test_cached = X_test
            self.y_test_cached = y_test
            
            # Train model
            self.train_model(
                X_train, y_train,
                use_smote=use_smote,
                n_estimators=n_estimators,
                max_depth=max_depth
            )
            
            # Evaluate model
            metrics = self.evaluate_model(X_test, y_test)
            
            # Detect drift if requested
            drift_results = None
            if detect_drift:
                drift_results = self.detect_concept_drift(X_test, X_train)
            
            # Prepare parameters for logging
            params = {
                'use_smote': use_smote,
                'n_estimators': n_estimators,
                'max_depth': max_depth,
                'test_size': test_size,
                'random_state': self.random_state,
                'total_samples': len(df),
                'train_samples': len(X_train),
                'test_samples': len(X_test),
                'fraud_rate': float(df['Class'].mean())
            }
            
            # Log to MLflow
            tags = {}
            if drift_results and drift_results['drift_detected']:
                tags['drift_detected'] = 'true'
                tags['drifted_features'] = str(drift_results['num_drifted_features'])
            
            run_id = self.log_to_mlflow(params, metrics, tags=tags)
            
            # Save model if requested
            model_path = None
            if save_model:
                model_path = self.save_model()
            
            # Prepare results
            results = {
                'success': True,
                'run_id': run_id,
                'metrics': metrics,
                'drift_results': drift_results,
                'model_path': model_path,
                'timestamp': datetime.now().isoformat()
            }
            
            logger.info("="*50)
            logger.info("Pipeline completed successfully!")
            logger.info("="*50)
            
            return results
            
        except Exception as e:
            logger.error(f"Pipeline failed: {str(e)}")
            raise


def main():
    """Main entry point for the pipeline."""
    import argparse
    
    parser = argparse.ArgumentParser(description='Fraud Detection Pipeline')
    parser.add_argument('--data-path', type=str, required=True,
                       help='Path to the fraud detection dataset')
    parser.add_argument('--mlflow-uri', type=str, default='http://localhost:5000',
                       help='MLflow tracking URI')
    parser.add_argument('--experiment-name', type=str, default='fraud_detection',
                       help='MLflow experiment name')
    parser.add_argument('--test-size', type=float, default=0.2,
                       help='Test set size')
    parser.add_argument('--use-smote', action='store_true', default=True,
                       help='Use SMOTE for handling imbalanced data')
    parser.add_argument('--n-estimators', type=int, default=100,
                       help='Number of trees in Random Forest')
    parser.add_argument('--max-depth', type=int, default=20,
                       help='Maximum depth of trees')
    parser.add_argument('--detect-drift', action='store_true',
                       help='Perform concept drift detection')
    parser.add_argument('--no-save-model', action='store_true',
                       help='Do not save model to disk')
    
    args = parser.parse_args()
    
    # Initialize and run pipeline
    pipeline = FraudDetectionPipeline(
        data_path=args.data_path,
        mlflow_tracking_uri=args.mlflow_uri,
        experiment_name=args.experiment_name
    )
    
    results = pipeline.run_pipeline(
        test_size=args.test_size,
        use_smote=args.use_smote,
        n_estimators=args.n_estimators,
        max_depth=args.max_depth,
        detect_drift=args.detect_drift,
        save_model=not args.no_save_model
    )
    
    # Print summary
    print("\n" + "="*50)
    print("PIPELINE RESULTS SUMMARY")
    print("="*50)
    print(f"MLflow Run ID: {results['run_id']}")
    print(f"ROC-AUC Score: {results['metrics']['roc_auc']:.4f}")
    print(f"Precision: {results['metrics']['precision']:.4f}")
    print(f"Recall: {results['metrics']['recall']:.4f}")
    if results['drift_results']:
        print(f"Drift Detected: {results['drift_results']['drift_detected']}")
    print("="*50)


if __name__ == "__main__":
    main()
```

```python
class TestPreprocessing:
    """Test suite for data preprocessing."""
    
    def test_preprocess_data_shapes(self, pipeline, sample_fraud_data):
        """Test that preprocessing returns correct shapes."""
        X_train, X_test, y_train, y_test = pipeline.preprocess_data(
            sample_fraud_data,
            test_size=0.2
        )
        
        total_samples = len(sample_fraud_data)
        expected_train_size = int(total_samples * 0.8)
        expected_test_size = total_samples - expected_train_size
        
        assert X_train.shape[0] == expected_train_size
        assert X_test.shape[0] == expected_test_size
        assert len(y_train) == expected_train_size
        assert len(y_test) == expected_test_size
        assert X_train.shape[1] == X_test.shape[1]
    
    def test_preprocess_data_stratification(self, pipeline, sample_fraud_data):
        """Test that stratification maintains class distribution."""
        original_fraud_rate = sample_fraud_data['Class'].mean()
        
        X_train, X_test, y_train, y_test = pipeline.preprocess_data(
            sample_fraud_data,
            test_size=0.2
        )
        
        train_fraud_rate = y_train.mean()
        test_fraud_rate = y_test.mean()
        
        # Fraud rates should be similar (within 50% relative difference)
        assert abs(train_fraud_rate - original_fraud_rate) / original_fraud_rate < 0.5
        assert abs(test_fraud_rate - original_fraud_rate) / original_fraud_rate < 0.5
    
    def test_preprocess_data_scaling(self, pipeline, sample_fraud_data):
        """Test that features are properly scaled."""
        X_train, X_test, y_train, y_test = pipeline.preprocess_data(
            sample_fraud_data,
            test_size=0.2
        )
        
        # Check that training data is approximately standardized
        assert np.abs(X_train.mean()) < 0.1
        assert np.abs(X_train.std() - 1.0) < 0.2
    
    def test_preprocess_handles_nan(self, pipeline):
        """Test that preprocessing handles NaN values."""
        df = pd.DataFrame({
            'V1': [1, 2, np.nan, 4, 5] * 200,
            'V2': [np.nan, 2, 3, 4, 5] * 200,
            'Class': [0, 1, 0, 1, 0] * 200
        })
        
        X_train, X_test, y_train, y_test = pipeline.preprocess_data(df)
        
        # No NaN values should remain
        assert not np.isnan(X_train).any()
        assert not np.isnan(X_test).any()

        """Test drift detection when drift exists."""

        X_reference = np.random.randn(1000, 10)

        drift_results = pipeline.detect_concept_drift(
            X_current, X_reference, threshold=0.05
        )
        
        # Should detect drift with such a large shift
        assert drift_results['drift_detected'] is True
        assert drift_results['num_drifted_features'] > 0
    
    def test_drift_detection_threshold_sensitivity(self, pipeline):
        """Test that drift detection is sensitive to threshold."""
        X_reference = np.random.randn(1000, 10)
        X_current = np.random.randn(1000, 10) + 0.5  # Small shift
        
        # Strict threshold
        drift_strict = pipeline.detect_concept_drift(
            X_current, X_reference, threshold=0.001
        )
        
        # Lenient threshold
        drift_lenient = pipeline.detect_concept_drift(
            X_current, X_reference, threshold=0.5
        )
        
        # Strict threshold should detect more drift
        assert drift_strict['num_drifted_features'] >= drift_lenient['num_drifted_features']

        """Test model saving functionality."""
        temp_dir = tempfile.mkdtemp()
        
        try:
            # Train model
            X_train, X_test, y_train, y_test = pipeline.preprocess_data(
                sample_fraud_data
            )
            pipeline.train_model(X_train, y_train)
            pipeline.metrics = pipeline.evaluate_model(X_test, y_test)
            
            # Save model
            model_path = pipeline.save_model(output_path=temp_dir)
            
            # Check that files were created
            assert os.path.exists(model_path)
            assert model_path.endswith('.pkl')
            
            # Check for metadata file
            metadata_files = [f for f in os.listdir(temp_dir) if f.startswith('metadata_')]
            assert len(metadata_files) > 0
            
        finally:
            shutil.rmtree(temp_dir)
    
    def test_save_and_load_model(self, pipeline, sample_fraud_data):
        """Test that saved model can be loaded and used."""
        import joblib
        temp_dir = tempfile.mkdtemp()
        
        try:
            # Train and save model
            X_train, X_test, y_train, y_test = pipeline.preprocess_data(
                sample_fraud_data
            )
            pipeline.train_model(X_train, y_train)
            pipeline.metrics = pipeline.evaluate_model(X_test, y_test)
            
            original_predictions = pipeline.model.predict(X_test[:10])
            
            model_path = pipeline.save_model(output_path=temp_dir)
            
            # Load model
            loaded_model = joblib.load(model_path)
            loaded_predictions = loaded_model.predict(X_test[:10])
            
            # Predictions should match
            np.testing.assert_array_equal(original_predictions, loaded_predictions)
            
        finally:
            shutil.rmtree(temp_dir)
```

```yaml
    name: Train Fraud Detection Model
    runs-on: ubuntu-latest
    needs: data-validation
    if: needs.data-validation.outputs.should_retrain == 'true'
    outputs:
      model_version: ${{ steps.train.outputs.model_version }}
      run_id: ${{ steps.train.outputs.run_id }}
    
    steps:
      - name: Checkout code
        uses: actions/checkout@v3
      
      - name: Set up Python
        uses: actions/setup-python@v4
        with:
          python-version: ${{ env.PYTHON_VERSION }}

          aws-region: ${{ env.AWS_REGION }}

        run: |

        run: |

            --test-size 0.2 \
            --use-smote \
            --n-estimators 100 \
            --max-depth 20 \
            --detect-drift \
            > training_log.txt 2>&1
          
          # Extract run ID from logs
          RUN_ID=$(grep "Run ID:" training_log.txt | awk '{print $NF}')
          MODEL_VERSION=$(date +%Y%m%d_%H%M%S)
          
          echo "run_id=$RUN_ID" >> $GITHUB_OUTPUT
          echo "model_version=$MODEL_VERSION" >> $GITHUB_OUTPUT

  # Job 4: Model Evaluation
  model-evaluation:
    name: Evaluate Model Performance
    runs-on: ubuntu-latest
    needs: model-training
    outputs:
      performance_acceptable: ${{ steps.evaluate.outputs.performance_acceptable }}
    
    steps:
      - name: Checkout code
        uses: actions/checkout@v3
      
      - name: Set up Python
        uses: actions/setup-python@v4
        with:
          python-version: ${{ env.PYTHON_VERSION }}
          cache: 'pip'
      
      - name: Install dependencies
        run: |
          python -m pip install --upgrade pip
          pip install -r requirements.txt
      
      - name: Configure AWS credentials
        uses: aws-actions/configure-aws-credentials@v2
        with:
          aws-access-key-id: ${{ secrets.AWS_ACCESS_KEY_ID }}
          aws-secret-access-key: ${{ secrets.AWS_SECRET_ACCESS_KEY }}
          aws-region: ${{ env.AWS_REGION }}
      
      - name: Download model and test data
        run: |
          mkdir -p data models
          aws s3 cp s3://${{ secrets.DATA_BUCKET }}/fraud_data_test.csv ./data/fraud_data_test.csv
          aws s3 cp s3://${{ secrets.MODEL_BUCKET }}/models/${{ needs.model-training.outputs.model_version }}/ ./models/ --recursive
      
      - name: Evaluate model
        id: evaluate
        env:
          MLFLOW_TRACKING_URI: ${{ secrets.MLFLOW_TRACKING_URI }}
        run: |
          python -c "
          import mlflow
          import json
          import os
          
          # Get model metrics from MLflow
          client = mlflow.tracking.MlflowClient()
          run = client.get_run('${{ needs.model-training.outputs.run_id }}')
          metrics = run.data.metrics
          
          # Performance thresholds
          MIN_ROC_AUC = 0.85
          MIN_RECALL = 0.70
          MIN_PRECISION = 0.10
          
          roc_auc = metrics.get('roc_auc', 0)
          recall = metrics.get('recall', 0)
          precision = metrics.get('precision', 0)
          
          print(f'ROC-AUC: {roc_auc:.4f} (threshold: {MIN_ROC_AUC})')
          print(f'Recall: {recall:.4f} (threshold: {MIN_RECALL})')
          print(f'Precision: {precision:.4f} (threshold: {MIN_PRECISION})')
          
          performance_acceptable = (
              roc_auc >= MIN_ROC_AUC and
              recall >= MIN_RECALL and
              precision >= MIN_PRECISION
          )
          
          # Save evaluation results
          eval_results = {
              'roc_auc': roc_auc,
              'recall': recall,
              'precision': precision,
              'performance_acceptable': performance_acceptable,
              'thresholds': {
                  'min_roc_auc': MIN_ROC_AUC,
                  'min_recall': MIN_RECALL,
                  'min_precision': MIN_PRECISION
              }
          }
          
          with open('evaluation_results.json', 'w') as f:
              json.dump(eval_results, f, indent=2)
          
          # Set output
          with open(os.environ['GITHUB_OUTPUT'], 'a') as f:
              f.write(f'performance_acceptable={str(performance_acceptable).lower()}\n')
          
          if not performance_acceptable:
              print('WARNING: Model performance below thresholds')
          else:
              print('Model performance acceptable')
          "
      
      - name: Upload evaluation results
        uses: actions/upload-artifact@v3
        with:
          name: evaluation-results
          path: evaluation_results.json
      
      - name: Comment on PR with results
        if: github.event_name == 'pull_request'
        uses: actions/github-script@v6
        with:
          script: |
            const fs = require('fs');
            const results = JSON.parse(fs.readFileSync('evaluation_results.json', 'utf8'));
            
            const comment = `## Model Evaluation Results
            
            | Metric | Value | Threshold | Status |
            |--------|-------|-----------|--------|
            | ROC-AUC | ${results.roc_auc.toFixed(4)} | ${results.thresholds.min_roc_auc} | ${results.roc_auc >= results.thresholds.min_roc_auc ? '✅' : '❌'} |
            | Recall | ${results.recall.toFixed(4)} | ${results.thresholds.min_recall} | ${results.recall >= results.thresholds.min_recall ? '✅' : '❌'} |
            | Precision | ${results.precision.toFixed(4)} | ${results.thresholds.min_precision} | ${results.precision >= results.thresholds.min_precision ? '✅' : '❌'} |
            
            **Overall Performance:** ${results.performance_acceptable ? '✅ Acceptable' : '❌ Below Threshold'}
            `;
            
            github.rest.issues.createComment({
              issue_number: context.issue.number,
              owner: context.repo.owner,
              repo: context.repo.repo,
              body: comment
            });
```
