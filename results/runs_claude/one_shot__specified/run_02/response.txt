class DataValidator:
    """Validate data quality and detect concept drift."""
    
    def __init__(self, reference_stats_path: Optional[str] = None):
        self.reference_stats_path = reference_stats_path
        self.reference_stats = self._load_reference_stats()
    
    def _load_reference_stats(self) -> Optional[Dict]:
        """Load reference statistics for drift detection."""
        if self.reference_stats_path and os.path.exists(self.reference_stats_path):
            try:
                with open(self.reference_stats_path, 'r') as f:
                    return json.load(f)
            except Exception as e:
                logger.warning(f"Could not load reference stats: {e}")
        return None
    
    def validate_data_quality(self, df: pd.DataFrame) -> Tuple[bool, Dict[str, Any]]:
        """Validate data quality checks."""
        issues = {}
        
        # Check for missing values
        missing_pct = (df.isnull().sum() / len(df) * 100).to_dict()
        if any(pct > 5 for pct in missing_pct.values()):
            issues['missing_values'] = {k: v for k, v in missing_pct.items() if v > 5}
        
        # Check for duplicate rows
        duplicates = df.duplicated().sum()
        if duplicates > 0:
            issues['duplicates'] = duplicates
        
        # Check fraud rate
        if 'Class' in df.columns:
            fraud_rate = df['Class'].mean() * 100
            if fraud_rate < 0.01 or fraud_rate > 5:
                issues['fraud_rate'] = f"{fraud_rate:.4f}% (expected 0.01-5%)"
        
        # Check for infinite values
        numeric_cols = df.select_dtypes(include=[np.number]).columns
        inf_counts = np.isinf(df[numeric_cols]).sum().to_dict()
        if any(count > 0 for count in inf_counts.values()):
            issues['infinite_values'] = {k: v for k, v in inf_counts.items() if v > 0}
        
        is_valid = len(issues) == 0
        return is_valid, issues
    
    def detect_concept_drift(
        self, 
        df: pd.DataFrame, 
        significance_level: float = 0.05
    ) -> Tuple[bool, Dict[str, Any]]:
        """Detect concept drift using Kolmogorov-Smirnov test."""
        if self.reference_stats is None:
            logger.info("No reference stats available. Saving current stats.")
            self._save_current_stats(df)
            return False, {}
        
        drift_detected = False
        drift_features = {}
        
        numeric_cols = df.select_dtypes(include=[np.number]).columns
        numeric_cols = [col for col in numeric_cols if col != 'Class']
        
        for col in numeric_cols:
            if col in self.reference_stats:
                # Kolmogorov-Smirnov test
                ref_mean = self.reference_stats[col]['mean']
                ref_std = self.reference_stats[col]['std']
                
                # Generate reference distribution
                ref_dist = np.random.normal(ref_mean, ref_std, size=len(df))
                current_dist = df[col].dropna().values
                
                ks_statistic, p_value = stats.ks_2samp(ref_dist, current_dist)
                
                if p_value < significance_level:
                    drift_detected = True
                    drift_features[col] = {
                        'ks_statistic': float(ks_statistic),
                        'p_value': float(p_value),
                        'ref_mean': ref_mean,
                        'current_mean': float(df[col].mean()),
                        'ref_std': ref_std,
                        'current_std': float(df[col].std())
                    }
        
        return drift_detected, drift_features
    
    def _save_current_stats(self, df: pd.DataFrame):
        """Save current data statistics as reference."""
        stats_dict = {}
        numeric_cols = df.select_dtypes(include=[np.number]).columns
        numeric_cols = [col for col in numeric_cols if col != 'Class']
        
        for col in numeric_cols:
            stats_dict[col] = {
                'mean': float(df[col].mean()),
                'std': float(df[col].std()),
                'min': float(df[col].min()),
                'max': float(df[col].max())
            }
        
        save_path = self.reference_stats_path or 'reference_stats.json'
        with open(save_path, 'w') as f:
            json.dump(stats_dict, f, indent=2)
        logger.info(f"Saved reference statistics to {save_path}")


class FraudDetectionPipeline:
    """Complete fraud detection pipeline with MLflow tracking."""
    
    def __init__(
        self,
        experiment_name: str = "fraud_detection",
        model_name: str = "fraud_detector_rf",
        random_state: int = 42
    ):
        self.experiment_name = experiment_name
        self.model_name = model_name
        self.random_state = random_state
        self.model = None
        self.scaler = None
        self.feature_names = None
        
        # Set up MLflow
        mlflow.set_experiment(experiment_name)
    
    def load_data(self, data_path: str) -> pd.DataFrame:
        """Load and validate fraud detection data."""
        logger.info(f"Loading data from {data_path}")
        
        try:
            if data_path.endswith('.csv'):
                df = pd.read_csv(data_path)
            elif data_path.endswith('.parquet'):
                df = pd.read_parquet(data_path)
            else:
                raise ValueError(f"Unsupported file format: {data_path}")
            
            logger.info(f"Loaded {len(df)} records with {len(df.columns)} features")
            logger.info(f"Fraud rate: {df['Class'].mean() * 100:.4f}%")
            
            return df
            
        except Exception as e:
            logger.error(f"Error loading data: {e}")
            raise
    
    def preprocess_data(
        self, 
        df: pd.DataFrame,
        test_size: float = 0.2
    ) -> Tuple[np.ndarray, np.ndarray, np.ndarray, np.ndarray]:
        """Preprocess data with train-test split."""
        logger.info("Preprocessing data")
        
        # Separate features and target
        X = df.drop('Class', axis=1)
        y = df['Class']
        
        self.feature_names = X.columns.tolist()
        
        # Train-test split with stratification
        X_train, X_test, y_train, y_test = train_test_split(
            X, y,
            test_size=test_size,
            random_state=self.random_state,
            stratify=y
        )
        
        logger.info(f"Train set: {len(X_train)} samples, Fraud rate: {y_train.mean() * 100:.4f}%")
        logger.info(f"Test set: {len(X_test)} samples, Fraud rate: {y_test.mean() * 100:.4f}%")
        
        return X_train, X_test, y_train, y_test
    
    def train_model(
        self,
        X_train: np.ndarray,
        y_train: np.ndarray,
        use_smote: bool = True,
        n_estimators: int = 100,
        max_depth: int = 20,
        min_samples_split: int = 10,
        min_samples_leaf: int = 4,
        class_weight: str = 'balanced'
    ) -> ImbPipeline:
        """Train Random Forest model with SMOTE for handling imbalanced data."""
        logger.info("Training Random Forest model")
        
        try:
            # Create pipeline
            pipeline_steps = [
                ('scaler', StandardScaler())
            ]
            
            if use_smote:
                # SMOTE for handling imbalanced data
                pipeline_steps.append(
                    ('smote', SMOTE(random_state=self.random_state, k_neighbors=5))
                )
            
            pipeline_steps.append(
                ('classifier', RandomForestClassifier(
                    n_estimators=n_estimators,
                    max_depth=max_depth,
                    min_samples_split=min_samples_split,
                    min_samples_leaf=min_samples_leaf,
                    class_weight=class_weight,
                    random_state=self.random_state,
                    n_jobs=-1,
                    verbose=0
                ))
            )
            
            pipeline = ImbPipeline(pipeline_steps)
            
            # Train model
            pipeline.fit(X_train, y_train)
            
            self.model = pipeline
            logger.info("Model training completed")
            
            return pipeline
            
        except Exception as e:
            logger.error(f"Error training model: {e}")
            raise
    
    def evaluate_model(
        self,
        X_test: np.ndarray,
        y_test: np.ndarray
    ) -> Dict[str, Any]:
        """Evaluate model performance with comprehensive metrics."""
        logger.info("Evaluating model")
        
        try:
            # Predictions
            y_pred = self.model.predict(X_test)
            y_pred_proba = self.model.predict_proba(X_test)[:, 1]
            
            # Calculate metrics
            metrics = {
                'accuracy': float(self.model.score(X_test, y_test)),
                'precision': float(precision_score(y_test, y_pred)),
                'recall': float(recall_score(y_test, y_pred)),
                'f1_score': float(f1_score(y_test, y_pred)),
                'roc_auc': float(roc_auc_score(y_test, y_pred_proba))
            }
            
            # Precision-Recall AUC
            precision, recall, _ = precision_recall_curve(y_test, y_pred_proba)
            metrics['pr_auc'] = float(auc(recall, precision))
            
            # Confusion matrix
            cm = confusion_matrix(y_test, y_pred)
            metrics['confusion_matrix'] = {
                'tn': int(cm[0, 0]),
                'fp': int(cm[0, 1]),
                'fn': int(cm[1, 0]),
                'tp': int(cm[1, 1])
            }
            
            # Classification report
            metrics['classification_report'] = classification_report(
                y_test, y_pred, output_dict=True
            )
            
            # Feature importance
            rf_model = self.model.named_steps['classifier']
            feature_importance = dict(zip(
                self.feature_names,
                rf_model.feature_importances_
            ))
            metrics['feature_importance'] = {
                k: float(v) for k, v in 
                sorted(feature_importance.items(), key=lambda x: x[1], reverse=True)[:10]
            }
            
            logger.info(f"Model Metrics - F1: {metrics['f1_score']:.4f}, "
                       f"ROC-AUC: {metrics['roc_auc']:.4f}, "
                       f"PR-AUC: {metrics['pr_auc']:.4f}")
            
            return metrics
            
        except Exception as e:
            logger.error(f"Error evaluating model: {e}")
            raise
    
    def log_to_mlflow(
        self,
        metrics: Dict[str, Any],
        params: Dict[str, Any],
        model_path: str = "models/fraud_detector"
    ):
        """Log model, metrics, and artifacts to MLflow."""
        logger.info("Logging to MLflow")
        
        try:
            with mlflow.start_run(run_name=f"run_{datetime.now().strftime('%Y%m%d_%H%M%S')}"):
                # Log parameters
                mlflow.log_params(params)
                
                # Log metrics
                mlflow.log_metric("accuracy", metrics['accuracy'])
                mlflow.log_metric("precision", metrics['precision'])
                mlflow.log_metric("recall", metrics['recall'])
                mlflow.log_metric("f1_score", metrics['f1_score'])
                mlflow.log_metric("roc_auc", metrics['roc_auc'])
                mlflow.log_metric("pr_auc", metrics['pr_auc'])
                
                # Log confusion matrix
                mlflow.log_metric("true_positives", metrics['confusion_matrix']['tp'])
                mlflow.log_metric("true_negatives", metrics['confusion_matrix']['tn'])
                mlflow.log_metric("false_positives", metrics['confusion_matrix']['fp'])
                mlflow.log_metric("false_negatives", metrics['confusion_matrix']['fn'])
                
                # Log model
                mlflow.sklearn.log_model(
                    self.model,
                    "model",
                    registered_model_name=self.model_name
                )
                
                # Save and log artifacts
                os.makedirs("artifacts", exist_ok=True)
                
                # Save metrics as JSON
                with open("artifacts/metrics.json", 'w') as f:
                    json.dump(metrics, f, indent=2)
                mlflow.log_artifact("artifacts/metrics.json")
                
                # Save model locally
                os.makedirs(os.path.dirname(model_path), exist_ok=True)
                joblib.dump(self.model, f"{model_path}.pkl")
                mlflow.log_artifact(f"{model_path}.pkl")
                
                run_id = mlflow.active_run().info.run_id
                logger.info(f"MLflow run completed. Run ID: {run_id}")
                
                return run_id
                
        except Exception as e:
            logger.error(f"Error logging to MLflow: {e}")
            raise
    
    def run_pipeline(
        self,
        data_path: str,
        reference_stats_path: Optional[str] = None,
        force_retrain: bool = False,
        **model_params
    ) -> Dict[str, Any]:
        """Run the complete pipeline."""
        logger.info("="*80)
        logger.info("Starting Fraud Detection Pipeline")
        logger.info("="*80)
        
        pipeline_results = {
            'success': False,
            'timestamp': datetime.now().isoformat(),
            'drift_detected': False,
            'retrain_triggered': False
        }
        
        try:
            # 1. Load data
            df = self.load_data(data_path)
            
            # 2. Validate data quality
            validator = DataValidator(reference_stats_path)
            is_valid, issues = validator.validate_data_quality(df)
            
            if not is_valid:
                logger.warning(f"Data quality issues detected: {issues}")
                pipeline_results['data_quality_issues'] = issues
            
            # 3. Detect concept drift
            drift_detected, drift_features = validator.detect_concept_drift(df)
            pipeline_results['drift_detected'] = drift_detected
            
            if drift_detected:
                logger.warning(f"Concept drift detected in {len(drift_features)} features")
                pipeline_results['drift_features'] = drift_features
                pipeline_results['retrain_triggered'] = True
            
            # 4. Decide whether to retrain
            should_retrain = force_retrain or drift_detected
            
            if not should_retrain:
                logger.info("No retraining needed. Pipeline complete.")
                pipeline_results['success'] = True
                return pipeline_results
            
            logger.info("Retraining model...")
            
            # 5. Preprocess data
            X_train, X_test, y_train, y_test = self.preprocess_data(df)
            
            # 6. Train model
            default_params = {
                'use_smote': True,
                'n_estimators': 100,
                'max_depth': 20,
                'min_samples_split': 10,
                'min_samples_leaf': 4,
                'class_weight': 'balanced'
            }
            default_params.update(model_params)
            
            self.train_model(X_train, y_train, **default_params)
            
            # 7. Evaluate model
            metrics = self.evaluate_model(X_test, y_test)
            pipeline_results['metrics'] = metrics
            
            # 8. Log to MLflow
            run_id = self.log_to_mlflow(
                metrics=metrics,
                params=default_params
            )
            pipeline_results['mlflow_run_id'] = run_id
            
            # 9. Update reference statistics if retraining occurred
            if drift_detected:
                validator._save_current_stats(df)
            
            pipeline_results['success'] = True
            logger.info("="*80)
            logger.info("Pipeline completed successfully")
            logger.info("="*80)
            
            return pipeline_results
            
        except Exception as e:
            logger.error(f"Pipeline failed: {e}", exc_info=True)
            pipeline_results['error'] = str(e)
            raise


def main():
    """Main entry point for the pipeline."""
    import argparse
    
    parser = argparse.ArgumentParser(description='Fraud Detection Pipeline')
    parser.add_argument('--data-path', type=str, required=True,
                       help='Path to the fraud detection dataset')
    parser.add_argument('--reference-stats', type=str, default='reference_stats.json',
                       help='Path to reference statistics for drift detection')
    parser.add_argument('--force-retrain', action='store_true',
                       help='Force model retraining')
    parser.add_argument('--experiment-name', type=str, default='fraud_detection',
                       help='MLflow experiment name')
    parser.add_argument('--n-estimators', type=int, default=100,
                       help='Number of trees in Random Forest')
    parser.add_argument('--max-depth', type=int, default=20,
                       help='Maximum depth of trees')
    
    args = parser.parse_args()
    
    # Initialize pipeline
    pipeline = FraudDetectionPipeline(
        experiment_name=args.experiment_name,
        model_name="fraud_detector_rf"
    )
    
    # Run pipeline
    results = pipeline.run_pipeline(
        data_path=args.data_path,
        reference_stats_path=args.reference_stats,
        force_retrain=args.force_retrain,
        n_estimators=args.n_estimators,
        max_depth=args.max_depth
    )
    
    # Print results
    print("\n" + "="*80)
    print("PIPELINE RESULTS")
    print("="*80)
    print(json.dumps(results, indent=2, default=str))
    
    # Exit with appropriate code
    sys.exit(0 if results['success'] else 1)


if __name__ == "__main__":
    main()
