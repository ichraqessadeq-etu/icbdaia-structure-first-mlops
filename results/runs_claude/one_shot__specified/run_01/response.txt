```python
# fraud_detection_pipeline.py
"""
Production-ready fraud detection pipeline with MLflow tracking.
Handles imbalanced data, concept drift detection, and comprehensive logging.
"""

import os
import sys
import logging
import pickle
from typing import Tuple, Dict, Any
from datetime import datetime
from pathlib import Path

import numpy as np
import pandas as pd
from sklearn.ensemble import RandomForestClassifier
from sklearn.model_selection import train_test_split, StratifiedKFold, cross_val_score
from sklearn.metrics import (
    classification_report, 
    confusion_matrix, 
    roc_auc_score,
    precision_recall_curve,
    f1_score,
    precision_score,
    recall_score,
    average_precision_score
)
from sklearn.preprocessing import StandardScaler
from imblearn.over_sampling import SMOTE
from imblearn.pipeline import Pipeline as ImbPipeline
import mlflow
import mlflow.sklearn
from scipy.stats import ks_2samp
import joblib

# Configure logging
logging.basicConfig(
    level=logging.INFO,
    format='%(asctime)s - %(name)s - %(levelname)s - %(message)s',
    handlers=[
        logging.FileHandler('fraud_detection_pipeline.log'),
        logging.StreamHandler(sys.stdout)
    ]
)
logger = logging.getLogger(__name__)


class FraudDetectionPipeline:
    """Complete pipeline for fraud detection model training and evaluation."""
    
    def __init__(
        self,
        experiment_name: str = "fraud_detection",
        random_state: int = 42,
        test_size: float = 0.2,
        drift_threshold: float = 0.05
    ):
        """
        Initialize the fraud detection pipeline.
        
        Args:
            experiment_name: MLflow experiment name
            random_state: Random seed for reproducibility
            test_size: Proportion of data for testing
            drift_threshold: P-value threshold for drift detection
        """
        self.experiment_name = experiment_name
        self.random_state = random_state
        self.test_size = test_size
        self.drift_threshold = drift_threshold
        self.model = None
        self.scaler = None
        self.feature_columns = None
        
        # Set MLflow tracking URI
        mlflow_uri = os.getenv('MLFLOW_TRACKING_URI', 'file:./mlruns')
        mlflow.set_tracking_uri(mlflow_uri)
        mlflow.set_experiment(experiment_name)
        
        logger.info(f"Initialized FraudDetectionPipeline with experiment: {experiment_name}")
    
    def load_data(self, data_path: str) -> pd.DataFrame:
        """
        Load and validate fraud detection dataset.
        
        Args:
            data_path: Path to the CSV file
            
        Returns:
            Loaded DataFrame
            
        Raises:
            FileNotFoundError: If data file doesn't exist
            ValueError: If data validation fails
        """
        try:
            if not os.path.exists(data_path):
                raise FileNotFoundError(f"Data file not found: {data_path}")
            
            logger.info(f"Loading data from {data_path}")
            df = pd.read_csv(data_path)
            
            # Validate data
            if df.empty:
                raise ValueError("Loaded DataFrame is empty")
            
            if 'Class' not in df.columns:
                raise ValueError("Target column 'Class' not found in dataset")
            
            # Log dataset statistics
            fraud_rate = df['Class'].mean()
            logger.info(f"Data loaded successfully. Shape: {df.shape}")
            logger.info(f"Fraud rate: {fraud_rate:.4%}")
            logger.info(f"Number of frauds: {df['Class'].sum()}")
            
            # Check for missing values
            missing_values = df.isnull().sum().sum()
            if missing_values > 0:
                logger.warning(f"Dataset contains {missing_values} missing values")
                df = df.dropna()
                logger.info(f"Dropped missing values. New shape: {df.shape}")
            
            return df
            
        except Exception as e:
            logger.error(f"Error loading data: {str(e)}")
            raise
    
    def preprocess_data(
        self, 
        df: pd.DataFrame
    ) -> Tuple[np.ndarray, np.ndarray, np.ndarray, np.ndarray]:
        """
        Preprocess data and split into train/test sets.
        
        Args:
            df: Input DataFrame
            
        Returns:
            Tuple of (X_train, X_test, y_train, y_test)
        """
        try:
            logger.info("Starting data preprocessing")
            
            # Separate features and target
            X = df.drop('Class', axis=1)
            y = df['Class']
            
            # Store feature columns
            self.feature_columns = X.columns.tolist()
            
            # Split data with stratification
            X_train, X_test, y_train, y_test = train_test_split(
                X, y, 
                test_size=self.test_size, 
                random_state=self.random_state,
                stratify=y
            )
            
            logger.info(f"Train set size: {X_train.shape[0]}, Test set size: {X_test.shape[0]}")
            logger.info(f"Train fraud rate: {y_train.mean():.4%}")
            logger.info(f"Test fraud rate: {y_test.mean():.4%}")
            
            return X_train, X_test, y_train, y_test
            
        except Exception as e:
            logger.error(f"Error in preprocessing: {str(e)}")
            raise
    
    def detect_concept_drift(
        self, 
        X_reference: np.ndarray, 
        X_current: np.ndarray
    ) -> Dict[str, Any]:
        """
        Detect concept drift using Kolmogorov-Smirnov test.
        
        Args:
            X_reference: Reference dataset (e.g., training data)
            X_current: Current dataset (e.g., new production data)
            
        Returns:
            Dictionary with drift detection results
        """
        try:
            logger.info("Starting concept drift detection")
            
            drift_detected = False
            drift_features = []
            p_values = []
            
            n_features = X_reference.shape[1]
            
            for i in range(n_features):
                statistic, p_value = ks_2samp(
                    X_reference[:, i], 
                    X_current[:, i]
                )
                p_values.append(p_value)
                
                if p_value < self.drift_threshold:
                    drift_detected = True
                    feature_name = self.feature_columns[i] if self.feature_columns else f"feature_{i}"
                    drift_features.append(feature_name)
                    logger.warning(
                        f"Drift detected in {feature_name}: "
                        f"p-value={p_value:.6f}"
                    )
            
            drift_info = {
                'drift_detected': drift_detected,
                'drift_features': drift_features,
                'min_p_value': min(p_values),
                'mean_p_value': np.mean(p_values),
                'n_drifted_features': len(drift_features)
            }
            
            if drift_detected:
                logger.warning(
                    f"Concept drift detected in {len(drift_features)} features. "
                    f"Retraining recommended."
                )
            else:
                logger.info("No significant concept drift detected")
            
            return drift_info
            
        except Exception as e:
            logger.error(f"Error in drift detection: {str(e)}")
            raise
    
    def train_model(
        self, 
        X_train: np.ndarray, 
        y_train: np.ndarray,
        use_smote: bool = True,
        n_estimators: int = 100,
        max_depth: int = 10,
        min_samples_split: int = 10,
        class_weight: str = 'balanced'
    ) -> ImbPipeline:
        """
        Train Random Forest model with SMOTE for handling imbalanced data.
        
        Args:
            X_train: Training features
            y_train: Training labels
            use_smote: Whether to use SMOTE oversampling
            n_estimators: Number of trees in the forest
            max_depth: Maximum depth of trees
            min_samples_split: Minimum samples required to split
            class_weight: Class weight strategy
            
        Returns:
            Trained pipeline
        """
        try:
            logger.info("Starting model training")
            
            # Create pipeline steps
            steps = [('scaler', StandardScaler())]
            
            if use_smote:
                # Calculate SMOTE sampling strategy
                fraud_count = y_train.sum()
                normal_count = len(y_train) - fraud_count
                # Oversample minority class to 20% of majority class
                sampling_strategy = min(0.2, fraud_count / normal_count * 5)
                
                logger.info(f"Applying SMOTE with sampling_strategy={sampling_strategy:.4f}")
                steps.append(('smote', SMOTE(
                    sampling_strategy=sampling_strategy,
                    random_state=self.random_state
                )))
            
            # Add Random Forest classifier
            rf_classifier = RandomForestClassifier(
                n_estimators=n_estimators,
                max_depth=max_depth,
                min_samples_split=min_samples_split,
                class_weight=class_weight,
                random_state=self.random_state,
                n_jobs=-1,
                verbose=0
            )
            steps.append(('classifier', rf_classifier))
            
            # Create and train pipeline
            pipeline = ImbPipeline(steps)
            
            logger.info("Training model...")
            pipeline.fit(X_train, y_train)
            
            # Perform cross-validation
            logger.info("Performing cross-validation...")
            cv_scores = cross_val_score(
                pipeline, X_train, y_train,
                cv=StratifiedKFold(n_splits=5, shuffle=True, random_state=self.random_state),
                scoring='roc_auc',
                n_jobs=-1
            )
            
            logger.info(f"Cross-validation ROC-AUC: {cv_scores.mean():.4f} (+/- {cv_scores.std():.4f})")
            
            self.model = pipeline
            return pipeline
            
        except Exception as e:
            logger.error(f"Error in model training: {str(e)}")
            raise
    
    def evaluate_model(
        self, 
        X_test: np.ndarray, 
        y_test: np.ndarray
    ) -> Dict[str, Any]:
        """
        Evaluate model performance with comprehensive metrics.
        
        Args:
            X_test: Test features
            y_test: Test labels
            
        Returns:
            Dictionary with evaluation metrics
        """
        try:
            if self.model is None:
                raise ValueError("Model not trained. Call train_model() first.")
            
            logger.info("Evaluating model performance")
            
            # Get predictions
            y_pred = self.model.predict(X_test)
            y_pred_proba = self.model.predict_proba(X_test)[:, 1]
            
            # Calculate metrics
            metrics = {
                'accuracy': self.model.score(X_test, y_test),
                'precision': precision_score(y_test, y_pred, zero_division=0),
                'recall': recall_score(y_test, y_pred, zero_division=0),
                'f1_score': f1_score(y_test, y_pred, zero_division=0),
                'roc_auc': roc_auc_score(y_test, y_pred_proba),
                'average_precision': average_precision_score(y_test, y_pred_proba)
            }
            
            # Confusion matrix
            cm = confusion_matrix(y_test, y_pred)
            metrics['confusion_matrix'] = cm.tolist()
            metrics['true_negatives'] = int(cm[0, 0])
            metrics['false_positives'] = int(cm[0, 1])
            metrics['false_negatives'] = int(cm[1, 0])
            metrics['true_positives'] = int(cm[1, 1])
            
            # Calculate cost-sensitive metrics (assuming fraud costs more)
            cost_fp = 1  # Cost of false positive
            cost_fn = 100  # Cost of false negative (missing fraud)
            total_cost = (metrics['false_positives'] * cost_fp + 
                         metrics['false_negatives'] * cost_fn)
            metrics['total_cost'] = total_cost
            
            # Log metrics
            logger.info("Model Evaluation Results:")
            logger.info(f"  Accuracy: {metrics['accuracy']:.4f}")
            logger.info(f"  Precision: {metrics['precision']:.4f}")
            logger.info(f"  Recall: {metrics['recall']:.4f}")
            logger.info(f"  F1-Score: {metrics['f1_score']:.4f}")
            logger.info(f"  ROC-AUC: {metrics['roc_auc']:.4f}")
            logger.info(f"  Average Precision: {metrics['average_precision']:.4f}")
            logger.info(f"  Total Cost: {total_cost}")
            logger.info(f"  Confusion Matrix:\n{cm}")
            
            return metrics
            
        except Exception as e:
            logger.error(f"Error in model evaluation: {str(e)}")
            raise
    
    def log_to_mlflow(
        self, 
        metrics: Dict[str, Any],
        params: Dict[str, Any],
        model: Any,
        artifacts: Dict[str, str] = None
    ) -> str:
        """
        Log model, metrics, and artifacts to MLflow.
        
        Args:
            metrics: Dictionary of metrics to log
            params: Dictionary of parameters to log
            model: Trained model to log
            artifacts: Dictionary of artifact paths to log
            
        Returns:
            Run ID
        """
        try:
            logger.info("Logging to MLflow")
            
            with mlflow.start_run() as run:
                # Log parameters
                mlflow.log_params(params)
                
                # Log metrics (filter out non-numeric values)
                numeric_metrics = {
                    k: v for k, v in metrics.items() 
                    if isinstance(v, (int, float, np.number))
                }
                mlflow.log_metrics(numeric_metrics)
                
                # Log confusion matrix as JSON
                if 'confusion_matrix' in metrics:
                    mlflow.log_dict(
                        {'confusion_matrix': metrics['confusion_matrix']},
                        'confusion_matrix.json'
                    )
                
                # Log model
                mlflow.sklearn.log_model(
                    model,
                    "model",
                    registered_model_name="fraud_detection_model"
                )
                
                # Log feature importance if available
                if hasattr(model.named_steps['classifier'], 'feature_importances_'):
                    feature_importance = pd.DataFrame({
                        'feature': self.feature_columns,
                        'importance': model.named_steps['classifier'].feature_importances_
                    }).sort_values('importance', ascending=False)
                    
                    feature_importance.to_csv('feature_importance.csv', index=False)
                    mlflow.log_artifact('feature_importance.csv')
                    os.remove('feature_importance.csv')
                
                # Log additional artifacts
                if artifacts:
                    for artifact_name, artifact_path in artifacts.items():
                        if os.path.exists(artifact_path):
                            mlflow.log_artifact(artifact_path)
                
                # Add tags
                mlflow.set_tags({
                    'model_type': 'RandomForestClassifier',
                    'problem_type': 'fraud_detection',
                    'framework': 'scikit-learn',
                    'timestamp': datetime.now().isoformat()
                })
                
                run_id = run.info.run_id
                logger.info(f"Successfully logged to MLflow. Run ID: {run_id}")
                
                return run_id
                
        except Exception as e:
            logger.error(f"Error logging to MLflow: {str(e)}")
            raise
    
    def save_model(self, model_path: str = 'models/fraud_detection_model.pkl'):
        """
        Save trained model to disk.
        
        Args:
            model_path: Path to save the model
        """
        try:
            if self.model is None:
                raise ValueError("No model to save. Train a model first.")
            
            # Create directory if it doesn't exist
            Path(model_path).parent.mkdir(parents=True, exist_ok=True)
            
            # Save model
            joblib.dump(self.model, model_path)
            logger.info(f"Model saved to {model_path}")
            
        except Exception as e:
            logger.error(f"Error saving model: {str(e)}")
            raise
    
    def load_model(self, model_path: str = 'models/fraud_detection_model.pkl'):
        """
        Load trained model from disk.
        
        Args:
            model_path: Path to the saved model
        """
        try:
            if not os.path.exists(model_path):
                raise FileNotFoundError(f"Model file not found: {model_path}")
            
            self.model = joblib.load(model_path)
            logger.info(f"Model loaded from {model_path}")
            
        except Exception as e:
            logger.error(f"Error loading model: {str(e)}")
            raise
    
    def run_pipeline(
        self,
        data_path: str,
        reference_data_path: str = None,
        model_params: Dict[str, Any] = None
    ) -> Dict[str, Any]:
        """
        Run the complete pipeline: load, preprocess, train, evaluate, and log.
        
        Args:
            data_path: Path to the training data
            reference_data_path: Path to reference data for drift detection
            model_params: Dictionary of model hyperparameters
            
        Returns:
            Dictionary with pipeline results
        """
        try:
            logger.info("=" * 80)
            logger.info("Starting Fraud Detection Pipeline")
            logger.info("=" * 80)
            
            # Load data
            df = self.load_data(data_path)
            
            # Preprocess
            X_train, X_test, y_train, y_test = self.preprocess_data(df)
            
            # Detect drift if reference data is provided
            drift_info = None
            if reference_data_path:
                logger.info("Checking for concept drift...")
                df_reference = self.load_data(reference_data_path)
                X_reference = df_reference.drop('Class', axis=1).values
                
                drift_info = self.detect_concept_drift(X_reference, X_train.values)
                
                if not drift_info['drift_detected']:
                    logger.info("No drift detected. Skipping retraining.")
                    return {
                        'status': 'skipped',
                        'reason': 'no_drift_detected',
                        'drift_info': drift_info
                    }
            
            # Set default model parameters
            if model_params is None:
                model_params = {
                    'n_estimators': 100,
                    'max_depth': 10,
                    'min_samples_split': 10,
                    'class_weight': 'balanced',
                    'use_smote': True
                }
            
            # Train model
            model = self.train_model(X_train.values, y_train.values, **model_params)
            
            # Evaluate model
            metrics = self.evaluate_model(X_test.values, y_test.values)
            
            # Prepare parameters for logging
            params = {
                'test_size': self.test_size,
                'random_state': self.random_state,
                'train_samples': X_train.shape[0],
                'test_samples': X_test.shape[0],
                'n_features': X_train.shape[1],
                **model_params
            }
            
            if drift_info:
                params.update({
                    'drift_detected': drift_info['drift_detected'],
                    'n_drifted_features': drift_info['n_drifted_features']
                })
            
            # Log to MLflow
            run_id = self.log_to_mlflow(metrics, params, model)
            
            # Save model
            self.save_model()
            
            logger.info("=" * 80)
            logger.info("Pipeline completed successfully")
            logger.info("=" * 80)
            
            return {
                'status': 'success',
                'run_id': run_id,
                'metrics': metrics,
                'drift_info': drift_info
            }
            
        except Exception as e:
            logger.error(f"Pipeline failed: {str(e)}")
            raise


def main():
    """Main entry point for the pipeline."""
    import argparse
    
    parser = argparse.ArgumentParser(description='Fraud Detection Pipeline')
    parser.add_argument('--data-path', type=str, required=True,
                       help='Path to the training data CSV file')
    parser.add_argument('--reference-data-path', type=str, default=None,
                       help='Path to reference data for drift detection')
    parser.add_argument('--experiment-name', type=str, default='fraud_detection',
                       help='MLflow experiment name')
    parser.add_argument('--n-estimators', type=int, default=100,
                       help='Number of trees in Random Forest')
    parser.add_argument('--max-depth', type=int, default=10,
                       help='Maximum depth of trees')
    parser.add_argument('--test-size', type=float, default=0.2,
                       help='Test set size proportion')
    
    args = parser.parse_args()
    
    # Initialize pipeline
    pipeline = FraudDetectionPipeline(
        experiment_name=args.experiment_name,
        test_size=args.test_size
    )
    
    # Run pipeline
    model_params = {
        'n_estimators': args.n_estimators,
        'max_depth': args.max_depth,
        'use_smote': True
    }
    
    results = pipeline.run_pipeline(
        data_path=args.data_path,
        reference_data_path=args.reference_data_path,
        model_params=model_params
    )
    
    print("\nPipeline Results:")
    print(f"Status: {results['status']}")
    if results['status'] == 'success':
        print(f"Run ID: {results['run_id']}")
        print(f"ROC-AUC: {results['metrics']['roc_auc']:.4f}")
        print(f"F1-Score: {results['metrics']['f1_score']:.4f}")


if __name__ == '__main__':
    main()
```

```python
class TestDataPreprocessing:
    """Test suite for data preprocessing."""
    
    def test_preprocess_data_split(self, pipeline, sample_fraud_data):
        """Test train/test split with stratification."""
        X_train, X_test, y_train, y_test = pipeline.preprocess_data(sample_fraud_data)
        
        # Check shapes
        assert X_train.shape[0] + X_test.shape[0] == len(sample_fraud_data)
        assert len(y_train) == X_train.shape[0]
        assert len(y_test) == X_test.shape[0]
        
        # Check test size
        expected_test_size = int(len(sample_fraud_data) * pipeline.test_size)
        assert abs(len(X_test) - expected_test_size) <= 1
    
    def test_preprocess_data_stratification(self, pipeline, sample_fraud_data):
        """Test that stratification maintains class distribution."""
        X_train, X_test, y_train, y_test = pipeline.preprocess_data(sample_fraud_data)
        
        original_fraud_rate = sample_fraud_data['Class'].mean()
        train_fraud_rate = y_train.mean()
        test_fraud_rate = y_test.mean()
        
        # Fraud rates should be similar (within 50% relative difference)
```
