class ConceptDriftDetector:
    """Detects concept drift using statistical tests."""
    
    def __init__(self, reference_data: pd.DataFrame, alpha: float = 0.05):
        """
        Initialize drift detector.
        
        Args:
            reference_data: Reference dataset for comparison
            alpha: Significance level for statistical tests
        """
        self.reference_data = reference_data
        self.alpha = alpha
        self.reference_stats = self._compute_statistics(reference_data)
    
    def _compute_statistics(self, data: pd.DataFrame) -> Dict[str, Any]:
        """Compute statistical properties of data."""
        stats_dict = {}
        for col in data.select_dtypes(include=[np.number]).columns:
            stats_dict[col] = {
                'mean': data[col].mean(),
                'std': data[col].std(),
                'median': data[col].median(),
                'min': data[col].min(),
                'max': data[col].max()
            }
        return stats_dict
    
    def detect_drift(self, new_data: pd.DataFrame) -> Tuple[bool, Dict[str, Any]]:
        """
        Detect drift using Kolmogorov-Smirnov test.
        
        Args:
            new_data: New dataset to check for drift
            
        Returns:
            Tuple of (drift_detected, drift_report)
        """
        drift_detected = False
        drift_report = {}
        
        numeric_cols = new_data.select_dtypes(include=[np.number]).columns
        
        for col in numeric_cols:
            if col in self.reference_data.columns:
                # Perform KS test
                statistic, p_value = stats.ks_2samp(
                    self.reference_data[col].dropna(),
                    new_data[col].dropna()
                )
                
                drift_report[col] = {
                    'ks_statistic': statistic,
                    'p_value': p_value,
                    'drift_detected': p_value < self.alpha
                }
                
                if p_value < self.alpha:
                    drift_detected = True
                    logger.warning(f"Drift detected in column {col}: p-value={p_value:.4f}")
        
        return drift_detected, drift_report


class FraudDetectionPipeline:
    """Complete pipeline for fraud detection model."""
    
    def __init__(
        self,
        mlflow_tracking_uri: str = "sqlite:///mlflow.db",
        experiment_name: str = "fraud_detection",
        random_state: int = 42
    ):
        """
        Initialize pipeline.
        
        Args:
            mlflow_tracking_uri: MLflow tracking server URI
            experiment_name: Name of MLflow experiment
            random_state: Random seed for reproducibility
        """
        self.random_state = random_state
        self.mlflow_tracking_uri = mlflow_tracking_uri
        self.experiment_name = experiment_name
        self.model = None
        self.scaler = None
        self.performance_threshold = {
            'roc_auc': 0.85,
            'precision': 0.70,
            'recall': 0.75,
            'f1': 0.70
        }
        
        # Setup MLflow
        mlflow.set_tracking_uri(mlflow_tracking_uri)
        mlflow.set_experiment(experiment_name)
        
        logger.info(f"Pipeline initialized with experiment: {experiment_name}")
    
    def load_data(self, file_path: str) -> pd.DataFrame:
        """
        Load and validate data.
        
        Args:
            file_path: Path to data file (CSV)
            
        Returns:
            Loaded DataFrame
        """
        try:
            logger.info(f"Loading data from {file_path}")
            
            if not os.path.exists(file_path):
                raise FileNotFoundError(f"Data file not found: {file_path}")
            
            data = pd.read_csv(file_path)
            
            # Data quality checks
            self._validate_data(data)
            
            logger.info(f"Data loaded successfully. Shape: {data.shape}")
            logger.info(f"Fraud rate: {data['Class'].mean():.4%}")
            
            return data
            
        except Exception as e:
            logger.error(f"Error loading data: {str(e)}")
            raise
    
    def _validate_data(self, data: pd.DataFrame) -> None:
        """
        Validate data quality.
        
        Args:
            data: DataFrame to validate
        """
        # Check for required columns
        if 'Class' not in data.columns:
            raise DataQualityError("Target column 'Class' not found in data")
        
        # Check for missing values
        missing_pct = data.isnull().sum().sum() / (data.shape[0] * data.shape[1])
        if missing_pct > 0.1:
            raise DataQualityError(f"Too many missing values: {missing_pct:.2%}")
        
        # Check for sufficient fraud cases
        fraud_count = data['Class'].sum()
        if fraud_count < 10:
            raise DataQualityError(f"Insufficient fraud cases: {fraud_count}")
        
        # Check for data leakage (Time column should be monotonic or removed)
        if 'Time' in data.columns:
            logger.warning("Time column present - ensure no data leakage")
        
        logger.info("Data validation passed")
    
    def preprocess_data(
        self, 
        data: pd.DataFrame
    ) -> Tuple[np.ndarray, np.ndarray, np.ndarray, np.ndarray]:
        """
        Preprocess data for training.
        
        Args:
            data: Input DataFrame
            
        Returns:
            Tuple of (X_train, X_test, y_train, y_test)
        """
        logger.info("Preprocessing data")
        
        # Separate features and target
        X = data.drop('Class', axis=1)
        y = data['Class']
        
        # Remove Time column if present (potential data leakage)
        if 'Time' in X.columns:
            X = X.drop('Time', axis=1)
            logger.info("Removed Time column")
        
        # Split data with stratification
        X_train, X_test, y_train, y_test = train_test_split(
            X, y, 
            test_size=0.2, 
            random_state=self.random_state,
            stratify=y
        )
        
        logger.info(f"Train set size: {X_train.shape[0]}, Test set size: {X_test.shape[0]}")
        logger.info(f"Train fraud rate: {y_train.mean():.4%}, Test fraud rate: {y_test.mean():.4%}")
        
        return X_train, X_test, y_train, y_test
    
    def train_model(
        self,
        X_train: np.ndarray,
        y_train: np.ndarray,
        hyperparameters: Dict[str, Any] = None
    ) -> ImbPipeline:
        """
        Train fraud detection model with SMOTE for handling imbalance.
        
        Args:
            X_train: Training features
            y_train: Training labels
            hyperparameters: Model hyperparameters
            
        Returns:
            Trained pipeline
        """
        logger.info("Training model")
        
        if hyperparameters is None:
            hyperparameters = {
                'n_estimators': 100,
                'max_depth': 10,
                'min_samples_split': 10,
                'min_samples_leaf': 4,
                'max_features': 'sqrt',
                'class_weight': 'balanced',
                'random_state': self.random_state,
                'n_jobs': -1
            }
        
        # Create pipeline with SMOTE and RandomForest
        pipeline = ImbPipeline([
            ('scaler', StandardScaler()),
            ('smote', SMOTE(random_state=self.random_state, k_neighbors=5)),
            ('classifier', RandomForestClassifier(**hyperparameters))
        ])
        
        try:
            # Train model
            pipeline.fit(X_train, y_train)
            
            # Perform cross-validation
            cv_scores = self._cross_validate(pipeline, X_train, y_train)
            logger.info(f"Cross-validation ROC-AUC: {cv_scores['test_roc_auc'].mean():.4f} "
                       f"(+/- {cv_scores['test_roc_auc'].std() * 2:.4f})")
            
            self.model = pipeline
            return pipeline
            
        except Exception as e:
            logger.error(f"Error training model: {str(e)}")
            raise
    
    def _cross_validate(
        self, 
        pipeline: ImbPipeline, 
        X: np.ndarray, 
        y: np.ndarray,
        cv: int = 5
    ) -> Dict[str, np.ndarray]:
        """
        Perform cross-validation.
        
        Args:
            pipeline: Model pipeline
            X: Features
            y: Labels
            cv: Number of folds
            
        Returns:
            Cross-validation scores
        """
        cv_strategy = StratifiedKFold(n_splits=cv, shuffle=True, random_state=self.random_state)
        
        scoring = {
            'roc_auc': 'roc_auc',
            'precision': 'precision',
            'recall': 'recall',
            'f1': 'f1'
        }
        
        scores = cross_validate(
            pipeline, X, y,
            cv=cv_strategy,
            scoring=scoring,
            return_train_score=False,
            n_jobs=-1
        )
        
        return scores
    
    def evaluate_model(
        self,
        pipeline: ImbPipeline,
        X_test: np.ndarray,
        y_test: np.ndarray
    ) -> Dict[str, Any]:
        """
        Evaluate model performance.
        
        Args:
            pipeline: Trained pipeline
            X_test: Test features
            y_test: Test labels
            
        Returns:
            Dictionary of metrics
        """
        logger.info("Evaluating model")
        
        try:
            # Predictions
            y_pred = pipeline.predict(X_test)
            y_pred_proba = pipeline.predict_proba(X_test)[:, 1]
            
            # Calculate metrics
            metrics = {
                'roc_auc': roc_auc_score(y_test, y_pred_proba),
                'average_precision': average_precision_score(y_test, y_pred_proba),
                'precision': precision_score(y_test, y_pred),
                'recall': recall_score(y_test, y_pred),
                'f1': f1_score(y_test, y_pred),
                'confusion_matrix': confusion_matrix(y_test, y_pred).tolist(),
                'classification_report': classification_report(y_test, y_pred, output_dict=True)
            }
            
            # Log metrics
            logger.info(f"ROC-AUC: {metrics['roc_auc']:.4f}")
            logger.info(f"Average Precision: {metrics['average_precision']:.4f}")
            logger.info(f"Precision: {metrics['precision']:.4f}")
            logger.info(f"Recall: {metrics['recall']:.4f}")
            logger.info(f"F1 Score: {metrics['f1']:.4f}")
            
            # Check performance thresholds
            self._check_performance_thresholds(metrics)
            
            return metrics
            
        except Exception as e:
            logger.error(f"Error evaluating model: {str(e)}")
            raise
    
    def _check_performance_thresholds(self, metrics: Dict[str, Any]) -> None:
        """
        Check if model meets performance thresholds.
        
        Args:
            metrics: Model metrics
        """
        failed_metrics = []
        
        for metric, threshold in self.performance_threshold.items():
            if metric in metrics and metrics[metric] < threshold:
                failed_metrics.append(f"{metric}: {metrics[metric]:.4f} < {threshold}")
        
        if failed_metrics:
            error_msg = "Model performance below threshold:\n" + "\n".join(failed_metrics)
            logger.error(error_msg)
            raise ModelPerformanceError(error_msg)
        
        logger.info("Model meets all performance thresholds")
    
    def log_to_mlflow(
        self,
        pipeline: ImbPipeline,
        metrics: Dict[str, Any],
        hyperparameters: Dict[str, Any],
        X_train: np.ndarray,
        additional_info: Dict[str, Any] = None
    ) -> str:
        """
        Log model and metrics to MLflow.
        
        Args:
            pipeline: Trained pipeline
            metrics: Model metrics
            hyperparameters: Model hyperparameters
            X_train: Training data for signature
            additional_info: Additional information to log
            
        Returns:
            Run ID
        """
        logger.info("Logging to MLflow")
        
        try:
            with mlflow.start_run() as run:
                # Log parameters
                mlflow.log_params(hyperparameters)
                
                # Log metrics
                mlflow.log_metric("roc_auc", metrics['roc_auc'])
                mlflow.log_metric("average_precision", metrics['average_precision'])
                mlflow.log_metric("precision", metrics['precision'])
                mlflow.log_metric("recall", metrics['recall'])
                mlflow.log_metric("f1_score", metrics['f1'])
                
                # Log confusion matrix
                cm = metrics['confusion_matrix']
                mlflow.log_dict({'confusion_matrix': cm}, "confusion_matrix.json")
                
                # Log classification report
                mlflow.log_dict(
                    metrics['classification_report'], 
                    "classification_report.json"
                )
                
                # Log additional info
                if additional_info:
                    mlflow.log_params(additional_info)
                
                # Log model
                signature = mlflow.models.infer_signature(
                    X_train, 
                    pipeline.predict(X_train)
                )
                
                mlflow.sklearn.log_model(
                    pipeline,
                    "model",
                    signature=signature,
                    registered_model_name="fraud_detection_model"
                )
                
                # Log tags
                mlflow.set_tag("model_type", "RandomForestClassifier")
                mlflow.set_tag("framework", "scikit-learn")
                mlflow.set_tag("task", "fraud_detection")
                mlflow.set_tag("training_date", datetime.now().isoformat())
                
                run_id = run.info.run_id
                logger.info(f"Model logged to MLflow. Run ID: {run_id}")
                
                return run_id
                
        except Exception as e:
            logger.error(f"Error logging to MLflow: {str(e)}")
            raise
    
    def save_model_artifacts(
        self,
        pipeline: ImbPipeline,
        metrics: Dict[str, Any],
        output_dir: str = "model_artifacts"
    ) -> None:
        """
        Save model artifacts locally.
        
        Args:
            pipeline: Trained pipeline
            metrics: Model metrics
            output_dir: Output directory
        """
        os.makedirs(output_dir, exist_ok=True)
        
        # Save metrics
        metrics_file = os.path.join(output_dir, "metrics.json")
        with open(metrics_file, 'w') as f:
            # Convert numpy arrays to lists for JSON serialization
            serializable_metrics = {}
            for key, value in metrics.items():
                if isinstance(value, np.ndarray):
                    serializable_metrics[key] = value.tolist()
                elif isinstance(value, (np.integer, np.floating)):
                    serializable_metrics[key] = float(value)
                else:
                    serializable_metrics[key] = value
            json.dump(serializable_metrics, f, indent=2)
        
        logger.info(f"Model artifacts saved to {output_dir}")
    
    def run_pipeline(
        self,
        data_path: str,
        hyperparameters: Dict[str, Any] = None,
        save_artifacts: bool = True
    ) -> Tuple[ImbPipeline, Dict[str, Any], str]:
        """
        Run complete pipeline.
        
        Args:
            data_path: Path to training data
            hyperparameters: Model hyperparameters
            save_artifacts: Whether to save artifacts locally
            
        Returns:
            Tuple of (pipeline, metrics, run_id)
        """
        try:
            logger.info("=" * 80)
            logger.info("Starting Fraud Detection Pipeline")
            logger.info("=" * 80)
            
            # Load data
            data = self.load_data(data_path)
            
            # Preprocess
            X_train, X_test, y_train, y_test = self.preprocess_data(data)
            
            # Train
            pipeline = self.train_model(X_train, y_train, hyperparameters)
            
            # Evaluate
            metrics = self.evaluate_model(pipeline, X_test, y_test)
            
            # Log to MLflow
            additional_info = {
                'train_size': len(X_train),
                'test_size': len(X_test),
                'fraud_rate': float(y_train.mean()),
                'n_features': X_train.shape[1]
            }
            
            run_id = self.log_to_mlflow(
                pipeline, 
                metrics, 
                hyperparameters or {}, 
                X_train,
                additional_info
            )
            
            # Save artifacts
            if save_artifacts:
                self.save_model_artifacts(pipeline, metrics)
            
            logger.info("=" * 80)
            logger.info("Pipeline completed successfully")
            logger.info("=" * 80)
            
            return pipeline, metrics, run_id
            
        except Exception as e:
            logger.error(f"Pipeline failed: {str(e)}")
            raise


def main():
    """Main entry point for pipeline."""
    import argparse

    parser.add_argument(

        default="sqlite:///mlflow.db",

        "--experiment-name",

    )
    
    pipeline.run_pipeline(args.data_path)


if __name__ == "__main__":
    main()

        numeric_cols = drifted_data.select_dtypes(include=[np.number]).columns

        # Should detect drift
        assert drift_detected is True
        assert len(drift_report) > 0
