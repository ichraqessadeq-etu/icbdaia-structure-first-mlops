def test_split_data_stratification(sample_fraud_data):
    """Test that splitting maintains class distribution."""
    X_train, X_test, y_train, y_test = split_data(sample_fraud_data, test_size=0.2)
    
    original_fraud_rate = sample_fraud_data['is_fraud'].mean()
    train_fraud_rate = y_train.mean()
    test_fraud_rate = y_test.mean()
    
    # Check that fraud rates are similar (within 10% relative difference)
    assert abs(train_fraud_rate - original_fraud_rate) / original_fraud_rate < 0.2
    assert abs(test_fraud_rate - original_fraud_rate) / original_fraud_rate < 0.2


def test_split_data_random_state(sample_fraud_data):
    """Test that random state produces reproducible splits."""

    pd.testing.assert_frame_equal(X_train1, X_train2)

  evaluate-model:
    name: Evaluate Model
    runs-on: ubuntu-latest

    steps:

        run: |
          python -c "
          from src.pipeline import load_model, load_data, split_data
          import matplotlib.pyplot as plt
          import seaborn as sns
          from sklearn.metrics import confusion_matrix, roc_curve, auc, precision_recall_curve
          import numpy as np
          import json
          
          # Load model and data
          model, metadata = load_model('models/fraud_model')
          data = load_data('data/fraud_data.csv')
          
          # Prepare test data
          X_train, X_test, y_train, y_test = split_data(data, test_size=0.2, random_state=42)
          
          # Drop categorical columns for evaluation
          X_test_numeric = X_test.select_dtypes(include=[np.number])
          
          # Make predictions
          y_pred = model.predict(X_test_numeric)
          y_pred_proba = model.predict_proba(X_test_numeric)[:, 1]
          
          # Create evaluation plots directory
          import os
          os.makedirs('evaluation_plots', exist_ok=True)
          
          # 1. Confusion Matrix
          plt.figure(figsize=(8, 6))
          cm = confusion_matrix(y_test, y_pred)
          sns.heatmap(cm, annot=True, fmt='d', cmap='Blues')
          plt.title('Confusion Matrix')
          plt.ylabel('True Label')
          plt.xlabel('Predicted Label')
          plt.savefig('evaluation_plots/confusion_matrix.png')
          plt.close()
          
          # 2. ROC Curve
          plt.figure(figsize=(8, 6))
          fpr, tpr, _ = roc_curve(y_test, y_pred_proba)
          roc_auc = auc(fpr, tpr)
          plt.plot(fpr, tpr, color='darkorange', lw=2, label=f'ROC curve (AUC = {roc_auc:.2f})')
          plt.plot([0, 1], [0, 1], color='navy', lw=2, linestyle='--')
          plt.xlim([0.0, 1.0])
          plt.ylim([0.0, 1.05])
          plt.xlabel('False Positive Rate')
          plt.ylabel('True Positive Rate')
          plt.title('Receiver Operating Characteristic (ROC) Curve')
          plt.legend(loc='lower right')
          plt.savefig('evaluation_plots/roc_curve.png')
          plt.close()
          
          # 3. Precision-Recall Curve
          plt.figure(figsize=(8, 6))
          precision, recall, _ = precision_recall_curve(y_test, y_pred_proba)
          plt.plot(recall, precision, color='blue', lw=2)
          plt.xlabel('Recall')
          plt.ylabel('Precision')
          plt.title('Precision-Recall Curve')
          plt.savefig('evaluation_plots/precision_recall_curve.png')
          plt.close()
          
          print('✓ Evaluation plots generated')
          "
