## GitHub Actions Workflow Structure

```yaml
# .github/workflows/ml-pipeline.yml

name: ML Pipeline CI/CD

on:
  push:
    branches: [main, develop]
  pull_request:
    branches: [main]
  schedule:
    - cron: '0 0 * * 0'  # Weekly retraining

env:
  PYTHON_VERSION: '3.9'
  MODEL_REGISTRY: 's3://fraud-models'

jobs:
  
  code-quality:
    name: Code Quality Checks
    runs-on: ubuntu-latest
    steps:
      - name: Checkout code
      - name: Set up Python
      - name: Install dependencies
      - name: Run black formatter check
      - name: Run flake8 linter
      - name: Run isort import check
      - name: Run mypy type checking
      - name: Run bandit security scan

  unit-tests:
    name: Unit Tests
    runs-on: ubuntu-latest
    needs: code-quality
    steps:
      - name: Checkout code
      - name: Set up Python
      - name: Install dependencies
      - name: Run pytest with coverage
      - name: Upload coverage to Codecov
      - name: Check coverage threshold

  data-validation:
    name: Data Validation
    runs-on: ubuntu-latest
    needs: code-quality
    steps:
      - name: Checkout code
      - name: Set up Python
      - name: Install dependencies
      - name: Download latest training data
      - name: Validate data schema
      - name: Check data quality metrics
      - name: Detect data drift
      - name: Generate data report

  model-training:
    name: Model Training
    runs-on: ubuntu-latest
    needs: [unit-tests, data-validation]
    steps:
      - name: Checkout code
      - name: Set up Python
      - name: Install dependencies
      - name: Download training data
      - name: Run training pipeline
      - name: Log experiment to MLflow
      - name: Save model artifacts
      - name: Upload model to artifact storage

  model-evaluation:
    name: Model Evaluation
    runs-on: ubuntu-latest
    needs: model-training
    steps:
      - name: Checkout code
      - name: Set up Python
      - name: Download trained model
      - name: Download test data
      - name: Evaluate model performance
      - name: Validate performance thresholds
      - name: Compare with champion model
      - name: Generate evaluation report
      - name: Upload metrics

  model-testing:
    name: Model Testing
    runs-on: ubuntu-latest
    needs: model-training
    steps:
      - name: Checkout code
      - name: Set up Python
      - name: Download trained model
      - name: Run model inference tests
      - name: Test model serialization
      - name: Test prediction latency
      - name: Test batch prediction
      - name: Validate output format

  security-scan:
    name: Security Scanning
    runs-on: ubuntu-latest
    needs: code-quality
    steps:
      - name: Checkout code
      - name: Run Trivy vulnerability scan
      - name: Scan dependencies for vulnerabilities
      - name: Check for secrets in code
      - name: Generate security report

  build-container:
    name: Build Docker Container
    runs-on: ubuntu-latest
    needs: [model-evaluation, model-testing]
    if: github.ref == 'refs/heads/main'
    steps:
      - name: Checkout code
      - name: Set up Docker Buildx
      - name: Login to container registry
      - name: Download model artifacts
      - name: Build Docker image
      - name: Scan image for vulnerabilities
      - name: Push image to registry
      - name: Tag image with version

  deploy-staging:
    name: Deploy to Staging
    runs-on: ubuntu-latest
    needs: build-container
    environment: staging
    steps:
      - name: Checkout code
      - name: Configure AWS credentials
      - name: Download model from registry
      - name: Deploy to staging endpoint
      - name: Run smoke tests
      - name: Validate endpoint health
      - name: Run integration tests

  deploy-production:
    name: Deploy to Production
    runs-on: ubuntu-latest
    needs: deploy-staging
    environment: production
    if: github.ref == 'refs/heads/main'
    steps:
      - name: Checkout code
      - name: Configure AWS credentials
      - name: Download approved model
      - name: Deploy canary release
      - name: Monitor canary metrics
      - name: Promote to full production
      - name: Update model registry
      - name: Generate model card
      - name: Notify stakeholders

  monitoring-setup:
    name: Setup Monitoring
    runs-on: ubuntu-latest
    needs: deploy-production
    steps:
      - name: Checkout code
      - name: Configure monitoring dashboards
      - name: Set up performance alerts
      - name: Set up drift detection alerts
      - name: Configure logging
      - name: Verify monitoring endpoints
```
