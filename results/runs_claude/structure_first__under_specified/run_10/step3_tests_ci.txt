        """Test basic data splitting."""

        total_samples = len(train) + len(val) + len(test)
        assert total_samples == len(sample_fraud_data)

        """Test data splitting maintains class distribution."""
        train, val, test = split_data(sample_fraud_data)
        
        original_fraud_rate = sample_fraud_data['is_fraud'].mean()
        train_fraud_rate = train['is_fraud'].mean()
        test_fraud_rate = test['is_fraud'].mean()
        
        # Check fraud rates are similar (within 5%)
        assert abs(train_fraud_rate - original_fraud_rate) < 0.05
        assert abs(test_fraud_rate - original_fraud_rate) < 0.05

        X_train, y_train = handle_class_imbalance(X, y, method='smote')
