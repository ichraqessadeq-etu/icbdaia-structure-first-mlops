class TestDataSplitting:
    """Test suite for data splitting functionality."""
    
    def test_split_data_basic(self, sample_fraud_data):
        """Test basic data splitting."""
        train_df, val_df, test_df = split_data(sample_fraud_data)
        
        # Check all splits are DataFrames
        assert isinstance(train_df, pd.DataFrame)
        assert isinstance(val_df, pd.DataFrame)
        assert isinstance(test_df, pd.DataFrame)
        
        # Check total size
        total_size = len(train_df) + len(val_df) + len(test_df)
        assert total_size == len(sample_fraud_data)
    
    def test_split_data_proportions(self, sample_fraud_data):
        """Test data split proportions are correct."""
        test_size = 0.2
        val_size = 0.1
        
        train_df, val_df, test_df = split_data(
            sample_fraud_data, 
            test_size=test_size,
            validation_size=val_size
        )
        
        total = len(sample_fraud_data)
        
        # Check approximate proportions (allow small variance due to rounding)
        assert abs(len(test_df) / total - test_size) < 0.02
        assert abs(len(val_df) / total - val_size) < 0.02
    
    def test_split_data_stratification(self, sample_fraud_data):
        """Test stratification maintains class distribution."""
        train_df, val_df, test_df = split_data(sample_fraud_data)
        
        original_fraud_rate = sample_fraud_data['is_fraud'].mean()
        
        # Check fraud rates are similar across splits
        train_fraud_rate = train_df['is_fraud'].mean()
        val_fraud_rate = val_df['is_fraud'].mean()
        test_fraud_rate = test_df['is_fraud'].mean()
        
        # Allow 2% variance
        assert abs(train_fraud_rate - original_fraud_rate) < 0.02
        assert abs(val_fraud_rate - original_fraud_rate) < 0.02
        assert abs(test_fraud_rate - original_fraud_rate) < 0.02
    
    def test_split_data_reproducibility(self, sample_fraud_data):
        """Test splitting is reproducible with same random state."""
        train1, val1, test1 = split_data(sample_fraud_data, random_state=42)
        train2, val2, test2 = split_data(sample_fraud_data, random_state=42)
        
        pd.testing.assert_frame_equal(train1, train2)
        pd.testing.assert_frame_equal(val1, val2)
        pd.testing.assert_frame_equal(test1, test2)
    
    def test_split_data_no_target_column(self):

class TestDataDrift:
    """Test suite for data drift detection."""
    
    def test_check_drift_no_drift(self):
        """Test drift detection when no drift exists."""
        np.random.seed(42)
        ref_data = pd.DataFrame({
            'feature1': np.random.normal(0, 1, 1000),
            'feature2': np.random.normal(5, 2, 1000)
        })
        
        curr_data = pd.DataFrame({
            'feature1': np.random.normal(0, 1, 1000),
            'feature2': np.random.normal(5, 2, 1000)
        })
        
        drift_results = check_data_drift(ref_data, curr_data, threshold=0.05)
        
        assert drift_results['drift_detected'] is False
        assert len(drift_results['drifted_features']) == 0
    
    def test_check_drift_with_drift(self):
        """Test drift detection when drift exists."""
        np.random.seed(42)
        ref_data = pd.DataFrame({
            'feature1': np.random.normal(0, 1, 1000),
            'feature2': np.random.normal(5, 2, 1000)
        })
        
        # Current data has shifted distribution
        curr_data = pd.DataFrame({
            'feature1': np.random.normal(3, 1, 1000),  # Mean shifted
            'feature2': np.random.normal(5, 2, 1000)
        })
        
        drift_results = check_data_drift(ref_data, curr_data, threshold=0.05)
        
        assert drift_results['drift_detected'] is True
        assert 'feature1' in drift_results['drifted_features']
    
    def test_check_drift_statistics(self):
        """Test drift detection includes statistical information."""
        np.random.seed(42)
        ref_data = pd.DataFrame({'feature1': np.random.normal(0, 1, 1000)})
        curr_data = pd.DataFrame({'feature1': np.random.normal(0, 1, 1000)})
        
        drift_results = check_data_drift(ref_data, curr_data)
        
        assert 'feature_statistics' in drift_results
        assert 'feature1' in drift_results['feature_statistics']
        
        stats = drift_results['feature_statistics']['feature1']
        assert 'ks_statistic' in stats
        assert 'p_value' in stats
        assert 'reference_mean' in stats
        assert 'current_mean' in stats
    
    def test_check_drift_handles_missing_columns(self):
        """Test drift detection handles missing columns gracefully."""
        ref_data = pd.DataFrame({
            'feature1': np.random.normal(0, 1, 100),
            'feature2': np.random.normal(0, 1, 100)
        })
        
        curr_data = pd.DataFrame({
            'feature1': np.random.normal(0, 1, 100)
            # feature2 missing
        })
        
        drift_results = check_data_drift(ref_data, curr_data)
        
        # Should only check common columns
        assert 'feature1' in drift_results['feature_statistics']
        assert 'feature2' not in drift_results['feature_statistics']

        model_path = temp_data_dir / "test_model"
        metadata = {'version': '1.0'}
        
        # Make predictions before saving
        X_test = np.random.rand(10, 5)
        predictions_before = sample_model.predict(X_test)
        
        # Save and load
        save_model(sample_model, model_path, metadata)

        # Make predictions after loading
        predictions_after = loaded_model.predict(X_test)
        
        # Predictions should be identical
        np.testing.assert_array_equal(predictions_before, predictions_after)

          # For demo, we'll skip if no sensitive features
          sensitive_features = []
          
          if sensitive_features:
              bias_metrics = check_model_bias(model, X_test, y_test, sensitive_features)
              
              with open('bias_report.json', 'w') as f:
                  json.dump(bias_metrics, f, indent=2)
              
              print('Bias evaluation completed')
              print(json.dumps(bias_metrics, indent=2))
          else:
              print('No sensitive features defined for bias checking')
              with open('bias_report.json', 'w') as f:
                  json.dump({'message': 'No sensitive features'}, f)
          "

      - name: Upload bias report
        uses: actions/upload-artifact@v3
        with:
          name: bias-report
          path: bias_report.json
          retention-days: 30

  drift-detection:
    name: Data Drift Detection
    runs-on: ubuntu-latest
    needs: model-training
    if: github.event_name == 'schedule' || github.event_name == 'workflow_dispatch'
    steps:

        uses: actions/checkout@v3

          pip install -r requirements.txt

      - name: Download reference data
        run: |
          mkdir -p data
          # In production, download reference data from storage
          # For demo, create reference data
          python -c "
          import pandas as pd
          import numpy as np
          np.random.seed(42)
          n = 5000
          df = pd.DataFrame({
              'amount': np.random.exponential(100, n),
              'time': np.random.randint(0, 172800, n),
              'customer_age': np.random.randint(18, 80, n),
              'is_fraud': np.random.choice([0, 1], n, p=[0.95, 0.05])
          })
          df.to_csv('data/reference_data.csv', index=False)
          "

      - name: Download current data
        run: |
          # In production, download latest production data
          # For demo, create slightly drifted data
          python -c "
          import pandas as pd
          import numpy as np
          np.random.seed(123)
          n = 5000
          df = pd.DataFrame({
              'amount': np.random.exponential(110, n),  # Slight drift
              'time': np.random.randint(0, 172800, n),
              'customer_age': np.random.randint(18, 80, n),
              'is_fraud': np.random.choice([0, 1], n, p=[0.94, 0.06])
          })
          df.to_csv('data/current_data.csv', index=False)
          "

      - name: Detect data drift
        id: drift
        run: |
          python -c "
          from pipeline import load_data, check_data_drift
          import json
          
          reference_data = load_data('data/reference_data.csv')
          current_data = load_data('data/current_data.csv')
          
          drift_results = check_data_drift(reference_data, current_data, threshold=0.05)
          
          with open('drift_report.json', 'w') as f:
              json.dump(drift_results, f, indent=2)
          
          print('Drift detection completed')
          print(json.dumps(drift_results, indent=2))
          
          # Set output
          with open('$GITHUB_OUTPUT', 'a') as f:
              f.write(f'drift_detected={str(drift_results[\"drift_detected\"]).lower()}\n')
              f.write(f'num_drifted_features={len(drift_results[\"drifted_features\"])}\n')
          "

      - name: Upload drift report
        uses: actions/upload-artifact@v3
        with:
          name: drift-report
          path: drift_report.json
          retention-days: 30

      - name: Create drift alert
        if: steps.drift.outputs.drift_detected == 'true'
        uses: actions/github-script@v6
        with:
          script: |
            const fs = require('fs');
            const drift = JSON.parse(fs.readFileSync('drift_report.json', 'utf8'));
            
            const issue_body = `
            # ⚠️ Data Drift Detected
            
            Data drift has been detected in the production data.
            
            **Drifted Features**: ${drift.drifted_features.join(', ')}
            
            **Action Required**: Review the drift report and consider retraining the model.
            
            See attached drift report for detailed statistics.
            `;
            
            github.rest.issues.create({
              owner: context.repo.owner,
              repo: context.repo.repo,
              title: '[ALERT] Data Drift Detected',
              body: issue_body,
              labels: ['data-drift', 'monitoring', 'high-priority']
            });
