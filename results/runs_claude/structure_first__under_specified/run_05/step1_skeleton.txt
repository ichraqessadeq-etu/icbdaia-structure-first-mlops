## GitHub Actions Workflow (`.github/workflows/ml-pipeline.yml`)

```yaml
name: ML Pipeline CI/CD

on:
  push:
    branches: [main, develop]
  pull_request:
    branches: [main]
  schedule:
    - cron: '0 0 * * 0'  # Weekly retraining

jobs:
  # Job 1: Code Quality & Linting
  code-quality:
    steps:
      - Checkout code
      - Setup Python
      - Install dependencies
      - Run black formatter check
      - Run flake8 linter
      - Run mypy type checking
      - Run bandit security scan

  # Job 2: Unit & Integration Tests
  test:
    steps:
      - Checkout code
      - Setup Python
      - Install dependencies
      - Run pytest with coverage
      - Upload coverage reports
      - Generate test report artifacts

  # Job 3: Data Validation
  data-validation:
    steps:
      - Checkout code
      - Setup Python & dependencies
      - Download latest data
      - Validate data schema
      - Check for data quality issues
      - Detect data drift
      - Generate data validation report

  # Job 4: Model Training
  train-model:
    needs: [code-quality, test, data-validation]
    steps:
      - Checkout code
      - Setup Python & dependencies
      - Load and preprocess data
      - Train model
      - Save model artifacts
      - Upload model to artifact storage

  # Job 5: Model Evaluation
  evaluate-model:
    needs: train-model
    steps:
      - Checkout code
      - Setup Python & dependencies
      - Download trained model
      - Load test dataset
      - Evaluate model performance
      - Check performance thresholds
      - Run bias detection
      - Generate evaluation report

  # Job 6: Model Validation & Testing
  model-validation:
    needs: evaluate-model
    steps:
      - Checkout code
      - Setup Python & dependencies
      - Download model artifacts
      - Run model unit tests
      - Test model inference
      - Validate prediction latency
      - Check model size constraints
      - Verify serialization/deserialization

  # Job 7: Security & Compliance Scan
  security-scan:
    needs: model-validation
    steps:
      - Checkout code
      - Scan dependencies for vulnerabilities
      - Check model for adversarial robustness
      - Validate compliance requirements
      - Generate security report

  # Job 8: Build & Package Model
  build-package:
    needs: [model-validation, security-scan]
    steps:
      - Checkout code
      - Download model artifacts
      - Generate model card
      - Create model package
      - Build Docker image
      - Tag image with version
      - Scan Docker image

  # Job 9: Deploy to Staging
  deploy-staging:
    needs: build-package
    if: github.ref == 'refs/heads/develop'
    steps:
      - Checkout code
      - Setup cloud CLI
      - Push Docker image to registry
      - Deploy to staging environment
      - Run smoke tests
      - Validate deployment health

  # Job 10: Production Deployment
  deploy-production:
    needs: build-package
    if: github.ref == 'refs/heads/main'
    environment: production
    steps:
      - Checkout code
      - Setup cloud CLI
      - Push Docker image to registry
      - Deploy with blue-green strategy
      - Run integration tests
      - Monitor deployment metrics
      - Rollback on failure

  # Job 11: Post-Deployment Monitoring
  post-deployment:
    needs: [deploy-staging, deploy-production]
    if: always()
    steps:
      - Setup monitoring alerts
      - Configure model performance tracking
      - Enable prediction logging
      - Create dashboard links
      - Send notification to team
```
