def load_data(data_path: Path, version: Optional[str] = None) -> pd.DataFrame:
    """Load fraud detection dataset from specified path.
    
    Args:
        data_path: Path to the data file (CSV, parquet, etc.)
        version: Optional data version identifier for DVC
        
    Returns:
        DataFrame containing the loaded data
        
    Raises:
        FileNotFoundError: If data file doesn't exist
        ValueError: If data format is unsupported
    """
    logger.info(f"Loading data from {data_path}")
    
    if version:
        logger.info(f"Requesting data version: {version}")
        # In production, this would checkout specific DVC version
        # subprocess.run(["dvc", "checkout", str(data_path), "--rev", version])
    
    if not data_path.exists():
        raise FileNotFoundError(f"Data file not found: {data_path}")
    
    # Support multiple file formats
    suffix = data_path.suffix.lower()
    
    if suffix == '.csv':
        df = pd.read_csv(data_path)
    elif suffix == '.parquet':
        df = pd.read_parquet(data_path)
    elif suffix == '.json':
        df = pd.read_json(data_path)
    else:
        raise ValueError(f"Unsupported file format: {suffix}")
    
    logger.info(f"Loaded {len(df)} records with {len(df.columns)} columns")
    return df


def validate_data(df: pd.DataFrame) -> Tuple[bool, Dict[str, Any]]:
    """Validate data schema, quality, and detect data drift.
    
    Args:
        df: DataFrame to validate
        
    Returns:
        Tuple of (is_valid, validation_report)
    """
    logger.info("Validating data quality and schema")
    
    validation_report = {
        "timestamp": datetime.now().isoformat(),
        "total_records": len(df),
        "total_columns": len(df.columns),
        "issues": [],
        "warnings": [],
        "statistics": {}
    }
    
    is_valid = True
    
    # Required columns for fraud detection
    required_columns = ['amount', 'is_fraud']
    missing_columns = [col for col in required_columns if col not in df.columns]
    
    if missing_columns:
        validation_report["issues"].append(
            f"Missing required columns: {missing_columns}"
        )
        is_valid = False
    
    # Check for excessive null values
    null_percentages = (df.isnull().sum() / len(df) * 100).to_dict()
    validation_report["statistics"]["null_percentages"] = null_percentages
    
    for col, null_pct in null_percentages.items():
        if null_pct > 50:
            validation_report["issues"].append(
                f"Column '{col}' has {null_pct:.2f}% null values"
            )
            is_valid = False
        elif null_pct > 20:
            validation_report["warnings"].append(
                f"Column '{col}' has {null_pct:.2f}% null values"
            )
    
    # Check data types
    validation_report["statistics"]["dtypes"] = df.dtypes.astype(str).to_dict()
    
    # Check for duplicate records
    duplicate_count = df.duplicated().sum()
    validation_report["statistics"]["duplicate_records"] = int(duplicate_count)
    
    if duplicate_count > len(df) * 0.1:
        validation_report["warnings"].append(
            f"Found {duplicate_count} duplicate records ({duplicate_count/len(df)*100:.2f}%)"
        )
    
    # Validate fraud label distribution
    if 'is_fraud' in df.columns:
        fraud_distribution = df['is_fraud'].value_counts().to_dict()
        validation_report["statistics"]["fraud_distribution"] = fraud_distribution
        
        fraud_rate = fraud_distribution.get(1, 0) / len(df)
        validation_report["statistics"]["fraud_rate"] = fraud_rate
        
        if fraud_rate < 0.001:
            validation_report["warnings"].append(
                f"Very low fraud rate: {fraud_rate*100:.4f}%"
            )
        elif fraud_rate > 0.5:
            validation_report["warnings"].append(
                f"Unusually high fraud rate: {fraud_rate*100:.2f}%"
            )
    
    # Check for data drift using statistical tests
    if 'amount' in df.columns:
        amount_stats = {
            "mean": float(df['amount'].mean()),
            "median": float(df['amount'].median()),
            "std": float(df['amount'].std()),
            "min": float(df['amount'].min()),
            "max": float(df['amount'].max())
        }
        validation_report["statistics"]["amount_stats"] = amount_stats
        
        # Check for outliers
        q1 = df['amount'].quantile(0.25)
        q3 = df['amount'].quantile(0.75)
        iqr = q3 - q1
        outliers = ((df['amount'] < (q1 - 3 * iqr)) | 
                   (df['amount'] > (q3 + 3 * iqr))).sum()
        
        validation_report["statistics"]["amount_outliers"] = int(outliers)
    
    validation_report["is_valid"] = is_valid
    
    logger.info(f"Validation complete. Valid: {is_valid}, "
                f"Issues: {len(validation_report['issues'])}, "
                f"Warnings: {len(validation_report['warnings'])}")
    
    return is_valid, validation_report


def preprocess_data(df: pd.DataFrame) -> Tuple[pd.DataFrame, Dict[str, Any]]:
    """Apply feature engineering and preprocessing transformations.
    
    Args:
        df: Raw DataFrame to preprocess
        
    Returns:
        Tuple of (preprocessed_df, preprocessing_metadata)
    """
    logger.info("Starting data preprocessing and feature engineering")
    
    df_processed = df.copy()
    metadata = {
        "timestamp": datetime.now().isoformat(),
        "original_shape": df.shape,
        "transformations": [],
        "feature_names": [],
        "encoders": {}
    }
    
    # Handle missing values
    numeric_columns = df_processed.select_dtypes(include=[np.number]).columns
    categorical_columns = df_processed.select_dtypes(include=['object']).columns
    
    # Fill numeric missing values with median
    for col in numeric_columns:
        if df_processed[col].isnull().any():
            median_value = df_processed[col].median()
            df_processed[col].fillna(median_value, inplace=True)
            metadata["transformations"].append(
                f"Filled {col} nulls with median: {median_value}"
            )
    
    # Fill categorical missing values with mode
    for col in categorical_columns:
        if df_processed[col].isnull().any():
            mode_value = df_processed[col].mode()[0] if not df_processed[col].mode().empty else "UNKNOWN"
            df_processed[col].fillna(mode_value, inplace=True)
            metadata["transformations"].append(
                f"Filled {col} nulls with mode: {mode_value}"
            )
    
    # Feature engineering for fraud detection
    if 'amount' in df_processed.columns:
        # Log transform for amount (handle skewness)
        df_processed['amount_log'] = np.log1p(df_processed['amount'])
        metadata["transformations"].append("Created amount_log feature")
        
        # Amount bins
        df_processed['amount_bin'] = pd.cut(
            df_processed['amount'],
            bins=[0, 50, 100, 500, 1000, np.inf],
            labels=['very_low', 'low', 'medium', 'high', 'very_high']
        )
        metadata["transformations"].append("Created amount_bin feature")
    
    # Time-based features if timestamp exists
    if 'timestamp' in df_processed.columns:
        df_processed['timestamp'] = pd.to_datetime(df_processed['timestamp'])
        df_processed['hour'] = df_processed['timestamp'].dt.hour
        df_processed['day_of_week'] = df_processed['timestamp'].dt.dayofweek
        df_processed['is_weekend'] = df_processed['day_of_week'].isin([5, 6]).astype(int)
        df_processed['is_night'] = df_processed['hour'].isin(range(0, 6)).astype(int)
        
        metadata["transformations"].append("Created time-based features")
    
    # Encode categorical variables
    label_encoders = {}
    for col in categorical_columns:
        if col in df_processed.columns and col != 'is_fraud':
            le = LabelEncoder()
            df_processed[f'{col}_encoded'] = le.fit_transform(df_processed[col].astype(str))
            label_encoders[col] = {
                "classes": le.classes_.tolist()
            }
            metadata["transformations"].append(f"Label encoded {col}")
    
    metadata["encoders"]["label_encoders"] = label_encoders
    
    # Create interaction features
    if 'amount' in df_processed.columns and 'hour' in df_processed.columns:
        df_processed['amount_hour_interaction'] = (
            df_processed['amount'] * df_processed['hour']
        )
        metadata["transformations"].append("Created amount_hour_interaction")
    
    # Remove original categorical columns (keep encoded versions)
    cols_to_drop = [col for col in categorical_columns if col != 'is_fraud']
    df_processed.drop(columns=cols_to_drop, inplace=True, errors='ignore')
    
    # Remove timestamp if exists (already extracted features)
    if 'timestamp' in df_processed.columns:
        df_processed.drop(columns=['timestamp'], inplace=True)
    
    metadata["final_shape"] = df_processed.shape
    metadata["feature_names"] = df_processed.columns.tolist()
    
    logger.info(f"Preprocessing complete. Shape: {df.shape} -> {df_processed.shape}")
    logger.info(f"Applied {len(metadata['transformations'])} transformations")
    
    return df_processed, metadata


def split_data(
    df: pd.DataFrame, 
    test_size: float = 0.2, 
    stratify_column: str = "is_fraud"
) -> Tuple[pd.DataFrame, pd.DataFrame, pd.DataFrame, pd.DataFrame]:
    """Split data into train and test sets with stratification.
    
    Args:
        df: DataFrame to split
        test_size: Proportion of data for test set
        stratify_column: Column to use for stratification
        
    Returns:
        Tuple of (X_train, X_test, y_train, y_test)
    """
    logger.info(f"Splitting data with test_size={test_size}, stratify={stratify_column}")
    
    if stratify_column not in df.columns:
        raise ValueError(f"Stratify column '{stratify_column}' not found in DataFrame")
    
    # Separate features and target
    y = df[stratify_column]
    X = df.drop(columns=[stratify_column])
    
    # Perform stratified split
    X_train, X_test, y_train, y_test = train_test_split(
        X, y,
        test_size=test_size,
        random_state=42,
        stratify=y
    )
    
    logger.info(f"Train set: {len(X_train)} samples")
    logger.info(f"Test set: {len(X_test)} samples")
    logger.info(f"Train fraud rate: {y_train.mean():.4f}")
    logger.info(f"Test fraud rate: {y_test.mean():.4f}")
    
    return X_train, X_test, y_train, y_test


def train_model(
    X_train: pd.DataFrame, 
    y_train: pd.DataFrame, 
    hyperparameters: Dict[str, Any]
) -> Any:
    """Train fraud detection model with given hyperparameters.
    
    Args:
        X_train: Training features
        y_train: Training labels
        hyperparameters: Model hyperparameters
        
    Returns:
        Trained model object
    """
    logger.info("Starting model training")
    logger.info(f"Hyperparameters: {hyperparameters}")
    
    # Default hyperparameters for fraud detection
    default_params = {
        'n_estimators': 100,
        'max_depth': 10,
        'min_samples_split': 5,
        'min_samples_leaf': 2,
        'class_weight': 'balanced',  # Important for imbalanced fraud data
        'random_state': 42,
        'n_jobs': -1
    }
    
    # Merge with provided hyperparameters
    model_params = {**default_params, **hyperparameters}
    
    # Initialize and train model
    model = RandomForestClassifier(**model_params)
    
    logger.info(f"Training on {len(X_train)} samples with {X_train.shape[1]} features")
    model.fit(X_train, y_train)
    
    # Log feature importances
    feature_importance = pd.DataFrame({
        'feature': X_train.columns,
        'importance': model.feature_importances_
    }).sort_values('importance', ascending=False)
    
    logger.info("Top 5 most important features:")
    for idx, row in feature_importance.head(5).iterrows():
        logger.info(f"  {row['feature']}: {row['importance']:.4f}")
    
    return model


def evaluate_model(
    model: Any, 
    X_test: pd.DataFrame, 
    y_test: pd.DataFrame
) -> Dict[str, float]:
    """Evaluate model performance with fraud-specific metrics.
    
    Args:
        model: Trained model to evaluate
        X_test: Test features
        y_test: Test labels
        
    Returns:
        Dictionary of evaluation metrics
    """
    logger.info("Evaluating model performance")
    
    # Generate predictions
    y_pred = model.predict(X_test)
    y_pred_proba = model.predict_proba(X_test)[:, 1]
    
    # Calculate metrics
    metrics = {
        'accuracy': float(model.score(X_test, y_test)),
        'precision': float(precision_score(y_test, y_pred, zero_division=0)),
        'recall': float(recall_score(y_test, y_pred, zero_division=0)),
        'f1_score': float(f1_score(y_test, y_pred, zero_division=0)),
        'roc_auc': float(roc_auc_score(y_test, y_pred_proba)),
        'average_precision': float(average_precision_score(y_test, y_pred_proba))
    }
    
    # Confusion matrix
    tn, fp, fn, tp = confusion_matrix(y_test, y_pred).ravel()
    metrics.update({
        'true_negatives': int(tn),
        'false_positives': int(fp),
        'false_negatives': int(fn),
        'true_positives': int(tp),
        'false_positive_rate': float(fp / (fp + tn)) if (fp + tn) > 0 else 0.0,
        'false_negative_rate': float(fn / (fn + tp)) if (fn + tp) > 0 else 0.0
    })
    
    # Fraud-specific metrics
    metrics['fraud_detection_rate'] = metrics['recall']  # Same as recall
    metrics['precision_at_1pct_fpr'] = float(_precision_at_fpr(y_test, y_pred_proba, 0.01))
    
    logger.info("Evaluation metrics:")
    for metric, value in metrics.items():
        if isinstance(value, float):
            logger.info(f"  {metric}: {value:.4f}")
        else:
            logger.info(f"  {metric}: {value}")
    
    return metrics


def _precision_at_fpr(y_true: np.ndarray, y_scores: np.ndarray, target_fpr: float) -> float:
    """Calculate precision at a specific false positive rate."""
    from sklearn.metrics import roc_curve
    
    fpr, tpr, thresholds = roc_curve(y_true, y_scores)
    
    # Find threshold closest to target FPR
    idx = np.argmin(np.abs(fpr - target_fpr))
    threshold = thresholds[idx]
    
    # Calculate precision at this threshold
    y_pred = (y_scores >= threshold).astype(int)
    precision = precision_score(y_true, y_pred, zero_division=0)
    
    return precision


def validate_model_performance(
    metrics: Dict[str, float], 
    thresholds: Dict[str, float]
) -> Tuple[bool, str]:
    """Validate model meets minimum performance thresholds.
    
    Args:
        metrics: Calculated model metrics
        thresholds: Minimum acceptable thresholds
        
    Returns:
        Tuple of (passes_validation, message)
    """
    logger.info("Validating model performance against thresholds")
    
    default_thresholds = {
        'precision': 0.75,
        'recall': 0.70,
        'f1_score': 0.70,
        'roc_auc': 0.85
    }
    
    # Merge with provided thresholds
    validation_thresholds = {**default_thresholds, **thresholds}
    
    failures = []
    
    for metric_name, threshold_value in validation_thresholds.items():
        if metric_name in metrics:
            actual_value = metrics[metric_name]
            if actual_value < threshold_value:
                failures.append(
                    f"{metric_name}: {actual_value:.4f} < {threshold_value:.4f}"
                )
                logger.warning(f"❌ {metric_name}: {actual_value:.4f} < {threshold_value:.4f}")
            else:
                logger.info(f"✓ {metric_name}: {actual_value:.4f} >= {threshold_value:.4f}")
    
    if failures:
        message = f"Model failed validation. Issues: {'; '.join(failures)}"
        logger.error(message)
        return False, message
    else:
        message = "Model passed all performance validations"
        logger.info(message)
        return True, message


def compare_with_baseline(
    current_metrics: Dict[str, float], 
    baseline_model_path: Path
) -> Tuple[bool, Dict[str, float]]:
    """Compare current model performance against baseline/production model.
    
    Args:
        current_metrics: Metrics from current model
        baseline_model_path: Path to baseline model metadata
        
    Returns:
        Tuple of (is_better, comparison_results)
    """
    logger.info(f"Comparing with baseline model at {baseline_model_path}")
    
    comparison = {
        'timestamp': datetime.now().isoformat(),
        'has_baseline': False,
        'is_better': False,
        'improvements': {},
        'regressions': {}
    }
    
    # Check if baseline exists
    baseline_metrics_path = baseline_model_path / "metrics.json"
    
    if not baseline_metrics_path.exists():
        logger.warning("No baseline model found. Current model will become baseline.")
        comparison['is_better'] = True
        comparison['message'] = "No baseline exists - current model accepted"
        return True, comparison
    
    # Load baseline metrics
    with open(baseline_metrics_path, 'r') as f:
        baseline_metrics = json.load(f)
    
    comparison['has_baseline'] = True
    comparison['baseline_metrics'] = baseline_metrics
    comparison['current_metrics'] = current_metrics
    
    # Compare key metrics
    key_metrics = ['precision', 'recall', 'f1_score', 'roc_auc']
    
    improvement_count = 0
    regression_count = 0
    
    for metric in key_metrics:
        if metric in current_metrics and metric in baseline_metrics:
            current_val = current_metrics[metric]
            baseline_val = baseline_metrics[metric]
            diff = current_val - baseline_val
            pct_change = (diff / baseline_val * 100) if baseline_val != 0 else 0
            
            comparison_detail = {
                'baseline': baseline_val,
                'current': current_val,
                'difference': diff,
                'percent_change': pct_change
            }
            
            if diff > 0.01:  # Improvement threshold of 1%
                comparison['improvements'][metric] = comparison_detail
                improvement_count += 1
                logger.info(f"✓ {metric} improved: {baseline_val:.4f} -> {current_val:.4f} "
                          f"(+{pct_change:.2f}%)")
            elif diff < -0.01:  # Regression threshold of -1%
                comparison['regressions'][metric] = comparison_detail
                regression_count += 1
                logger.warning(f"❌ {metric} regressed: {baseline_val:.4f} -> {current_val:.4f} "
                             f"({pct_change:.2f}%)")
            else:
                logger.info(f"≈ {metric} unchanged: {baseline_val:.4f} -> {current_val:.4f}")
    
    # Decision logic: model is better if more improvements than regressions
    # and no critical metric (recall, roc_auc) regresses significantly
    critical_regressions = any(
        metric in comparison['regressions'] and 
        comparison['regressions'][metric]['difference'] < -0.05
        for metric in ['recall', 'roc_auc']
    )
    
    if critical_regressions:
        comparison['is_better'] = False
        comparison['message'] = "Critical metrics regressed significantly"
        logger.error("Model rejected: Critical metrics regressed")
    elif improvement_count >= regression_count:
        comparison['is_better'] = True
        comparison['message'] = f"{improvement_count} improvements, {regression_count} regressions"
        logger.info(f"Model accepted: {improvement_count} improvements vs {regression_count} regressions")
    else:
        comparison['is_better'] = False
        comparison['message'] = f"More regressions ({regression_count}) than improvements ({improvement_count})"
        logger.warning("Model rejected: More regressions than improvements")
    
    return comparison['is_better'], comparison


def save_model(
    model: Any, 
    model_path: Path, 
    metadata: Dict[str, Any]
) -> None:
    """Save trained model with metadata and versioning.
    
    Args:
        model: Trained model to save
        model_path: Directory path to save model
        metadata: Model metadata to save
    """
    logger.info(f"Saving model to {model_path}")
    
    # Create directory if it doesn't exist
    model_path.mkdir(parents=True, exist_ok=True)
    
    # Save model artifact
    model_file = model_path / "model.joblib"
    joblib.dump(model, model_file)
    logger.info(f"Model saved to {model_file}")
    
    # Enhance metadata
    full_metadata = {
        **metadata,
        'saved_at': datetime.now().isoformat(),
        'model_type': type(model).__name__,
        'model_file': str(model_file.name)
    }
    
    # Save metadata
    metadata_file = model_path / "metadata.json"
    with open(metadata_file, 'w') as f:
        json.dump(full_metadata, f, indent=2)
    logger.info(f"Metadata saved to {metadata_file}")
    
    # Save metrics if present
    if 'metrics' in metadata:
        metrics_file = model_path / "metrics.json"
        with open(metrics_file, 'w') as f:
            json.dump(metadata['metrics'], f, indent=2)
        logger.info(f"Metrics saved to {metrics_file}")
    
    # Save model version info
    version_file = model_path / "VERSION"
    version = metadata.get('version', '1.0.0')
    with open(version_file, 'w') as f:
        f.write(version)
    
    logger.info("Model save complete")


def load_model(model_path: Path) -> Tuple[Any, Dict[str, Any]]:
    """Load trained model and associated metadata.
    
    Args:
        model_path: Directory path containing saved model
        
    Returns:
        Tuple of (model, metadata)
        
    Raises:
        FileNotFoundError: If model files don't exist
    """
    logger.info(f"Loading model from {model_path}")
    
    if not model_path.exists():
        raise FileNotFoundError(f"Model path not found: {model_path}")
    
    # Load model
    model_file = model_path / "model.joblib"
    if not model_file.exists():
        raise FileNotFoundError(f"Model file not found: {model_file}")
    
    model = joblib.load(model_file)
    logger.info(f"Model loaded from {model_file}")
    
    # Load metadata
    metadata_file = model_path / "metadata.json"
    if metadata_file.exists():
        with open(metadata_file, 'r') as f:
            metadata = json.load(f)
        logger.info(f"Metadata loaded from {metadata_file}")
    else:
        logger.warning("No metadata file found")
        metadata = {}
    
    # Load metrics if available
    metrics_file = model_path / "metrics.json"
    if metrics_file.exists():
        with open(metrics_file, 'r') as f:
            metadata['metrics'] = json.load(f)
    
    return model, metadata

def register_model(
    model_path: Path, 
    model_name: str, 
    registry_uri: str
) -> str:
    """Register model to MLflow or model registry.
    
    Args:
        model_path: Path to saved model
        model_name: Name for registered model
        registry_uri: URI of model registry
        
    Returns:
        Model version string
    """
    logger.info(f"Registering model '{model_name}' to registry at {registry_uri}")
    
    try:
        import mlflow
        from mlflow.tracking import MlflowClient
        
        # Set tracking URI
        mlflow.set_tracking_uri(registry_uri)
        
        # Load model metadata
        metadata_file = model_path / "metadata.json"
        with open(metadata_file, 'r') as f:
            metadata = json.load(f)
        
        # Log model to MLflow
        with mlflow.start_run(run_name=f"{model_name}_registration") as run:
            # Log model
            mlflow.sklearn.log_model(
                sk_model=joblib.load(model_path / "model.joblib"),
                artifact_path="model",
                registered_model_name=model_name
            )
            
            # Log metrics
            if 'metrics' in metadata:
                mlflow.log_metrics(metadata['metrics'])
            
            # Log parameters
            if 'hyperparameters' in metadata:
                mlflow.log_params(metadata['hyperparameters'])
            
            # Log metadata as tags
            for key, value in metadata.items():
                if isinstance(value, (str, int, float, bool)):
                    mlflow.set_tag(key, value)
            
            run_id = run.info.run_id
        
        # Get the latest version
        client = MlflowClient()
        model_versions = client.search_model_versions(f"name='{model_name}'")
        latest_version = max([int(mv.version) for mv in model_versions])
        
        version_string = f"{model_name}/v{latest_version}"
        logger.info(f"Model registered successfully: {version_string}")
        
        return version_string
        
    except ImportError:
        logger.warning("MLflow not available, using local registry simulation")
        
        # Simulate registry with local storage
        registry_path = Path(registry_uri.replace('file://', ''))
        registry_path.mkdir(parents=True, exist_ok=True)
        
        # Create version directory
        existing_versions = list(registry_path.glob(f"{model_name}_v*"))
        version_num = len(existing_versions) + 1
        version_string = f"{model_name}/v{version_num}"
        
        version_path = registry_path / f"{model_name}_v{version_num}"
        version_path.mkdir(parents=True, exist_ok=True)
        
        # Copy model files
        import shutil
        for file in model_path.glob("*"):
            shutil.copy(file, version_path / file.name)
        
        logger.info(f"Model registered locally: {version_string}")
        return version_string

def deploy_model(
    model_uri: str, 
    deployment_target: str, 
    config: Dict[str, Any]
) -> Dict[str, str]:
    """Deploy model to specified target environment.
    
    Args:
        model_uri: URI of model to deploy
        deployment_target: Target environment (staging, production, etc.)
        config: Deployment configuration
        
    Returns:
        Dictionary with deployment information
    """
    logger.info(f"Deploying model {model_uri} to {deployment_target}")
    
    deployment_info = {
        'model_uri': model_uri,
        'target': deployment_target,
        'timestamp': datetime.now().isoformat(),
        'status': 'pending'
    }
    
    # Simulate deployment based on target
    if deployment_target == 'local':
        # Local deployment (Docker container)
        try:
            container_name = config.get('container_name', 'fraud-detection-model')
            port = config.get('port', 8000)
            
            # Run Docker container
            result = subprocess.run(
                [
                    "docker", "run", "-d",
                    "--name", container_name,
                    "-p", f"{port}:8000",
                    model_uri
                ],
                capture_output=True,
                text=True
            )
            
            if result.returncode == 0:
                deployment_info['status'] = 'success'
                deployment_info['endpoint'] = f"http://localhost:{port}"
                deployment_info['container_id'] = result.stdout.strip()
                logger.info(f"Deployed to local container: {container_name}")
            else:
                deployment_info['status'] = 'failed'
                deployment_info['error'] = result.stderr
                logger.error(f"Deployment failed: {result.stderr}")
                
        except FileNotFoundError:
            logger.warning("Docker not available for local deployment")
            deployment_info['status'] = 'simulated'
            deployment_info['endpoint'] = f"http://localhost:8000"
    
    elif deployment_target in ['staging', 'production']:
        # Cloud deployment simulation
        deployment_info['status'] = 'simulated'
        deployment_info['endpoint'] = f"https://{deployment_target}.example.com/api/v1/predict"
        deployment_info['deployment_id'] = f"deploy-{datetime.now().strftime('%Y%m%d%H%M%S')}"
        
        # In production, this would call cloud provider APIs
        logger.info(f"Simulated deployment to {deployment_target}")
        logger.info(f"Endpoint: {deployment_info['endpoint']}")
    
    else:
        deployment_info['status'] = 'failed'
        deployment_info['error'] = f"Unknown deployment target: {deployment_target}"
        logger.error(deployment_info['error'])
    
    return deployment_info


def run_smoke_tests(endpoint_url: str) -> Tuple[bool, Dict[str, Any]]:
    """Run smoke tests against deployed model endpoint.
    
    Args:
        endpoint_url: URL of deployed model endpoint
        
    Returns:
        Tuple of (all_tests_passed, test_results)
    """
    logger.info(f"Running smoke tests against {endpoint_url}")
    
    test_results = {
        'timestamp': datetime.now().isoformat(),
        'endpoint': endpoint_url,
        'tests': {},
        'all_passed': False
    }
    
    # Test 1: Health check
    try:
        health_url = endpoint_url.replace('/predict', '/health')
        response = requests.get(health_url, timeout=5)
        
        test_results['tests']['health_check'] = {
            'passed': response.status_code == 200,
            'status_code': response.status_code,
            'response_time_ms': response.elapsed.total_seconds() * 1000
        }
        logger.info(f"Health check: {'✓' if response.status_code == 200 else '✗'}")
        
    except requests.RequestException as e:
        test_results['tests']['health_check'] = {
            'passed': False,
            'error': str(e)
        }
        logger.error(f"Health check failed: {e}")
    
    # Test 2: Valid prediction request
    try:
        test_payload = {
            'features': {

                'amount_log': 4.61,
                'hour': 14,
                'day_of_week': 3,
                'is_weekend': 0,
                'is_night': 0
            }
        }
        
        response = requests.post(endpoint_url, json=test_payload, timeout=10)
        
        test_passed = (
            response.status_code == 200 and
            'is_fraud' in response.json() and
            'fraud_probability' in response.json()
        )
        
        test_results['tests']['valid_prediction'] = {
            'passed': test_passed,
            'status_code': response.status_code,
            'response_time_ms': response.elapsed.total_seconds() * 1000,
            'response': response.json() if response.status_code == 200 else None
        }
        logger.info(f"Valid prediction: {'✓' if test_passed else '✗'}")
        
    except requests.RequestException as e:
        test_results['tests']['valid_prediction'] = {
            'passed': False,
            'error': str(e)
        }
        logger.error(f"Valid prediction test failed: {e}")
    
    # Test 3: Latency check (should respond in <1000ms)
    if 'valid_prediction' in test_results['tests']:
        latency = test_results['tests']['valid_prediction'].get('response_time_ms', float('inf'))
        latency_passed = latency < 1000
        
        test_results['tests']['latency_check'] = {
            'passed': latency_passed,
            'latency_ms': latency,
            'threshold_ms': 1000
        }
        logger.info(f"Latency check: {'✓' if latency_passed else '✗'} ({latency:.2f}ms)")
    
    # Test 4: Invalid input handling
    try:
        invalid_payload = {'features': {}}
        response = requests.post(endpoint_url, json=invalid_payload, timeout=10)
        
        # Should return error status
        test_passed = response.status_code in [400, 422, 500]
        
        test_results['tests']['invalid_input_handling'] = {
            'passed': test_passed,
            'status_code': response.status_code
        }
        logger.info(f"Invalid input handling: {'✓' if test_passed else '✗'}")
        
    except requests.RequestException as e:
        test_results['tests']['invalid_input_handling'] = {
            'passed': False,
            'error': str(e)
        }
        logger.error(f"Invalid input test failed: {e}")
    
    # Determine overall result
    all_passed = all(
        test.get('passed', False) 
        for test in test_results['tests'].values()
    )
    
    test_results['all_passed'] = all_passed
    test_results['passed_count'] = sum(
        1 for test in test_results['tests'].values() if test.get('passed', False)
    )
    test_results['total_count'] = len(test_results['tests'])
    
    logger.info(f"Smoke tests complete: {test_results['passed_count']}/{test_results['total_count']} passed")
    
    return all_passed, test_results


def monitor_model_drift(
    reference_data: pd.DataFrame, 
    current_data: pd.DataFrame
) -> Dict[str, Any]:
    """Monitor for data and prediction drift.
    
    Args:
        reference_data: Reference/baseline data
        current_data: Current production data
        
    Returns:
        Dictionary with drift detection results
    """
    logger.info("Monitoring for data and prediction drift")
    
    drift_report = {
        'timestamp': datetime.now().isoformat(),
        'reference_size': len(reference_data),
        'current_size': len(current_data),
        'drift_detected': False,
        'feature_drift': {},
        'distribution_drift': {}
    }
    
    # Get common columns
    common_columns = list(set(reference_data.columns) & set(current_data.columns))
    numeric_columns = reference_data[common_columns].select_dtypes(include=[np.number]).columns
    
    # Statistical tests for drift
    for col in numeric_columns:
        ref_values = reference_data[col].dropna()
        curr_values = current_data[col].dropna()
        
        # Kolmogorov-Smirnov test
        ks_statistic, ks_pvalue = stats.ks_2samp(ref_values, curr_values)
        
        # Calculate distribution statistics
        ref_mean = float(ref_values.mean())
        curr_mean = float(curr_values.mean())
        mean_change = ((curr_mean - ref_mean) / ref_mean * 100) if ref_mean != 0 else 0
        
        ref_std = float(ref_values.std())
        curr_std = float(curr_values.std())
        
        drift_detected = ks_pvalue < 0.05  # 5% significance level
        
        drift_report['feature_drift'][col] = {
            'ks_statistic': float(ks_statistic),
            'ks_pvalue': float(ks_pvalue),
            'drift_detected': drift_detected,
            'reference_mean': ref_mean,
            'current_mean': curr_mean,
            'mean_change_pct': float(mean_change),
            'reference_std': ref_std,
            'current_std': curr_std
        }
        
        if drift_detected:
            drift_report['drift_detected'] = True
            logger.warning(f"Drift detected in {col}: KS p-value={ks_pvalue:.4f}")
    
    # Overall distribution drift
    if drift_report['drift_detected']:
        drifted_features = [
            col for col, info in drift_report['feature_drift'].items()
            if info['drift_detected']
        ]
        drift_report['summary'] = {
            'total_features': len(numeric_columns),
            'drifted_features': len(drifted_features),
            'drift_percentage': len(drifted_features) / len(numeric_columns) * 100,
            'drifted_feature_names': drifted_features
        }
        logger.warning(f"Drift detected in {len(drifted_features)}/{len(numeric_columns)} features")
    else:
        drift_report['summary'] = {
            'total_features': len(numeric_columns),
            'drifted_features': 0,
            'drift_percentage': 0.0,
            'drifted_feature_names': []
        }
        logger.info("No significant drift detected")
    
    return drift_report


def run_pipeline(
    config_path: Path, 
    mode: str = "train"
) -> Dict[str, Any]:
    """Execute end-to-end ML pipeline orchestration.
    
    Args:
        config_path: Path to pipeline configuration file
        mode: Pipeline mode ('train', 'deploy', 'evaluate')
        
    Returns:
        Dictionary with pipeline execution results
    """
    logger.info(f"Starting pipeline in {mode} mode")
    logger.info(f"Configuration: {config_path}")
    
    pipeline_results = {
        'mode': mode,
        'start_time': datetime.now().isoformat(),
        'config_path': str(config_path),
        'status': 'running',
        'steps': {}
    }
    
    try:
        # Load configuration
        from .config import load_config
        config = load_config(config_path)
        
        if mode == 'train':
            # Training pipeline
            logger.info("=== Starting Training Pipeline ===")
            
            # Step 1: Load data
            logger.info("Step 1: Loading data")
            df = load_data(config.data.data_path, config.data.version)
            pipeline_results['steps']['load_data'] = {'status': 'success', 'records': len(df)}
            
            # Step 2: Validate data
            logger.info("Step 2: Validating data")
            is_valid, validation_report = validate_data(df)
            pipeline_results['steps']['validate_data'] = {
                'status': 'success' if is_valid else 'failed',
                'report': validation_report
            }
            
            if not is_valid:
                raise ValueError("Data validation failed")
            
            # Step 3: Preprocess data
            logger.info("Step 3: Preprocessing data")
            df_processed, preprocessing_metadata = preprocess_data(df)
            pipeline_results['steps']['preprocess_data'] = {
                'status': 'success',
                'metadata': preprocessing_metadata
            }
            
            # Step 4: Split data
            logger.info("Step 4: Splitting data")
            X_train, X_test, y_train, y_test = split_data(
                df_processed,
                test_size=config.data.test_size,
                stratify_column='is_fraud'
            )
            pipeline_results['steps']['split_data'] = {
                'status': 'success',
                'train_size': len(X_train),
                'test_size': len(X_test)
            }
            
            # Step 5: Train model
            logger.info("Step 5: Training model")
            model = train_model(X_train, y_train, config.model.hyperparameters)
            pipeline_results['steps']['train_model'] = {'status': 'success'}
            
            # Step 6: Evaluate model
            logger.info("Step 6: Evaluating model")
            metrics = evaluate_model(model, X_test, y_test)
            pipeline_results['steps']['evaluate_model'] = {
                'status': 'success',
                'metrics': metrics
            }
            
            # Step 7: Validate performance
            logger.info("Step 7: Validating performance")
            passes_validation, validation_message = validate_model_performance(
                metrics,
                config.validation.thresholds
            )
            pipeline_results['steps']['validate_performance'] = {
                'status': 'success' if passes_validation else 'failed',
                'message': validation_message
            }
            
            if not passes_validation:
                raise ValueError(f"Performance validation failed: {validation_message}")
            
            # Step 8: Compare with baseline
            logger.info("Step 8: Comparing with baseline")
            is_better, comparison = compare_with_baseline(
                metrics,
                config.validation.baseline_model_path
            )
            pipeline_results['steps']['compare_baseline'] = {
                'status': 'success',
                'is_better': is_better,
                'comparison': comparison
            }
            
            if not is_better and comparison.get('has_baseline', False):
                logger.warning("Model did not improve over baseline")
            
            # Step 9: Save model
            logger.info("Step 9: Saving model")
            model_metadata = {
                'version': config.model.version,
                'hyperparameters': config.model.hyperparameters,
                'metrics': metrics,
                'preprocessing': preprocessing_metadata,
                'validation': validation_report,
                'comparison': comparison
            }
            save_model(model, config.model.model_path, model_metadata)
            pipeline_results['steps']['save_model'] = {
                'status': 'success',
                'path': str(config.model.model_path)
            }
            
            # Step 10: Generate model card
            logger.info("Step 10: Generating model card")
            model_card = generate_model_card(model, metrics, model_metadata)
            model_card_path = config.model.model_path / "MODEL_CARD.md"
            with open(model_card_path, 'w') as f:
                f.write(model_card)
            pipeline_results['steps']['generate_model_card'] = {
                'status': 'success',
                'path': str(model_card_path)
            }
            
            pipeline_results['status'] = 'success'
            logger.info("=== Training Pipeline Complete ===")
        
        elif mode == 'deploy':
            # Deployment pipeline
            logger.info("=== Starting Deployment Pipeline ===")
            
            # Step 1: Load model
            logger.info("Step 1: Loading model")
            model, metadata = load_model(config.deployment.model_path)
            pipeline_results['steps']['load_model'] = {'status': 'success'}
            
            # Step 2: Register model
            logger.info("Step 2: Registering model")
            model_version = register_model(
                config.deployment.model_path,
                config.deployment.model_name,
                config.deployment.registry_uri
            )
            pipeline_results['steps']['register_model'] = {
                'status': 'success',
                'version': model_version
            }
            
            # Step 3: Create Docker image
            logger.info("Step 3: Creating Docker image")
            image_name = create_docker_image(
                config.deployment.model_path,
                config.deployment.image_name,
                config.deployment.image_tag
            )
            pipeline_results['steps']['create_image'] = {
                'status': 'success',
                'image': image_name
            }
            
            # Step 4: Deploy model
            logger.info("Step 4: Deploying model")
            deployment_info = deploy_model(
                image_name,
                config.deployment.target,
                config.deployment.config
            )
            pipeline_results['steps']['deploy_model'] = deployment_info
            
            # Step 5: Run smoke tests
            logger.info("Step 5: Running smoke tests")
            if deployment_info.get('endpoint'):
                tests_passed, test_results = run_smoke_tests(deployment_info['endpoint'])
                pipeline_results['steps']['smoke_tests'] = {
                    'status': 'success' if tests_passed else 'failed',
                    'results': test_results
                }
                
                if not tests_passed:
                    raise ValueError("Smoke tests failed")
            
            pipeline_results['status'] = 'success'
            logger.info("=== Deployment Pipeline Complete ===")
        
        elif mode == 'evaluate':
            # Evaluation/monitoring pipeline
            logger.info("=== Starting Evaluation Pipeline ===")
            
            # Load reference and current data
            reference_data = load_data(config.data.reference_data_path)
            current_data = load_data(config.data.current_data_path)
            
            # Monitor drift
            drift_report = monitor_model_drift(reference_data, current_data)
            pipeline_results['steps']['monitor_drift'] = {
                'status': 'success',
                'report': drift_report
            }
            
            if drift_report['drift_detected']:
                logger.warning("Significant drift detected - consider retraining")
            
            pipeline_results['status'] = 'success'
            logger.info("=== Evaluation Pipeline Complete ===")
        
        else:
            raise ValueError(f"Unknown pipeline mode: {mode}")
    
    except Exception as e:
        logger.error(f"Pipeline failed: {str(e)}", exc_info=True)
        pipeline_results['status'] = 'failed'
        pipeline_results['error'] = str(e)
    
    finally:
        pipeline_results['end_time'] = datetime.now().isoformat()
        
        # Calculate duration
        start = datetime.fromisoformat(pipeline_results['start_time'])
        end = datetime.fromisoformat(pipeline_results['end_time'])
        pipeline_results['duration_seconds'] = (end - start).total_seconds()
    
    logger.info(f"Pipeline {mode} completed with status: {pipeline_results['status']}")
    logger.info(f"Duration: {pipeline_results['duration_seconds']:.2f} seconds")
    
    return pipeline_results
