        assert len(report['issues']) == 0

        df_missing = sample_fraud_data.drop(columns=['is_fraud'])

        assert is_valid is False
        assert any('Missing required columns' in issue for issue in report['issues'])

        """Test validation detects excessive null values."""
        df_with_nulls = sample_fraud_data.copy()
        df_with_nulls.loc[:600, 'amount'] = np.nan
        
        is_valid, report = validate_data(df_with_nulls)
        
        assert is_valid is False
        assert any('null values' in issue for issue in report['issues'])
    
    def test_validate_duplicate_records(self, sample_fraud_data):
        """Test validation detects duplicate records."""
        df_with_dupes = pd.concat([sample_fraud_data] * 2, ignore_index=True)
        
        is_valid, report = validate_data(df_with_dupes)
        
        assert report['statistics']['duplicate_records'] > 0
    
    def test_validate_fraud_distribution(self, sample_fraud_data):
        """Test validation checks fraud distribution."""
        is_valid, report = validate_data(sample_fraud_data)
        
        assert 'fraud_distribution' in report['statistics']
        assert 'fraud_rate' in report['statistics']
        assert 0 <= report['statistics']['fraud_rate'] <= 1
    
    def test_validate_outlier_detection(self, sample_fraud_data):
        """Test validation detects outliers in amount."""
        is_valid, report = validate_data(sample_fraud_data)
        
        assert 'amount_stats' in report['statistics']
        assert 'amount_outliers' in report['statistics']
        assert report['statistics']['amount_outliers'] >= 0
    
    def test_validate_very_low_fraud_rate(self):
        """Test validation warns about very low fraud rate."""
        df = pd.DataFrame({

        })

class TestDataSplitting:
    """Test suite for data splitting functionality."""
    
    def test_split_basic(self, sample_processed_data):
        """Test basic train/test split."""
        X_train, X_test, y_train, y_test = split_data(sample_processed_data)
        
        assert len(X_train) + len(X_test) == len(sample_processed_data)
        assert len(y_train) == len(X_train)
        assert len(y_test) == len(X_test)
        assert 'is_fraud' not in X_train.columns
        assert 'is_fraud' not in X_test.columns
    
    def test_split_stratification(self, sample_processed_data):
        """Test stratified split maintains class distribution."""
        X_train, X_test, y_train, y_test = split_data(sample_processed_data)
        
        train_fraud_rate = y_train.mean()
        test_fraud_rate = y_test.mean()
        overall_fraud_rate = sample_processed_data['is_fraud'].mean()
        
        assert abs(train_fraud_rate - overall_fraud_rate) < 0.02
        assert abs(test_fraud_rate - overall_fraud_rate) < 0.02
    
    def test_split_custom_test_size(self, sample_processed_data):
        """Test split with custom test size."""
        X_train, X_test, y_train, y_test = split_data(
            sample_processed_data, 
            test_size=0.3
        )
        
        expected_test_size = int(len(sample_processed_data) * 0.3)
        assert abs(len(X_test) - expected_test_size) <= 1
    
    def test_split_missing_stratify_column(self, sample_processed_data):
        """Test split fails with missing stratify column."""
        with pytest.raises(ValueError, match="Stratify column"):
            split_data(sample_processed_data, stratify_column='nonexistent')
    
    def test_split_reproducibility(self, sample_processed_data):
        """Test split is reproducible with same random state."""
        X_train1, X_test1, y_train1, y_test1 = split_data(sample_processed_data)
        X_train2, X_test2, y_train2, y_test2 = split_data(sample_processed_data)
        
        pd.testing.assert_frame_equal(X_train1, X_train2)
        pd.testing.assert_frame_equal(X_test1, X_test2)

class TestBaselineComparison:
    """Test suite for baseline model comparison."""
    
    def test_compare_no_baseline(self, sample_metrics, temp_dir):
        """Test comparison when no baseline exists."""
        baseline_path = temp_dir / 'baseline'
        
        is_better, comparison = compare_with_baseline(sample_metrics, baseline_path)
        
        assert is_better is True
        assert comparison['has_baseline'] is False
        assert 'No baseline' in comparison['message']
    
    def test_compare_better_model(self, sample_metrics, temp_dir):
        """Test comparison when current model is better."""
        baseline_path = temp_dir / 'baseline'
        baseline_path.mkdir()
        
        baseline_metrics = {
            'precision': 0.70,
            'recall': 0.65,
            'f1_score': 0.67,
            'roc_auc': 0.80
        }
        
        with open(baseline_path / 'metrics.json', 'w') as f:
            json.dump(baseline_metrics, f)
        
        is_better, comparison = compare_with_baseline(sample_metrics, baseline_path)
        
        assert is_better is True
        assert comparison['has_baseline'] is True
        assert len(comparison['improvements']) > 0
    
    def test_compare_worse_model(self, sample_metrics, temp_dir):
        """Test comparison when current model is worse."""
        baseline_path = temp_dir / 'baseline'
        baseline_path.mkdir()
        
        better_baseline = {
            'precision': 0.95,
            'recall': 0.90,
            'f1_score': 0.92,
            'roc_auc': 0.98
        }
        
        with open(baseline_path / 'metrics.json', 'w') as f:
            json.dump(better_baseline, f)
        
        is_better, comparison = compare_with_baseline(sample_metrics, baseline_path)
        
        assert is_better is False
        assert len(comparison['regressions']) > 0
    
    def test_compare_critical_regression(self, sample_metrics, temp_dir):
        """Test comparison rejects critical metric regression."""
        baseline_path = temp_dir / 'baseline'
        baseline_path.mkdir()
        
        baseline_metrics = {
            'precision': 0.70,
            'recall': 0.95,  # Much higher recall
            'f1_score': 0.70,
            'roc_auc': 0.80
        }
        
        with open(baseline_path / 'metrics.json', 'w') as f:
            json.dump(baseline_metrics, f)
        
        is_better, comparison = compare_with_baseline(sample_metrics, baseline_path)
        
        assert is_better is False
        assert 'Critical metrics regressed' in comparison['message']

  # ============================================================================
  # Model Evaluation
  # ============================================================================
  
  evaluate-model:
    name: Evaluate Model
    runs-on: ubuntu-latest
    needs: [train-model]
    
    steps:
      - name: Checkout code
        uses: actions/checkout@v4
      
      - name: Set up Python
        uses: actions/setup-python@v4
        with:
          python-version: ${{ env.PYTHON_VERSION }}
          cache: 'pip'
      
      - name: Install dependencies
        run: |
          pip install -r requirements.txt
      
      - name: Download model artifacts
        uses: actions/download-artifact@v3
        with:
          name: trained-model
          path: models/
      
      - name: Evaluate model performance
        run: |
          python -c "
          from pipeline import load_model, evaluate_model, load_data, preprocess_data, split_data
          from pathlib import Path
          import json
          
          # Load model
          model, metadata = load_model(Path('models/fraud_detection_model'))
          
          # Load test data
          df = load_data(Path('data/fraud_data.csv'))
          df_processed, _ = preprocess_data(df)
          _, X_test, _, y_test = split_data(df_processed)
          
          # Evaluate
          metrics = evaluate_model(model, X_test, y_test)
          
          print('Model Performance:')
          for metric, value in metrics.items():
              if isinstance(value, float):
                  print(f'  {metric}: {value:.4f}')
          
          # Save metrics
          with open('evaluation_metrics.json', 'w') as f:
              json.dump(metrics, f, indent=2)
          "
      
      - name: Check performance thresholds
        run: |
          python -c "
          from pipeline import validate_model_performance
          import json
          
          with open('evaluation_metrics.json', 'r') as f:
              metrics = json.load(f)
          
          thresholds = {
              'precision': 0.75,
              'recall': 0.70,
              'f1_score': 0.70,
              'roc_auc': 0.85
          }
          
          passes, message = validate_model_performance(metrics, thresholds)
          
          print(f'Validation: {message}')
          
          if not passes:
              print('Model did not meet performance thresholds!')
              exit(1)
          "
      
      - name: Compare with baseline
        run: |
          python -c "
          from pipeline import compare_with_baseline
          from pathlib import Path
          import json
          
          with open('evaluation_metrics.json', 'r') as f:
              current_metrics = json.load(f)
          
          is_better, comparison = compare_with_baseline(
              current_metrics,
              Path('models/baseline')
          )
          
          print(f'Comparison result: {comparison[\"message\"]}')
          print(f'Is better: {is_better}')
          
          if comparison['has_baseline']:
              print(f'Improvements: {len(comparison[\"improvements\"])}')
              print(f'Regressions: {len(comparison[\"regressions\"])}')
          
          with open('comparison_results.json', 'w') as f:
              json.dump(comparison, f, indent=2)
          "
      
      - name: Generate model card
        run: |
          python -c "
          from pipeline import load_model, generate_model_card
          from pathlib import Path
          import json
          
          model, metadata = load_model(Path('models/fraud_detection_model'))
          
          with open('evaluation_metrics.json', 'r') as f:
              metrics = json.load(f)
          
          model_card = generate_model_card(model, metrics, metadata)
          
          with open('MODEL_CARD.md', 'w') as f:
              f.write(model_card)
          "
      
      - name: Upload evaluation results
        uses: actions/upload-artifact@v3
        with:
          name: evaluation-results
          path: |
            evaluation_metrics.json
            comparison_results.json
            MODEL_CARD.md
      
      - name: Comment PR with results
        if: github.event_name == 'pull_request'
        uses: actions/github-script@v6
        with:
          script: |
            const fs = require('fs');
            const metrics = JSON.parse(fs.readFileSync('evaluation_metrics.json', 'utf8'));
            
            const comment = `## 🤖 Model Evaluation Results
            
            | Metric | Value |
            |--------|-------|
            | Accuracy | ${metrics.accuracy.toFixed(4)} |
            | Precision | ${metrics.precision.toFixed(4)} |
            | Recall | ${metrics.recall.toFixed(4)} |
            | F1 Score | ${metrics.f1_score.toFixed(4)} |
            | ROC-AUC | ${metrics.roc_auc.toFixed(4)} |
            
            ✅ Model passed performance validation
            `;
            
            github.rest.issues.createComment({
              issue_number: context.issue.number,
              owner: context.repo.owner,
              repo: context.repo.repo,
              body: comment
            });

          reference_data = load_data(Path('data/reference_data.csv'))
          current_data = load_data(Path('data/current_data.csv'))
          
          drift_report = monitor_model_drift(reference_data, current_data)
          
          print(f'Drift detected: {drift_report[\"drift_detected\"]}')
          print(f'Drifted features: {drift_report[\"summary\"][\"drifted_features\"]}')
          
          with open('drift_report.json', 'w') as f:
              json.dump(drift_report, f, indent=2)
          
          if drift_report['drift_detected']:
              print('⚠️ Significant drift detected - retraining recommended')
              exit(1)
          "
      
      - name: Upload drift report
        uses: actions/upload-artifact@v3
        if: always()
        with:
          name: drift-report
          path: drift_report.json
      
      - name: Create issue if drift detected
        if: failure()
        uses: actions/github-script@v6
        with:
          script: |
            github.rest.issues.create({
              owner: context.repo.owner,
              repo: context.repo.repo,
              title: '⚠️ Data Drift Detected',
              body: 'Significant data drift has been detected. Model retraining is recommended.\n\nSee drift report artifact for details.',
              labels: ['drift', 'monitoring', 'ml-ops']
