## GitHub Actions Workflow (`..github/workflows/ci-cd.yml`)

```yaml
name: Fraud Detection Model CI/CD

on:
  push:
    branches: [main, develop]
  pull_request:
    branches: [main]
  workflow_dispatch:

env:
  PYTHON_VERSION: "3.10"
  MODEL_NAME: fraud-detection-model

jobs:
  # Job 1: Code Quality and Linting
  code-quality:
    steps:
      - Checkout code
      - Setup Python
      - Install dependencies
      - Run black formatter check
      - Run flake8 linting
      - Run mypy type checking
      - Run bandit security scan

  # Job 2: Unit and Integration Tests
  test:
    steps:
      - Checkout code
      - Setup Python
      - Install dependencies
      - Run pytest with coverage
      - Generate coverage report
      - Upload coverage to Codecov

  # Job 3: Data Validation
  data-validation:
    steps:
      - Checkout code
      - Setup Python and DVC
      - Pull data from DVC
      - Run data validation tests
      - Check for data drift
      - Generate data quality report

  # Job 4: Model Training
  train-model:
    needs: [code-quality, test, data-validation]
    steps:
      - Checkout code
      - Setup Python and dependencies
      - Configure MLflow tracking
      - Pull training data
      - Train model with hyperparameters
      - Log metrics to MLflow
      - Save model artifacts

  # Job 5: Model Evaluation
  evaluate-model:
    needs: train-model
    steps:
      - Checkout code
      - Load trained model
      - Run model evaluation
      - Compare with baseline model
      - Validate performance thresholds
      - Generate model card
      - Upload evaluation artifacts

  # Job 6: Model Registration
  register-model:
    needs: evaluate-model
    if: github.ref == 'refs/heads/main'
    steps:
      - Checkout code
      - Load model artifacts
      - Register model to MLflow registry
      - Tag model version
      - Transition model stage

  # Job 7: Build Docker Image
  build-image:
    needs: register-model
    if: github.ref == 'refs/heads/main'
    steps:
      - Checkout code
      - Setup Docker Buildx
      - Login to container registry
      - Build model serving image
      - Run container security scan
      - Push image to registry

  # Job 8: Deploy to Staging
  deploy-staging:
    needs: build-image
    if: github.ref == 'refs/heads/main'
    environment: staging
    steps:
      - Checkout code
      - Configure cloud credentials
      - Deploy model to staging
      - Run smoke tests
      - Run integration tests
      - Monitor deployment health

  # Job 9: Deploy to Production
  deploy-production:
    needs: deploy-staging
    if: github.ref == 'refs/heads/main'
    environment: production
    steps:
      - Checkout code
      - Configure cloud credentials
      - Deploy with blue-green strategy
      - Run smoke tests
      - Run canary deployment
      - Monitor metrics
      - Complete rollout or rollback

  # Job 10: Post-Deployment Monitoring
  post-deployment:
    needs: deploy-production
    if: github.ref == 'refs/heads/main'
    steps:
      - Setup monitoring dashboards
      - Configure alerting rules
      - Run drift detection
      - Generate deployment report
      - Notify team via Slack
```
