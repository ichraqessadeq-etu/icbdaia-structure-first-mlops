## GitHub Actions Workflow (`.github/workflows/ci-cd.yml`)

```yaml
name: Fraud Detection Model CI/CD

on:
  push:
    branches: [main, develop]
  pull_request:
    branches: [main]
  schedule:
    - cron: '0 0 * * 0'  # Weekly model retraining

jobs:
  # Job 1: Code Quality & Linting
  code-quality:
    steps:
      - checkout-code
      - setup-python
      - install-dependencies
      - run-black-formatter-check
      - run-flake8-linter
      - run-mypy-type-check
      - run-bandit-security-scan

  # Job 2: Unit & Integration Tests
  test:
    steps:
      - checkout-code
      - setup-python
      - install-dependencies
      - run-pytest-with-coverage
      - upload-coverage-report
      - check-coverage-threshold

  # Job 3: Data Validation
  data-validation:
    steps:
      - checkout-code
      - setup-python
      - install-dependencies
      - download-latest-data
      - validate-data-schema
      - check-data-quality
      - detect-data-drift
      - upload-validation-report

  # Job 4: Model Training
  train-model:
    needs: [code-quality, test, data-validation]
    steps:
      - checkout-code
      - setup-python
      - install-dependencies
      - download-training-data
      - run-training-pipeline
      - save-training-artifacts
      - upload-model-artifacts

  # Job 5: Model Evaluation & Validation
  evaluate-model:
    needs: [train-model]
    steps:
      - checkout-code
      - setup-python
      - install-dependencies
      - download-model-artifacts
      - download-test-data
      - run-model-evaluation
      - check-performance-thresholds
      - check-model-bias
      - generate-evaluation-report
      - upload-evaluation-artifacts

  # Job 6: Model Registry & Versioning
  register-model:
    needs: [evaluate-model]
    if: github.ref == 'refs/heads/main'
    steps:
      - checkout-code
      - setup-python
      - install-dependencies
      - download-model-artifacts
      - register-model-version
      - tag-model-metadata
      - generate-model-card
      - upload-to-model-registry

  # Job 7: Deploy to Staging
  deploy-staging:
    needs: [register-model]
    if: github.ref == 'refs/heads/main'
    environment: staging
    steps:
      - checkout-code
      - setup-cloud-credentials
      - download-model-from-registry
      - deploy-to-staging-endpoint
      - run-smoke-tests
      - update-staging-config

  # Job 8: Integration Tests (Staging)
  integration-test-staging:
    needs: [deploy-staging]
    steps:
      - checkout-code
      - setup-python
      - install-dependencies
      - run-api-integration-tests
      - run-performance-tests
      - run-load-tests
      - validate-prediction-latency

  # Job 9: Deploy to Production
  deploy-production:
    needs: [integration-test-staging]
    if: github.ref == 'refs/heads/main'
    environment: production
    steps:
      - checkout-code
      - setup-cloud-credentials
      - download-model-from-registry
      - deploy-canary-release
      - monitor-canary-metrics
      - promote-to-full-deployment
      - update-production-config

  # Job 10: Post-Deployment Monitoring
  post-deployment-validation:
    needs: [deploy-production]
    steps:
      - checkout-code
      - setup-monitoring-tools
      - validate-production-endpoint
      - setup-performance-alerts
      - setup-drift-monitoring
      - create-deployment-summary
      - notify-team
```
