class TestModelRegistry:
    """Test suite for model registry module."""
    
    def test_register_model_local(self, trained_model, temp_dir):
        """Test model registration with local registry."""
        model_path = Path(temp_dir) / 'model.joblib'
        joblib.dump(trained_model, model_path)
        
        metrics = {'f1_score': 0.85, 'roc_auc': 0.90}
        metadata = {'model_name': 'test-model', 'version': '1'}
        
        version = register_model(
            str(model_path),
            metrics,
            metadata,
            str(temp_dir)
        )
        
        assert version is not None
        assert (Path(temp_dir) / 'test-model' / 'versions' / version).exists()
    
    def test_get_latest_model_version(self, trained_model, temp_dir):
        """Test getting latest model version."""
        model_path = Path(temp_dir) / 'model.joblib'
        joblib.dump(trained_model, model_path)
        
        metrics = {'f1_score': 0.85}
        metadata = {'model_name': 'test-model'}
        
        # Register two versions
        register_model(str(model_path), metrics, metadata, str(temp_dir))
        register_model(str(model_path), metrics, metadata, str(temp_dir))
        
        latest_version = get_latest_model_version('test-model', str(temp_dir))
        
        assert latest_version == '2'
    
    def test_promote_model_to_stage(self, trained_model, temp_dir):
        """Test model promotion to stage."""
        model_path = Path(temp_dir) / 'model.joblib'
        joblib.dump(trained_model, model_path)
        
        metrics = {'f1_score': 0.85}
        metadata = {'model_name': 'test-model'}
        
        version = register_model(str(model_path), metrics, metadata, str(temp_dir))
        
        promote_model_to_stage('test-model', version, 'production', str(temp_dir))
        
        # Check metadata was updated
        metadata_path = Path(temp_dir) / 'test-model' / 'versions' / version / 'metadata.json'
        with open(metadata_path, 'r') as f:
            updated_metadata = json.load(f)
        
        assert updated_metadata['stage'] == 'production'
    
    def test_compare_model_versions(self, trained_model, temp_dir):
        """Test comparing two model versions."""
        model_path = Path(temp_dir) / 'model.joblib'
        joblib.dump(trained_model, model_path)
        
        metrics1 = {'f1_score': 0.80, 'roc_auc': 0.85}
        metrics2 = {'f1_score': 0.85, 'roc_auc': 0.90}
        metadata = {'model_name': 'test-model'}
        
        v1 = register_model(str(model_path), metrics1, metadata, str(temp_dir))
        v2 = register_model(str(model_path), metrics2, metadata, str(temp_dir))
        
        comparison = compare_model_versions('test-model', v1, v2, str(temp_dir))
        
        assert comparison['winner'] == v2
        assert 'metrics_comparison' in comparison
    
    def test_archive_old_models(self, trained_model, temp_dir):
        """Test archiving old model versions."""
        model_path = Path(temp_dir) / 'model.joblib'
        joblib.dump(trained_model, model_path)
        
        metrics = {'f1_score': 0.85}
        metadata = {'model_name': 'test-model'}
        
        # Register 5 versions
        for _ in range(5):
            register_model(str(model_path), metrics, metadata, str(temp_dir))
        
        # Archive, keeping only 2 recent
        archive_old_models('test-model', keep_n_versions=2, registry_uri=str(temp_dir))
        
        versions_dir = Path(temp_dir) / 'test-model' / 'versions'
        remaining_versions = [d for d in versions_dir.iterdir() if d.is_dir()]
        
        assert len(remaining_versions) == 2
