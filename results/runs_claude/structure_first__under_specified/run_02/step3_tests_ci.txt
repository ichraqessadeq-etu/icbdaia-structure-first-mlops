class TestHandleImbalance:
    """Test suite for handle_imbalance function."""
    
    def test_handle_imbalance_smote(self, sample_fraud_data):
        """Test SMOTE balancing."""
        df_preprocessed = preprocess_data(sample_fraud_data)
        X_train, X_test, y_train, y_test = split_data(df_preprocessed)
        
        X_resampled, y_resampled = handle_imbalance(X_train, y_train, strategy='smote')
        
        # Check that classes are more balanced
        class_counts = y_resampled.value_counts()
        assert len(class_counts) == 2
        # Classes should be equal or close after SMOTE
        assert abs(class_counts.iloc[0] - class_counts.iloc[1]) < 100
    
    def test_handle_imbalance_undersample(self, sample_fraud_data):
        """Test undersampling."""
        df_preprocessed = preprocess_data(sample_fraud_data)
        X_train, X_test, y_train, y_test = split_data(df_preprocessed)
        
        X_resampled, y_resampled = handle_imbalance(X_train, y_train, strategy='undersample')
        
        # Check that total samples decreased
        assert len(X_resampled) < len(X_train)
        
        # Classes should be balanced
        class_counts = y_resampled.value_counts()
        assert class_counts.iloc[0] == class_counts.iloc[1]
    
    def test_handle_imbalance_oversample(self, sample_fraud_data):
        """Test oversampling."""
        df_preprocessed = preprocess_data(sample_fraud_data)
        X_train, X_test, y_train, y_test = split_data(df_preprocessed)
        
        X_resampled, y_resampled = handle_imbalance(X_train, y_train, strategy='oversample')
        
        # Check that total samples increased
        assert len(X_resampled) >= len(X_train)
        
        # Classes should be balanced
        class_counts = y_resampled.value_counts()
        assert class_counts.iloc[0] == class_counts.iloc[1]
    
    def test_handle_imbalance_invalid_strategy(self, sample_fraud_data):
        """Test error handling for invalid strategy."""
        df_preprocessed = preprocess_data(sample_fraud_data)
        X_train, X_test, y_train, y_test = split_data(df_preprocessed)
        
        with pytest.raises(ValueError, match="Unknown strategy"):
            handle_imbalance(X_train, y_train, strategy='invalid')

        version = '1.5.0'
        versioned_path = version_model(model_path, registry_url, version)
        
        # Load metadata
        metadata_path = versioned_path.replace('.pkl', '_metadata.json')
        with open(metadata_path, 'r') as f:
            metadata = json.load(f)

        assert metadata['registry_url'] == registry_url
