class TestModelTraining:
    """Test suite for model training module."""
    
    def test_split_data(self, sample_data):
        """Test data splitting."""
        X_train, X_test, y_train, y_test = split_data(

        assert abs(train_fraud_rate - test_fraud_rate) < 0.02
    
    def test_split_data_different_sizes(self, sample_data):
        """Test data splitting with different test sizes."""
        X_train, X_test, y_train, y_test = split_data(
            sample_data, test_size=0.3, random_state=42
        )
        
        assert len(X_train) == 700
        assert len(X_test) == 300
    
    def test_handle_class_imbalance_smote(self, sample_data):
        """Test SMOTE resampling."""
        X_train, _, y_train, _ = split_data(sample_data, test_size=0.2, random_state=42)
        
        # Remove categorical columns for SMOTE
        X_train_numeric = X_train.select_dtypes(include=[np.number])
        
        X_resampled, y_resampled = handle_class_imbalance(
            X_train_numeric, y_train, strategy='smote'
        )
        
        # Check that classes are more balanced
        fraud_rate = y_resampled.mean()
        assert fraud_rate > 0.3  # Should be more balanced than original ~5%
    
    def test_handle_class_imbalance_undersample(self, sample_data):
        """Test undersampling."""
        X_train, _, y_train, _ = split_data(sample_data, test_size=0.2, random_state=42)
        X_train_numeric = X_train.select_dtypes(include=[np.number])
        
        initial_count = len(X_train_numeric)
        
        X_resampled, y_resampled = handle_class_imbalance(
            X_train_numeric, y_train, strategy='undersample'
        )
        
        # Undersampling should reduce total samples
        assert len(X_resampled) < initial_count
        
        # Classes should be balanced
        fraud_rate = y_resampled.mean()
        assert 0.4 < fraud_rate < 0.6
    
    def test_handle_class_imbalance_none(self, sample_data):
        """Test no resampling."""
        X_train, _, y_train, _ = split_data(sample_data, test_size=0.2, random_state=42)
        X_train_numeric = X_train.select_dtypes(include=[np.number])
        
        X_resampled, y_resampled = handle_class_imbalance(
            X_train_numeric, y_train, strategy='none'
        )
        
        # Should be unchanged
        assert len(X_resampled) == len(X_train_numeric)
        pd.testing.assert_frame_equal(X_resampled, X_train_numeric)

class TestModelEvaluation:

    def test_evaluate_model(self, trained_model, sample_data):
        """Test model evaluation."""
        X_test = sample_data.drop(columns=['is_fraud'])
        y_test = sample_data['is_fraud']
        
        # Encode categorical
        X_test_encoded = X_test.copy()

        ).codes

        assert 'precision' in metrics
        assert 'recall' in metrics
        assert 'f1_score' in metrics
        assert 'roc_auc' in metrics
        
        # All metrics should be between 0 and 1
        for metric, value in metrics.items():
            if not metric.startswith('true_') and not metric.startswith('false_'):
                assert 0 <= value <= 1
    
    def test_calculate_fraud_metrics(self):
        """Test fraud metrics calculation."""
        y_true = np.array([0, 0, 1, 1, 0, 1, 0, 1])
        y_pred = np.array([0, 0, 1, 0, 0, 1, 1, 1])
        y_pred_proba = np.array([0.1, 0.2, 0.8, 0.4, 0.3, 0.9, 0.6, 0.7])
        
        metrics = calculate_fraud_metrics(y_true, y_pred, y_pred_proba)
        
        assert metrics['accuracy'] == 0.75
        assert metrics['precision'] > 0
        assert metrics['recall'] > 0
        assert 'true_positives' in metrics
        assert 'false_positives' in metrics
        assert 'mcc' in metrics
    
    def test_compute_confusion_matrix(self):
        """Test confusion matrix computation."""
        y_true = np.array([0, 0, 1, 1, 0, 1, 0, 1])
        y_pred = np.array([0, 0, 1, 0, 0, 1, 1, 1])
        
        cm = compute_confusion_matrix(y_true, y_pred)
        
        assert cm.shape == (2, 2)
        assert cm.sum() == len(y_true)
    
    def test_calculate_business_metrics(self):
        """Test business metrics calculation."""
        y_true = np.array([0, 0, 1, 1, 0, 1, 0, 1])
        y_pred = np.array([0, 0, 1, 0, 0, 1, 1, 1])
        transaction_amounts = np.array([100, 200, 500, 300, 150, 400, 250, 600])
        
        metrics = calculate_business_metrics(y_true, y_pred, transaction_amounts)
        
        assert 'total_fraud_amount' in metrics
        assert 'detected_fraud_amount' in metrics
        assert 'missed_fraud_amount' in metrics
        assert 'investigation_cost' in metrics
        assert 'net_savings' in metrics
        assert 'roi' in metrics
        
        # Total fraud should equal detected + missed
        assert abs(
            metrics['total_fraud_amount'] - 
            (metrics['detected_fraud_amount'] + metrics['missed_fraud_amount'])
        ) < 0.01
