# src/pipeline/data_validation.py
"""Data validation module for fraud detection pipeline."""

import logging
from pathlib import Path
from typing import Dict, Tuple, Any
import pandas as pd
import numpy as np
from scipy import stats
from sklearn.preprocessing import StandardScaler, LabelEncoder
import json

logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)


def load_data(data_path: str) -> pd.DataFrame:
    """Load training or validation data from specified path.
    
    Args:
        data_path: Path to the data file (CSV, parquet, or JSON)
        
    Returns:
        DataFrame containing the loaded data
        
    Raises:
        FileNotFoundError: If the data file does not exist
        ValueError: If the file format is not supported
    """
    path = Path(data_path)
    
    if not path.exists():
        raise FileNotFoundError(f"Data file not found: {data_path}")
    
    logger.info(f"Loading data from {data_path}")
    
    # Determine file type and load accordingly
    if path.suffix == '.csv':
        df = pd.read_csv(data_path)
    elif path.suffix == '.parquet':
        df = pd.read_parquet(data_path)
    elif path.suffix == '.json':
        df = pd.read_json(data_path)
    else:
        raise ValueError(f"Unsupported file format: {path.suffix}")
    
    logger.info(f"Loaded {len(df)} rows and {len(df.columns)} columns")
    return df


def validate_schema(df: pd.DataFrame, expected_schema: Dict[str, str]) -> bool:
    """Validate that dataframe matches expected schema.
    
    Args:
        df: DataFrame to validate
        expected_schema: Dictionary mapping column names to expected data types
        
    Returns:
        True if schema is valid, False otherwise
    """
    logger.info("Validating data schema")
    
    # Check if all expected columns are present
    missing_columns = set(expected_schema.keys()) - set(df.columns)
    if missing_columns:
        logger.error(f"Missing columns: {missing_columns}")
        return False
    
    # Check if there are unexpected columns
    extra_columns = set(df.columns) - set(expected_schema.keys())
    if extra_columns:
        logger.warning(f"Extra columns found: {extra_columns}")
    
    # Validate data types
    type_mapping = {
        'int': ['int64', 'int32', 'int16', 'int8'],
        'float': ['float64', 'float32', 'float16'],
        'string': ['object', 'string'],
        'bool': ['bool'],
        'datetime': ['datetime64[ns]']
    }
    
    for col, expected_type in expected_schema.items():
        actual_type = str(df[col].dtype)
        
        # Map expected type to allowed pandas types
        allowed_types = type_mapping.get(expected_type, [expected_type])
        
        if actual_type not in allowed_types:
            logger.error(f"Column '{col}' has type '{actual_type}', expected '{expected_type}'")
            return False
    
    logger.info("Schema validation passed")
    return True


def check_data_quality(df: pd.DataFrame, quality_thresholds: Dict[str, float]) -> Dict[str, Any]:
    """Check data quality metrics against defined thresholds.
    
    Args:
        df: DataFrame to check
        quality_thresholds: Dictionary of quality metrics and their thresholds
            Example: {
                'missing_threshold': 0.1,  # Max 10% missing values
                'duplicate_threshold': 0.05,  # Max 5% duplicates
                'min_rows': 1000  # Minimum number of rows
            }
            
    Returns:
        Dictionary containing quality metrics and pass/fail status
    """
    logger.info("Checking data quality")
    
    results = {
        'passed': True,
        'metrics': {},
        'issues': []
    }
    
    # Check minimum rows
    min_rows = quality_thresholds.get('min_rows', 0)
    actual_rows = len(df)
    results['metrics']['row_count'] = actual_rows
    
    if actual_rows < min_rows:
        results['passed'] = False
        results['issues'].append(f"Insufficient rows: {actual_rows} < {min_rows}")
    
    # Check missing values
    missing_threshold = quality_thresholds.get('missing_threshold', 1.0)
    missing_pct = df.isnull().sum() / len(df)
    results['metrics']['missing_values'] = missing_pct.to_dict()
    
    for col, pct in missing_pct.items():
        if pct > missing_threshold:
            results['passed'] = False
            results['issues'].append(f"Column '{col}' has {pct:.2%} missing values (threshold: {missing_threshold:.2%})")
    
    # Check duplicates
    duplicate_threshold = quality_thresholds.get('duplicate_threshold', 1.0)
    duplicate_count = df.duplicated().sum()
    duplicate_pct = duplicate_count / len(df) if len(df) > 0 else 0
    results['metrics']['duplicate_percentage'] = duplicate_pct
    
    if duplicate_pct > duplicate_threshold:
        results['passed'] = False
        results['issues'].append(f"Duplicate rows: {duplicate_pct:.2%} (threshold: {duplicate_threshold:.2%})")
    
    # Check for infinite values in numeric columns
    numeric_cols = df.select_dtypes(include=[np.number]).columns
    for col in numeric_cols:
        inf_count = np.isinf(df[col]).sum()
        if inf_count > 0:
            results['passed'] = False
            results['issues'].append(f"Column '{col}' has {inf_count} infinite values")
    
    # Check variance (avoid zero-variance features)
    variance_threshold = quality_thresholds.get('min_variance', 0.0)
    for col in numeric_cols:
        variance = df[col].var()
        if variance < variance_threshold:
            results['issues'].append(f"Column '{col}' has low variance: {variance}")
    
    results['metrics']['total_issues'] = len(results['issues'])
    
    if results['passed']:
        logger.info("Data quality checks passed")
    else:
        logger.warning(f"Data quality checks failed with {len(results['issues'])} issues")
    
    return results


def detect_data_drift(current_data: pd.DataFrame, reference_data: pd.DataFrame) -> Dict[str, float]:
    """Detect statistical drift between current and reference datasets.
    
    Args:
        current_data: Current dataset
        reference_data: Reference/baseline dataset
        
    Returns:
        Dictionary mapping column names to drift scores (p-values from KS test)
    """
    logger.info("Detecting data drift")
    
    drift_results = {}
    
    # Get common numeric columns
    numeric_cols = set(current_data.select_dtypes(include=[np.number]).columns) & \
                   set(reference_data.select_dtypes(include=[np.number]).columns)
    
    for col in numeric_cols:
        # Remove NaN values for comparison
        current_values = current_data[col].dropna()
        reference_values = reference_data[col].dropna()
        
        if len(current_values) > 0 and len(reference_values) > 0:
            # Kolmogorov-Smirnov test for distribution comparison
            ks_statistic, p_value = stats.ks_2samp(current_values, reference_values)
            drift_results[col] = {
                'ks_statistic': float(ks_statistic),
                'p_value': float(p_value),
                'drift_detected': p_value < 0.05  # 5% significance level
            }
            
            if p_value < 0.05:
                logger.warning(f"Drift detected in column '{col}': p-value={p_value:.4f}")
    
    # Check categorical columns using chi-square test
    categorical_cols = set(current_data.select_dtypes(include=['object', 'category']).columns) & \
                       set(reference_data.select_dtypes(include=['object', 'category']).columns)
    
    for col in categorical_cols:
        current_counts = current_data[col].value_counts()
        reference_counts = reference_data[col].value_counts()
        
        # Align the categories
        all_categories = set(current_counts.index) | set(reference_counts.index)
        current_aligned = [current_counts.get(cat, 0) for cat in all_categories]
        reference_aligned = [reference_counts.get(cat, 0) for cat in all_categories]
        
        if sum(current_aligned) > 0 and sum(reference_aligned) > 0:
            # Chi-square test
            chi2_stat, p_value = stats.chisquare(current_aligned, reference_aligned)
            drift_results[col] = {
                'chi2_statistic': float(chi2_stat),
                'p_value': float(p_value),
                'drift_detected': p_value < 0.05
            }
            
            if p_value < 0.05:
                logger.warning(f"Drift detected in categorical column '{col}': p-value={p_value:.4f}")
    
    logger.info(f"Drift detection complete. Checked {len(drift_results)} columns")
    return drift_results


def validate_feature_distributions(df: pd.DataFrame, reference_stats: Dict[str, Any]) -> bool:
    """Validate feature distributions against reference statistics.
    
    Args:
        df: DataFrame to validate
        reference_stats: Dictionary containing reference statistics for each feature
            Example: {
                'feature_name': {
                    'mean': 0.5,
                    'std': 0.2,
                    'min': 0.0,
                    'max': 1.0,
                    'quantiles': [0.25, 0.5, 0.75]
                }
            }
            
    Returns:
        True if distributions are within acceptable bounds, False otherwise
    """
    logger.info("Validating feature distributions")
    
    validation_passed = True
    tolerance = 0.2  # 20% tolerance for distribution differences
    
    for col, stats in reference_stats.items():
        if col not in df.columns:
            logger.warning(f"Column '{col}' not found in data")
            continue
        
        col_data = df[col].dropna()
        
        if len(col_data) == 0:
            logger.warning(f"Column '{col}' has no valid data")
            validation_passed = False
            continue
        
        # Check mean
        if 'mean' in stats:
            actual_mean = col_data.mean()
            expected_mean = stats['mean']
            mean_diff = abs(actual_mean - expected_mean) / (abs(expected_mean) + 1e-10)
            
            if mean_diff > tolerance:
                logger.warning(f"Column '{col}' mean deviation: {mean_diff:.2%} (actual: {actual_mean:.4f}, expected: {expected_mean:.4f})")
                validation_passed = False
        
        # Check standard deviation
        if 'std' in stats:
            actual_std = col_data.std()
            expected_std = stats['std']
            std_diff = abs(actual_std - expected_std) / (abs(expected_std) + 1e-10)
            
            if std_diff > tolerance:
                logger.warning(f"Column '{col}' std deviation: {std_diff:.2%} (actual: {actual_std:.4f}, expected: {expected_std:.4f})")
                validation_passed = False
        
        # Check min/max bounds
        if 'min' in stats:
            actual_min = col_data.min()
            expected_min = stats['min']
            if actual_min < expected_min * (1 - tolerance):
                logger.warning(f"Column '{col}' min value out of bounds: {actual_min} < {expected_min}")
                validation_passed = False
        
        if 'max' in stats:
            actual_max = col_data.max()
            expected_max = stats['max']
            if actual_max > expected_max * (1 + tolerance):
                logger.warning(f"Column '{col}' max value out of bounds: {actual_max} > {expected_max}")
                validation_passed = False
    
    if validation_passed:
        logger.info("Feature distribution validation passed")
    else:
        logger.warning("Feature distribution validation failed")
    
    return validation_passed


def preprocess_data(df: pd.DataFrame) -> Tuple[pd.DataFrame, Dict[str, Any]]:
    """Apply preprocessing transformations to the data.
    
    Args:
        df: DataFrame to preprocess
        
    Returns:
        Tuple of (preprocessed DataFrame, preprocessing metadata)
    """
    logger.info("Preprocessing data")
    
    df_processed = df.copy()
    metadata = {
        'transformations': [],
        'columns_dropped': [],
        'columns_created': [],
        'encoders': {},
        'scalers': {}
    }
    
    # Handle missing values
    numeric_cols = df_processed.select_dtypes(include=[np.number]).columns
    categorical_cols = df_processed.select_dtypes(include=['object', 'category']).columns
    
    # Fill numeric missing values with median
    for col in numeric_cols:
        if df_processed[col].isnull().any():
            median_value = df_processed[col].median()
            df_processed[col].fillna(median_value, inplace=True)
            metadata['transformations'].append({
                'column': col,
                'operation': 'fill_missing',
                'value': float(median_value)
            })
    
    # Fill categorical missing values with mode
    for col in categorical_cols:
        if df_processed[col].isnull().any():
            mode_value = df_processed[col].mode()[0] if len(df_processed[col].mode()) > 0 else 'Unknown'
            df_processed[col].fillna(mode_value, inplace=True)
            metadata['transformations'].append({
                'column': col,
                'operation': 'fill_missing',
                'value': str(mode_value)
            })
    
    # Remove duplicate rows
    initial_rows = len(df_processed)
    df_processed.drop_duplicates(inplace=True)
    duplicates_removed = initial_rows - len(df_processed)
    
    if duplicates_removed > 0:
        logger.info(f"Removed {duplicates_removed} duplicate rows")
        metadata['transformations'].append({
            'operation': 'remove_duplicates',
            'rows_removed': duplicates_removed
        })
    
    # Encode categorical variables
    for col in categorical_cols:
        if col in df_processed.columns:
            le = LabelEncoder()
            df_processed[f'{col}_encoded'] = le.fit_transform(df_processed[col].astype(str))
            metadata['encoders'][col] = {
                'classes': le.classes_.tolist(),
                'encoded_column': f'{col}_encoded'
            }
            metadata['columns_created'].append(f'{col}_encoded')
    
    # Clip outliers (using IQR method)
    for col in numeric_cols:
        Q1 = df_processed[col].quantile(0.25)
        Q3 = df_processed[col].quantile(0.75)
        IQR = Q3 - Q1
        lower_bound = Q1 - 3 * IQR
        upper_bound = Q3 + 3 * IQR
        
        outliers_count = ((df_processed[col] < lower_bound) | (df_processed[col] > upper_bound)).sum()
        
        if outliers_count > 0:
            df_processed[col] = df_processed[col].clip(lower=lower_bound, upper=upper_bound)
            metadata['transformations'].append({
                'column': col,
                'operation': 'clip_outliers',
                'lower_bound': float(lower_bound),
                'upper_bound': float(upper_bound),
                'outliers_clipped': int(outliers_count)
            })
    
    # Scale numeric features
    scaler = StandardScaler()
    numeric_cols_list = numeric_cols.tolist()
    
    if len(numeric_cols_list) > 0:
        df_processed[numeric_cols_list] = scaler.fit_transform(df_processed[numeric_cols_list])
        metadata['scalers']['standard_scaler'] = {
            'columns': numeric_cols_list,
            'mean': scaler.mean_.tolist(),
            'scale': scaler.scale_.tolist()
        }
        metadata['transformations'].append({
            'operation': 'standard_scaling',
            'columns': numeric_cols_list
        })
    
    logger.info(f"Preprocessing complete. Applied {len(metadata['transformations'])} transformations")
    
    return df_processed, metadata

"""Model training module for fraud detection pipeline."""

import logging
from typing import Dict, Any, Tuple
import pandas as pd
import numpy as np
from sklearn.model_selection import train_test_split, GridSearchCV, StratifiedKFold
from sklearn.ensemble import RandomForestClassifier, GradientBoostingClassifier
from sklearn.linear_model import LogisticRegression
from sklearn.base import BaseEstimator
from imblearn.over_sampling import SMOTE
from imblearn.under_sampling import RandomUnderSampler
from imblearn.combine import SMOTETomek
import joblib
import json
from pathlib import Path
from datetime import datetime

logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)


def split_data(
    df: pd.DataFrame, 
    test_size: float, 
    random_state: int
) -> Tuple[pd.DataFrame, pd.DataFrame, pd.Series, pd.Series]:
    """Split data into training and validation sets.
    
    Args:
        df: DataFrame containing features and target
        test_size: Proportion of data to use for testing (0.0 to 1.0)
        random_state: Random seed for reproducibility
        
    Returns:
        Tuple of (X_train, X_test, y_train, y_test)
    """
    logger.info(f"Splitting data with test_size={test_size}, random_state={random_state}")
    
    # Assume the last column or 'is_fraud' column is the target
    if 'is_fraud' in df.columns:
        target_col = 'is_fraud'
    elif 'fraud' in df.columns:
        target_col = 'fraud'
    else:
        target_col = df.columns[-1]
        logger.warning(f"Target column not explicitly named, using last column: {target_col}")
    
    X = df.drop(columns=[target_col])
    y = df[target_col]
    
    # Stratified split to maintain class distribution
    X_train, X_test, y_train, y_test = train_test_split(
        X, y, 
        test_size=test_size, 
        random_state=random_state,
        stratify=y
    )
    
    logger.info(f"Training set: {len(X_train)} samples")
    logger.info(f"Test set: {len(X_test)} samples")
    logger.info(f"Training set fraud rate: {y_train.mean():.2%}")
    logger.info(f"Test set fraud rate: {y_test.mean():.2%}")
    
    return X_train, X_test, y_train, y_test


def handle_class_imbalance(
    X_train: pd.DataFrame, 
    y_train: pd.Series, 
    strategy: str
) -> Tuple[pd.DataFrame, pd.Series]:
    """Handle class imbalance using specified strategy.
    
    Args:
        X_train: Training features
        y_train: Training labels
        strategy: Resampling strategy ('smote', 'undersample', 'smotetomek', 'none')
        
    Returns:
        Tuple of (resampled X_train, resampled y_train)
    """
    logger.info(f"Handling class imbalance using strategy: {strategy}")
    
    initial_count = len(X_train)
    initial_fraud_rate = y_train.mean()
    
    logger.info(f"Initial dataset: {initial_count} samples, fraud rate: {initial_fraud_rate:.2%}")
    
    if strategy.lower() == 'none':
        logger.info("No resampling applied")
        return X_train, y_train
    
    elif strategy.lower() == 'smote':
        smote = SMOTE(random_state=42, k_neighbors=5)
        X_resampled, y_resampled = smote.fit_resample(X_train, y_train)
        
    elif strategy.lower() == 'undersample':
        undersampler = RandomUnderSampler(random_state=42)
        X_resampled, y_resampled = undersampler.fit_resample(X_train, y_train)
        
    elif strategy.lower() == 'smotetomek':
        smotetomek = SMOTETomek(random_state=42)
        X_resampled, y_resampled = smotetomek.fit_resample(X_train, y_train)
        
    else:
        raise ValueError(f"Unknown resampling strategy: {strategy}")
    
    # Convert back to DataFrame/Series with original column names
    X_resampled = pd.DataFrame(X_resampled, columns=X_train.columns)
    y_resampled = pd.Series(y_resampled, name=y_train.name)
    
    final_count = len(X_resampled)
    final_fraud_rate = y_resampled.mean()
    
    logger.info(f"Resampled dataset: {final_count} samples, fraud rate: {final_fraud_rate:.2%}")
    logger.info(f"Sample count change: {final_count - initial_count:+d}")
    
    return X_resampled, y_resampled


def train_model(
    X_train: pd.DataFrame, 
    y_train: pd.Series, 
    model_params: Dict[str, Any]
) -> BaseEstimator:
    """Train fraud detection model with specified parameters.
    
    Args:
        X_train: Training features
        y_train: Training labels
        model_params: Dictionary containing model configuration
            Example: {
                'model_type': 'random_forest',
                'n_estimators': 100,
                'max_depth': 10,
                'random_state': 42
            }
            
    Returns:
        Trained model
    """
    logger.info("Training fraud detection model")
    
    model_type = model_params.get('model_type', 'random_forest').lower()
    random_state = model_params.get('random_state', 42)
    
    # Create model based on type
    if model_type == 'random_forest':
        model = RandomForestClassifier(
            n_estimators=model_params.get('n_estimators', 100),
            max_depth=model_params.get('max_depth', 10),
            min_samples_split=model_params.get('min_samples_split', 2),
            min_samples_leaf=model_params.get('min_samples_leaf', 1),
            max_features=model_params.get('max_features', 'sqrt'),
            random_state=random_state,
            n_jobs=-1,
            class_weight=model_params.get('class_weight', 'balanced')
        )
        
    elif model_type == 'gradient_boosting':
        model = GradientBoostingClassifier(
            n_estimators=model_params.get('n_estimators', 100),
            max_depth=model_params.get('max_depth', 5),
            learning_rate=model_params.get('learning_rate', 0.1),
            subsample=model_params.get('subsample', 0.8),
            random_state=random_state
        )
        
    elif model_type == 'logistic_regression':
        model = LogisticRegression(
            C=model_params.get('C', 1.0),
            penalty=model_params.get('penalty', 'l2'),
            solver=model_params.get('solver', 'lbfgs'),
            max_iter=model_params.get('max_iter', 1000),
            random_state=random_state,
            class_weight=model_params.get('class_weight', 'balanced'),
            n_jobs=-1
        )
        
    else:
        raise ValueError(f"Unknown model type: {model_type}")
    
    logger.info(f"Training {model_type} model with {len(X_train)} samples")
    
    # Train the model
    model.fit(X_train, y_train)
    
    # Log feature importances if available
    if hasattr(model, 'feature_importances_'):
        feature_importance = pd.DataFrame({
            'feature': X_train.columns,
            'importance': model.feature_importances_
        }).sort_values('importance', ascending=False)
        
        logger.info("Top 10 most important features:")
        for idx, row in feature_importance.head(10).iterrows():
            logger.info(f"  {row['feature']}: {row['importance']:.4f}")
    
    logger.info("Model training complete")
    
    return model


def tune_hyperparameters(
    X_train: pd.DataFrame, 
    y_train: pd.Series, 
    param_grid: Dict[str, list]
) -> Dict[str, Any]:
    """Perform hyperparameter tuning using cross-validation.
    
    Args:
        X_train: Training features
        y_train: Training labels
        param_grid: Dictionary of parameters to search
            Example: {
                'model_type': ['random_forest'],
                'n_estimators': [50, 100, 200],
                'max_depth': [5, 10, 15],
                'min_samples_split': [2, 5, 10]
            }
            
    Returns:
        Dictionary containing best parameters and cross-validation results
    """
    logger.info("Starting hyperparameter tuning")
    
    model_type = param_grid.get('model_type', ['random_forest'])[0]
    
    # Create base model
    if model_type == 'random_forest':
        base_model = RandomForestClassifier(random_state=42, n_jobs=-1)
    elif model_type == 'gradient_boosting':
        base_model = GradientBoostingClassifier(random_state=42)
    elif model_type == 'logistic_regression':
        base_model = LogisticRegression(random_state=42, n_jobs=-1)
    else:
        raise ValueError(f"Unknown model type: {model_type}")
    
    # Remove model_type from param_grid for GridSearchCV
    search_params = {k: v for k, v in param_grid.items() if k != 'model_type'}
    
    # Setup cross-validation
    cv = StratifiedKFold(n_splits=5, shuffle=True, random_state=42)
    
    # Perform grid search
    grid_search = GridSearchCV(
        estimator=base_model,
        param_grid=search_params,
        cv=cv,
        scoring='f1',  # Use F1 score for fraud detection
        n_jobs=-1,
        verbose=1
    )
    
    logger.info(f"Searching {len(search_params)} parameters with {cv.n_splits}-fold CV")
    grid_search.fit(X_train, y_train)
    
    results = {
        'best_params': grid_search.best_params_,
        'best_score': float(grid_search.best_score_),
        'cv_results': {
            'mean_test_score': grid_search.cv_results_['mean_test_score'].tolist(),
            'std_test_score': grid_search.cv_results_['std_test_score'].tolist(),
            'params': [str(p) for p in grid_search.cv_results_['params']]
        }
    }
    
    logger.info(f"Best parameters: {results['best_params']}")
    logger.info(f"Best CV F1 score: {results['best_score']:.4f}")
    
    return results

def calculate_fraud_metrics(
    y_true: np.ndarray, 
    y_pred: np.ndarray, 
    y_pred_proba: np.ndarray
) -> Dict[str, float]:
    """Calculate fraud-specific metrics.
    
    Args:
        y_true: True labels
        y_pred: Predicted labels
        y_pred_proba: Predicted probabilities
        
    Returns:
        Dictionary containing fraud detection metrics
    """
    metrics = {}
    
    # Basic classification metrics
    metrics['accuracy'] = float(accuracy_score(y_true, y_pred))
    metrics['precision'] = float(precision_score(y_true, y_pred, zero_division=0))
    metrics['recall'] = float(recall_score(y_true, y_pred, zero_division=0))
    metrics['f1_score'] = float(f1_score(y_true, y_pred, zero_division=0))
    
    # ROC AUC score
    if len(np.unique(y_true)) > 1:
        metrics['roc_auc'] = float(roc_auc_score(y_true, y_pred_proba))
        metrics['average_precision'] = float(average_precision_score(y_true, y_pred_proba))
    else:
        metrics['roc_auc'] = 0.0
        metrics['average_precision'] = 0.0
    
    # Confusion matrix components
    tn, fp, fn, tp = confusion_matrix(y_true, y_pred).ravel()
    
    metrics['true_positives'] = int(tp)
    metrics['true_negatives'] = int(tn)
    metrics['false_positives'] = int(fp)
    metrics['false_negatives'] = int(fn)
    
    # Additional fraud-specific metrics
    metrics['specificity'] = float(tn / (tn + fp)) if (tn + fp) > 0 else 0.0
    metrics['false_positive_rate'] = float(fp / (fp + tn)) if (fp + tn) > 0 else 0.0
    metrics['false_negative_rate'] = float(fn / (fn + tp)) if (fn + tp) > 0 else 0.0
    
    # Matthews correlation coefficient
    denominator = np.sqrt((tp + fp) * (tp + fn) * (tn + fp) * (tn + fn))
    metrics['mcc'] = float((tp * tn - fp * fn) / denominator) if denominator > 0 else 0.0
    
    return metrics


def compute_confusion_matrix(y_true: np.ndarray, y_pred: np.ndarray) -> np.ndarray:
    """Compute confusion matrix for fraud predictions.
    
    Args:
        y_true: True labels
        y_pred: Predicted labels
        
    Returns:
        Confusion matrix as numpy array
    """
    cm = confusion_matrix(y_true, y_pred)
    
    logger.info("Confusion Matrix:")
    logger.info(f"  TN: {cm[0, 0]}, FP: {cm[0, 1]}")
    logger.info(f"  FN: {cm[1, 0]}, TP: {cm[1, 1]}")
    
    return cm


def calculate_business_metrics(
    y_true: np.ndarray, 
    y_pred: np.ndarray, 
    transaction_amounts: np.ndarray
) -> Dict[str, float]:
    """Calculate business impact metrics.
    
    Args:
        y_true: True labels
        y_pred: Predicted labels
        transaction_amounts: Transaction amounts for each sample
        
    Returns:
        Dictionary containing business metrics
    """
    logger.info("Calculating business impact metrics")
    
    # Confusion matrix
    tn, fp, fn, tp = confusion_matrix(y_true, y_pred).ravel()
    
    # Define costs (these should be configurable based on business requirements)
    fraud_investigation_cost = 25  # Cost to investigate a flagged transaction
    fraud_loss_rate = 1.0  # Percentage of transaction amount lost to fraud
    
    # Calculate business metrics
    metrics = {}
    
    # Total fraud amount (actual fraud in dataset)
    actual_fraud_mask = y_true == 1
    metrics['total_fraud_amount'] = float(transaction_amounts[actual_fraud_mask].sum())
    
    # Detected fraud amount (true positives)
    detected_fraud_mask = (y_true == 1) & (y_pred == 1)
    metrics['detected_fraud_amount'] = float(transaction_amounts[detected_fraud_mask].sum())
    
    # Missed fraud amount (false negatives)
    missed_fraud_mask = (y_true == 1) & (y_pred == 0)
    metrics['missed_fraud_amount'] = float(transaction_amounts[missed_fraud_mask].sum())
    
    # False positive amount (legitimate transactions flagged as fraud)
    false_positive_mask = (y_true == 0) & (y_pred == 1)
    metrics['false_positive_amount'] = float(transaction_amounts[false_positive_mask].sum())
    
    # Cost calculations
    metrics['investigation_cost'] = float((tp + fp) * fraud_investigation_cost)
    metrics['fraud_losses'] = float(metrics['missed_fraud_amount'] * fraud_loss_rate)
    metrics['total_cost'] = metrics['investigation_cost'] + metrics['fraud_losses']
    
    # Savings and ROI
    metrics['fraud_prevented'] = metrics['detected_fraud_amount']
    metrics['net_savings'] = metrics['fraud_prevented'] - metrics['total_cost']
    
    if metrics['total_fraud_amount'] > 0:
        metrics['fraud_detection_rate'] = metrics['detected_fraud_amount'] / metrics['total_fraud_amount']
    else:
        metrics['fraud_detection_rate'] = 0.0
    
    # ROI calculation
    baseline_cost = metrics['total_fraud_amount']  # Cost without any fraud detection
    if baseline_cost > 0:
        metrics['roi'] = (baseline_cost - metrics['total_cost']) / baseline_cost
    else:
        metrics['roi'] = 0.0
    
    logger.info(f"Total fraud amount: ${metrics['total_fraud_amount']:,.2f}")
    logger.info(f"Detected fraud: ${metrics['detected_fraud_amount']:,.2f}")
    logger.info(f"Missed fraud: ${metrics['missed_fraud_amount']:,.2f}")
    logger.info(f"Net savings: ${metrics['net_savings']:,.2f}")
    logger.info(f"ROI: {metrics['roi']:.2%}")
    
    return metrics


def compare_model_performance(
    current_metrics: Dict[str, float], 
    baseline_metrics: Dict[str, float], 
    threshold: float
) -> bool:
    """Compare current model performance against baseline.
    
    Args:
        current_metrics: Metrics from current model
        baseline_metrics: Metrics from baseline model
        threshold: Minimum improvement threshold (e.g., 0.05 for 5% improvement)
        
    Returns:
        True if current model is better than baseline, False otherwise
    """
    logger.info("Comparing model performance against baseline")
    
    # Key metrics to compare (weighted by importance)
    key_metrics = {
        'f1_score': 0.3,
        'roc_auc': 0.3,
        'recall': 0.2,
        'precision': 0.2
    }
    
    weighted_improvement = 0.0
    
    for metric, weight in key_metrics.items():
        if metric in current_metrics and metric in baseline_metrics:
            current_value = current_metrics[metric]
            baseline_value = baseline_metrics[metric]
            
            if baseline_value > 0:
                improvement = (current_value - baseline_value) / baseline_value
            else:
                improvement = current_value
            
            weighted_improvement += improvement * weight
            
            logger.info(f"  {metric}: {baseline_value:.4f} -> {current_value:.4f} ({improvement:+.2%})")
    
    is_better = weighted_improvement >= threshold
    
    logger.info(f"Weighted improvement: {weighted_improvement:+.2%} (threshold: {threshold:+.2%})")
    logger.info(f"Model comparison result: {'PASS' if is_better else 'FAIL'}")
    
    return is_better

def validate_model_fairness(
    model: BaseEstimator, 
    X_test: pd.DataFrame, 
    y_test: pd.Series, 
    sensitive_features: List[str]
) -> Dict[str, float]:
    """Validate model fairness across sensitive feature groups.
    
    Args:
        model: Trained model
        X_test: Test features
        y_test: Test labels
        sensitive_features: List of sensitive feature column names
        
    Returns:
        Dictionary containing fairness metrics for each sensitive feature
    """
    logger.info("Validating model fairness")
    
    fairness_metrics = {}
    
    # Generate predictions
    y_pred = model.predict(X_test)
    y_pred_proba = model.predict_proba(X_test)[:, 1] if hasattr(model, 'predict_proba') else y_pred
    
    for feature in sensitive_features:
        if feature not in X_test.columns:
            logger.warning(f"Sensitive feature '{feature}' not found in test data")
            continue
        
        logger.info(f"Analyzing fairness for feature: {feature}")
        
        # Get unique groups
        groups = X_test[feature].unique()
        group_metrics = {}
        
        for group in groups:
            # Filter data for this group
            group_mask = X_test[feature] == group
            y_true_group = y_test[group_mask]
            y_pred_group = y_pred[group_mask]
            y_pred_proba_group = y_pred_proba[group_mask]
            
            # Calculate metrics for this group
            if len(y_true_group) > 0:
                group_metrics[str(group)] = {
                    'sample_size': int(group_mask.sum()),
                    'precision': float(precision_score(y_true_group, y_pred_group, zero_division=0)),
                    'recall': float(recall_score(y_true_group, y_pred_group, zero_division=0)),
                    'f1_score': float(f1_score(y_true_group, y_pred_group, zero_division=0)),
                    'false_positive_rate': float(
                        (y_pred_group[y_true_group == 0] == 1).sum() / max((y_true_group == 0).sum(), 1)
                    )
                }
        
        # Calculate disparity metrics
        if len(group_metrics) >= 2:
            # Calculate max disparity in false positive rates
            fpr_values = [m['false_positive_rate'] for m in group_metrics.values()]
            max_fpr_disparity = max(fpr_values) - min(fpr_values)
            
            # Calculate max disparity in recall
            recall_values = [m['recall'] for m in group_metrics.values()]
            max_recall_disparity = max(recall_values) - min(recall_values)
            
            fairness_metrics[feature] = {
                'groups': group_metrics,
                'max_fpr_disparity': float(max_fpr_disparity),
                'max_recall_disparity': float(max_recall_disparity),
                'is_fair': max_fpr_disparity < 0.1 and max_recall_disparity < 0.1  # 10% threshold
            }
            
            logger.info(f"  Max FPR disparity: {max_fpr_disparity:.4f}")
            logger.info(f"  Max recall disparity: {max_recall_disparity:.4f}")
            logger.info(f"  Fairness assessment: {'PASS' if fairness_metrics[feature]['is_fair'] else 'FAIL'}")
    
    return fairness_metrics

# src/pipeline/model_registry.py
"""Model registry module for fraud detection pipeline."""

import logging
from typing import Dict, Any, Optional, Tuple, List
from sklearn.base import BaseEstimator
import joblib
import json
from pathlib import Path
from datetime import datetime
import shutil

logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)


class ModelRegistry:
    """Simple file-based model registry."""
    
    def __init__(self, registry_path: str = "models/registry"):
        self.registry_path = Path(registry_path)
        self.registry_path.mkdir(parents=True, exist_ok=True)
        self.index_file = self.registry_path / "index.json"
        self._load_index()
    
    def _load_index(self):
        """Load registry index."""
        if self.index_file.exists():
            with open(self.index_file, 'r') as f:
                self.index = json.load(f)
        else:
            self.index = {}
    
    def _save_index(self):
        """Save registry index."""
        with open(self.index_file, 'w') as f:
            json.dump(self.index, f, indent=2)
    
    def _get_model_path(self, model_name: str, version: str) -> Path:
        """Get path for a specific model version."""
        return self.registry_path / model_name / version


# Global registry instance
_registry = None


def _get_registry() -> ModelRegistry:
    """Get or create global registry instance."""
    global _registry
    if _registry is None:
        _registry = ModelRegistry()
    return _registry


def register_model(
    model: BaseEstimator, 
    model_name: str, 
    version: str, 
    metrics: Dict[str, float], 
    tags: Dict[str, str]
) -> str:
    """Register model in model registry with metadata.
    
    Args:
        model: Trained model to register
        model_name: Name of the model
        version: Version string (e.g., "1.0.0", "v2023-01-15")
        metrics: Dictionary of model performance metrics
        tags: Dictionary of tags for the model
        
    Returns:
        Model URI (path to registered model)
    """
    logger.info(f"Registering model: {model_name} version {version}")
    
    registry = _get_registry()
    model_path = registry._get_model_path(model_name, version)
    model_path.mkdir(parents=True, exist_ok=True)
    
    # Save model
    model_file = model_path / "model.joblib"
    joblib.dump(model, model_file)
    
    # Create metadata
    metadata = {
        'model_name': model_name,
        'version': version,
        'registered_at': datetime.now().isoformat(),
        'metrics': metrics,
        'tags': tags,
        'model_class': model.__class__.__name__,
        'stage': 'none',
        'model_file': str(model_file)
    }
    
    # Save metadata
    metadata_file = model_path / "metadata.json"
    with open(metadata_file, 'w') as f:
        json.dump(metadata, f, indent=2)
    
    # Update registry index
    if model_name not in registry.index:
        registry.index[model_name] = {'versions': {}}
    
    registry.index[model_name]['versions'][version] = {
        'registered_at': metadata['registered_at'],
        'stage': 'none',
        'metrics': metrics,
        'path': str(model_path)
    }
    
    registry._save_index()
    
    logger.info(f"Model registered successfully at {model_path}")
    
    return str(model_path)


def get_model_version(
    model_name: str, 
    version: Optional[str] = None
) -> Tuple[BaseEstimator, Dict[str, Any]]:
    """Retrieve specific model version from registry.
    
    Args:
        model_name: Name of the model
        version: Version string (if None, returns latest version)
        
    Returns:
        Tuple of (model, metadata)
        
    Raises:
        ValueError: If model or version not found
    """
    logger.info(f"Retrieving model: {model_name} version {version or 'latest'}")
    
    registry = _get_registry()
    
    if model_name not in registry.index:
        raise ValueError(f"Model '{model_name}' not found in registry")
    
    # Get version
    if version is None:
        # Get latest version
        versions = registry.index[model_name]['versions']
        if not versions:
            raise ValueError(f"No versions found for model '{model_name}'")
        
        # Sort by registration time and get latest
        version = max(
            versions.items(),
            key=lambda x: x[1]['registered_at']
        )[0]
        logger.info(f"Using latest version: {version}")
    
    if version not in registry.index[model_name]['versions']:
        raise ValueError(f"Version '{version}' not found for model '{model_name}'")
    
    # Load model
    model_path = registry._get_model_path(model_name, version)
    model_file = model_path / "model.joblib"
    metadata_file = model_path / "metadata.json"
    
    model = joblib.load(model_file)
    
    with open(metadata_file, 'r') as f:
        metadata = json.load(f)
    
    logger.info(f"Model loaded: {model.__class__.__name__}")
    
    return model, metadata


def promote_model(model_name: str, version: str, stage: str) -> None:
    """Promote model to specified stage.
    
    Args:
        model_name: Name of the model
        version: Version string
        stage: Target stage ('staging', 'production', 'archived')
        
    Raises:
        ValueError: If model/version not found or invalid stage
    """
    valid_stages = ['none', 'staging', 'production', 'archived']
    
    if stage not in valid_stages:
        raise ValueError(f"Invalid stage '{stage}'. Must be one of {valid_stages}")
    
    logger.info(f"Promoting model {model_name} version {version} to {stage}")
    
    registry = _get_registry()
    
    if model_name not in registry.index:
        raise ValueError(f"Model '{model_name}' not found in registry")
    
    if version not in registry.index[model_name]['versions']:
        raise ValueError(f"Version '{version}' not found for model '{model_name}'")
    
    # If promoting to production, demote current production model
    if stage == 'production':
        for ver, info in registry.index[model_name]['versions'].items():
            if info['stage'] == 'production':
                logger.info(f"Demoting version {ver} from production to staging")
                info['stage'] = 'staging'
    
    # Update stage
    registry.index[model_name]['versions'][version]['stage'] = stage
    
    # Update metadata file
    model_path = registry._get_model_path(model_name, version)
    metadata_file = model_path / "metadata.json"
    
    with open(metadata_file, 'r') as f:
        metadata = json.load(f)
    
    metadata['stage'] = stage
    metadata['promoted_at'] = datetime.now().isoformat()
    
    with open(metadata_file, 'w') as f:
        json.dump(metadata, f, indent=2)
    
    registry._save_index()
    
    logger.info(f"Model promoted to {stage} successfully")


def compare_model_versions(
    model_name: str, 
    version1: str, 
    version2: str
) -> Dict[str, Any]:
    """Compare metrics between two model versions.
    
    Args:
        model_name: Name of the model
        version1: First version to compare
        version2: Second version to compare
        
    Returns:
        Dictionary containing comparison results
    """
    logger.info(f"Comparing {model_name} versions {version1} vs {version2}")
    
    registry = _get_registry()
    
    if model_name not in registry.index:
        raise ValueError(f"Model '{model_name}' not found in registry")
    
    versions = registry.index[model_name]['versions']
    
    if version1 not in versions:
        raise ValueError(f"Version '{version1}' not found")
    if version2 not in versions:
        raise ValueError(f"Version '{version2}' not found")
    
    metrics1 = versions[version1]['metrics']
    metrics2 = versions[version2]['metrics']
    
    comparison = {
        'version1': version1,
        'version2': version2,
        'metrics_comparison': {},
        'winner': None
    }
    
    # Compare each metric
    score1 = 0
    score2 = 0
    
    for metric in set(metrics1.keys()) | set(metrics2.keys()):
        val1 = metrics1.get(metric, 0)
        val2 = metrics2.get(metric, 0)
        
        comparison['metrics_comparison'][metric] = {
            'version1': val1,
            'version2': val2,
            'difference': val2 - val1,
            'percent_change': ((val2 - val1) / val1 * 100) if val1 != 0 else 0
        }
        
        # Simple scoring (higher is better for most metrics)
        if val1 > val2:
            score1 += 1
        elif val2 > val1:
            score2 += 1
    
    if score1 > score2:
        comparison['winner'] = version1
    elif score2 > score1:
        comparison['winner'] = version2
    else:
        comparison['winner'] = 'tie'
    
    logger.info(f"Comparison complete. Winner: {comparison['winner']}")
    
    return comparison


def get_production_model(model_name: str) -> Tuple[BaseEstimator, Dict[str, Any]]:
    """Retrieve current production model.
    
    Args:
        model_name: Name of the model

def archive_old_models(model_name: str, keep_latest_n: int) -> List[str]:
    """Archive old model versions, keeping only the latest N versions.
    
    Args:
        model_name: Name of the model
        keep_latest_n: Number of latest versions to keep
        
    Returns:
        List of archived version strings
    """
    logger.info(f"Archiving old versions of {model_name}, keeping latest {keep_latest_n}")
    
    registry = _get_registry()
    
    if model_name not in registry.index:
        raise ValueError(f"Model '{model_name}' not found in registry")
    
    versions = registry.index[model_name]['versions']
    
    # Sort versions by registration time
    sorted_versions = sorted(
        versions.items(),
        key=lambda x: x[1]['registered_at'],
        reverse=True
    )
    
    archived_versions = []
    
    # Archive versions beyond keep_latest_n
    for i, (version, info) in enumerate(sorted_versions):
        if i >= keep_latest_n and info['stage'] not in ['production', 'staging']:
            # Archive this version
            info['stage'] = 'archived'
            archived_versions.append(version)
            
            # Update metadata
            model_path = registry._get_model_path(model_name, version)
            metadata_file = model_path / "metadata.json"
            
            with open(metadata_file, 'r') as f:
                metadata = json.load(f)
            
            metadata['stage'] = 'archived'
            metadata['archived_at'] = datetime.now().isoformat()
            
            with open(metadata_file, 'w') as f:
                json.dump(metadata, f, indent=2)
    
    registry._save_index()
    
    logger.info(f"Archived {len(archived_versions)} versions: {archived_versions}")
    
    return archived_versions

def create_deployment_package(
    model_path: str, 
    dependencies: list, 
    output_path: str
) -> str:
    """Create deployment package with model and dependencies.
    
    Args:
        model_path: Path to the trained model
        dependencies: List of dependency files/directories to include
        output_path: Path where deployment package should be created
        
    Returns:
        Path to created deployment package
    """
    logger.info(f"Creating deployment package at {output_path}")
    
    output_path = Path(output_path)
    output_path.parent.mkdir(parents=True, exist_ok=True)
    
    # Create temporary directory for package contents
    package_dir = output_path.parent / f"package_tmp_{datetime.now().strftime('%Y%m%d_%H%M%S')}"
    package_dir.mkdir(exist_ok=True)
    
    try:
        # Copy model
        model_src = Path(model_path)
        if model_src.is_file():
            shutil.copy(model_src, package_dir / "model.joblib")
        else:
            shutil.copytree(model_src, package_dir / "model")
        
        # Copy metadata if exists
        metadata_path = model_src.parent / f"{model_src.stem}_metadata.json"
        if metadata_path.exists():
            shutil.copy(metadata_path, package_dir / "metadata.json")
        
        # Copy dependencies
        for dep in dependencies:
            dep_path = Path(dep)
            if dep_path.exists():
                if dep_path.is_file():
                    shutil.copy(dep_path, package_dir / dep_path.name)
                else:
                    shutil.copytree(dep_path, package_dir / dep_path.name)
        
        # Create package manifest
        manifest = {
            'created_at': datetime.now().isoformat(),
            'model_path': str(model_path),
            'dependencies': dependencies,
            'contents': [str(p.relative_to(package_dir)) for p in package_dir.rglob('*') if p.is_file()]
        }
        
        with open(package_dir / "manifest.json", 'w') as f:
            json.dump(manifest, f, indent=2)
        
        # Create tarball
        with tarfile.open(output_path, 'w:gz') as tar:
            tar.add(package_dir, arcname='.')
        
        logger.info(f"Deployment package created: {output_path}")
        
        return str(output_path)
    
    finally:
        # Clean up temporary directory
        if package_dir.exists():
            shutil.rmtree(package_dir)

                    'passed': False
                }

            validation_results['checks'].append(check)
    
    # Check memory (simplified - in production would check actual system resources)

    Returns:
        Deployment ID or endpoint URL
    """
    logger.info(f"Deploying model to {environment}")
    
    # Create deployment directory
    deployment_base = Path(deployment_config.get('deployment_path', 'deployments'))
    deployment_dir = deployment_base / environment / datetime.now().strftime('%Y%m%d_%H%M%S')
    deployment_dir.mkdir(parents=True, exist_ok=True)
    
    # Copy model to deployment directory
    model_src = Path(model_path)

        shutil.copytree(model_src, deployment_dir / "model")
        model_dest = deployment_dir / "model"

    deployment_metadata = {
        'deployment_id': deployment_dir.name,
        'environment': environment,
        'model_path': str(model_path),
        'deployed_at': datetime.now().isoformat(),

        """Test SMOTE resampling."""
        X_train, _, y_train, _ = split_data(sample_data, test_size=0.2, random_state=42)
        
        # Remove categorical columns for SMOTE
        X_train_numeric = X_train.select_dtypes(include=[np.number])
        
        X_resampled, y_resampled = handle_class_imbalance(
            X_train_numeric, y_train, strategy='smote'
        )
        
        # Check that classes are more balanced
        fraud_rate = y_resampled.mean()
        assert fraud_rate > 0.3  # Should be more balanced than original ~5%

        """Test undersampling."""
        X_train, _, y_train, _ = split_data(sample_data, test_size=0.2, random_state=42)
        X_train_numeric = X_train.select_dtypes(include=[np.number])
        
        initial_count = len(X_train_numeric)
        
        X_resampled, y_resampled = handle_class_imbalance(
            X_train_numeric, y_train, strategy='undersample'

        # Undersampling should reduce total samples
        assert len(X_resampled) < initial_count
        
        # Classes should be balanced
        fraud_rate = y_resampled.mean()
        assert 0.4 < fraud_rate < 0.6
