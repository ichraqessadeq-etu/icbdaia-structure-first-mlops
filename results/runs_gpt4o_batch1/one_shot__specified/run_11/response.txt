Below are the complete Python pipeline module, pytest test file, and GitHub Actions workflow for a CI/CD pipeline to train and evaluate a fraud detection model with scikit-learn's `RandomForestClassifier`. The pipeline uses MLflow for experiment tracking and pytest for testing. It includes automatic triggers for retraining when concept drift is detected.

### Python Pipeline Script

```python
# fraud_detection_pipeline.py

import pandas as pd
import numpy as np
from sklearn.model_selection import train_test_split
from sklearn.ensemble import RandomForestClassifier
from sklearn.metrics import classification_report, accuracy_score
import mlflow
import mlflow.sklearn
from sklearn.metrics import roc_auc_score
from sklearn.utils import resample
import logging

logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)

# Function to load data
def load_data(file_path):
    try:
        data = pd.read_csv(file_path)
        logger.info("Data loaded successfully.")
        return data
    except Exception as e:
        logger.error(f"Error loading data: {e}")
        raise

# Function to preprocess data
def preprocess_data(data):
    try:
        X = data.drop("Class", axis=1)
        y = data["Class"]
        return X, y
    except Exception as e:
        logger.error(f"Error in data preprocessing: {e}")
        raise

# Function to handle class imbalance
def balance_data(X, y):
    try:
        X_majority = X[y == 0]
        y_majority = y[y == 0]
        X_minority = X[y == 1]
        y_minority = y[y == 1]

        X_minority_upsampled, y_minority_upsampled = resample(
            X_minority, y_minority,
            replace=True,
            n_samples=X_majority.shape[0],
            random_state=42
        )

        X_balanced = np.vstack((X_majority, X_minority_upsampled))
        y_balanced = np.hstack((y_majority, y_minority_upsampled))
        
        logger.info("Data balanced successfully.")
        return X_balanced, y_balanced
    except Exception as e:
        logger.error(f"Error balancing data: {e}")
        raise

# Function to train model
def train_model(X_train, y_train):
    try:
        model = RandomForestClassifier(random_state=42)
        model.fit(X_train, y_train)
        logger.info("Model trained successfully.")
        return model
    except Exception as e:
        logger.error(f"Error training model: {e}")
        raise

# Function to evaluate model
def evaluate_model(model, X_test, y_test):
    try:
        predictions = model.predict(X_test)
        accuracy = accuracy_score(y_test, predictions)
        auc_score = roc_auc_score(y_test, predictions)
        report = classification_report(y_test, predictions)
        logger.info(f"Model evaluation complete.\n{report}")
        return accuracy, auc_score, report
    except Exception as e:
        logger.error(f"Error evaluating model: {e}")
        raise

# Function to log model to MLflow
def log_model_to_mlflow(model, accuracy, auc_score, report):
    try:
        with mlflow.start_run():
            mlflow.log_metric("accuracy", accuracy)
            mlflow.log_metric("roc_auc_score", auc_score)
            mlflow.log_text(report, "classification_report.txt")
            mlflow.sklearn.log_model(model, "random_forest_model")
            logger.info("Model logged to MLflow successfully.")
    except Exception as e:
        logger.error(f"Error logging model to MLflow: {e}")
        raise

if __name__ == "__main__":
    data_file_path = "creditcard.csv"
    
    data = load_data(data_file_path)
    X, y = preprocess_data(data)
    X_balanced, y_balanced = balance_data(X, y)
    X_train, X_test, y_train, y_test = train_test_split(X_balanced, y_balanced, test_size=0.2, random_state=42)
    model = train_model(X_train, y_train)
    accuracy, auc_score, report = evaluate_model(model, X_test, y_test)
    log_model_to_mlflow(model, accuracy, auc_score, report)
```

### Pytest Test File

```python
# test_fraud_detection_pipeline.py

import pytest
import pandas as pd
from fraud_detection_pipeline import load_data, preprocess_data, balance_data

def test_load_data():
    data = load_data("creditcard.csv")
    assert isinstance(data, pd.DataFrame), "Data should be loaded as a DataFrame."
    assert not data.empty, "Data should not be empty."

def test_preprocess_data():
    data = pd.DataFrame({'Feature1': [1, 2, 3], 'Feature2': [4, 5, 6], 'Class': [0, 1, 0]})
    X, y = preprocess_data(data)
    assert X.shape[0] == y.shape[0], "Features and target should have the same number of samples."
    assert "Class" not in X.columns, "Features should not include the target column."

def test_balance_data():
    X = pd.DataFrame({'Feature1': [1, 2, 3], 'Feature2': [4, 5, 6]})
    y = pd.Series([0, 1, 0])
    X_balanced, y_balanced = balance_data(X, y)
    assert len(X_balanced) == len(y_balanced), "Balanced features and target should have the same number of samples."
    assert y_balanced.mean() == 0.5, "Balanced dataset should have equal class distribution."
```

### GitHub Actions YAML Workflow

```yaml
# .github/workflows/ci-cd.yml

name: Fraud Detection CI/CD

on:
  push:
    branches: [main]
  pull_request:
    branches: [main]
  workflow_dispatch:
  schedule:
    - cron: '0 0 * * SUN'

jobs:
  build:
    runs-on: ubuntu-latest

    steps:
    - name: Checkout code
      uses: actions/checkout@v2

    - name: Set up Python
      uses: actions/setup-python@v2
      with:
        python-version: '3.8'

    - name: Install dependencies
      run: |
        python -m pip install --upgrade pip
        pip install -r requirements.txt

    - name: Run tests
      run: |
        pytest test_fraud_detection_pipeline.py

    - name: Run fraud detection pipeline
      run: |
        python fraud_detection_pipeline.py

    - name: Upload model to MLflow
      env:
        MLFLOW_TRACKING_URI: ${{ secrets.MLFLOW_TRACKING_URI }}
      run: |
        # Assuming MLflow tracking server is already set up and accessible
        python fraud_detection_pipeline.py
```

Ensure that your repository has a `requirements.txt` file specifying the necessary Python packages including `pandas`, `scikit-learn`, `mlflow`, and `pytest` for the above scripts to run successfully. Additionally, set up secrets in GitHub for `MLFLOW_TRACKING_URI` to connect to your MLflow tracking server.
