Below is a complete CI/CD pipeline configuration for a fraud detection model using `scikit-learn`'s `RandomForestClassifier`. This includes a Python script for the pipeline, a pytest file for testing, and a GitHub Actions workflow for CI/CD.

### Python Pipeline Script

```python
# fraud_detection_pipeline.py

import pandas as pd
from sklearn.ensemble import RandomForestClassifier
from sklearn.model_selection import train_test_split
from sklearn.metrics import classification_report
import mlflow
import mlflow.sklearn
from sklearn.utils import resample
import logging
import numpy as np

logging.basicConfig(level=logging.INFO)

def load_data(file_path):
    try:
        data = pd.read_csv(file_path)
        return data
    except Exception as e:
        logging.error(f"Error loading data: {e}")
        raise

def preprocess_data(data):
    try:
        # Assuming 'Class' is the target column
        X = data.drop('Class', axis=1)
        y = data['Class']
        
        # Handle imbalance using resampling
        data_majority = data[data.Class == 0]
        data_minority = data[data.Class == 1]
        
        data_minority_upsampled = resample(data_minority, 
                                           replace=True,
                                           n_samples=len(data_majority), 
                                           random_state=42)
        
        data_upsampled = pd.concat([data_majority, data_minority_upsampled])
        
        X = data_upsampled.drop('Class', axis=1)
        y = data_upsampled['Class']
        
        return train_test_split(X, y, test_size=0.25, random_state=42)
    except Exception as e:
        logging.error(f"Error in data preprocessing: {e}")
        raise

def train_model(X_train, y_train):
    try:
        model = RandomForestClassifier(random_state=42)
        model.fit(X_train, y_train)
        return model
    except Exception as e:
        logging.error(f"Error during model training: {e}")
        raise

def evaluate_model(model, X_test, y_test):
    try:
        y_pred = model.predict(X_test)
        report = classification_report(y_test, y_pred)
        logging.info(f"Classification Report: \n{report}")
        return report
    except Exception as e:
        logging.error(f"Error during model evaluation: {e}")
        raise

def log_experiment(model, X_test, y_test):
    try:
        with mlflow.start_run():
            mlflow.sklearn.log_model(model, "random_forest_model")
            y_pred = model.predict(X_test)
            accuracy = np.mean(y_pred == y_test)
            mlflow.log_metric("accuracy", accuracy)
    except Exception as e:
        logging.error(f"Error logging experiment to MLflow: {e}")
        raise

def main(file_path):
    data = load_data(file_path)
    X_train, X_test, y_train, y_test = preprocess_data(data)
    model = train_model(X_train, y_train)
    evaluate_model(model, X_test, y_test)
    log_experiment(model, X_test, y_test)

if __name__ == "__main__":
    main("creditcard.csv")
```

### GitHub Actions Workflow

```yaml
# .github/workflows/ci-cd-pipeline.yml

name: CI/CD Pipeline

on:
  push:
    branches:
      - main
  pull_request:
    branches:
      - main
  workflow_dispatch:
  schedule:
    - cron: '0 0 * * 0'  # Weekly schedule at midnight on Sunday

jobs:
  build:
    runs-on: ubuntu-latest

    steps:
    - name: Checkout code
      uses: actions/checkout@v2

    - name: Set up Python
      uses: actions/setup-python@v2
      with:
        python-version: '3.8'

    - name: Install dependencies
      run: |
        python -m pip install --upgrade pip
        pip install -r requirements.txt

    - name: Run tests
      run: |
        pytest test_fraud_detection_pipeline.py

    - name: Run Pipeline
      run: |
        python fraud_detection_pipeline.py
```

Make sure to create a `requirements.txt` file with the necessary dependencies for your Python environment:

```plaintext
pandas
scikit-learn
mlflow
pytest
```

This setup includes error handling for data loading, preprocessing, training, evaluation, and logging steps, ensuring that the pipeline is robust and ready for production use. The GitHub Actions workflow will automatically run tests and the pipeline script on every push or pull request to the main branch, and it can also be triggered manually or on a schedule.
