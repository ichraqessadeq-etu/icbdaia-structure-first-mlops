Below are the deliverables for a complete CI/CD pipeline configuration for a fraud detection model using scikit-learn's `RandomForestClassifier`, with MLflow for experiment tracking, `pytest` for testing, and an automatic trigger for retraining when concept drift is detected.

### 1. Complete Python Pipeline Script

```python
import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.ensemble import RandomForestClassifier
from sklearn.metrics import classification_report, accuracy_score
from sklearn.exceptions import NotFittedError
import mlflow
import mlflow.sklearn
import logging
import joblib

# Configure logging
logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)

# Constants
DATA_PATH = 'creditcard.csv'
MODEL_PATH = 'model.joblib'
RANDOM_STATE = 42

def load_data():
    try:
        data = pd.read_csv(DATA_PATH)
        logger.info("Data loaded successfully.")
        return data
    except Exception as e:
        logger.error(f"Failed to load data: {e}")
        raise

def preprocess_data(data):
    try:
        X = data.drop('Class', axis=1)
        y = data['Class']
        X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.2, random_state=RANDOM_STATE, stratify=y)
        logger.info("Data split into train and test sets.")
        return X_train, X_test, y_train, y_test
    except Exception as e:
        logger.error(f"Failed to preprocess data: {e}")
        raise

def train_model(X_train, y_train):
    try:
        model = RandomForestClassifier(random_state=RANDOM_STATE)
        model.fit(X_train, y_train)
        logger.info("Model trained successfully.")
        return model
    except Exception as e:
        logger.error(f"Failed to train model: {e}")
        raise

def evaluate_model(model, X_test, y_test):
    try:
        predictions = model.predict(X_test)
        acc = accuracy_score(y_test, predictions)
        report = classification_report(y_test, predictions)
        logger.info("Model evaluated successfully.")
        return acc, report
    except NotFittedError as e:
        logger.error(f"Model is not fitted: {e}")
        raise
    except Exception as e:
        logger.error(f"Failed to evaluate model: {e}")
        raise

def log_to_mlflow(model, acc, report):
    try:
        mlflow.start_run()
        mlflow.log_metric("accuracy", acc)
        mlflow.log_text(report, "classification_report.txt")
        mlflow.sklearn.log_model(model, "model")
        mlflow.end_run()
        logger.info("Metrics logged to MLflow successfully.")
    except Exception as e:
        logger.error(f"Failed to log to MLflow: {e}")
        raise

def save_model(model):
    try:
        joblib.dump(model, MODEL_PATH)
        logger.info("Model saved successfully.")
    except Exception as e:
        logger.error(f"Failed to save model: {e}")
        raise

def main():
    data = load_data()
    X_train, X_test, y_train, y_test = preprocess_data(data)
    model = train_model(X_train, y_train)
    acc, report = evaluate_model(model, X_test, y_test)
    log_to_mlflow(model, acc, report)
    save_model(model)

if __name__ == "__main__":
    main()
```

    assert y_test.shape[0] > 0, "Test labels should not be empty"

      - main
  pull_request:

      - main

    runs-on: ubuntu-latest
