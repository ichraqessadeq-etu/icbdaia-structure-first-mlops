Below is a complete setup for a CI/CD pipeline for a fraud detection model using `scikit-learn`, `MLflow`, and `pytest`. The pipeline includes a Python module for data loading, model training, evaluation, and logging to MLflow, a pytest test file, and a GitHub Actions YAML workflow.

### 1. Python Pipeline Script

```python
# pipeline.py

import os
import pandas as pd
import numpy as np
from sklearn.ensemble import RandomForestClassifier
from sklearn.model_selection import train_test_split
from sklearn.metrics import classification_report, confusion_matrix
import mlflow
import mlflow.sklearn
from mlflow.exceptions import MlflowException

RANDOM_STATE = 42
MLFLOW_TRACKING_URI = os.getenv("MLFLOW_TRACKING_URI", "http://localhost:5000")
MLFLOW_EXPERIMENT_NAME = "Credit Card Fraud Detection"

def load_data():
    try:
        df = pd.read_csv('creditcard.csv')
        return df
    except FileNotFoundError as e:
        print(f"Data file not found: {e}")
        raise

def preprocess_data(df):
    try:
        X = df.drop('Class', axis=1)
        y = df['Class']
        return train_test_split(X, y, test_size=0.2, random_state=RANDOM_STATE)
    except KeyError as e:
        print(f"Column not found in the dataset: {e}")
        raise

def train_model(X_train, y_train):
    clf = RandomForestClassifier(random_state=RANDOM_STATE)
    clf.fit(X_train, y_train)
    return clf

def evaluate_model(clf, X_test, y_test):
    predictions = clf.predict(X_test)
    report = classification_report(y_test, predictions, output_dict=True)
    cm = confusion_matrix(y_test, predictions)
    return report, cm

def log_to_mlflow(clf, report, cm):
    try:
        mlflow.set_tracking_uri(MLFLOW_TRACKING_URI)
        mlflow.set_experiment(MLFLOW_EXPERIMENT_NAME)

        with mlflow.start_run():
            mlflow.sklearn.log_model(clf, "model")
            mlflow.log_metrics({
                "precision": report['1']['precision'],
                "recall": report['1']['recall'],
                "f1-score": report['1']['f1-score'],
            })
            mlflow.log_artifact("confusion_matrix.txt")

            with open("confusion_matrix.txt", "w") as f:
                f.write(np.array2string(cm))

    except MlflowException as e:
        print(f"Failed to log to MLflow: {e}")
        raise

if __name__ == "__main__":
    df = load_data()
    X_train, X_test, y_train, y_test = preprocess_data(df)
    clf = train_model(X_train, y_train)
    report, cm = evaluate_model(clf, X_test, y_test)
    log_to_mlflow(clf, report, cm)
```

### 2. Pytest Test File

```python
# test_pipeline.py

import pytest
import pandas as pd
from pipeline import load_data, preprocess_data, train_model, evaluate_model

def test_load_data():
    df = load_data()
    assert isinstance(df, pd.DataFrame)

def test_preprocess_data():
    df = load_data()
    X_train, X_test, y_train, y_test = preprocess_data(df)
    assert len(X_train) > 0
    assert len(X_test) > 0
    assert len(y_train) > 0
    assert len(y_test) > 0

def test_train_model():
    df = load_data()
    X_train, X_test, y_train, y_test = preprocess_data(df)
    clf = train_model(X_train, y_train)
    assert hasattr(clf, "predict"), "Model should have a predict method"
```

### 3. GitHub Actions YAML Workflow

```yaml
# .github/workflows/ci-cd.yml

name: CI/CD Pipeline

on:
  push:
    branches:
      - main
  pull_request:
    branches:
      - main
  schedule:
    - cron: '0 0 * * *' # Daily check for concept drift

jobs:
  build-and-test:
    runs-on: ubuntu-latest

    steps:
    - name: Checkout code
      uses: actions/checkout@v2

    - name: Set up Python
      uses: actions/setup-python@v2
      with:
        python-version: '3.8'

    - name: Install dependencies
      run: |
        python -m pip install --upgrade pip
        pip install -r requirements.txt

    - name: Run tests
      run: |
        pytest test_pipeline.py

    - name: Run pipeline
      run: |
        python pipeline.py

  concept-drift-check:
    runs-on: ubuntu-latest
    if: github.event_name == 'schedule'

    steps:
    - name: Checkout code
      uses: actions/checkout@v2

    - name: Set up Python
      uses: actions/setup-python@v2
      with:
        python-version: '3.8'

    - name: Install dependencies
      run: |
        python -m pip install --upgrade pip
        pip install -r requirements.txt

    - name: Check for concept drift
      run: |
        python check_concept_drift.py # This script should be implemented to detect concept drift
        # If concept drift is detected, re-trigger the pipeline
```

### Notes:
- Make sure to have a `requirements.txt` file listing all necessary Python packages.
- The `check_concept_drift.py` script should be implemented to check for concept drift and trigger the pipeline when detected.
- Ensure that the MLflow tracking server is correctly set up and accessible via the provided `MLFLOW_TRACKING_URI`.
