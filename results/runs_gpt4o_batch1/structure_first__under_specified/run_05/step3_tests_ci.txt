# tests/test_pipeline.py

import pytest
import pandas as pd
from src.data_preprocessing import load_data, clean_data, feature_engineering
from src.model_training import split_data, train_model
from src.model_evaluation import evaluate_model, generate_report
from sklearn.ensemble import RandomForestClassifier

# Sample data for testing
@pytest.fixture
def sample_data():
    return pd.DataFrame({
        'feature1': [1.0, 2.0, 3.0, 4.0],
        'feature2': [5.0, 6.0, 7.0, 8.0],
        'label': [0, 1, 0, 1]
    })

@pytest.fixture
def sample_labels(sample_data):
    return sample_data['label']

@pytest.fixture
def sample_features(sample_data):
    return sample_data.drop(columns=['label'])

def test_load_data(tmp_path):
    file_path = tmp_path / "test.csv"
    sample_data = "feature1,feature2,label\n1.0,5.0,0\n2.0,6.0,1\n3.0,7.0,0\n4.0,8.0,1\n"
    file_path.write_text(sample_data)

    data = load_data(file_path)
    assert not data.empty, "Data should not be empty after loading."

def test_clean_data(sample_data):
    raw_data = sample_data.append({'feature1': None, 'feature2': None, 'label': None}, ignore_index=True)
    cleaned_data = clean_data(raw_data)
    assert cleaned_data.shape[0] == 4, "Cleaned data should not have NaN values."

def test_feature_engineering(sample_features):
    features = feature_engineering(sample_features)
    assert features.shape[1] == 2, "Feature engineering should not change the number of features."
    assert not features.isna().any().any(), "Features should not contain NaN values."

def test_split_data(sample_features, sample_labels):
    X_train, X_test, y_train, y_test = split_data(sample_features, sample_labels, test_size=0.5)
    assert len(X_train) == 2, "Training set should have 2 samples."
    assert len(X_test) == 2, "Test set should have 2 samples."

def test_train_model(sample_features, sample_labels):
    X_train, _, y_train, _ = split_data(sample_features, sample_labels, test_size=0.5)
    model = train_model(X_train, y_train)
    assert isinstance(model, RandomForestClassifier), "The model should be a RandomForestClassifier."

def test_evaluate_model(sample_features, sample_labels):
    X_train, X_test, y_train, y_test = split_data(sample_features, sample_labels, test_size=0.5)
    model = train_model(X_train, y_train)
    results = evaluate_model(model, X_test, y_test)
    assert 'accuracy' in results, "Evaluation results should contain accuracy."
    assert 'report' in results, "Evaluation results should contain a classification report."

def test_generate_report(capsys):
    evaluation_results = {'accuracy': 0.75, 'report': 'Sample Report'}
    generate_report(evaluation_results)
    captured = capsys.readouterr()
    assert "Accuracy: 0.75" in captured.out, "Report should contain accuracy."
    assert "Sample Report" in captured.out, "Report should contain the classification report."

### (b) GitHub Actions Workflow

Create a GitHub Actions workflow file named `python-app.yml` in the `.github/workflows` directory to automate testing.

```yaml
# .github/workflows/python-app.yml

name: Python application

on:
  push:
    branches: [main]
  pull_request:
    branches: [main]

jobs:
  build:
    runs-on: ubuntu-latest

    steps:
    - uses: actions/checkout@v2

    - name: Set up Python
      uses: actions/setup-python@v2
      with:
        python-version: '3.8'

    - name: Install dependencies
      run: |
        python -m pip install --upgrade pip
        pip install pytest pandas scikit-learn

    - name: Run tests
      run: |
        pytest tests
```

This workflow will trigger on pushes and pull requests to the `main` branch, set up Python 3.8, install the necessary dependencies, and run the tests using `pytest`.
