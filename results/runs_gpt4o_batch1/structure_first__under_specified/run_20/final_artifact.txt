```python
import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.ensemble import RandomForestClassifier
from sklearn.metrics import accuracy_score
import joblib

def load_data(file_path: str) -> pd.DataFrame:
    """Load data from a file path."""
    try:
        data = pd.read_csv(file_path)
        return data
    except Exception as e:
        raise RuntimeError(f"Failed to load data from {file_path}: {e}")

def preprocess_data(data: pd.DataFrame) -> pd.DataFrame:
    """Preprocess the input data."""
    # Example preprocessing: drop missing values and encode categorical variables
    data = data.dropna()
    categorical_columns = data.select_dtypes(include=['object']).columns
    data = pd.get_dummies(data, columns=categorical_columns)
    return data

def split_data(data: pd.DataFrame, test_size: float) -> tuple:
    """Split the data into training and testing sets."""
    X = data.drop('label', axis=1)  # Assume 'label' is the target column
    y = data['label']
    X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=test_size, random_state=42)
    return X_train, X_test, y_train, y_test

def train_model(train_data: tuple, model_params: dict) -> RandomForestClassifier:
    """Train the fraud detection model."""
    X_train, _, y_train, _ = train_data
    model = RandomForestClassifier(**model_params)
    model.fit(X_train, y_train)
    return model

def evaluate_model(model: RandomForestClassifier, test_data: tuple) -> float:
    """Evaluate the trained model on test data."""
    _, X_test, _, y_test = test_data
    predictions = model.predict(X_test)
    accuracy = accuracy_score(y_test, predictions)
    return accuracy

def save_model(model: RandomForestClassifier, save_path: str) -> None:
    """Save the trained model to a file."""
    joblib.dump(model, save_path)

def load_model(load_path: str) -> RandomForestClassifier:
    """Load a model from a file."""
    try:
        model = joblib.load(load_path)
        return model
    except Exception as e:
        raise RuntimeError(f"Failed to load the model from {load_path}: {e}")

def predict(model: RandomForestClassifier, input_data: pd.DataFrame) -> pd.Series:
    """Make predictions using the loaded model."""
    predictions = model.predict(input_data)
    return pd.Series(predictions)
```

### Explanation
- **Loading Data:** The `load_data` function reads a CSV file into a DataFrame.
- **Preprocessing Data:** The `preprocess_data` function handles missing values and encodes categorical features.
- **Splitting Data:** The `split_data` function splits the dataset into training and testing sets, assuming a column named 'label' as the target.
- **Training Model:** The `train_model` function trains a RandomForestClassifier using the provided parameters.
- **Evaluating Model:** The `evaluate_model` function computes the accuracy of the model on the test dataset.
- **Saving and Loading Model:** The `save_model` and `load_model` functions handle model persistence using `joblib`.
- **Predicting:** The `predict` function generates predictions for the input data using the trained model.
