```python
# tests/test_pipeline.py

import pytest
import pandas as pd
from sklearn.ensemble import RandomForestClassifier
from fraud_detection.pipeline.data_processing import load_data, preprocess_data, split_data
from fraud_detection.pipeline.model_training import train_model, save_model
from fraud_detection.pipeline.model_evaluation import evaluate_model

@pytest.fixture
def sample_data():
    """Fixture for creating a sample DataFrame."""
    data = {
        'feature1': [1, 2, 3, 4, 5],
        'feature2': ['A', 'B', 'A', 'B', 'A'],
        'target': [0, 1, 0, 1, 0]
    }
    return pd.DataFrame(data)

def test_load_data(tmp_path):
    """Test the load_data function."""
    # Create a temporary CSV file
    file_path = tmp_path / "data.csv"
    df = pd.DataFrame({
        'feature1': [1, 2],
        'feature2': ['A', 'B'],
        'target': [0, 1]
    })
    df.to_csv(file_path, index=False)

    loaded_df = load_data(file_path)
    pd.testing.assert_frame_equal(df, loaded_df)

def test_preprocess_data(sample_data):
    """Test the preprocess_data function."""
    processed_df = preprocess_data(sample_data)
    assert 'feature2_A' in processed_df.columns
    assert 'feature2_B' in processed_df.columns
    assert processed_df.isnull().sum().sum() == 0

def test_split_data(sample_data):
    """Test the split_data function."""
    processed_df = preprocess_data(sample_data)
    X_train, X_test, y_train, y_test = split_data(processed_df, test_size=0.2)
    
    assert len(X_train) + len(X_test) == len(processed_df)
    assert len(y_train) + len(y_test) == len(sample_data)

def test_train_model(sample_data):
    """Test the train_model function."""
    processed_df = preprocess_data(sample_data)
    X_train, _, y_train, _ = split_data(processed_df, test_size=0.2)
    model = train_model(X_train, y_train)
    
    assert isinstance(model, RandomForestClassifier)

def test_save_model(tmp_path, sample_data):
    """Test the save_model function."""
    model_file_path = tmp_path / "model.pkl"
    processed_df = preprocess_data(sample_data)
    X_train, _, y_train, _ = split_data(processed_df, test_size=0.2)
    model = train_model(X_train, y_train)
    
    save_model(model, model_file_path)
    assert model_file_path.exists()

def test_evaluate_model(sample_data):
    """Test the evaluate_model function."""
    processed_df = preprocess_data(sample_data)
    X_train, X_test, y_train, y_test = split_data(processed_df, test_size=0.2)
    model = train_model(X_train, y_train)
    accuracy = evaluate_model(model, X_test, y_test)
    
    assert 0.0 <= accuracy <= 1.0
```

### (b) GitHub Actions Workflow

Here's a GitHub Actions workflow for running the tests:

```yaml
# .github/workflows/test_pipeline.yml

name: Test Pipeline

on: [push, pull_request]

jobs:
  test:
    runs-on: ubuntu-latest
    steps:
    - name: Checkout code
      uses: actions/checkout@v2

    - name: Set up Python
      uses: actions/setup-python@v2
      with:
        python-version: '3.8'

    - name: Install dependencies
      run: |
        python -m pip install --upgrade pip
        pip install pytest pandas scikit-learn joblib

    - name: Run tests
      run: |
        pytest tests/
```

In the above workflow, the jobs will be triggered on every push and pull request to the repository. The tests are run in a virtual environment with Python 3.8, and dependencies such as `pytest`, `pandas`, `scikit-learn`, and `joblib` are installed before running the tests.
