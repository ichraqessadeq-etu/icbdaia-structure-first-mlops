```python
import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.ensemble import RandomForestClassifier
from sklearn.metrics import accuracy_score
import joblib

def load_data(file_path: str) -> pd.DataFrame:
    """Load data from the specified file path."""
    data = pd.read_csv(file_path)
    return data

def preprocess_data(data: pd.DataFrame) -> pd.DataFrame:
    """Perform preprocessing on the data."""
    # Example preprocessing steps
    # Handle missing values, encode categorical variables, etc.
    data = data.dropna()  # Drop missing values
    # Assuming 'label' is the target column and others are features
    features = data.drop('label', axis=1)
    target = data['label']
    return features, target

def train_model(features: pd.DataFrame, target: pd.Series) -> RandomForestClassifier:
    """Train the fraud detection model."""
    X_train, X_test, y_train, y_test = train_test_split(features, target, test_size=0.2, random_state=42)
    model = RandomForestClassifier(n_estimators=100, random_state=42)
    model.fit(X_train, y_train)
    return model, X_test, y_test

def evaluate_model(model: RandomForestClassifier, X_test: pd.DataFrame, y_test: pd.Series) -> float:
    """Evaluate the model using test data."""
    predictions = model.predict(X_test)
    accuracy = accuracy_score(y_test, predictions)
    print(f"Model Accuracy: {accuracy}")
    return accuracy

def save_model(model: RandomForestClassifier, file_path: str) -> None:
    """Save the trained model to the specified file path."""
    joblib.dump(model, file_path)

def main():
    """Main function to execute the pipeline."""
    data_path = 'path/to/your/data.csv'
    model_path = 'path/to/save/your/model.pkl'
    
    data = load_data(data_path)
    features, target = preprocess_data(data)
    model, X_test, y_test = train_model(features, target)
    evaluate_model(model, X_test, y_test)
    save_model(model, model_path)

if __name__ == "__main__":
    main()
```

### Explanation:

- **Loading Data**: Assumes the data is in CSV format and uses `pandas` to load it into a DataFrame.
- **Preprocessing Data**: Includes handling missing values and separating features from the target. You might need more advanced preprocessing depending on your dataset.
- **Training the Model**: Uses a RandomForestClassifier from `sklearn` to train the model.
- **Evaluating the Model**: Prints and returns the accuracy of the model.
- **Saving the Model**: Uses `joblib` to save the trained model for later use.
- **Main Function**: Orchestrates the entire pipeline from loading data to saving the model.

You can modify the paths and preprocessing steps according to your specific needs and dataset characteristics.
