# test_pipeline_module.py

import pytest
import pandas as pd
from pipeline_module import load_data, preprocess_data, train_model, evaluate_model, save_model, load_model, predict
from sklearn.linear_model import LogisticRegression
import os
import numpy as np

# Sample data for testing
sample_data = pd.DataFrame({
    'feature1': [1, 2, 3, 4, 5],
    'feature2': [5, 4, 3, 2, 1],
    'target': [0, 1, 0, 1, 0]
})

def test_load_data(tmpdir):
    # Create a temporary CSV file
    file_path = tmpdir.join("sample.csv")
    sample_data.to_csv(file_path, index=False)

    # Test loading the data
    data = load_data(file_path)
    pd.testing.assert_frame_equal(data, sample_data)

def test_preprocess_data():
    X_train, X_test, y_train, y_test = preprocess_data(sample_data)

    assert X_train.shape[0] > 0
    assert X_test.shape[0] > 0
    assert y_train.shape[0] > 0
    assert y_test.shape[0] > 0

def test_train_model():
    preprocessed_data = preprocess_data(sample_data)
    model = train_model(preprocessed_data)
    assert isinstance(model, LogisticRegression)

def test_evaluate_model():
    preprocessed_data = preprocess_data(sample_data)
    model = train_model(preprocessed_data)
    accuracy = evaluate_model(model, preprocessed_data)
    assert isinstance(accuracy, float)
    assert 0 <= accuracy <= 1

def test_save_and_load_model(tmpdir):
    # Train a model for testing
    preprocessed_data = preprocess_data(sample_data)
    model = train_model(preprocessed_data)
    
    # Save the model
    file_path = tmpdir.join("model.pkl")
    save_model(model, file_path)
    
    # Load the model
    loaded_model = load_model(file_path)
    
    # Check if the loaded model is the same as the saved model
    assert isinstance(loaded_model, LogisticRegression)

def test_predict():
    preprocessed_data = preprocess_data(sample_data)
    model = train_model(preprocessed_data)
    X_train, _, _, _ = preprocessed_data
    
    predictions = predict(model, X_train)
    assert len(predictions) == len(X_train)
    assert all(isinstance(pred, np.integer) for pred in predictions)

### (b) GitHub Actions Workflow

```yaml
name: Python Package

on:
  push:
    branches: [ main ]
  pull_request:
    branches: [ main ]

jobs:
  build:

    runs-on: ubuntu-latest

    steps:
    - name: Checkout code
      uses: actions/checkout@v2

    - name: Set up Python
      uses: actions/setup-python@v2
      with:
        python-version: '3.8'

    - name: Install dependencies
      run: |
        python -m pip install --upgrade pip
        pip install pandas scikit-learn joblib pytest

    - name: Run tests
      run: |
        pytest test_pipeline_module.py
```
