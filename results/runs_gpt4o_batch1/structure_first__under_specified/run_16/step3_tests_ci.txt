# test_pipeline.py

import pytest
import pandas as pd
from sklearn.ensemble import RandomForestClassifier
from pipeline_module import load_data, preprocess_data, train_model, evaluate_model, save_model

def test_load_data():
    file_path = 'sample_data.csv'
    # Create a sample CSV to test loading
    sample_data = pd.DataFrame({'feature1': [1, 2, 3], 'label': [0, 1, 0]})
    sample_data.to_csv(file_path, index=False)

    data = load_data(file_path)
    assert isinstance(data, pd.DataFrame)
    assert not data.empty

def test_preprocess_data():
    data = pd.DataFrame({'feature1': [1, None, 3], 'label': [0, 1, 0]})
    preprocessed_data = preprocess_data(data)
    
    assert not preprocessed_data.isnull().values.any()

def test_train_model():
    data = pd.DataFrame({'feature1': [1, 2, 3, 4], 'label': [0, 1, 0, 1]})
    preprocessed_data = preprocess_data(data)
    model = train_model(preprocessed_data)
    
    assert isinstance(model, RandomForestClassifier)

def test_evaluate_model():
    data = pd.DataFrame({'feature1': [1, 2, 3, 4], 'label': [0, 1, 0, 1]})
    preprocessed_data = preprocess_data(data)
    model = train_model(preprocessed_data)
    accuracy = evaluate_model(model, preprocessed_data)
    
    assert isinstance(accuracy, float)
    assert 0 <= accuracy <= 1

def test_save_model(tmp_path):
    model = RandomForestClassifier()
    file_path = tmp_path / "model.joblib"
    save_model(model, file_path)
    
    assert file_path.exists()

### (b) GitHub Actions Workflow

```yaml
# .github/workflows/python-package.yml

name: Python package

on:
  push:
    branches: [ main ]
  pull_request:
    branches: [ main ]

jobs:
  build:

    runs-on: ubuntu-latest

    steps:
    - uses: actions/checkout@v2

    - name: Set up Python
      uses: actions/setup-python@v2
      with:
        python-version: '3.8'

    - name: Install dependencies
      run: |
        python -m pip install --upgrade pip
        pip install pytest
        pip install pandas
        pip install scikit-learn
        pip install joblib

    - name: Run tests
      run: |
        pytest
```
