### GitHub Actions Workflow (`ci_cd_pipeline.yml`)
```yaml
name: CI/CD Pipeline

on:
  push:
    branches:
      - main
  pull_request:
    branches:
      - main

jobs:
  setup:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout code
        uses: actions/checkout@v2

      - name: Set up Python
        uses: actions/setup-python@v2
        with:
          python-version: '3.x'

  install_dependencies:
    runs-on: ubuntu-latest
    needs: setup
    steps:
      - name: Install dependencies
        run: |
          python -m pip install --upgrade pip
          pip install -r requirements.txt

  lint:
    runs-on: ubuntu-latest
    needs: install_dependencies
    steps:
      - name: Run code linting
        run: |
          pip install flake8
          flake8 fraud_detection/

  test:
    runs-on: ubuntu-latest
    needs: install_dependencies
    steps:
      - name: Run tests
        run: |
          pip install pytest
          pytest tests/

  build_and_deploy:
    runs-on: ubuntu-latest
    needs: test
    steps:
      - name: Build and deploy model
        run: |
          python fraud_detection/pipeline/pipeline_module.py
          # Add deployment steps here
```
