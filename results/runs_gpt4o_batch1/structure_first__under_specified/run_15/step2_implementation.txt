import pandas as pd
from sklearn.preprocessing import StandardScaler

def load_data(file_path: str) -> pd.DataFrame:

    df = pd.read_csv(file_path)

    df = df.dropna()  # Drop missing values

    return df

def feature_engineering(df: pd.DataFrame) -> pd.DataFrame:
    """Apply feature engineering to the input DataFrame."""
    scaler = StandardScaler()
    df[df.columns] = scaler.fit_transform(df[df.columns])
    return df

# pipeline/model_training.py

from sklearn.model_selection import train_test_split
from sklearn.ensemble import RandomForestClassifier

def split_data(df: pd.DataFrame, test_size: float, random_state: int) -> tuple:
    """Split the data into train and test sets."""
    X = df.drop('target', axis=1)
    y = df['target']
    X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=test_size, random_state=random_state)
    return X_train, X_test, y_train, y_test

def train_model(X_train: pd.DataFrame, y_train: pd.Series) -> RandomForestClassifier:
    """Train the model using the training data."""
    model = RandomForestClassifier(random_state=42)
    model.fit(X_train, y_train)
    return model

# pipeline/model_evaluation.py

from sklearn.metrics import accuracy_score, precision_score, recall_score, f1_score

def evaluate_model(model: RandomForestClassifier, X_test: pd.DataFrame, y_test: pd.Series) -> dict:
    """Evaluate the model on the test data and return performance metrics."""
    y_pred = model.predict(X_test)
    metrics = {
        'accuracy': accuracy_score(y_test, y_pred),
        'precision': precision_score(y_test, y_pred, average='binary'),
        'recall': recall_score(y_test, y_pred, average='binary'),
        'f1_score': f1_score(y_test, y_pred, average='binary')
    }
    return metrics

# pipeline/model_deployment.py

import joblib

def save_model(model: RandomForestClassifier, file_path: str) -> None:
    """Save the trained model to a specified file path."""
    joblib.dump(model, file_path)

def load_model(file_path: str) -> RandomForestClassifier:
    """Load a model from a specified file path."""
    model = joblib.load(file_path)
    return model
