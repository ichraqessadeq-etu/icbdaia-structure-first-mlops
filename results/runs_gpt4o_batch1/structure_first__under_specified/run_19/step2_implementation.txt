# fraud_detection_pipeline/pipeline/data_processing.py

import pandas as pd
from sklearn.model_selection import train_test_split

def load_data(source: str) -> pd.DataFrame:
    """Load data from the specified source."""
    try:
        data = pd.read_csv(source)
        return data
    except Exception as e:
        raise ValueError(f"Error loading data from {source}: {e}")

def preprocess_data(raw_data: pd.DataFrame) -> pd.DataFrame:
    """Preprocess the raw data for model consumption."""
    try:
        # Example preprocessing: handle missing values and encode categorical variables
        processed_data = raw_data.fillna(method='ffill')
        processed_data = pd.get_dummies(processed_data)
        return processed_data
    except Exception as e:
        raise ValueError(f"Error preprocessing data: {e}")

def split_data(processed_data: pd.DataFrame, test_size: float) -> tuple:
    """Split the data into training and testing sets."""
    try:
        train_data, test_data = train_test_split(processed_data, test_size=test_size)
        return train_data, test_data
    except Exception as e:
        raise ValueError(f"Error splitting data: {e}")

# fraud_detection_pipeline/pipeline/model_training.py

from sklearn.ensemble import RandomForestClassifier
import joblib

def train_model(training_data: pd.DataFrame, model_params: dict) -> RandomForestClassifier:
    """Train the fraud detection model with given parameters."""
    try:
        features = training_data.drop('target', axis=1)
        target = training_data['target']
        model = RandomForestClassifier(**model_params)
        model.fit(features, target)
        return model
    except Exception as e:
        raise ValueError(f"Error training model: {e}")

def save_model(model: RandomForestClassifier, destination: str) -> None:
    """Save the trained model to the specified destination."""
    try:
        joblib.dump(model, destination)
    except Exception as e:
        raise ValueError(f"Error saving model to {destination}: {e}")

# fraud_detection_pipeline/pipeline/model_evaluation.py

from sklearn.metrics import classification_report

def evaluate_model(model: RandomForestClassifier, test_data: pd.DataFrame) -> dict:
    """Evaluate the model using the test data and return metrics."""
    try:
        features = test_data.drop('target', axis=1)
        target = test_data['target']
        predictions = model.predict(features)
        report = classification_report(target, predictions, output_dict=True)
        return report
    except Exception as e:
        raise ValueError(f"Error evaluating model: {e}")

def log_evaluation_results(results: dict, log_destination: str) -> None:
    """Log the evaluation results to the specified destination."""
    try:
        with open(log_destination, 'w') as file:
            for key, value in results.items():
                file.write(f"{key}: {value}\n")
    except Exception as e:
        raise ValueError(f"Error logging evaluation results to {log_destination}: {e}")
