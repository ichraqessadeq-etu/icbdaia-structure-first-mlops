# tests/test_pipeline.py

import pytest
import pandas as pd
from sklearn.ensemble import RandomForestClassifier
from fraud_detection.pipeline.data_preprocessing import load_data, clean_data, split_data
from fraud_detection.pipeline.feature_engineering import generate_features, select_features
from fraud_detection.pipeline.model_training import train_model, save_model
from fraud_detection.pipeline.model_evaluation import evaluate_model

@pytest.fixture
def raw_data():
    return pd.DataFrame({
        'feature': [1, 2, 3, None, 5],
        'feature1': [10, 20, 30, 40, 50],
        'feature3': [100, 200, 300, 400, 500],
        'label': [0, 1, 0, 1, 0]
    })

def test_load_data(raw_data, tmpdir):
    file_path = tmpdir.join('test_data.csv')
    raw_data.to_csv(file_path, index=False)
    loaded_data = load_data(file_path)
    pd.testing.assert_frame_equal(raw_data, loaded_data)

def test_clean_data(raw_data):
    cleaned_data = clean_data(raw_data)
    assert cleaned_data.isnull().sum().sum() == 0

def test_split_data(raw_data):
    cleaned_data = clean_data(raw_data)
    train_data, test_data = split_data(cleaned_data)
    assert len(train_data) + len(test_data) == len(cleaned_data)

def test_generate_features(raw_data):
    features = generate_features(raw_data)
    assert 'feature_squared' in features.columns
    assert all(features['feature_squared'] == features['feature'] ** 2)

def test_select_features(raw_data):
    features = generate_features(raw_data)
    selected_features = select_features(features)
    assert list(selected_features.columns) == ['feature1', 'feature_squared', 'feature3']

def test_train_model(raw_data):
    features = generate_features(raw_data)
    selected_features = select_features(features)
    labels = raw_data['label']
    model = train_model(selected_features, labels)
    assert isinstance(model, RandomForestClassifier)

def test_evaluate_model(raw_data):
    features = generate_features(raw_data)
    selected_features = select_features(features)
    labels = raw_data['label']
    model = train_model(selected_features, labels)
    train_data, test_data = split_data(raw_data)
    test_features = select_features(generate_features(test_data))
    test_labels = test_data['label']
    
    evaluation_metrics = evaluate_model(model, test_features, test_labels)
    assert 'accuracy' in evaluation_metrics
    assert 'precision' in evaluation_metrics
    assert 'recall' in evaluation_metrics

### GitHub Actions Workflow File

```yaml
# .github/workflows/python-app.yml

name: Python application

on:
  push:
    branches: [main]
  pull_request:
    branches: [main]

jobs:
  build:

    runs-on: ubuntu-latest

    steps:
    - uses: actions/checkout@v2

    - name: Set up Python
      uses: actions/setup-python@v2
      with:
        python-version: '3.8'

    - name: Install dependencies
      run: |
        python -m pip install --upgrade pip
        pip install pytest pandas scikit-learn joblib

    - name: Test with pytest
      run: |
        pytest tests
```

### Explanation

1. **Pytest Test File**: This file defines tests for each of the functionalities in the pipeline module. It uses pytest fixtures to create a mock dataset and then tests each function from loading data to model evaluation.

2. **GitHub Actions Workflow File**: This workflow is triggered on push and pull requests to the `main` branch. It sets up a Python environment, installs the necessary dependencies, and runs the tests using `pytest`.
