### GitHub Actions YAML Workflow

#### `.github/workflows/ci_cd_pipeline.yml`
```yaml
name: CI/CD Pipeline for Fraud Detection

on:
  push:
    branches:
      - main
  pull_request:
    branches:
      - main
  schedule:
    - cron: '0 0 * * 0' # Weekly run
  workflow_dispatch:

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout code
        uses: actions/checkout@v2

      - name: Set up Python
        uses: actions/setup-python@v2
        with:
          python-version: '3.8'

      - name: Install dependencies
        run: |
          python -m pip install --upgrade pip
          pip install -r requirements.txt

      - name: Run tests
        run: |
          pytest tests/

  deploy:
    runs-on: ubuntu-latest
    needs: build
    if: success()
    steps:
      - name: Deploy model
        run: |
          # Deployment steps
          ...
```
