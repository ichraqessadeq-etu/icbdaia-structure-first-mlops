### GitHub Actions Workflow

#### `.github/workflows/ci_cd_pipeline.yml`

```yaml
name: CI/CD Pipeline

on:
  push:
    branches:
      - main
  pull_request:
    branches:
      - main
  schedule:
    - cron: '0 0 * * 0'  # Weekly run

jobs:
  setup:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout repository
        uses: actions/checkout@v2

      - name: Set up Python
        uses: actions/setup-python@v2
        with:
          python-version: '3.8'

      - name: Install dependencies
        run: pip install -r requirements.txt

  test:
    runs-on: ubuntu-latest
    needs: setup
    steps:
      - name: Run Pytest
        run: pytest tests/

  drift-detection:
    runs-on: ubuntu-latest
    needs: test
    steps:
      - name: Detect Concept Drift
        run: python fraud_detection_pipeline/drift_detector.py

  retrain:
    runs-on: ubuntu-latest
    if: needs.drift-detection.outputs.drift_detected == 'true'
    steps:
      - name: Retrain Model
        run: python fraud_detection_pipeline/main_pipeline.py --file-path data/credit_card.csv
```

This structure provides a clear outline for organizing the pipeline, testing, and CI/CD processes. Each component is modular and can be developed independently while integrating seamlessly into the overall system.
