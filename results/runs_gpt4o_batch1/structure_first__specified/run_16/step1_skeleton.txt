### GitHub Actions Workflow (`.github/workflows/ci_cd_pipeline.yml`)
```yaml
name: CI/CD Pipeline

on:
  push:
    branches:
      - main
  pull_request:
    branches:
      - main
  workflow_dispatch:

jobs:
  setup:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout repository
        uses: actions/checkout@v2
      
      - name: Set up Python
        uses: actions/setup-python@v2
        with:
          python-version: '3.8'

      - name: Install dependencies
        run: |
          python -m pip install --upgrade pip
          pip install -r fraud_detection_pipeline/requirements.txt

  test:
    runs-on: ubuntu-latest
    needs: setup
    steps:
      - name: Checkout repository
        uses: actions/checkout@v2
      
      - name: Run tests
        run: |
          pytest fraud_detection_pipeline/test_pipeline.py

  deploy:
    runs-on: ubuntu-latest
    needs: test
    steps:
      - name: Checkout repository
        uses: actions/checkout@v2
      
      - name: Execute pipeline
        run: |
          python fraud_detection_pipeline/pipeline.py --file-path <path-to-data>
```
