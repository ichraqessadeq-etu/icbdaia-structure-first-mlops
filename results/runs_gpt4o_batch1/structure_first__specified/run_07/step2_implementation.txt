# pipeline/data_loader.py
import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler

def load_data(file_path: str) -> tuple:
    """Load the credit card fraud dataset."""
    data = pd.read_csv(file_path)
    return data

def preprocess_data(data: pd.DataFrame) -> tuple:
    """Preprocess the data to handle imbalance and prepare features."""
    X = data.drop('Class', axis=1)
    y = data['Class']

    # Feature scaling
    scaler = StandardScaler()
    X_scaled = scaler.fit_transform(X)

    # Splitting the dataset
    X_train, X_test, y_train, y_test = train_test_split(X_scaled, y, test_size=0.2, random_state=42, stratify=y)
    return X_train, X_test, y_train, y_test

# pipeline/model_trainer.py
from sklearn.ensemble import RandomForestClassifier
from sklearn.metrics import accuracy_score
import numpy as np

def train_model(X_train: pd.DataFrame, y_train: pd.Series) -> RandomForestClassifier:
    """Train a RandomForestClassifier model."""
    model = RandomForestClassifier(n_estimators=100, random_state=42)
    model.fit(X_train, y_train)
    return model

def check_concept_drift(X_new: pd.DataFrame, model: RandomForestClassifier) -> bool:
    """Check for concept drift between old and new data."""
    try:
        predictions = model.predict(X_new)
        accuracy = accuracy_score(y_new, predictions)
        # Placeholder threshold for drift detection
        drift_detected = accuracy < 0.8
        return drift_detected
    except Exception as e:
        print(f"Error checking concept drift: {e}")
        return False

# pipeline/model_evaluator.py
from sklearn.metrics import classification_report, roc_auc_score

def evaluate_model(model: RandomForestClassifier, X_test: pd.DataFrame, y_test: pd.Series) -> dict:
    """Evaluate the model and return metrics."""
    predictions = model.predict(X_test)
    report = classification_report(y_test, predictions, output_dict=True)
    roc_auc = roc_auc_score(y_test, model.predict_proba(X_test)[:, 1])
    report['roc_auc'] = roc_auc
    return report

import mlflow
import mlflow.sklearn

def log_to_mlflow(params: dict, metrics: dict, model: RandomForestClassifier):
    """Log the model, parameters, and metrics to MLflow."""
    mlflow.start_run()
    mlflow.log_params(params)
    mlflow.log_metrics(metrics)
    mlflow.sklearn.log_model(model, "model")
    mlflow.end_run()
