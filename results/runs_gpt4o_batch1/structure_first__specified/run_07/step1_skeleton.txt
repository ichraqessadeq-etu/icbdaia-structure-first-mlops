### GitHub Actions Workflow

#### `.github/workflows/ci_cd_pipeline.yml`
```yaml
name: CI/CD Pipeline

on:
  push:
    branches:
      - main
  pull_request:
    branches:
      - main
  schedule:
    - cron: '0 0 * * 0'  # Every Sunday at midnight

jobs:
  setup:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout code
        uses: actions/checkout@v2

      - name: Set up Python
        uses: actions/setup-python@v2
        with:
          python-version: '3.8'

      - name: Install dependencies
        run: pip install -r requirements.txt

  test:
    runs-on: ubuntu-latest
    needs: setup
    steps:
      - name: Run tests
        run: pytest tests/

  train-and-log:
    runs-on: ubuntu-latest
    needs: test
    steps:
      - name: Train and log model
        run: python -m pipeline.model_trainer

  detect-concept-drift:
    runs-on: ubuntu-latest
    needs: train-and-log
    steps:
      - name: Detect concept drift
        run: python -m pipeline.model_trainer --check-drift
```

This structure ensures that the pipeline is modular, testable, and easily integrated into a CI/CD framework using GitHub Actions.
