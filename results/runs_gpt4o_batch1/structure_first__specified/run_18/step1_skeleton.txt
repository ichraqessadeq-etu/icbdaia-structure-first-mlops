### ci_cd_pipeline.yml

```yaml
name: CI/CD Pipeline

on:
  push:
    branches:
      - main
  pull_request:
    branches:
      - main
  schedule:
    - cron: '0 0 * * 0'  # Weekly schedule for concept drift check

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout code
        uses: actions/checkout@v2
      
      - name: Set up Python
        uses: actions/setup-python@v2
        with:
          python-version: '3.8'
      
      - name: Install dependencies
        run: |
          python -m pip install --upgrade pip
          pip install -r requirements.txt
      
      - name: Run tests
        run: |
          pytest --cov=fraud_detection_pipeline test_pipeline.py
      
      - name: Detect concept drift
        run: |
          python -c "from pipeline import detect_concept_drift; detect_concept_drift()"
      
      - name: Run pipeline
        if: ${{ github.event_name == 'push' || github.event_name == 'pull_request' || steps.drift.outputs.concept_drift_detected == 'true' }}
        run: |
          python pipeline.py

  deploy:
    runs-on: ubuntu-latest
    needs: build
    if: github.ref == 'refs/heads/main'
    steps:
      - name: Checkout code
        uses: actions/checkout@v2

      - name: Deploy model
        run: echo "Deploying model..."
```
