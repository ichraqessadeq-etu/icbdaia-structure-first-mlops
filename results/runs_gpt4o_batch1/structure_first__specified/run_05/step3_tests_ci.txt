```python
from src.pipeline import main

@patch('src.data_loader.load_data')

@patch('src.evaluator.evaluate_model')
@patch('src.evaluator.log_to_mlflow')

                            mock_detect_concept_drift, mock_preprocess_data, mock_load_data):

    mock_load_data.return_value = 'mocked data'
```

```yaml
    - name: Install dependencies
      run: |
        python -m pip install --upgrade pip
        pip install -r requirements.txt

    - name: Run tests
      run: pytest tests/
```
