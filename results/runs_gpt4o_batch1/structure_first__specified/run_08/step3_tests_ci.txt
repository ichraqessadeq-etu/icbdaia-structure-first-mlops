### (a) Complete `pytest` Test File

Let's create a test file named `test_pipeline.py` that tests the different components of the pipeline.

```python
# test/test_pipeline.py
import pytest
import pandas as pd
from sklearn.datasets import make_classification
from pipeline.data_loading import load_data
from pipeline.model_training import train_model
from pipeline.model_evaluation import evaluate_model

@pytest.fixture
def generate_synthetic_data():
    """Fixture to create synthetic data for testing."""
    X, y = make_classification(n_samples=100, n_features=20, n_informative=15, n_redundant=5, random_state=42)
    df = pd.DataFrame(X, columns=[f'feature_{i}' for i in range(X.shape[1])])
    df['target'] = y
    return df

def test_load_data(generate_synthetic_data, tmp_path):
    """Test the data loading function."""
    file_path = tmp_path / "synthetic_data.csv"
    generate_synthetic_data.to_csv(file_path, index=False)

    X_train, X_test, y_train, y_test = load_data(file_path)
    
    assert X_train.shape[0] > 0
    assert X_test.shape[0] > 0
    assert y_train.shape[0] > 0
    assert y_test.shape[0] > 0
    assert X_train.shape[1] == generate_synthetic_data.shape[1] - 1

def test_train_model(generate_synthetic_data, tmp_path):
    """Test the model training function."""
    file_path = tmp_path / "synthetic_data.csv"
    generate_synthetic_data.to_csv(file_path, index=False)

    X_train, X_test, y_train, y_test = load_data(file_path)
    
    model = train_model(X_train, y_train)
    
    assert model is not None
    assert hasattr(model, "predict")

def test_evaluate_model(generate_synthetic_data, tmp_path):
    """Test the model evaluation function."""
    file_path = tmp_path / "synthetic_data.csv"
    generate_synthetic_data.to_csv(file_path, index=False)

    X_train, X_test, y_train, y_test = load_data(file_path)
    model = train_model(X_train, y_train)
    metrics = evaluate_model(model, X_test, y_test)
    
    assert 'accuracy' in metrics
    assert 'precision' in metrics
    assert 'recall' in metrics
    assert 'f1_score' in metrics
    for key, value in metrics.items():
        assert isinstance(value, float)
        assert 0.0 <= value <= 1.0
```

### (b) Complete GitHub Actions Workflow

Below is a YAML file for GitHub Actions to automate the testing of your pipeline. Save this as `.github/workflows/pipeline.yml`.

```yaml
name: MLOps Pipeline CI

on:
  push:
    branches:
      - main
  pull_request:
    branches:
      - main

jobs:
  test:
    runs-on: ubuntu-latest

    steps:
    - name: Checkout code
      uses: actions/checkout@v2

    - name: Set up Python
      uses: actions/setup-python@v2
      with:
        python-version: '3.8'

    - name: Install dependencies
      run: |
        python -m pip install --upgrade pip
        pip install pandas scikit-learn pytest

    - name: Run tests
      run: |
        pytest test/test_pipeline.py
```

This setup assumes you have a `requirements.txt` or equivalent for dependencies, but for simplicity, the required packages are installed directly in the workflow. Adjust the Python version and dependencies as necessary for your specific environment.
