# pipeline.py

import argparse
from src.data_loader import load_data
from src.train_model import train_random_forest
from src.evaluate import evaluate_model
from src.detect_drift import detect_concept_drift

def run_pipeline(data_path: str):
    """Execute the full pipeline: load data, train model, evaluate, and log results."""
    
    # Step 1: Load and preprocess the data
    print("Loading and preprocessing data...")
    (X_train, y_train), (X_test, y_test) = load_data(data_path)
    print("Data loaded successfully.")
    
    # Step 2: Train the model
    print("Training the model...")
    model = train_random_forest(X_train, y_train)
    print("Model trained successfully.")

    # Step 3: Evaluate the model
    print("Evaluating the model...")
    evaluation_metrics = evaluate_model(model, X_test, y_test)
    print("Model evaluation completed.")
    print("Evaluation Metrics:", evaluation_metrics)

    # Step 4: Detect concept drift (as an example, using X_test as new data)
    print("Detecting concept drift...")
    reference_stats = {"mean": X_train.mean(), "std": X_train.std()}  # Example reference stats
    drift_detected = detect_concept_drift(X_test, reference_stats)
    print(f"Concept drift detected: {drift_detected}")
    
    # Log results (for demonstration purposes, we're simply printing them)
    print("Pipeline execution completed.")
    if drift_detected:
        print("Warning: Concept drift detected. Consider retraining the model with new data.")
    else:
        print("No concept drift detected. Model performance is stable.")

if __name__ == "__main__":
    parser = argparse.ArgumentParser(description='Run the fraud detection pipeline.')
