### GitHub Actions Workflow (`.github/workflows/ci_cd_pipeline.yml`)

```yaml
name: CI/CD Pipeline

on:
  push:
    branches:
      - main
  pull_request:
    branches:
      - main
  schedule:
    - cron: '0 0 * * 0'  # Weekly schedule
  workflow_dispatch:

jobs:
  test:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout code
        uses: actions/checkout@v2

      - name: Set up Python
        uses: actions/setup-python@v2
        with:
          python-version: '3.8'

      - name: Install dependencies
        run: pip install -r requirements.txt

      - name: Run tests
        run: pytest tests/

  train_and_log:
    runs-on: ubuntu-latest
    needs: test
    steps:
      - name: Checkout code
        uses: actions/checkout@v2

      - name: Set up Python
        uses: actions/setup-python@v2
        with:
          python-version: '3.8'

      - name: Install dependencies
        run: pip install -r requirements.txt

      - name: Detect concept drift
        run: python -c "from src.concept_drift_detector import detect_concept_drift; exit(0) if detect_concept_drift() else exit(1)"

      - name: Run pipeline script
        if: success()
        run: python -m src.pipeline

  deploy:
    runs-on: ubuntu-latest
    needs: train_and_log
    steps:
      - name: Deploy model
        run: echo "Deploy model step here"
```
